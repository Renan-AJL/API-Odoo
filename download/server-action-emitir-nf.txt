# --- Validacoes ---

if record.move_type not in ('out_invoice', 'out_refund'):
    raise UserError('Selecione uma fatura de saida (venda ou devolucao).')

if record.x_studio_status_emissao == 'pendente':
    raise UserError('Ja existe uma emissao em andamento para esta fatura. Aguarde o processamento.')

if record.x_studio_status_emissao == 'autorizada':
    raise UserError('Esta fatura ja possui NF autorizada. Numero: ' + str(record.x_studio_numero_nf or ''))

if not record.partner_id:
    raise UserError('A fatura precisa ter um cliente (parceiro) definido.')

if not record.invoice_line_ids:
    raise UserError('A fatura nao possui linhas de itens.')

# --- Detectar tipo: NF-e (produto) ou NFS-e (servico) ---

tem_servico = False
tem_produto = False

for linha in record.invoice_line_ids:
    if linha.display_type:
        continue
    if linha.product_id:
        if linha.product_id.detailed_type == 'service':
            tem_servico = True
        else:
            tem_produto = True
    else:
        tem_produto = True

if tem_servico and not tem_produto:
    tipo_label = 'NFS-e (Servico)'
    tipo_nf = 'nfse'
elif tem_produto and not tem_servico:
    tipo_label = 'NF-e (Produto)'
    tipo_nf = 'nfe'
else:
    tipo_label = 'NF-e (Misto)'
    tipo_nf = 'nfe'

# --- Marcar como pendente e postar no chatter ---

record.write({
    'x_studio_status_emissao': 'pendente',
})

record.message_post(
    body='<b>Emissao de NF Solicitada</b><br/>'
         'Tipo: ' + tipo_label + '<br/>'
         'Fatura: ' + str(record.name or '') + '<br/>'
         'Valor: R$ ' + str(record.amount_total or 0) + '<br/>'
         '<br/>'
         'Status: <b>Aguardando processamento pelo middleware...</b><br/>'
         '<i>O XML e o PDF (DANFE) aparecerao aqui assim que a nota for autorizada pela SEFAZ.</i>',
    message_type='comment',
    subtype_xmlid='mail.mt_note',
)
