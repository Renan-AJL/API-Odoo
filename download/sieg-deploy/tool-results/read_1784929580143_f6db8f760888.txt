   104→   104→// XML-RPC connection (singleton)
   105→   105→// ============================================================
   106→   106→var _cached = null;
   107→   107→
   108→   108→function getConn() {
   109→   109→  if (_cached && _cached.uid) return _cached;
   110→   110→  var odooConfig = config.odoo;
   111→   111→  var base = odooConfig.url.replace(/\/+$/, '');
   112→   112→  var host = base.replace('https://', '').replace('http://', '');
   113→   113→  var common = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/common', port: 443 });
   114→   114→  var models = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
   115→   115→  _cached = { uid: null, common: common, models: models, db: odooConfig.db, password: odooConfig.password };
   116→   116→  return _cached;
   117→   117→}
   118→   118→
   119→   119→function authenticate() {
   120→   120→  return new Promise(function(resolve, reject) {
   121→   121→    var conn = getConn();
   122→   122→    conn.common.methodCall('authenticate', [conn.db, config.odoo.user, conn.password, {}], function(err, uid) {
   123→   123→      if (err) { reject(new Error('Auth Odoo falhou: ' + (err.message || JSON.stringify(err)))); }
   124→   124→      else if (uid === false || uid === null) { reject(new Error('Auth Odoo: credenciais invalidas (UID=false)')); }
   125→   125→      else { conn.uid = uid; logger.info('[ODOO-TE] Autenticado. UID: ' + uid); resolve(uid); }
   126→   126→    });
   127→   127→  });
   128→   128→}
   129→   129→
   130→   130→function ensureAuth() {
   131→   131→  var conn = getConn();
   132→   132→  if (conn.uid) return Promise.resolve(conn.uid);
   133→   133→  return authenticate();
   134→   134→}
   135→   135→
   136→   136→function executeKw(model, method, args, kwargs) {
   137→   137→  return new Promise(function(resolve, reject) {
   138→   138→    ensureAuth().then(function(uid) {
   139→   139→      var conn = getConn();
   140→   140→      var params = [conn.db, uid, conn.password, model, method, args || []];
   141→   141→      if (kwargs) params.push(kwargs);
   142→   142→      conn.models.methodCall('execute_kw', params, function(err, result) {
   143→   143→        if (err) { reject(new Error(model + '.' + method + ': ' + (err.message || JSON.stringify(err)))); }
   144→   144→        else { resolve(result); }
   145→   145→      });
   146→   146→    }).catch(reject);
   147→   147→  });
   148→   148→}

   152→   152→// ============================================================
   153→   153→var _customFieldsCache = {};
   154→   154→
   155→   155→async function safeReadCustom(model, ids, nativeFields, customFields) {
   156→   156→  try {
   157→   157→    return await executeKw(model, 'read', [ids], { fields: nativeFields.concat(customFields) });
   158→   158→  } catch (err) {
   159→   159→    if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   160→   160→      if (!_customFieldsCache[model]) {
   161→   161→        _customFieldsCache[model] = { available: false, checked: true };
   162→   162→        logger.warn('[ODOO-TE] Campos x_studio_* nao encontrados em ' + model + '. Usando campos nativos.');
   163→   163→      }
   164→   164→      return await executeKw(model, 'read', [ids], { fields: nativeFields });
   165→   165→    }
   166→   166→    throw err;
   167→   167→  }
   168→   168→}
   169→   169→
   170→   170→async function safeWriteCustom(model, ids, vals) {
   171→   171→  var customVals = {};
   172→   172→  var nativeVals = {};
   173→   173→  Object.keys(vals).forEach(function(k) {
   174→   174→    if (k.indexOf('x_studio_') === 0) customVals[k] = vals[k];
   175→   175→    else nativeVals[k] = vals[k];
   176→   176→  });
   177→   177→  if (Object.keys(nativeVals).length > 0) await executeKw(model, 'write', [ids, nativeVals]);
   178→   178→  if (Object.keys(customVals).length > 0) {
   179→   179→    try { await executeKw(model, 'write', [ids, customVals]); }
   180→   180→    catch (err) {
   181→   181→      if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   182→   182→        if (!_customFieldsCache[model] || !_customFieldsCache[model].warned) {
   183→   183→          logger.warn('[ODOO-TE] Campo x_studio_* nao existe em ' + model + '. Gravacao ignorada.');
   184→   184→          if (!_customFieldsCache[model]) _customFieldsCache[model] = {};
   185→   185→          _customFieldsCache[model].warned = true;
   186→   186→        }
   187→   187→      } else throw err;
   188→   188→    }
   189→   189→  }
   190→   190→}

   205→   205→}
