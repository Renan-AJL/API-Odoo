   103→// ============================================================
   104→// XML-RPC connection (singleton)
   105→// ============================================================
   106→var _cached = null;
   107→
   108→function getConn() {
   109→  if (_cached && _cached.uid) return _cached;
   110→  var odooConfig = config.odoo;
   111→  var base = odooConfig.url.replace(/\/+$/, '');
   112→  var host = base.replace('https://', '').replace('http://', '');
   113→  var common = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/common', port: 443 });
   114→  var models = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
   115→  _cached = { uid: null, common: common, models: models, db: odooConfig.db, password: odooConfig.password };
   116→  return _cached;
   117→}
   118→
   119→function authenticate() {
   120→  return new Promise(function(resolve, reject) {
   121→    var conn = getConn();
   122→    conn.common.methodCall('authenticate', [conn.db, config.odoo.user, conn.password, {}], function(err, uid) {
   123→      if (err) { reject(new Error('Auth Odoo falhou: ' + (err.message || JSON.stringify(err)))); }
   124→      else if (uid === false || uid === null) { reject(new Error('Auth Odoo: credenciais invalidas (UID=false)')); }
   125→      else { conn.uid = uid; logger.info('[ODOO-TE] Autenticado. UID: ' + uid); resolve(uid); }
   126→    });
   127→  });
   128→}
   129→
   130→function ensureAuth() {
   131→  var conn = getConn();
   132→  if (conn.uid) return Promise.resolve(conn.uid);
   133→  return authenticate();
   134→}
   135→
   136→function executeKw(model, method, args, kwargs) {
   137→  return new Promise(function(resolve, reject) {
   138→    ensureAuth().then(function(uid) {
   139→      var conn = getConn();
   140→      var params = [conn.db, uid, conn.password, model, method, args || []];
   141→      if (kwargs) params.push(kwargs);
   142→      conn.models.methodCall('execute_kw', params, function(err, result) {
   143→        if (err) { reject(new Error(model + '.' + method + ': ' + (err.message || JSON.stringify(err)))); }
   144→        else { resolve(result); }
   145→      });
   146→    }).catch(reject);
   147→  });
   148→}

   150→// ============================================================
   151→// Safe read/write — tenta com x_studio, fallback sem
   152→// ============================================================
   153→var _customFieldsCache = {};
   154→
   155→async function safeReadCustom(model, ids, nativeFields, customFields) {
   156→  try {
   157→    return await executeKw(model, 'read', [ids], { fields: nativeFields.concat(customFields) });
   158→  } catch (err) {
   159→    if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   160→      if (!_customFieldsCache[model]) {
   161→        _customFieldsCache[model] = { available: false, checked: true };
   162→        logger.warn('[ODOO-TE] Campos x_studio_* nao encontrados em ' + model + '. Usando campos nativos.');
   163→      }
   164→      return await executeKw(model, 'read', [ids], { fields: nativeFields });
   165→    }
   166→    throw err;
   167→  }
   168→}
   169→
   170→async function safeWriteCustom(model, ids, vals) {
   171→  var customVals = {};
   172→  var nativeVals = {};
   173→  Object.keys(vals).forEach(function(k) {
   174→    if (k.indexOf('x_studio_') === 0) customVals[k] = vals[k];
   175→    else nativeVals[k] = vals[k];
   176→  });
   177→  if (Object.keys(nativeVals).length > 0) await executeKw(model, 'write', [ids, nativeVals]);
   178→  if (Object.keys(customVals).length > 0) {
   179→    try { await executeKw(model, 'write', [ids, customVals]); }
   180→    catch (err) {
   181→      if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   182→        if (!_customFieldsCache[model] || !_customFieldsCache[model].warned) {
   183→          logger.warn('[ODOO-TE] Campo x_studio_* nao existe em ' + model + '. Gravacao ignorada.');
   184→          if (!_customFieldsCache[model]) _customFieldsCache[model] = {};
   185→          _customFieldsCache[model].warned = true;
   186→        }
   187→      } else throw err;
   188→    }
   189→  }
   190→}

   199→async function searchRead(model, domain, fields, limit, offset, order) {
   200→  var kwargs = { fields: fields, limit: limit || 0, offset: offset || 0 };
   201→  if (order) kwargs.order = order;
   202→  var ids = await executeKw(model, 'search', [domain], kwargs);
   203→  if (!ids || !ids.length) return [];
   204→  return await executeKw(model, 'read', [ids], { fields: fields });
   205→}

   291→async function findSaleOrderByInvoice(invoiceId) {
   292→  console.log('[ODOO-TE] findSaleOrderByInvoice: invoiceId=' + invoiceId);
   293→
   294→  // Estrategia 1: sale.order.invoice_ids
   295→  try {
   296→    var saleIds = await executeKw('sale.order', 'search', [['invoice_ids', 'in', [invoiceId]]]);
   297→    if (saleIds && saleIds.length) {
   298→      var orders = await executeKw('sale.order', 'read', [saleIds], { fields: ['id', 'name'] });
   299→      if (orders && orders[0]) { console.log('[ODOO-TE] Estrategia 1 (invoice_ids) encontrou: ' + orders[0].name); return orders[0]; }
   300→    }
   301→    console.log('[ODOO-TE] Estrategia 1 (invoice_ids): sem resultados');
   302→  } catch (err) { console.warn('[ODOO-TE] Estrategia 1 falhou: ' + err.message); }
   303→
   304→  // Estrategia 2: sale_order_ids da fatura (campo computado)
   305→  try {
   306→    var invData = await executeKw('account.move', 'read', [[invoiceId]], { fields: ['id', 'partner_id', 'sale_order_ids'] });
   307→    var inv = invData && invData[0];
   308→    console.log('[ODOO-TE] Fatura lida: id=' + (inv ? inv.id : 'null') + ', partner=' + JSON.stringify(inv ? inv.partner_id : null) + ', sale_order_ids=' + JSON.stringify(inv ? inv.sale_order_ids : null));
   309→    if (inv && inv.sale_order_ids && inv.sale_order_ids.length) {
   310→      var soId = Array.isArray(inv.sale_order_ids[0]) ? inv.sale_order_ids[0][0] : inv.sale_order_ids[0];
   311→      var orders = await executeKw('sale.order', 'read', [[soId]], { fields: ['id', 'name'] });
   312→      if (orders && orders[0]) { console.log('[ODOO-TE] Estrategia 2 (sale_order_ids) encontrou: ' + orders[0].name); return orders[0]; }
   313→    }
   314→    console.log('[ODOO-TE] Estrategia 2 (sale_order_ids): sem resultados');
   315→  } catch (err) { console.warn('[ODOO-TE] Estrategia 2 falhou: ' + err.message); }
   316→
   317→  // Estrategia 3: account.move.line -> sale_line_ids -> order_id
   318→  try {
   319→    var invData = await executeKw('account.move', 'read', [[invoiceId]], { fields: ['line_ids'] });
   320→    var inv = invData && invData[0];
   321→    var lineIds = (inv && inv.line_ids) || [];
   322→    console.log('[ODOO-TE] Estrategia 3: ' + lineIds.length + ' linhas da fatura');
   323→    if (lineIds.length) {
   324→      var lines = await executeKw('account.move.line', 'read', [lineIds], { fields: ['sale_line_ids'] });
   325→      if (lines) {
   326→        for (var i = 0; i < lines.length; i++) {
   327→          var sli = lines[i].sale_line_ids;
   328→          if (sli && sli.length) {
   329→            var saleLineId = Array.isArray(sli[0]) ? sli[0][0] : sli[0];
   330→            var saleLines = await executeKw('sale.order.line', 'read', [[saleLineId]], { fields: ['order_id'] });
   331→            if (saleLines && saleLines[0] && saleLines[0].order_id) {
   332→              var orderId = Array.isArray(saleLines[0].order_id) ? saleLines[0].order_id[0] : saleLines[0].order_id;
   333→              var orders = await executeKw('sale.order', 'read', [[orderId]], { fields: ['id', 'name'] });
   334→              if (orders && orders[0]) { console.log('[ODOO-TE] Estrategia 3 (lines) encontrou: ' + orders[0].name); return orders[0]; }
   335→            }
   336→          }
   337→        }
   338→      }
   339→    }
   340→    console.log('[ODOO-TE] Estrategia 3 (lines): sem resultados');
   341→  } catch (err) { console.warn('[ODOO-TE] Estrategia 3 falhou: ' + err.message); }
   342→
   343→  // Estrategia 4: partner_id fallback
   344→  try {
   345→    var invData = await executeKw('account.move', 'read', [[invoiceId]], { fields: ['partner_id'] });
   346→    var inv = invData && invData[0];
   347→    var partnerId = inv && inv.partner_id ? (Array.isArray(inv.partner_id) ? inv.partner_id[0] : inv.partner_id) : null;
   348→    if (partnerId) {
   349→      console.log('[ODOO-TE] Estrategia 4: buscando sale.order por partner_id=' + partnerId);
   350→      var orders = await safeReadCustom('sale.order', null, ['id', 'name', 'state'], []);
   351→      // search + read
   352→      var ids = await executeKw('sale.order', 'search', [
   353→        ['partner_id', '=', partnerId], ['state', 'in', ['sale', 'done']],
   354→      ], { limit: 5, order: 'id desc' });
   355→      if (ids && ids.length) {
   356→        var orders = await executeKw('sale.order', 'read', [ids], { fields: ['id', 'name'] });
   357→        if (orders && orders[0]) { console.log('[ODOO-TE] Estrategia 4 (partner) encontrou: ' + orders[0].name); return orders[0]; }
   358→      }
   359→    }
   360→  } catch (err) { console.warn('[ODOO-TE] Estrategia 4 falhou: ' + err.message); }
   361→
   362→  console.error('[ODOO-TE] Nenhuma estrategia encontrou sale.order para invoice ' + invoiceId);
   363→  return null;
   364→}
