    17→// POST /api/v1/te/deliveries/send — Envia pedidos do ODOO ao TE
    18→// Body: { orderIds?: number[], pickingIds?: number[], orderType?: 1|2|3|4|5 }
    19→// -------------------------------------------------------
    20→router.post('/deliveries/send', async (req, res, next) => {
    21→  try {
    22→    const { orderIds, pickingIds, orderType = 1 } = req.body;
    23→
    24→    if (!orderIds?.length && !pickingIds?.length) {
    25→      return res.status(400).json({ success: false, error: 'Informe orderIds ou pickingIds' });
    26→    }
    27→
    28→    // 1. Busca pedidos ODOO
    29→    const pickings = [];
    30→    const saleOrders = [];
    31→
    32→    if (pickingIds?.length) {
    33→      const found = await odooTe.read('stock.picking', pickingIds, [
    34→        'id', 'name', 'partner_id', 'scheduled_date', 'move_line_count', 'origin',
    35→      ]);
    36→      pickings.push(...found);
    37→    }
    38→
    39→    if (orderIds?.length) {
    40→      const soFields = odooTe.getStudioFields('sale.order');
    41→      const found = await odooTe.read('sale.order', orderIds, [
    42→        'id', 'name', 'state', 'partner_id', 'amount_total', 'note', ...soFields,
    43→      ]);
    44→      saleOrders.push(...found);
    45→    }
    46→
    47→    // 2. Busca parceiros
    48→    const partnerIds = [
    49→      ...pickings.map(p => p.partner_id?.[0]),
    50→      ...saleOrders.map(o => o.partner_id?.[0]),
    51→    ].filter(Boolean);
    52→
    53→    const partners = partnerIds.length > 0
    54→      ? await odooTe.read('res.partner', [...new Set(partnerIds)], [
    55→          'id', 'name', 'cnpj_cpf', 'vat', 'phone', 'mobile', 'email',
    56→          'street', 'street_number', 'street2', 'zip', 'city',
    57→          'l10n_br_district', 'partner_latitude', 'partner_longitude',
    58→          'state_id', 'country_id',
    59→          ...odooTe.getStudioFields('res.partner'),
    60→        ])
    61→      : [];
    62→
    63→    const partnerMap = {};
    64→    partners.forEach(p => { partnerMap[p.id] = p; });
    65→
    66→    // 3. Mapeia para entregas TE
    67→    const teDeliveries = [];
    68→
    69→    for (const so of saleOrders) {
    70→      const partner = partnerMap[so.partner_id?.[0]] || {};
    71→      const picking = pickings.find(p => p.origin === so.name) || null;
    72→      const teDelivery = Mapper.odooToTeDelivery(so, partner, picking);
    73→      teDelivery.OrderType = orderType;
    74→      teDeliveries.push(teDelivery);
    75→    }
    76→
    77→    for (const picking of pickings) {
    78→      if (saleOrders.some(so => so.name === picking.origin)) continue; // ja mapeado
    79→      const partner = partnerMap[picking.partner_id?.[0]] || {};
    80→      const fakeSo = { name: picking.origin || `WH-${picking.id}`, note: '', id: 0 };
    81→      const teDelivery = Mapper.odooToTeDelivery(fakeSo, partner, picking);
    82→      teDelivery.OrderType = orderType;
    83→      teDeliveries.push(teDelivery);
    84→    }
    85→
    86→    if (teDeliveries.length === 0) {
    87→      return res.json({ success: true, sent: 0, message: 'Nenhuma entrega para enviar' });
    88→    }
    89→
    90→    // 4. Envia ao TE em lotes de 50
    91→    const results = [];
    92→    const BATCH = 50;
    93→
    94→    for (let i = 0; i < teDeliveries.length; i += BATCH) {
    95→      const batch = teDeliveries.slice(i, i + BATCH);
    96→      const result = await teClient.createDeliveries(batch);
    97→      results.push(result);
    98→
    99→      // 5. Marca como sincronizado no ODOO
   100→      const batchOrderNums = batch.map(d => d.OrderNumber);
   101→
   102→      const syncedSo = saleOrders.filter(so =>
   103→        batchOrderNums.includes(so.name) || batchOrderNums.includes(`WH-${so.id}`)
   104→      );
   105→      const syncedPickings = pickings.filter(p =>
   106→        batchOrderNums.includes(p.origin) || batchOrderNums.includes(`WH-${p.id}`)
   107→      );
   108→
   109→      if (syncedSo.length > 0) {
   110→        await odooTe.markSaleOrdersSynced(syncedSo.map(s => s.id), batchOrderNums[0], orderType);
   111→      }
   112→      if (syncedPickings.length > 0) {
   113→        await odooTe.markPickingsSynced(syncedPickings.map(p => p.id), batchOrderNums[0]);
   114→      }
   115→    }
   116→
   117→    console.log(`[TE-DELIVERY] Enviadas ${teDeliveries.length} entregas ao TE`);
   118→    res.json({
   119→      success: true,
   120→      sent: teDeliveries.length,
   121→      batches: results.length,
   122→      results,
   123→    });
   124→  } catch (err) {
   125→    console.error('[TE-DELIVERY] Erro:', err.message);
   126→    const status = err.isAxiosError ? (err.response?.status || 502) : 500;
   127→    res.status(status).json({ success: false, error: err.message });
   128→  }
   129→});

   183→    res.json({
   184→      success: true,

   186→      situation: mapped.situation,
   187→      situationLabel: mapped.situationLabel,
   188→      trackingCode: mapped.trackingCode,
   189→      driverName: mapped.driverName,
   190→      odooUpdated: !!(picking || saleOrder),
   191→      raw: delivery,
   192→    });
   193→  } catch (err) {
   194→    console.error('[TE-DELIVERY] Erro status:', err.message);

   280→// GET /api/v1/te/deliveries/pull — Puxa status e atualiza ODOO
   281→// -------------------------------------------------------
   282→router.get('/deliveries/pull', async (req, res) => {
   283→  try {
   284→    const { dateFrom, dateTo, situation } = req.query;
   285→
   286→    const params = { page: 1 };
   287→    if (dateFrom) params.dateFrom = dateFrom;
   288→    if (dateTo) params.dateTo = dateTo;
   289→    if (situation) params.situation = parseInt(situation, 10);
   290→
   291→    const allDeliveries = await teClient.fetchAllPages('/api/Entregas', params);
   292→
   293→    let updated = 0;
   294→    let notFound = 0;
   295→
   296→    for (const d of allDeliveries) {
   297→      const orderNum = d.OrderNumber || d.orderNumber;
   298→      const picking = await odooTe.findPickingByTeId(orderNum);
   299→
   300→      if (picking) {
   301→        const mapped = Mapper.teToOdooPicking(d);
   302→        await odooTe.updatePickingTeData(picking.id, mapped);
   303→        updated++;
   304→      } else {
   305→        const so = await odooTe.findSaleOrderByTeId(orderNum);
   306→        if (so) {
   307→          await odooTe.updateSaleOrderTeData(so.id, Mapper.teToOdooSaleOrder(d));
   308→          updated++;
   309→        } else {
   310→          notFound++;
   311→        }
   312→      }
   313→    }
   314→
   315→    res.json({
   316→      success: true,
   317→      totalFetched: allDeliveries.length,
   318→      updatedInOdoo: updated,
   319→      notFoundInOdoo: notFound,
   320→    });
   321→  } catch (err) {
   322→    console.error('[TE-DELIVERY] Erro pull:', err.message);
   323→    res.status(500).json({ success: false, error: err.message });
   324→  }
   325→});

   359→function fmtDate(dateStr) {
   360→  if (!dateStr) return '';
   361→  try {
   362→    const d = new Date(dateStr);
   363→    if (isNaN(d.getTime())) return dateStr;
   364→    return d.toLocaleString('pt-BR', { timeZone: 'America/Sao_Paulo',
   365→      day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit' });
   366→  } catch { return dateStr; }
   367→}

   814→// POST /api/v1/te/delivery-status — Busca status TE e grava card HTML
   815→// Body: { saleOrderId: number }
   816→// ============================================================
   817→router.post('/delivery-status', async (req, res) => {
   818→  try {
   819→    const { saleOrderId } = req.body;
   820→    if (!saleOrderId) {
   821→      return res.status(400).json({ success: false, error: 'Informe saleOrderId' });
   822→    }
   823→
   824→    // 1. Busca sale order no Odoo
   825→    const so = await odooTe.findSaleOrderById(saleOrderId);
   826→    if (!so) {
   827→      return res.status(404).json({ success: false, error: `Sale Order ${saleOrderId} nao encontrada` });
   828→    }
   829→
   830→    const f = odooTe.FIELDS['sale.order'];
   831→    const orderName = so.name;
   832→    const teOrderId = so[f.teOrderId];
   833→    const trackingCode = so[f.teTrackingCode];
   834→
   835→    if (!teOrderId && !trackingCode) {
   836→      // Nenhuma entrega enviada ao TE ainda
   837→      const noCard = buildErrorCard(
   838→        'Sem Entrega TE',
   839→        `Pedido ${orderName} ainda nao foi enviado ao TudoEntregue. Envie a entrega primeiro.`
   840→      );
   841→      await odooTe.updateSaleOrderStatusHtml(saleOrderId, noCard);
   842→      return res.json({ success: true, message: 'Pedido ainda nao enviado ao TE', card: noCard });
   843→    }
   844→
   845→    // 2. Busca dados no TE — /api/Entregas (dados completos da entrega)
   846→    let deliveryData = null;
   847→    if (teOrderId) {
   848→      const result = await teClient.getDeliveries({ orderNumber: teOrderId, page: 1 });
   849→      deliveryData = result?.Result?.[0] || null;
   850→    }
   851→
   852→    // 3. Busca situacao detalhada — /orders/situation
   853→    let situationData = null;
   854→    if (teOrderId) {
   855→      try {
   856→        situationData = await teClient.getOrderSituation({
   857→          orderType: 1,
   858→          orderID: teOrderId,
   859→        });
   860→      } catch (err) {
   861→        console.warn(`[TE-DELIVERY-STATUS] /orders/situation falhou: ${err.message}`);
   862→      }
   863→    }
   864→
   865→    // 4. Busca tracking detalhado (ocorrencias com fotos)
   866→    let trackingData = null;
   867→    if (trackingCode) {
   868→      try {
   869→        trackingData = await teClient.getTracking(trackingCode);
   870→      } catch (err) {
   871→        console.warn(`[TE-DELIVERY-STATUS] /tracking falhou: ${err.message}`);
   872→      }
   873→    }
   874→
   875→    // 5. Monta card HTML
   876→    const cardHtml = buildTeCard(deliveryData || {}, trackingData, situationData);
   877→
   878→    // 6. Tambem atualiza campos padrao do TE
   879→    if (deliveryData) {
   880→      const mapped = Mapper.teToOdooPicking(deliveryData);
   881→      const soMapped = Mapper.teToOdooSaleOrder(deliveryData);
   882→      await odooTe.updateSaleOrderTeData(saleOrderId, soMapped);
   883→    }
   884→
   885→    // 7. Atualiza campo de motorista (selection) a partir do TE
   886→    let motoristaAtualizado = null;
   887→    if (deliveryData) {
   888→      const driverName = deliveryData.DriverName || deliveryData.driverName || '';
   889→      if (driverName) {
   890→        motoristaAtualizado = matchMotoristaSelection(driverName);
   891→        if (motoristaAtualizado) {
   892→          await odooTe.updateSaleOrderMotorista(saleOrderId, motoristaAtualizado);
   893→          console.log(`[TE-DELIVERY-STATUS] Motorista: ${driverName} -> ${motoristaAtualizado}`);
   894→        }
   895→      }
   896→    }
   897→
   898→    // 8. Grava card no campo HTML
   899→    await odooTe.updateSaleOrderStatusHtml(saleOrderId, cardHtml);
   900→
   901→    console.log(`[TE-DELIVERY-STATUS] Card atualizado: SO ${orderName} (${saleOrderId})`);
   902→    res.json({
   903→      success: true,
   904→      orderName,
   905→      teOrderId,
   906→      trackingCode,
   907→      motorista: motoristaAtualizado || null,
   908→      hasDeliveryData: !!deliveryData,
   909→      hasSituationData: !!situationData,
   910→      hasTrackingData: !!trackingData,
   911→      card: cardHtml,
   912→    });
   913→  } catch (err) {
   914→    console.error('[TE-DELIVERY-STATUS] Erro:', err.message);
   915→    // Grava card de erro no campo
   916→    try {
   917→      await odooTe.updateSaleOrderStatusHtml(req.body.saleOrderId, buildErrorCard('Erro ao Buscar Status', err.message));
   918→    } catch {}
   919→    const status = err.isAxiosError ? (err.response?.status || 502) : 500;
   920→    res.status(status).json({ success: false, error: err.message });
   921→  }
   922→});

   924→// ============================================================
   925→// MATCH MOTORISTA — Faz match do nome do TE com as opcoes do campo selection
   926→// O campo x_studio_motorista e um selection com 17 opcoes (key/label)
   927→// ============================================================
   928→const MOTORISTA_OPTIONS = [
   929→  'adelson', 'adriano', 'anderlucio', 'antonio', 'carlos', 'daniel',
   930→  'diego', 'eduardo', 'fellipe', 'gerson', 'glauber',
   931→  'henrique', 'italo', 'jeferson', 'joao_pedro', 'lucas',
   932→  'matheus', 'rafael',
   933→];
   934→
   935→function matchMotoristaSelection(driverNameFromTe) {
   936→  if (!driverNameFromTe) return null;
   937→  const name = driverNameFromTe.toUpperCase().trim();
   938→  // Remove sufixos comuns como "MOTORISTA", "FRETE", etc
   939→  const cleanName = name
   940→    .replace(/\b(MOTORISTA|FRETE|ENTREGAS|MOTOBOY)\b/g, '')
   941→    .replace(/[^A-ZÀ-Ú\s]/g, '')
   942→    .trim();
   943→  const parts = cleanName.split(/\s+/).filter(Boolean);
   944→
   945→  // Tenta match exato com a key ou com o primeiro/ultimo nome
   946→  for (const key of MOTORISTA_OPTIONS) {
   947→    const keyUpper = key.toUpperCase();
   948→    if (name.includes(keyUpper) || keyUpper.includes(name)) return key;
   949→  }
   950→  // Tenta match com qualquer parte do nome
   951→  for (const part of parts) {
   952→    if (part.length < 3) continue;
   953→    for (const key of MOTORISTA_OPTIONS) {
   954→      if (key.toUpperCase() === part || part.includes(key.toUpperCase())) return key;
   955→    }
   956→  }
   957→  return null;
   958→}
