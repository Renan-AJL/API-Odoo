    17→// POST /api/v1/te/deliveries/send — Envia pedidos do ODOO ao TE
    18→// Body: { orderIds?: number[], pickingIds?: number[], orderType?: 1|2|3|4|5 }
    19→// -------------------------------------------------------
    20→router.post('/deliveries/send', async (req, res, next) => {
    21→  try {
    22→    const { orderIds, pickingIds, orderType = 1 } = req.body;
    23→
    24→    if (!orderIds?.length && !pickingIds?.length) {
    25→      return res.status(400).json({ success: false, error: 'Informe orderIds ou pickingIds' });
    26→    }
    27→
    28→    // 1. Busca pedidos ODOO
    29→    const pickings = [];
    30→    const saleOrders = [];
    31→
    32→    if (pickingIds?.length) {
    33→      const found = await odooTe.read('stock.picking', pickingIds, [
    34→        'id', 'name', 'partner_id', 'scheduled_date', 'move_line_count', 'origin',
    35→      ]);
    36→      pickings.push(...found);
    37→    }
    38→
    39→    if (orderIds?.length) {
    40→      const soFields = odooTe.getStudioFields('sale.order');
    41→      const found = await odooTe.read('sale.order', orderIds, [
    42→        'id', 'name', 'state', 'partner_id', 'amount_total', 'note', ...soFields,
    43→      ]);
    44→      saleOrders.push(...found);
    45→    }
    46→
    47→    // 2. Busca parceiros
    48→    const partnerIds = [
    49→      ...pickings.map(p => p.partner_id?.[0]),
    50→      ...saleOrders.map(o => o.partner_id?.[0]),
    51→    ].filter(Boolean);
    52→
    53→    const partners = partnerIds.length > 0
    54→      ? await odooTe.read('res.partner', [...new Set(partnerIds)], [
    55→          'id', 'name', 'cnpj_cpf', 'vat', 'phone', 'mobile', 'email',
    56→          'street', 'street_number', 'street2', 'zip', 'city',
    57→          'l10n_br_district', 'partner_latitude', 'partner_longitude',
    58→          'state_id', 'country_id',
    59→          ...odooTe.getStudioFields('res.partner'),
    60→        ])
    61→      : [];
    62→
    63→    const partnerMap = {};
    64→    partners.forEach(p => { partnerMap[p.id] = p; });
    65→
    66→    // 3. Mapeia para entregas TE
    67→    const teDeliveries = [];
    68→
    69→    for (const so of saleOrders) {
    70→      const partner = partnerMap[so.partner_id?.[0]] || {};
    71→      const picking = pickings.find(p => p.origin === so.name) || null;
    72→      const teDelivery = Mapper.odooToTeDelivery(so, partner, picking);
    73→      teDelivery.OrderType = orderType;
    74→      teDeliveries.push(teDelivery);
    75→    }
    76→
    77→    for (const picking of pickings) {
    78→      if (saleOrders.some(so => so.name === picking.origin)) continue; // ja mapeado
    79→      const partner = partnerMap[picking.partner_id?.[0]] || {};
    80→      const fakeSo = { name: picking.origin || `WH-${picking.id}`, note: '', id: 0 };
    81→      const teDelivery = Mapper.odooToTeDelivery(fakeSo, partner, picking);
    82→      teDelivery.OrderType = orderType;
    83→      teDeliveries.push(teDelivery);
    84→    }
    85→
    86→    if (teDeliveries.length === 0) {
    87→      return res.json({ success: true, sent: 0, message: 'Nenhuma entrega para enviar' });
    88→    }
    89→
    90→    // 4. Envia ao TE em lotes de 50
    91→    const results = [];
    92→    const BATCH = 50;
    93→
    94→    for (let i = 0; i < teDeliveries.length; i += BATCH) {
    95→      const batch = teDeliveries.slice(i, i + BATCH);
    96→      const result = await teClient.createDeliveries(batch);
    97→      results.push(result);
    98→
    99→      // 5. Marca como sincronizado no ODOO
   100→      const batchOrderNums = batch.map(d => d.OrderNumber);
   101→
   102→      const syncedSo = saleOrders.filter(so =>
   103→        batchOrderNums.includes(so.name) || batchOrderNums.includes(`WH-${so.id}`)
   104→      );
   105→      const syncedPickings = pickings.filter(p =>
   106→        batchOrderNums.includes(p.origin) || batchOrderNums.includes(`WH-${p.id}`)
   107→      );
   108→
   109→      if (syncedSo.length > 0) {
   110→        await odooTe.markSaleOrdersSynced(syncedSo.map(s => s.id), batchOrderNums[0], orderType);
   111→      }
   112→      if (syncedPickings.length > 0) {
   113→        await odooTe.markPickingsSynced(syncedPickings.map(p => p.id), batchOrderNums[0]);
   114→      }
   115→    }
   116→
   117→    console.log(`[TE-DELIVERY] Enviadas ${teDeliveries.length} entregas ao TE`);
   118→    res.json({
   119→      success: true,
   120→      sent: teDeliveries.length,
   121→      batches: results.length,
   122→      results,
   123→    });
   124→  } catch (err) {
   125→    console.error('[TE-DELIVERY] Erro:', err.message);
   126→    const status = err.isAxiosError ? (err.response?.status || 502) : 500;
   127→    res.status(status).json({ success: false, error: err.message });
   128→  }
   129→});

   183→    res.json({
   184→      success: true,

   186→      situation: mapped.situation,
   187→      situationLabel: mapped.situationLabel,
   188→      trackingCode: mapped.trackingCode,
   189→      driverName: mapped.driverName,
   190→      odooUpdated: !!(picking || saleOrder),
   191→      raw: delivery,
   192→    });
   193→  } catch (err) {
   194→    console.error('[TE-DELIVERY] Erro status:', err.message);

   280→// GET /api/v1/te/deliveries/pull — Puxa status e atualiza ODOO
   281→// -------------------------------------------------------
   282→router.get('/deliveries/pull', async (req, res) => {
   283→  try {
   284→    const { dateFrom, dateTo, situation } = req.query;
   285→
   286→    const params = { page: 1 };
   287→    if (dateFrom) params.dateFrom = dateFrom;
   288→    if (dateTo) params.dateTo = dateTo;
   289→    if (situation) params.situation = parseInt(situation, 10);
   290→
   291→    const allDeliveries = await teClient.fetchAllPages('/api/Entregas', params);
   292→
   293→    let updated = 0;
   294→    let notFound = 0;
   295→
   296→    for (const d of allDeliveries) {
   297→      const orderNum = d.OrderNumber || d.orderNumber;
   298→      const picking = await odooTe.findPickingByTeId(orderNum);
   299→
   300→      if (picking) {
   301→        const mapped = Mapper.teToOdooPicking(d);
   302→        await odooTe.updatePickingTeData(picking.id, mapped);
   303→        updated++;
   304→      } else {
   305→        const so = await odooTe.findSaleOrderByTeId(orderNum);
   306→        if (so) {
   307→          await odooTe.updateSaleOrderTeData(so.id, Mapper.teToOdooSaleOrder(d));
   308→          updated++;
   309→        } else {
   310→          notFound++;
   311→        }
   312→      }
   313→    }
   314→
   315→    res.json({
   316→      success: true,
   317→      totalFetched: allDeliveries.length,
   318→      updatedInOdoo: updated,
   319→      notFoundInOdoo: notFound,
   320→    });
   321→  } catch (err) {
   322→    console.error('[TE-DELIVERY] Erro pull:', err.message);
   323→    res.status(500).json({ success: false, error: err.message });
   324→  }
   325→});

   359→function fmtDate(dateStr) {
   360→  if (!dateStr) return '';
   361→  try {
   362→    const d = new Date(dateStr);
   363→    if (isNaN(d.getTime())) return dateStr;
   364→    return d.toLocaleString('pt-BR', { timeZone: 'America/Sao_Paulo',
   365→      day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit' });
   366→  } catch { return dateStr; }
   367→}

   539→router.post('/send-invoice', async (req, res) => {
   540→  console.log('[TE-SEND-INVOICE] Body recebido: ' + JSON.stringify(req.body));
   541→  const invId = req.body.invoice_id || req.body.id;
   542→  if (!invId) return res.status(400).json({ success: false, error: 'invoice_id obrigatorio' });
   543→
   544→  try {
   545→    // 1. Le a fatura completa
   546→    const invoice = await odooTe.readInvoiceFull(invId);
   547→    if (!invoice) {
   548→      return res.status(404).json({ success: false, error: 'Fatura ' + invId + ' nao encontrada' });
   549→    }
   550→    console.log('[TE-SEND-INVOICE] Fatura: ' + invoice.name + ' (id=' + invoice.id + ', partner=' + JSON.stringify(invoice.partner_id) + ')');
   551→
   552→    // 2. Le o partner da fatura
   553→    const partnerId = invoice.partner_id ? (Array.isArray(invoice.partner_id) ? invoice.partner_id[0] : invoice.partner_id) : null;
   554→    if (!partnerId) {
   555→      return res.status(400).json({ success: false, error: 'Fatura sem parceiro' });
   556→    }
   557→    const partnerArr = await odooTe.read('res.partner', [partnerId], [
   558→      'id', 'name', 'cnpj_cpf', 'vat', 'phone', 'mobile', 'email',
   559→      'street', 'street_number', 'street2', 'zip', 'city',
   560→      'l10n_br_district', 'partner_latitude', 'partner_longitude', 'state_id', 'country_id',
   561→      ...odooTe.getStudioFields('res.partner'),
   562→    ]);

   564→

   566→    let saleOrder = null;
   567→    let saleFull = null;
   568→    let picking = null;
   569→
   570→    // 3a. Se Odoo enviou sale_order_id diretamente no body
   571→    if (req.body.sale_order_id) {
   572→      console.log('[TE-SEND-INVOICE] sale_order_id direto: ' + req.body.sale_order_id);
   573→      saleOrder = await odooTe.findSaleOrderById(req.body.sale_order_id);
   574→    }
   575→
   576→    // 3b. Busca automatica via relacao invoice -> sale.order
   577→    if (!saleOrder) {
   578→      saleOrder = await odooTe.findSaleOrderByInvoice(invId);
   579→    }
   580→
   581→    // 4. Se tem venda, busca dados completos (picking, linhas, motorista)
   582→    let orderLines = [];
   583→    let productsMap = {};
   584→
   585→    if (saleOrder) {
   586→      console.log('[TE-SEND-INVOICE] Venda encontrada: ' + saleOrder.name + ' (id=' + saleOrder.id + ')');
   587→      saleFull = await odooTe.readSaleOrderFull(saleOrder.id);
   588→      picking = await odooTe.findDeliveryPicking(saleOrder.id);
   589→      if (picking) {
   590→        console.log('[TE-SEND-INVOICE] Picking: ' + picking.name + ' (id=' + picking.id + ')');
   591→      } else {
   592→        console.log('[TE-SEND-INVOICE] Picking nao encontrado, usando fatura como base');
   593→      }
   594→      orderLines = await odooTe.getSaleOrderLines(saleOrder.id);

   597→      // Usa linhas da fatura como orderLines
   598→      try {
   599→        orderLines = await odooTe.getInvoiceLines(invId);
   600→        console.log('[TE-SEND-INVOICE] ' + orderLines.length + ' linhas da fatura com produto');
   601→      } catch (err) {

   607→    const productIds = [];

   614→    const company = await odooTe.getCompany();

   617→    const delivery = Mapper.odooToTeDelivery({
   618→      picking: picking,
   619→      partner: partnerData,

   621→      invoice: invoice,

   623→      companyCnpj: config.empresa.cnpj,
   624→      orderLines: orderLines,
   625→      productsMap: productsMap,
   626→    });

   628→      return res.status(500).json({ success: false, error: 'Falha no mapeamento dos dados' });
   629→    }

   631→    // 8. Chatter: enviando (so em account.move + se tiver, sale.order/picking)

   654→    const teResp = Array.isArray(teResult) ? teResult[0] : teResult;
   655→

   658→      const odooData = Mapper.teCreateToOdoo(teResp);
   659→      await odooTe.markInvoiceSynced([invId], teResp.OrderID || null);
   660→      if (picking) {
   661→        if (Object.keys(odooData).length) await odooTe.updatePickingTeData(picking.id, odooData);
   662→        await odooTe.markPickingsSynced([picking.id], teResp.OrderID || null);

   664→      if (saleOrder) {
   665→        if (Object.keys(odooData).length) await odooTe.updateSaleOrderTeData(saleOrder.id, odooData);
   666→        await odooTe.markSaleOrdersSynced([saleOrder.id], teResp.OrderID || null);
   667→      }
   668→
   669→      var resultMsg = Mapper.chatterCreateMessage(teResp, delivery);
   670→      await odooTe.postChatter('account.move', invId, resultMsg);
   671→      if (picking) await odooTe.postChatter('stock.picking', picking.id, resultMsg);
   672→      if (saleOrder) await odooTe.postChatter('sale.order', saleOrder.id, resultMsg);
   673→
   674→      res.json({
   675→        success: true,
   676→        invoice_id: invId,
   677→        sale: saleOrder ? saleOrder.name : null,
   678→        picking: picking ? picking.name : null,
   679→        te_order_id: teResp.OrderID,
   680→        te_received: teResp.Received,
   681→        te_tracking: teResp.TrackingCode,
   682→      });
   683→    } else {
   684→      await odooTe.postChatter('account.move', invId, '<b>TudoEntregue - ERRO</b><br/>TE nao retornou dados.');
   685→      res.status(502).json({ success: false, error: 'TE nao retornou dados' });
   686→    }
   687→  } catch (err) {
   688→    console.error('[TE-SEND-INVOICE] Erro:', err.message);
   689→    try {
   690→      await odooTe.postChatter('account.move', invId, '<b>TudoEntregue - ERRO</b><br/>' + err.message);
   691→    } catch {}
   692→    const status = err.isAxiosError ? (err.response?.status || 502) : 500;
   693→    res.status(status).json({ success: false, error: err.message });
   694→  }

   808→// ============================================================
   809→// MATCH MOTORISTA — Faz match do nome do TE com as opcoes do campo selection
   810→// O campo x_studio_motorista e um selection com 17 opcoes (key/label)
   811→// ============================================================
   812→const MOTORISTA_OPTIONS = [
   813→  'adelson', 'anderlucio', 'antonio', 'carlos', 'daniel',
   814→  'diego', 'eduardo', 'fellipe', 'gerson', 'glauber',
   815→  'henrique', 'italo', 'jeferson', 'joao_pedro', 'lucas',
   816→  'matheus', 'rafael',
   817→];
   818→
   819→function matchMotoristaSelection(driverNameFromTe) {
   820→  if (!driverNameFromTe) return null;
   821→  const name = driverNameFromTe.toUpperCase().trim();
   822→  // Remove sufixos comuns como "MOTORISTA", "FRETE", etc
   823→  const cleanName = name
   824→    .replace(/\b(MOTORISTA|FRETE|ENTREGAS|MOTOBOY)\b/g, '')
   825→    .replace(/[^A-ZÀ-Ú\s]/g, '')
   826→    .trim();
   827→  const parts = cleanName.split(/\s+/).filter(Boolean);
   828→
   829→  // Tenta match exato com a key ou com o primeiro/ultimo nome
   830→  for (const key of MOTORISTA_OPTIONS) {
   831→    const keyUpper = key.toUpperCase();
   832→    if (name.includes(keyUpper) || keyUpper.includes(name)) return key;
   833→  }
   834→  // Tenta match com qualquer parte do nome
   835→  for (const part of parts) {
   836→    if (part.length < 3) continue;
   837→    for (const key of MOTORISTA_OPTIONS) {
   838→      if (key.toUpperCase() === part || part.includes(key.toUpperCase())) return key;
   839→    }
   840→  }
   841→  return null;
   842→}
