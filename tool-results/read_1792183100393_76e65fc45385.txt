   105→   105→// XML-RPC connection (singleton)
   106→   106→// ============================================================
   107→   107→var _cached = null;
   108→   108→
   109→   109→function getConn() {
   110→   110→  if (_cached && _cached.uid) return _cached;
   111→   111→  var odooConfig = config.odoo;
   112→   112→  var base = odooConfig.url.replace(/\/+$/, '');
   113→   113→  var host = base.replace('https://', '').replace('http://', '');
   114→   114→  var common = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/common', port: 443 });
   115→   115→  var models = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
   116→   116→  _cached = { uid: null, common: common, models: models, db: odooConfig.db, password: odooConfig.password };
   117→   117→  return _cached;
   118→   118→}
   119→   119→
   120→   120→function authenticate() {
   121→   121→  return new Promise(function(resolve, reject) {
   122→   122→    var conn = getConn();
   123→   123→    conn.common.methodCall('authenticate', [conn.db, config.odoo.user, conn.password, {}], function(err, uid) {
   124→   124→      if (err) { reject(new Error('Auth Odoo falhou: ' + (err.message || JSON.stringify(err)))); }
   125→   125→      else if (uid === false || uid === null) { reject(new Error('Auth Odoo: credenciais invalidas (UID=false)')); }
   126→   126→      else { conn.uid = uid; logger.info('[ODOO-TE] Autenticado. UID: ' + uid); resolve(uid); }
   127→   127→    });
   128→   128→  });
   129→   129→}
   130→   130→
   131→   131→function ensureAuth() {
   132→   132→  var conn = getConn();
   133→   133→  if (conn.uid) return Promise.resolve(conn.uid);
   134→   134→  return authenticate();
   135→   135→}
   136→   136→
   137→   137→function executeKw(model, method, args, kwargs) {
   138→   138→  return new Promise(function(resolve, reject) {
   139→   139→    ensureAuth().then(function(uid) {
   140→   140→      var conn = getConn();
   141→   141→      var params = [conn.db, uid, conn.password, model, method, args || []];
   142→   142→      if (kwargs) params.push(kwargs);
   143→   143→      // Debug log para metodos read/search (nao logar write/create com dados sensiveis)
   144→   144→      if (method === 'read' || method === 'search' || method === 'search_read') {
   145→   145→        logger.info('[ODOO-TE] executeKw: ' + model + '.' + method + ' args=' + JSON.stringify(args).substring(0, 200));
   146→   146→      }
   147→   147→      conn.models.methodCall('execute_kw', params, function(err8, result) {
   148→   148→        if (err8) {
   149→   149→          logger.error('[ODOO-TE] executeKw ERRO: ' + model + '.' + method + ': ' + (err8.message || JSON.stringify(err8)).substring(0, 300));
   150→   150→          reject(new Error(model + '.' + method + ': ' + (err8.message || JSON.stringify(err8))));
   151→   151→        }
   152→   152→        else { resolve(result); }
   153→   153→      });
   154→   154→    }).catch(reject);
   155→   155→  });
   156→   156→}

   161→   161→var _customFieldsCache = {};
   162→   162→
   163→   163→async function safeReadCustom(model, ids, nativeFields, customFields) {
   164→   164→  try {
   165→   165→    return await executeKw(model, 'read', [ids, nativeFields.concat(customFields)]);
   166→   166→  } catch (err) {
   167→   167→    if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   168→   168→      if (!_customFieldsCache[model]) {
   169→   169→        _customFieldsCache[model] = { available: false, checked: true };
   170→   170→        logger.warn('[ODOO-TE] Campos x_studio_* nao encontrados em ' + model + '. Usando campos nativos.');
   171→   171→      }
   172→   172→      return await executeKw(model, 'read', [ids, nativeFields]);
   173→   173→    }
   174→   174→    throw err;
   175→   175→  }
   176→   176→}
   177→   177→
   178→   178→async function safeWriteCustom(model, ids, vals) {
   179→   179→  var customVals = {};
   180→   180→  var nativeVals = {};
   181→   181→  Object.keys(vals).forEach(function(k) {
   182→   182→    if (k.indexOf('x_studio_') === 0) customVals[k] = vals[k];
   183→   183→    else nativeVals[k] = vals[k];
   184→   184→  });
   185→   185→  if (Object.keys(nativeVals).length > 0) await executeKw(model, 'write', [ids, nativeVals]);
   186→   186→  if (Object.keys(customVals).length > 0) {
   187→   187→    try { await executeKw(model, 'write', [ids, customVals]); }
   188→   188→    catch (err) {
   189→   189→      if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   190→   190→        if (!_customFieldsCache[model] || !_customFieldsCache[model].warned) {
   191→   191→          logger.warn('[ODOO-TE] Campo x_studio_* nao existe em ' + model + '. Gravacao ignorada.');
   192→   192→          if (!_customFieldsCache[model]) _customFieldsCache[model] = {};
   193→   193→          _customFieldsCache[model].warned = true;
   194→   194→        }
   195→   195→      } else throw err;
   196→   196→    }
   197→   197→  }
   198→   198→}

   409→   409→async function getInvoiceLines(invoiceId) {
   410→   410→  try {
   411→   411→    // Evita account.move.line.search — Odoo 19.2 SaaS tem override de
   412→   412→    // search_fetch com _get_recon_limit_from_domain que rejeita certos dominios.
   413→   413→    // Em vez disso, le line_ids direto da fatura.
   414→   414→    var invData = await executeKw('account.move', 'read', [[invoiceId], ['line_ids']]);
   415→   415→    var inv = invData && invData[0];
   416→   416→    if (!inv || !inv.line_ids || !inv.line_ids.length) return [];
   417→   417→
   418→   418→    // line_ids pode vir como [id, id, ...] ou [[id, name], ...]
   419→   419→    var lineIds = inv.line_ids.map(function(item) {
   420→   420→      return Array.isArray(item) ? item[0] : item;
   421→   421→    });
   422→   422→
   423→   423→    var lines = await executeKw('account.move.line', 'read', [lineIds, INVOICE_LINE_FIELDS]);
   424→   424→    if (!lines || !lines.length) return [];
   425→   425→
   426→   426→    // Filtra apenas linhas com produto (product_id != false)
   427→   427→    return lines.filter(function(line) {
   428→   428→      if (!line.product_id) return false;
   429→   429→      // product_id pode ser [id, name] ou false
   430→   430→      if (Array.isArray(line.product_id) && line.product_id[0]) return true;
   431→   431→      if (typeof line.product_id === 'number' && line.product_id) return true;
   432→   432→      return false;
   433→   433→    });
   434→   434→  } catch (err) {
   435→   435→    logger.warn('[ODOO-TE] getInvoiceLines falhou: ' + err.message);
   436→   436→    return [];
   437→   437→  }
   438→   438→}
