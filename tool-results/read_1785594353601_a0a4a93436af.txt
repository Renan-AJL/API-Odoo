    34→    34→    34→function createClient(url) {
    35→    35→    35→  var base = url.replace(/\/+$/, '');
    36→    36→    36→  var host = base.replace('https://', '').replace('http://', '');
    37→    37→    37→  var port = base.startsWith('https') ? 443 : 80;
    38→    38→    38→  var isSecure = base.startsWith('https');
    39→    39→    39→  var createFn = isSecure ? xmlrpc.createSecureClient : xmlrpc.createClient;
    40→    40→    40→  return {
    41→    41→    41→    common: createFn({ host: host, path: '/xmlrpc/2/common', port: port }),
    42→    42→    42→    models: createFn({ host: host, path: '/xmlrpc/2/object', port: port }),
    43→    43→    43→  };
    44→    44→    44→}
    45→    45→    45→
    46→    46→    46→function authenticate(client, db, login, password) {
    47→    47→    47→  return new Promise(function(resolve, reject) {
    48→    48→    48→    client.common.methodCall('authenticate', [db, login, password, {}], function(err, uid) {
    49→    49→    49→      if (err) reject(new Error('Auth Odoo falhou: ' + (err.message || JSON.stringify(err))));
    50→    50→    50→      else if (uid === false || uid === null) reject(new Error('Credenciais Odoo invalidas'));
    51→    51→    51→      else resolve(uid);
    52→    52→    52→    });
    53→    53→    53→  });
    54→    54→    54→}
    55→    55→    55→
    56→    56→    56→function executeKw(client, db, uid, password, model, method, args, kwargs) {
    57→    57→    57→  return new Promise(function(resolve, reject) {
    58→    58→    58→    var params = [db, uid, password, model, method, args || []];
    59→    59→    59→    if (kwargs) params.push(kwargs);
    60→    60→    60→    client.models.methodCall('execute_kw', params, function(err, result) {
    61→    61→    61→      if (err) reject(new Error(model + '.' + method + ': ' + (err.message || JSON.stringify(err))));
    62→    62→    62→      else resolve(result);
    63→    63→    63→    });
    64→    64→    64→  });
    65→    65→    65→}

    70→    70→    70→async function processPendingEmissions() {
    71→    71→    71→  var odoo = config.odoo;
    72→    72→    72→  if (!odoo || !odoo.enabled || !odoo.url) {
    73→    73→    73→    console.log('[SIEG-EMIT] Odoo nao configurado');
    74→    74→    74→    return { processed: 0, reason: 'odoo_not_configured' };
    75→    75→    75→  }
    76→    76→    76→
    77→    77→    77→  var client = createClient(odoo.url);
    78→    78→    78→  var uid = await authenticate(client, odoo.db, odoo.user, odoo.password);
    79→    79→    79→  var db = odoo.db;
    80→    80→    80→  var pwd = odoo.password;
    81→    81→    81→
    82→    82→    82→  console.log('[SIEG-EMIT] Buscando faturas pendentes...');
    83→    83→    83→
    84→    84→    84→  // Buscar NF-e pendentes
    85→    85→    85→  var nfeIds = await executeKw(client, db, uid, pwd, 'account.move', 'search', [[
    86→    86→    86→    ['move_type', '=', 'out_invoice'],
    87→    87→    87→    ['x_studio_nfe_status', '=', 'pendente'],
    88→    88→    88→  ]], { order: 'id asc', limit: 5 });
    89→    89→    89→
    90→    90→    90→  // Buscar NFS-e pendentes
    91→    91→    91→  var nfseIds = await executeKw(client, db, uid, pwd, 'account.move', 'search', [[
    92→    92→    92→    ['move_type', '=', 'out_invoice'],
    93→    93→    93→    ['x_studio_nfse_status', '=', 'pendente'],
    94→    94→    94→  ]], { order: 'id asc', limit: 5 });
    95→    95→    95→
    96→    96→    96→  var allPending = [];
    97→    97→    97→  for (var i = 0; i < nfeIds.length; i++) {
    98→    98→    98→    allPending.push({ id: nfeIds[i], tipo: 'nfe' });
    99→    99→    99→  }
   100→   100→   100→  for (var j = 0; j < nfseIds.length; j++) {
   101→   101→   101→    allPending.push({ id: nfseIds[j], tipo: 'nfse' });
   102→   102→   102→  }
   103→   103→   103→
   104→   104→   104→  if (!allPending.length) {
   105→   105→   105→    console.log('[SIEG-EMIT] Nenhuma fatura pendente');
   106→   106→   106→    return { processed: 0 };
   107→   107→   107→  }
   108→   108→   108→
   109→   109→   109→  console.log('[SIEG-EMIT] ' + allPending.length + ' fatura(s) pendente(s)');
   110→   110→   110→  var results = [];
   111→   111→   111→
   112→   112→   112→  for (var k = 0; k < allPending.length; k++) {
   113→   113→   113→    var item = allPending[k];
   114→   114→   114→    try {
   115→   115→   115→      var r = await processOne(client, db, uid, pwd, item.id, item.tipo);
   116→   116→   116→      results.push(r);
   117→   117→   117→    } catch (err) {
   118→   118→   118→      console.error('[SIEG-EMIT] ERRO fatura ' + item.id + ':', err.message);
   119→   119→   119→      await safeUpdateError(client, db, uid, pwd, item.id, item.tipo, err.message);
   120→   120→   120→      results.push({ move_id: item.id, sucesso: false, tipo: item.tipo, erro: err.message });
   121→   121→   121→    }
   122→   122→   122→  }
   123→   123→   123→
   124→   124→   124→  var ok = results.filter(function(r) { return r.sucesso; }).length;
   125→   125→   125→  console.log('[SIEG-EMIT] Concluido: ' + ok + '/' + results.length + ' sucesso');
   126→   126→   126→  return { processed: results.length, sucesso: ok, detalhes: results };
   127→   127→   127→}

   132→   132→   132→async function processOne(client, db, uid, pwd, moveId, tipo) {
   133→   133→   133→  console.log('[SIEG-EMIT] --- Fatura ID: ' + moveId + ' | Tipo: ' + tipo + ' ---');
   134→   134→   134→
   135→   135→   135→  // 1. Read account.move (sem invoice_line_ids — pode ser computed nao-stored no Odoo 19)
   136→   136→   136→  var moves = await executeKw(client, db, uid, pwd, 'account.move', 'read', [[moveId], [
   137→   137→   137→    'name', 'partner_id', 'company_id', 'invoice_date', 'date',
   138→   138→   138→    'amount_total', 'narration',
   139→   139→   139→    'x_studio_nfe_status', 'x_studio_nfse_status', 'payment_state',
   140→   140→   140→  ]]);
   141→   141→   141→  if (!moves || !moves.length) throw new Error('Fatura ' + moveId + ' nao encontrada');
   142→   142→   142→  var move = moves[0];
   143→   143→   143→  console.log('[SIEG-EMIT] Fatura: ' + move.name + ' | R$ ' + (move.amount_total || 0));
   144→   144→   144→
   145→   145→   145→  // 2. Read company
   146→   146→   146→  var companyId = tupId(move.company_id);
   147→   147→   147→  var company = await readCompany(client, db, uid, pwd, companyId);
   148→   148→   148→
   149→   149→   149→  // 3. Read partner
   150→   150→   150→  var partnerId = tupId(move.partner_id);
   151→   151→   151→  if (!partnerId) throw new Error('Fatura sem parceiro');
   152→   152→   152→  var partner = await readPartner(client, db, uid, pwd, partnerId);
   153→   153→   153→
   154→   154→   154→  // 4. Read invoice lines diretamente de account.move.line (evita computed field)
   155→   155→   155→  var allLineIds = await executeKw(client, db, uid, pwd, 'account.move.line', 'search', [[
   156→   156→   156→    ['move_id', '=', moveId],
   157→   157→   157→  ]], { order: 'id asc' });
   158→   158→   158→  console.log('[SIEG-EMIT] Linhas encontradas na fatura: ' + allLineIds.length);
   159→   159→   159→  var rawLines = await executeKw(client, db, uid, pwd, 'account.move.line', 'read', [allLineIds, [
   160→   160→   160→    'display_type', 'product_id', 'name', 'quantity', 'price_unit',
   161→   161→   161→    'price_subtotal', 'tax_ids', 'discount',
   162→   162→   162→  ]]);
   163→   163→   163→  // Debug: log todas as linhas
   164→   164→   164→  for (var li = 0; li < rawLines.length; li++) {
   165→   165→   165→    var l = rawLines[li];
   166→   166→   166→    var pName = 'N/A';
   167→   167→   167→    if (l.product_id) pName = Array.isArray(l.product_id) ? l.product_id[1] : String(l.product_id);
   168→   168→   168→    console.log('[SIEG-EMIT]   Line ' + li + ': display_type=' + JSON.stringify(l.display_type) + ' product=' + pName + ' qty=' + l.quantity + ' price=' + l.price_unit);
   169→   169→   169→  }
   170→   170→   170→  // Filtro principal: linhas com produto associado (mais confiavel que display_type)
   171→   171→   171→  var invoiceLines = rawLines.filter(function(l) { return l.product_id; });
   172→   172→   172→  // Fallback: se nenhuma tem product_id, tenta por display_type
   173→   173→   173→  if (!invoiceLines.length) {
   174→   174→   174→    invoiceLines = rawLines.filter(function(l) {
   175→   175→   175→      var dt = l.display_type;
   176→   176→   176→      return !dt || dt === 'product' || dt === 'service';
   177→   177→   177→    });
   178→   178→   178→  }
   179→   179→   179→  if (!invoiceLines.length) throw new Error('Fatura sem linhas de produto/servico (total: ' + rawLines.length + ', com product: 0)');
   180→   180→   180→  console.log('[SIEG-EMIT] Linhas de produto: ' + invoiceLines.length);
   181→   181→   181→
   182→   182→   182→  // 5. Get next NF number from company
   183→   183→   183→  var serie, numField, nextNum;
   184→   184→   184→  if (tipo === 'nfe') {
   185→   185→   185→    serie = String(company.x_studio_sieg_serie_nfe || '100');
   186→   186→   186→    numField = 'x_studio_sieg_ultimo_nfe';
   187→   187→   187→  } else {
   188→   188→   188→    serie = String(company.x_studio_sieg_serie_nfse || '1');
   189→   189→   189→    numField = 'x_studio_sieg_ultimo_nfse';
   190→   190→   190→  }
   191→   191→   191→  nextNum = (parseInt(company[numField]) || 0) + 1;
   192→   192→   192→  console.log('[SIEG-EMIT] Serie: ' + serie + ' | nNF: ' + nextNum);
   193→   193→   193→
   194→   194→   194→  // 6. Build line data with taxes + product x_studio_ fields
   195→   195→   195→  var linesData = [];
   196→   196→   196→  for (var i = 0; i < invoiceLines.length; i++) {
   197→   197→   197→    var ld = await buildLineData(client, db, uid, pwd, invoiceLines[i]);
   198→   198→   198→    linesData.push(ld);
   199→   199→   199→  }
   200→   200→   200→
   201→   201→   201→  // 7. Build emission payload
   202→   202→   202→  var emitData = {
   203→   203→   203→    company: company,
   204→   204→   204→    partner: partner,
   205→   205→   205→    order: {
   206→   206→   206→      name: move.name,
   207→   207→   207→      number: String(nextNum),
   208→   208→   208→      date_order: move.invoice_date || move.date,
   209→   209→   209→      amount_total: move.amount_total,
   210→   210→   210→      note: move.narration || '',
   211→   211→   211→    },
   212→   212→   212→    lines: linesData,
   213→   213→   213→    config: {
   214→   214→   214→      serie: serie,
   215→   215→   215→      tpAmb: process.env.SIEG_TP_AMB || config.sieg.tpAmb || '2',
   216→   216→   216→      natOp: 'Venda de Mercadoria',
   217→   217→   217→      mod: tipo === 'nfe' ? '55' : '01',
   218→   218→   218→    },
   219→   219→   219→    tipo: tipo,
   220→   220→   220→  };
   221→   221→   221→
   222→   222→   222→  // NFS-e: build service block from product x_studio_ fields
   223→   223→   223→  if (tipo === 'nfse') {
   224→   224→   224→    emitData.service = buildServiceBlock(linesData, move);
   225→   225→   225→  }
   226→   226→   226→
   227→   227→   227→  // 8. Set status to 'processando'
   228→   228→   228→  var processVals = {};
   229→   229→   229→  if (tipo === 'nfe') processVals.x_studio_nfe_status = 'processando';
   230→   230→   230→  else processVals.x_studio_nfse_status = 'processando';
   231→   231→   231→  await executeKw(client, db, uid, pwd, 'account.move', 'write', [[moveId], processVals]);
   232→   232→   232→
   233→   233→   233→  // 9. Call SIEG API
   234→   234→   234→  console.log('[SIEG-EMIT] Enviando ao SIEG...');
   235→   235→   235→  var resultado;
   236→   236→   236→  try {
   237→   237→   237→    resultado = await emitirNota(emitData);
   238→   238→   238→  } catch (siegErr) {
   239→   239→   239→    // Captura erros HTTP (ex: 409 com detalhes no body)
   240→   240→   240→    var siegDetail = siegErr.response && siegErr.response.data;
   241→   241→   241→    var errMsg = 'SIEG HTTP ' + (siegErr.response ? siegErr.response.status : 'erro') + ': ';
   242→   242→   242→    if (siegDetail) {
   243→   243→   243→      errMsg += typeof siegDetail === 'string' ? siegDetail : JSON.stringify(siegDetail).slice(0, 500);
   244→   244→   244→    } else {
   245→   245→   245→      errMsg += siegErr.message;
   246→   246→   246→    }
   247→   247→   247→    console.error('[SIEG-EMIT] Erro SIEG:', errMsg);
   248→   248→   248→    throw new Error(errMsg);
   249→   249→   249→  }
   250→   250→   250→  console.log('[SIEG-EMIT] SIEG retornou - sucesso: ' + resultado.sucesso + (resultado.httpStatus ? ' (HTTP ' + resultado.httpStatus + ')' : ''));
   251→   251→   251→
   252→   252→   252→  // 10. Parse result
   253→   253→   253→  var info = parseResult(resultado, tipo);
   254→   254→   254→  var xmlRetornoText = '';
   255→   255→   255→  var rawXmlAuth = resultado.resposta && (resultado.resposta.Xml || resultado.resposta.xml || resultado.resposta.xmlBase64);
   256→   256→   256→  if (rawXmlAuth) {
   257→   257→   257→    xmlRetornoText = (typeof rawXmlAuth === 'string' && rawXmlAuth.length > 200)
   258→   258→   258→      ? rawXmlAuth
   259→   259→   259→      : Buffer.from(rawXmlAuth, 'base64').toString('utf-8');
   260→   260→   260→  }
   261→   261→   261→
   262→   262→   262→  // 11. Update account.move with results
   263→   263→   263→  var updateVals = {};
   264→   264→   264→  if (info.sucesso) {
   265→   265→   265→    if (tipo === 'nfe') {
   266→   266→   266→      updateVals.x_studio_nfe_status = 'autorizada';
   267→   267→   267→      updateVals.x_studio_nfe_chave = info.chave;
   268→   268→   268→      updateVals.x_studio_nfe_protocolo = info.protocolo;
   269→   269→   269→    } else {
   270→   270→   270→      updateVals.x_studio_nfse_status = 'autorizada';
   271→   271→   271→      updateVals.x_studio_nfse_numero = String(nextNum);
   272→   272→   272→    }
   273→   273→   273→
   274→   274→   274→    // Increment NF number (only on authorization)
   275→   275→   275→    try {
   276→   276→   276→      var companyUpdate = {};
   277→   277→   277→      companyUpdate[numField] = nextNum;
   278→   278→   278→      await executeKw(client, db, uid, pwd, 'res.company', 'write', [[companyId], companyUpdate]);
   279→   279→   279→      console.log('[SIEG-EMIT] Sequencia ' + numField + ' = ' + nextNum);
   280→   280→   280→    } catch (seqErr) {
   281→   281→   281→      console.error('[SIEG-EMIT] Erro ao atualizar sequencia:', seqErr.message);
   282→   282→   282→    }
   283→   283→   283→  } else {
   284→   284→   284→    if (tipo === 'nfe') {
   285→   285→   285→      updateVals.x_studio_nfe_status = info.cStat + ' - ' + info.motivo;
   286→   286→   286→    } else {
   287→   287→   287→      updateVals.x_studio_nfse_status = info.motivo;
   288→   288→   288→    }
   289→   289→   289→  }
   290→   290→   290→
   291→   291→   291→  await executeKw(client, db, uid, pwd, 'account.move', 'write', [[moveId], updateVals]);
   292→   292→   292→  console.log('[SIEG-EMIT] Fatura atualizada');
   293→   293→   293→
   294→   294→   294→  // 12. Post XML + PDF to Chatter
   295→   295→   295→  await postChatterResult(client, db, uid, pwd, moveId, move.name, tipo, info, resultado, xmlRetornoText);
   296→   296→   296→
   297→   297→   297→  return {
   298→   298→   298→    move_id: moveId, sucesso: info.sucesso, tipo: tipo,
   299→   299→   299→    chave: info.chave, numero: nextNum, serie: serie,
   300→   300→   300→  };
   301→   301→   301→}

   304→   304→   304→// Post XML + PDF to Odoo Chatter

   306→   306→   306→async function postChatterResult(client, db, uid, pwd, moveId, moveName, tipo, info, resultado, xmlRetornoText) {
   307→   307→   307→  var attachIds = [];
   308→   308→   308→  var safeName = (moveName || 'Fatura').replace(/[^a-zA-Z0-9\-_]/g, '_');
   309→   309→   309→  var nfLabel = tipo === 'nfe' ? 'NF-e' : 'NFS-e';
   310→   310→   310→  var xmlContent = xmlRetornoText || resultado.xmlEnviado || '';
   311→   311→   311→  var pdfBase64 = resultado.pdfBase64 || '';
   312→   312→   312→
   313→   313→   313→  // Attach XML
   314→   314→   314→  if (xmlContent) {
   315→   315→   315→    try {
   316→   316→   316→      var xmlB64 = Buffer.from(xmlContent, 'utf-8').toString('base64');
   317→   317→   317→      var xmlAttach = await executeKw(client, db, uid, pwd, 'ir.attachment', 'create', [{
   318→   318→   318→        name: nfLabel + '_' + safeName + '.xml',
   319→   319→   319→        datas: xmlB64,
   320→   320→   320→        res_model: 'account.move',
   321→   321→   321→        res_id: moveId,
   322→   322→   322→        mimetype: 'application/xml',
   323→   323→   323→      }]);
   324→   324→   324→      attachIds.push(xmlAttach);
   325→   325→   325→      console.log('[SIEG-EMIT] XML anexado (ID: ' + xmlAttach + ')');
   326→   326→   326→    } catch (e) { console.error('[SIEG-EMIT] Erro anexar XML:', e.message); }
   327→   327→   327→  }
   328→   328→   328→
   329→   329→   329→  // Attach PDF (DANFE / DANFSE)
   330→   330→   330→  if (pdfBase64) {
   331→   331→   331→    try {
   332→   332→   332→      var pdfLabel = tipo === 'nfe' ? 'DANFE' : 'DANFSE';
   333→   333→   333→      var pdfAttach = await executeKw(client, db, uid, pwd, 'ir.attachment', 'create', [{
   334→   334→   334→        name: pdfLabel + '_' + safeName + '.pdf',
   335→   335→   335→        datas: pdfBase64,
   336→   336→   336→        res_model: 'account.move',
   337→   337→   337→        res_id: moveId,
   338→   338→   338→        mimetype: 'application/pdf',

   615→   615→   615→async function extractTaxes(client, db, uid, pwd, line) {
   616→   616→   616→  var result = {
   617→   617→   617→    csosn: '103', cst_icms: '', vbc: 0, vicms: 0, picms: 0,
   618→   618→   618→    cst_pis: '99', vbc_pis: 0, ppis: 0, vpis: 0,
   619→   619→   619→    cst_cofins: '99', vbc_cofins: 0, pcofins: 0, vcofins: 0,
   620→   620→   620→  };
   621→   621→   621→
   622→   622→   622→  var taxIds = (line.tax_ids || []).map(function(t) { return Array.isArray(t) ? t[0] : t; }).filter(Boolean);
   623→   623→   623→  if (!taxIds.length) return result;
   624→   624→   624→
   625→   625→   625→  try {
   626→   626→   626→    var taxes = await executeKw(client, db, uid, pwd, 'account.tax', 'read', [taxIds, [
   627→   627→   627→      'name', 'amount', 'amount_type', 'description',
   628→   628→   628→    ]]);
   629→   629→   629→    if (!taxes || !taxes.length) return result;
   630→   630→   630→
   631→   631→   631→    var base = parseFloat(line.price_subtotal) || 0;
   632→   632→   632→    var totalIcms = 0, totalPis = 0, totalCofins = 0;
   633→   633→   633→
   634→   634→   634→    for (var i = 0; i < taxes.length; i++) {
   635→   635→   635→      var tax = taxes[i];
   636→   636→   636→      var n = ((tax.name || tax.description || '').toUpperCase());
   637→   637→   637→      var amt = parseFloat(tax.amount) || 0;
   638→   638→   638→      if (n.indexOf('ICMS') >= 0 || n.indexOf('CSOSN') >= 0) {
   639→   639→   639→        totalIcms += amt;
   640→   640→   640→        var m = n.match(/(10[0-9]|20[0-9]|300|400|500|900)/);
   641→   641→   641→        if (m) result.csosn = m[1];
   642→   642→   642→      } else if (n.indexOf('PIS') >= 0) { totalPis += amt; }
   643→   643→   643→      else if (n.indexOf('COFINS') >= 0) { totalCofins += amt; }
   644→   644→   644→    }
   645→   645→   645→
   646→   646→   646→    result.picms = totalIcms; result.vbc = base; result.vicms = base * (totalIcms / 100);
   647→   647→   647→    result.ppis = totalPis; result.vbc_pis = base; result.vpis = base * (totalPis / 100);
   648→   648→   648→    result.pcofins = totalCofins; result.vbc_cofins = base; result.vcofins = base * (totalCofins / 100);
   649→   649→   649→  } catch (e) { console.warn('[SIEG-EMIT] Erro ao extrair impostos:', e.message); }
   650→   650→   650→
   651→   651→   651→  return result;
   652→   652→   652→}
