    31→var BANDEIRAS = ['VISA', 'MASTER', 'ELO', 'AMEX', 'HIPERCARD', 'HIPER'];

    33→function detectarCartao(formaPag) {
    34→  if (!formaPag) return null;
    35→  var n = formaPag.trim().toUpperCase();
    36→  for (var i = 0; i < BANDEIRAS.length; i++) {
    37→    var b = BANDEIRAS[i];
    38→    if (n.indexOf(b) === 0 || n.indexOf(b + ' ') === 0) {
    39→      var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
    40→      var parcelas = 1;
    41→      var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
    42→      if (match) parcelas = parseInt(match[1]);
    43→      return { bandeira: b, parcelas: parcelas, debito: debito };
    44→    }
    45→  }
    46→  // Generico: CARTAO, CARTAO CREDITO, CREDITO, etc.
    47→  if (n.indexOf('CART') === 0 || n.indexOf('CREDITO') === 0) {
    48→    var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
    49→    var parcelas = 1;
    50→    var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
    51→    if (match) parcelas = parseInt(match[1]);
    52→    return { bandeira: 'CARTAO', parcelas: parcelas, debito: debito };
    53→  }
    54→  return null;
    55→}

    66→

    72→    console.error('[API/CARTAO] Rede nao configurada. PV:', config.rede.pv ? 'OK' : 'faltando',

    74→    return res.json({
    75→      success: false,
    76→      message: 'Rede nao configurada. Defina REDE_PV e REDE_CHAVE_INTEGRACAO nas variaveis de ambiente.',
    77→    });

    79→
    80→  try {
    81→    var linkResult = await criarLinkPagamento({
    82→      valor: valorTotal,
    83→      seu_numero: faturaName || String(Date.now()),
    84→      descricao: 'Fatura ' + faturaName,
    85→      parcelas: cartaoInfo.parcelas,
    86→      nome_pagador: pag.nome || d.pagador_nome || '',
    87→      cpf_cnpj_pagador: pag.cpf_cnpj || d.pagador_cpf || '',
    88→      email_pagador: pag.email || '',
    89→      expiracao: 7 * 86400,
    90→      host: req.protocol + '://' + req.get('host'),

    92→

    94→
    95→    if (!checkoutUrl) {
    96→      console.error('[API/CARTAO] Checkout criado mas sem URL. Resposta bruta:', JSON.stringify(linkResult.raw));
    97→      return res.json({
    98→        success: false,
    99→        message: 'Checkout criado porem sem URL de pagamento. Verificar resposta da Rede.',
   100→        detail: linkResult.raw,
   101→      });
   102→    }
   103→
   104→    console.log('[API/CARTAO] Checkout URL gerada:', checkoutUrl.substring(0, 80) + '...');
   105→
   106→    res.json({
   107→      success: true,
   108→      data: {
   109→        forma_pagamento: d.forma_pagamento,
   110→        total_parcelas: 1,
   111→        valor_total: valorTotal.toFixed(2),
   112→        fatura_name: faturaName || '(nao informado)',
   113→        fatura_id: faturaId || 0,
   114→        pagamentos: [{
   115→          tipo: cartaoInfo.debito ? 'cartao_debito' : 'cartao_credito',
   116→          parcela: 1,
   117→          total_parcelas: 1,
   118→          valor_titulo: valorTotal.toFixed(2),
   119→          bandeira: cartaoInfo.bandeira,
   120→          parcelas: cartaoInfo.parcelas,
   121→          pix_copia_cola: checkoutUrl,
   122→          link_checkout: checkoutUrl,
   123→          checkout_id: linkResult.id || '',

   125→        odoo_push: 'nao_aplicavel',

   129→  } catch (err) {

   139→function calcDataVenc(base, dias) {
   140→  var db = base ? new Date(base + 'T12:00:00') : new Date();
   141→  var hj = new Date(); hj.setHours(0,0,0,0);
   142→  if (db <= hj) db = new Date();
   143→  db.setDate(db.getDate() + dias);
   144→  var a = db.getFullYear();
   145→  var m = String(db.getMonth()+1).padStart(2,'0');
   146→  var d = String(db.getDate()).padStart(2,'0');
   147→  return a+'-'+m+'-'+d;
   148→}

   435→    var parc = req.body.parcela || '';
   436→    var tp = req.body.total_parcelas || '';

   438→    if (!nn && !ld) {
   439→      res.setHeader('Content-Type', 'text/plain; charset=utf-8');
   440→      res.send('ERRO|Campos obrigatorios: nosso_numero ou linha_digitavel');

   444→    var dados = {
   445→      nosso_numero: nn, linha_digitavel: ld, codigo_barras: cb,
   446→      pix_copia_cola: pix, valor_titulo: vd, data_vencimento: vc,
   447→      nome_pagador: pn, cpf_cnpj_pagador: pc, seu_numero: sn,
   448→      parcela: parc ? parseInt(parc) : 0, total_parcelas: tp ? parseInt(tp) : 0,
   449→    };

   451→    var pdfBuf = await generatePdfFromFields(dados);

   454→    res.setHeader('Content-Type', 'text/plain; charset=utf-8');

   462→// === PIX: handler para /pagar ===
   463→async function handlePix(req, res, d) {
   464→  var fat = d.fatura || {};
   465→  var faturaName = fat.name || fat.seu_numero || d.fatura_name || '';
   466→  var faturaId = (fat.id || d.fatura_id) ? parseInt(String(fat.id || d.fatura_id)) : 0;
   467→  var valorTotal = parseFloat(fat.valor_nominal || d.fatura_valor) || 0;
   468→  var pag = d.pagador || {};
   469→  var pagNome = pag.nome || d.pagador_nome || '';
   470→  var pagCpf = pag.cpf_cnpj || d.pagador_cpf || '';
   471→
   472→  console.log('[API/PIX] Fatura:', faturaName, '| Valor:', valorTotal);
   473→
   474→  if (valorTotal <= 0) {
   475→    return res.json({ success: false, message: 'Valor invalido para PIX' });
   476→  }
   477→
   478→  // === DEDUPLICACAO: verificar se ja tem requisicao em curso ou resultado recente ===
   479→  var lockKey = getPixLockKey(faturaId, faturaName);
   480→
   481→  // Se tem resultado cacheado (mesma fatura, menos de 60s atras), reusa
   482→  if (_pixCache[lockKey]) {
   483→    var cached = _pixCache[lockKey];
   484→    var age = Date.now() - cached.timestamp;
   485→    if (age < PIX_CACHE_TTL) {
   486→      console.log('[API/PIX] REUSANDO resultado cacheado (' + (age / 1000).toFixed(1) + 's atras) para', lockKey);
   487→      return res.json(cached.response);
   488→    }
   489→    delete _pixCache[lockKey];
   490→  }
   491→
   492→  // Se tem requisicao em curso, espera ela terminar
   493→  if (_pixLocks[lockKey]) {
   494→    console.log('[API/PIX] AGUARDANDO lock existente para', lockKey);
   495→    try {
   496→      var cachedResult = await _pixLocks[lockKey];
   497→      console.log('[API/PIX] Lock liberado, retornando resultado da requisicao original');
   498→      return res.json(cachedResult);
   499→    } catch (lockErr) {
   500→      console.log('[API/PIX] Requisicao original falhou, tentando novamente...');
   501→    }
   502→  }
   503→
   504→  // Criar lock
   505→  var lockResolve;
   506→  _pixLocks[lockKey] = new Promise(function(resolve) { lockResolve = resolve; });
   507→
   508→  // Auto-limpar lock apos TTL
   509→  setTimeout(function() {
   510→    if (_pixLocks[lockKey]) {
   511→      console.warn('[API/PIX] Lock expirou por TTL para', lockKey);
   512→      delete _pixLocks[lockKey];
   513→    }
   514→  }, PIX_LOCK_TTL);
   515→
   516→  try {
   517→    var isCpf = pagCpf.replace(/\\D/g, '').length <= 11;
   518→    var pixResult = await criarCobrancaPix({
   519→      valor: valorTotal,
   520→      devedor: pagNome ? {
   521→        nome: pagNome,
   522→        cpf: isCpf ? pagCpf.replace(/\\D/g, '') : null,
   523→        cnpj: !isCpf ? pagCpf.replace(/\\D/g, '') : null,
   524→      } : undefined,
   525→      solicitacaoPagador: faturaName || '',
   526→      expiracao: 86400,
   527→    });
   528→
   529→    console.log('[API/PIX] PIX criado: TXID=' + (pixResult.txid || 'N/A'));
   530→
   531→    var pixCopiaCola = pixResult.pixCopiaECola || '';
   532→    var qrcodeBase64 = '';
   533→    var htmlPix = '';
   534→
   535→    // Gerar QR code a partir do pixCopiaECola (bwip-js v4 e async)
   536→    if (pixCopiaCola) {
   537→      try {
   538→        var qrPng = await bwipjs.toBuffer({
   539→          bcid: 'qrcode',
   540→          text: pixCopiaCola,
   541→          scale: 5,
   542→          width: 12,
   543→          height: 12,
   544→        });
   545→        qrcodeBase64 = 'data:image/png;base64,' + qrPng.toString('base64');
   546→        console.log('[API/PIX] QR code gerado:', (qrPng.length / 1024).toFixed(1) + 'KB');
   547→      } catch (qrErr) {
   548→        console.error('[API/PIX] Erro ao gerar QR code:', qrErr.message);
   549→      }
   550→    }
   551→
   552→    // Montar HTML para o campo Odoo
   553→    var valorFmt = 'R$ ' + valorTotal.toFixed(2).replace('.', ',');
   554→    htmlPix = '<div style="text-align:center; font-family:Arial,sans-serif; padding:10px;">' +
   555→      '<div style="font-size:18px; font-weight:bold; color:#333; margin-bottom:8px;">PIX</div>' +
   556→      '<div style="font-size:14px; color:#666; margin-bottom:12px;">' + faturaName + ' - ' + valorFmt + '</div>' +
   557→      (qrcodeBase64 ? '<img src="' + qrcodeBase64 + '" style="width:200px; height:200px; margin:0 auto 12px auto; display:block;" />' : '') +
   558→      (pixCopiaCola ? '<div style="font-size:10px; color:#999; word-break:break-all; max-width:400px; margin:0 auto; line-height:1.4;">' + pixCopiaCola + '</div>' : '') +
   559→      '</div>';
   560→
   561→    // Push PIX para Odoo (grava campos x_studio_*)
   562→    var pushResult = { pushed: false, reason: 'not_called' };
   563→    try {
   564→      pushResult = await pushPixToOdoo({
   565→        faturaId: faturaId,
   566→        faturaName: faturaName,
   567→        valor: valorTotal.toFixed(2).replace('.', ','),
   568→        pix: {
   569→          txid: pixResult.txid || '',
   570→          pix_copia_cola: pixCopiaCola,
   571→          qrcode_base64: qrcodeBase64,
   572→          html_pix: htmlPix,
   573→        },
   574→      });
   575→    } catch (pushErr) {
   576→      pushResult = { pushed: false, reason: pushErr.message };
   577→    }
   578→
   579→    var responseData = {
   580→      success: true,
   581→      data: {
   582→        forma_pagamento: 'PIX',
   583→        total_parcelas: 1,
   584→        valor_total: valorTotal.toFixed(2),
   585→        fatura_name: faturaName || '(nao informado)',
   586→        fatura_id: faturaId || 0,
   587→        pagamentos: [{
   588→          tipo: 'pix',
   589→          parcela: 1,
   590→          total_parcelas: 1,
   591→          valor_titulo: valorTotal.toFixed(2),
   592→          txid: pixResult.txid || '',
   593→          pix_copia_cola: pixCopiaCola,
   594→          qrcode_base64: qrcodeBase64,
   595→          html_pix: htmlPix,
   596→        }],
   597→        odoo_push: pushResult.pushed ? 'OK' : 'falhou_' + (pushResult.reason || 'unknown'),
   598→      },
   599→    };
   600→
   601→    // Cache resultado e liberar lock
   602→    _pixCache[lockKey] = { timestamp: Date.now(), response: responseData };
   603→    delete _pixLocks[lockKey];
   604→    lockResolve(responseData);
   605→
   606→    res.json(responseData);
   607→  } catch (err) {
   608→    console.error('[API/PIX] ERRO:', err.message);
   609→    var errResponse = { success: false, message: 'Erro ao criar PIX: ' + (err.message || 'Erro desconhecido') };
   610→    delete _pixLocks[lockKey];
   611→    lockResolve(errResponse);
   612→    res.json(errResponse);
   613→  }
   614→}
