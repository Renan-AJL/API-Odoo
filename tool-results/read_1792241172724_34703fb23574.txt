    46→// Retorna: { podeEmitir: true } ou { podeEmitir: false, motivo: string }
    47→async function checkBoletoAtivo(faturaId, faturaName) {
    48→  var config = require('../config');
    49→  var odooConfig = config.odoo;
    50→
    51→  if (!odooConfig || !odooConfig.enabled || !odooConfig.url || faturaId <= 0) {
    52→    // Sem Odoo ou sem faturaId, nao podemos verificar — permite emissao
    53→    return { podeEmitir: true };
    54→  }
    55→
    56→  try {
    57→    var xmlrpc = require('xmlrpc');
    58→    var base = odooConfig.url.replace(/\/+$/, '');
    59→    var host = base.replace('https://', '');
    60→    var client = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/common', port: 443 });
    61→
    62→    // Auth
    63→    var uid = await new Promise(function(ok, fail) {
    64→      client.methodCall('authenticate', [odooConfig.db, odooConfig.user, odooConfig.password, {}], function(e, r) {
    65→        if (e || !r) return fail(e || new Error('Auth failed'));
    66→        ok(r);
    67→      });
    68→    });
    69→
    70→    var objClient = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
    71→
    72→    function execKw(model, method, args, kwargs) {
    73→      return new Promise(function(ok, fail) {
    74→        var params = [odooConfig.db, uid, odooConfig.password, model, method, args];
    75→        if (kwargs) params.push(kwargs);
    76→        objClient.methodCall('execute_kw', params, function(e, r) {
    77→          if (e) return fail(e);
    78→          ok(r);
    79→        });
    80→      });
    81→    }
    82→
    83→    // Ler campos da fatura
    84→    var fieldsToRead = ['x_studio_itau_situacao', 'x_studio_itau_nosso_numero',
    85→      'x_studio_itau_boleto_status', 'x_studio_boleto_cancelado'];
    86→    var invoiceData = await execKw('account.move', 'read', [[faturaId]], { fields: fieldsToRead });
    87→
    88→    if (!invoiceData || !invoiceData.length) {
    89→      return { podeEmitir: true }; // Fatura nao encontrada, permite
    90→    }
    91→
    92→    var inv = invoiceData[0];
    93→    var situacao = (inv.x_studio_itau_situacao || '').toLowerCase();
    94→    var boletoStatus = (inv.x_studio_itau_boleto_status || '').toLowerCase();
    95→    var nossoNumero = inv.x_studio_itau_nosso_numero || '';
    96→    var boletoCancelado = inv.x_studio_boleto_cancelado || false;
    97→
    98→    console.log('[BOLETO-CHECK] Fatura ' + faturaId + ': situacao=' + situacao +
    99→      ' | boleto_status=' + boletoStatus + ' | nn=' + nossoNumero +
   100→      ' | cancelado=' + boletoCancelado);
   101→
   102→    // Se boleto foi cancelado ou expirado, pode emitir novo
   103→    if (boletoCancelado || situacao === 'cancelado' || situacao === 'expirado' ||
   104→        boletoStatus === 'cancelado' || boletoStatus === 'expirado') {
   105→      console.log('[BOLETO-CHECK] Boleto anterior cancelado/expirado. Pode emitir novo.');
   106→      return { podeEmitir: true };
   107→    }
   108→
   109→    // Se nao tem nosso_numero, nao tem boleto emitido
   110→    if (!nossoNumero) {
   111→      console.log('[BOLETO-CHECK] Sem nosso_numero. Pode emitir.');
   112→      return { podeEmitir: true };
   113→    }
   114→
   115→    // Se situacao = emitido ou em_aberto, verificar no Itau se ainda ativo
   116→    if (situacao === 'emitido' || boletoStatus === 'em_aberto' || situacao === '' && nossoNumero) {
   117→      // Tentar consultar boleto no Itau
   118→      try {
   119→        var nnFirst = nossoNumero.split(' | ')[0].trim(); // Pega primeiro se multiplos
   120→        var consultaResult = await consultarBoletoPorNossoNumero(nnFirst);
   121→
   122→        if (consultaResult && consultaResult.dados) {
   123→          var dados = consultaResult.dados;
   124→          // Verificar situacao do boleto no Itau
   125→          var situacaoItau = '';
   126→          if (dados.data && dados.data.dado_boleto) {
   127→            situacaoItau = (dados.data.dado_boleto.situacao_boleto || '').toLowerCase();
   128→          }
   129→          console.log('[BOLETO-CHECK] Itau situacao_boleto: ' + situacaoItau);
   130→
   131→          // Se boleto esta ativo (em_aberto, registrado, etc), BLOQUEAR
   132→          if (situacaoItau === 'em_aberto' || situacaoItau === 'registrado' ||
   133→              situacaoItau === '' && dados.data) {
   134→            return {
   135→              podeEmitir: false,
   136→              motivo: 'Primeiro, cancele o boleto no Itaú para nova emissão. Nosso Número: ' + nnFirst,
   137→              nossoNumero: nnFirst,
   138→            };
   139→          }
   140→
   141→          // Se baixado/cancelado no Itau, pode emitir novo
   142→          if (situacaoItau.indexOf('cancelado') >= 0 || situacaoItau.indexOf('baixado') >= 0 ||
   143→              situacaoItau.indexOf('pago') >= 0) {
   144→            console.log('[BOLETO-CHECK] Boleto ' + situacaoItau + ' no Itau. Pode emitir novo.');
   145→            return { podeEmitir: true };
   146→          }
   147→        }
   148→      } catch (itauErr) {
   149→        console.warn('[BOLETO-CHECK] Erro ao consultar Itau: ' + itauErr.message);
   150→        // Se nao conseguiu consultar o Itau, e temos situacao=emitido no Odoo,
   151→        // BLOQUEIA por seguranca (nao sabemos se o boleto ainda esta ativo)
   152→        return {
   153→          podeEmitir: false,
   154→          motivo: 'Primeiro, cancele o boleto no Itaú para nova emissão. Nosso Número: ' + nossoNumero.split(' | ')[0] + ' (não foi possível verificar status no Itaú)',
   155→          nossoNumero: nossoNumero.split(' | ')[0],
   156→        };
   157→      }
   158→    }
   159→
   160→    // Se situacao = pago, pode emitir novo (boleto ja foi pago)
   161→    if (situacao === 'pago' || boletoStatus === 'pago') {
   162→      console.log('[BOLETO-CHECK] Boleto anterior ja pago. Pode emitir novo.');
   163→      return { podeEmitir: true };
   164→    }
   165→
   166→    // Default: se tem nosso_numero e situacao nao e cancelado/expirado/pago, bloqueia
   167→    if (nossoNumero && situacao !== 'cancelado' && situacao !== 'expirado' && situacao !== 'pago' &&
   168→        boletoStatus !== 'cancelado' && boletoStatus !== 'expirado' && boletoStatus !== 'pago') {
   169→      return {
   170→        podeEmitir: false,
   171→        motivo: 'Primeiro, cancele o boleto no Itaú para nova emissão. Nosso Número: ' + nossoNumero.split(' | ')[0],
   172→        nossoNumero: nossoNumero.split(' | ')[0],
   173→      };
   174→    }
   175→
   176→    return { podeEmitir: true };
   177→  } catch (err) {
   178→    console.warn('[BOLETO-CHECK] Erro na verificacao: ' + err.message);
   179→    // Falha FECHADA: se nao conseguimos verificar, BLOQUEIA emissao por seguranca.
   180→    // Evita duplicacao quando Odoo ou API Itau estao instaveis.
   181→    return {
   182→      podeEmitir: false,
   183→      motivo: 'Primeiro, cancele o boleto no Itaú para nova emissão. Não foi possível verificar o status do boleto (' + (err.message || 'erro desconhecido') + ')',
   184→    };
   185→  }
   186→}

   230→// --- Deteccao de cartao por bandeira ---
   231→var BANDEIRAS = ['VISA', 'MASTER', 'ELO', 'AMEX', 'HIPERCARD', 'HIPER'];
   232→
   233→function detectarCartao(formaPag) {
   234→  if (!formaPag) return null;
   235→  var n = formaPag.trim().toUpperCase();
   236→  for (var i = 0; i < BANDEIRAS.length; i++) {
   237→    var b = BANDEIRAS[i];
   238→    if (n.indexOf(b) === 0 || n.indexOf(b + ' ') === 0) {
   239→      var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
   240→      var parcelas = 1;
   241→      var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
   242→      if (match) parcelas = parseInt(match[1]);
   243→      return { bandeira: b, parcelas: parcelas, debito: debito };
   244→    }
   245→  }
   246→  // Generico: CARTAO, CARTAO CREDITO, CREDITO, etc.
   247→  if (n.indexOf('CART') === 0 || n.indexOf('CREDITO') === 0) {
   248→    var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
   249→    var parcelas = 1;
   250→    var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
   251→    if (match) parcelas = parseInt(match[1]);
   252→    return { bandeira: 'CARTAO', parcelas: parcelas, debito: debito };
   253→  }
   254→  return null;
   255→}

   258→async function handleCartao(req, res, d, cartaoInfo) {
   259→  var fat = d.fatura || {};
   260→  var faturaName = fat.name || fat.seu_numero || d.fatura_name || '';
   261→  var faturaId = (fat.id || d.fatura_id) ? parseInt(String(fat.id || d.fatura_id)) : 0;
   262→  var valorTotal = parseFloat(fat.valor_nominal || d.fatura_valor) || 0;
   263→  var pag = d.pagador || {};
   264→
   265→  console.log('[API/CARTAO] Fatura:', faturaName, '| Valor:', valorTotal, '| Bandeira:', cartaoInfo.bandeira);
   266→
   267→  if (valorTotal <= 0) {
   268→    return res.json({ success: false, message: 'Valor invalido para pagamento em cartao' });
   269→  }
   270→
   271→  if (!config.rede.pv || !config.rede.chaveIntegracao) {
   272→    console.error('[API/CARTAO] Rede nao configurada. PV:', config.rede.pv ? 'OK' : 'faltando',
   273→      '| Chave:', config.rede.chaveIntegracao ? 'OK' : 'faltando');
   274→    return res.json({
   275→      success: false,
   276→      message: 'Rede nao configurada. Defina REDE_PV e REDE_CHAVE_INTEGRACAO nas variaveis de ambiente.',
   277→    });
   278→  }
   279→
   280→  try {
   281→    var linkResult = await criarLinkPagamento({
   282→      valor: valorTotal,
   283→      seu_numero: faturaName || String(Date.now()),
   284→      descricao: 'Fatura ' + faturaName,
   285→      parcelas: cartaoInfo.parcelas,
   286→      nome_pagador: pag.nome || d.pagador_nome || '',
   287→      cpf_cnpj_pagador: pag.cpf_cnpj || d.pagador_cpf || '',
   288→      email_pagador: pag.email || '',
   289→      expiracao: 7 * 86400,
   290→      host: req.protocol + '://' + req.get('host'),
   291→    });
   292→
   293→    var checkoutUrl = linkResult.link || '';
   294→
   295→    if (!checkoutUrl) {
   296→      console.error('[API/CARTAO] Checkout criado mas sem URL. Resposta bruta:', JSON.stringify(linkResult.raw));
   297→      return res.json({
   298→        success: false,
   299→        message: 'Checkout criado porem sem URL de pagamento. Verificar resposta da Rede.',
   300→        detail: linkResult.raw,
   301→      });
   302→    }
   303→
   304→    console.log('[API/CARTAO] Checkout URL gerada:', checkoutUrl.substring(0, 80) + '...');
   305→
   306→    res.json({
   307→      success: true,
   308→      data: {
   309→        forma_pagamento: d.forma_pagamento,
   310→        total_parcelas: 1,
   311→        valor_total: valorTotal.toFixed(2),
   312→        fatura_name: faturaName || '(nao informado)',
   313→        fatura_id: faturaId || 0,
   314→        pagamentos: [{
   315→          tipo: cartaoInfo.debito ? 'cartao_debito' : 'cartao_credito',
   316→          parcela: 1,
   317→          total_parcelas: 1,
   318→          valor_titulo: valorTotal.toFixed(2),
   319→          bandeira: cartaoInfo.bandeira,
   320→          parcelas: cartaoInfo.parcelas,
   321→          pix_copia_cola: checkoutUrl,
   322→          link_checkout: checkoutUrl,
   323→          checkout_id: linkResult.id || '',
   324→        }],
   325→        odoo_push: 'nao_aplicavel',

   329→  } catch (err) {

   339→function calcDataVenc(base, dias) {
   340→  var db = base ? new Date(base + 'T12:00:00') : new Date();
   341→  var hj = new Date(); hj.setHours(0,0,0,0);
   342→  if (db <= hj) db = new Date();
   343→  db.setDate(db.getDate() + dias);
   344→  var a = db.getFullYear();
   345→  var m = String(db.getMonth()+1).padStart(2,'0');
   346→  var d = String(db.getDate()).padStart(2,'0');
   347→  return a+'-'+m+'-'+d;

   364→    }

   367→    if (formaPag.trim().toUpperCase().indexOf('PIX') === 0) {
   368→      return await handlePix(req, res, d);
   369→    }
   370→
   371→    var fat = d.fatura || {};
   372→    var fatNameFromNested = fat.name || fat.seu_numero || '';
   373→    var fatNameFromFlat = d.fatura_name || d.invoice_name || '';
   374→    var faturaName = fatNameFromNested || fatNameFromFlat;
   375→    var faturaId = (fat.id || d.fatura_id || d.invoice_id) ? parseInt(String(fat.id || d.fatura_id || d.invoice_id)) : 0;
   376→    var valorTotal = parseFloat(fat.valor_nominal || d.fatura_valor) || 0;
   377→    var dataVencBase = fat.data_vencimento || d.fatura_vencimento || '';
   378→

   380→

   382→    console.log('[API] Plano parsed:', JSON.stringify(plano));
   383→
   384→    if (plano.tipo !== 'boleto' || plano.parcelas.length === 0) {
   385→      return res.json({ success: false, message: 'Forma de pagamento nao suportada para boleto: ' + formaPag });
   386→    }
   387→
   388→    // === GUARDA: Verificar boleto ativo antes de emitir ===
   389→    var boletoCheck = await checkBoletoAtivo(faturaId, faturaName);
   390→    if (!boletoCheck.podeEmitir) {
   391→      console.warn('[API/BOLETO] Boleto ativo encontrado. Bloqueando emissao.');
   392→      // Postar chatter na fatura para rastreabilidade
   393→      try { await postBoletoBlockedChatter(faturaId, boletoCheck.motivo); } catch(cErr) {}
   394→      return res.json({ success: false, message: boletoCheck.motivo, nosso_numero: boletoCheck.nossoNumero });
   395→    }
   396→
   397→    // === DEDUPLICACAO: Lock anti-concorrencia boleto ===
   398→    var boletoLockKey = getBoletoLockKey(faturaId, faturaName);
   399→
   400→    // Se tem resultado cacheado (mesma fatura, menos de 90s atras), reusa
   401→    if (_boletoCache[boletoLockKey]) {
   402→      var boletoCached = _boletoCache[boletoLockKey];
   403→      var boletoAge = Date.now() - boletoCached.timestamp;
   404→      if (boletoAge < BOLETO_CACHE_TTL) {
   405→        console.log('[API/BOLETO] REUSANDO resultado cacheado (' + (boletoAge / 1000).toFixed(1) + 's atras) para', boletoLockKey);
   406→        return res.json(boletoCached.response);
   407→      }
   408→      delete _boletoCache[boletoLockKey];
   409→    }
   410→
   411→    // Se tem requisicao em curso, espera ela terminar
   412→    if (_boletoLocks[boletoLockKey]) {
   413→      console.log('[API/BOLETO] AGUARDANDO lock existente para', boletoLockKey);
   414→      try {
   415→        var boletoCachedResult = await _boletoLocks[boletoLockKey];
   416→        console.log('[API/BOLETO] Lock liberado, retornando resultado da requisicao original');
   417→        return res.json(boletoCachedResult);
   418→      } catch (boletoLockErr) {
   419→        console.log('[API/BOLETO] Requisicao original falhou, tentando novamente...');
   420→      }
   421→    }
   422→

   902→async function handlePix(req, res, d) {
   903→  var fat = d.fatura || {};
   904→  var faturaName = fat.name || fat.seu_numero || d.fatura_name || '';
   905→  var faturaId = (fat.id || d.fatura_id) ? parseInt(String(fat.id || d.fatura_id)) : 0;
   906→  var valorTotal = parseFloat(fat.valor_nominal || d.fatura_valor) || 0;
   907→  var pag = d.pagador || {};
   908→  var pagNome = pag.nome || d.pagador_nome || '';
   909→  var pagCpf = pag.cpf_cnpj || d.pagador_cpf || '';
   910→
   911→  console.log('[API/PIX] Fatura:', faturaName, '| Valor:', valorTotal);
   912→
   913→  if (valorTotal <= 0) {
   914→    return res.json({ success: false, message: 'Valor invalido para PIX' });
   915→  }
   916→
   917→  // === DEDUPLICACAO: verificar se ja tem requisicao em curso ou resultado recente ===
   918→  var lockKey = getPixLockKey(faturaId, faturaName);
   919→
   920→  // Se tem resultado cacheado (mesma fatura, menos de 60s atras), reusa
   921→  if (_pixCache[lockKey]) {
   922→    var cached = _pixCache[lockKey];
   923→    var age = Date.now() - cached.timestamp;
   924→    if (age < PIX_CACHE_TTL) {
   925→      console.log('[API/PIX] REUSANDO resultado cacheado (' + (age / 1000).toFixed(1) + 's atras) para', lockKey);
   926→      return res.json(cached.response);
   927→    }
   928→    delete _pixCache[lockKey];
   929→  }
   930→
   931→  // Se tem requisicao em curso, espera ela terminar
   932→  if (_pixLocks[lockKey]) {
   933→    console.log('[API/PIX] AGUARDANDO lock existente para', lockKey);
   934→    try {
   935→      var cachedResult = await _pixLocks[lockKey];
   936→      console.log('[API/PIX] Lock liberado, retornando resultado da requisicao original');
   937→      return res.json(cachedResult);
   938→    } catch (lockErr) {
   939→      console.log('[API/PIX] Requisicao original falhou, tentando novamente...');
   940→    }
   941→  }
   942→
   943→  // Criar lock
   944→  var lockResolve;
   945→  _pixLocks[lockKey] = new Promise(function(resolve) { lockResolve = resolve; });
   946→
   947→  // Auto-limpar lock apos TTL
   948→  setTimeout(function() {
   949→    if (_pixLocks[lockKey]) {
   950→      console.warn('[API/PIX] Lock expirou por TTL para', lockKey);
   951→      delete _pixLocks[lockKey];
   952→    }
   953→  }, PIX_LOCK_TTL);
   954→
   955→  try {
   956→    var devedor = buildDevedorFromDoc(pagNome, pagCpf);
   957→    if (!devedor && pagNome) {

   961→      valor: valorTotal,
   962→      devedor: devedor,

   964→      expiracao: 86400,

   977→          bcid: 'qrcode',

   980→          width: 12,
   981→          height: 12,
   982→        });
   983→        qrcodeBase64 = 'data:image/png;base64,' + qrPng.toString('base64');
   984→        console.log('[API/PIX] QR code gerado:', (qrPng.length / 1024).toFixed(1) + 'KB');
   985→      } catch (qrErr) {
   986→        console.error('[API/PIX] Erro ao gerar QR code:', qrErr.message);

   997→      '</div>';

  1000→    var responseData = {
  1001→      success: true,

  1087→  function odooRpc(model, method, args, kwargs) {
  1088→    var xmlrpc = require('xmlrpc');
  1089→    var base = oc.url.replace(/\/+$/, '');
  1090→    var host = base.replace('https://', '');

  1092→    var modelsCli = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
  1093→    return new Promise(function(ok, fail) {
  1094→      commonCli.methodCall('authenticate', [oc.db, oc.user, oc.password, {}], function(e, uid) {
  1095→        if (e || !uid) return fail(e || new Error('auth failed'));
  1096→        var params = [oc.db, uid, oc.password, model, method, args || []];
  1097→        if (kwargs) params.push(kwargs);
  1098→        modelsCli.methodCall('execute_kw', params, function(e2, r) {
  1099→          if (e2) fail(e2); else ok(r);

  1101→      });
  1102→    });
