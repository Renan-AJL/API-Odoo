    49→  if (!odooConfig || !odooConfig.enabled || !odooConfig.url || faturaId <= 0) {

    57→    var host = base.replace('https://', '');
    58→    var client = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/common', port: 443 });
    59→
    60→    // Auth
    61→    var uid = await new Promise(function(ok, fail) {
    62→      client.methodCall('authenticate', [odooConfig.db, odooConfig.user, odooConfig.password, {}], function(e, r) {
    63→        if (e || !r) return fail(e || new Error('Auth failed'));
    64→        ok(r);
    65→      });
    66→    });
    67→
    68→    var objClient = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
    69→
    70→    function execKw(model, method, args, kwargs) {
    71→      return new Promise(function(ok, fail) {
    72→        var params = [odooConfig.db, uid, odooConfig.password, model, method, args];
    73→        if (kwargs) params.push(kwargs);
    74→        objClient.methodCall('execute_kw', params, function(e, r) {
    75→          if (e) return fail(e);
    76→          ok(r);
    77→        });
    78→      });
    79→    }
    80→

    92→    var boletoStatus = (inv.x_studio_itau_boleto_status || '').toLowerCase();

    94→    var boletoCancelado = inv.x_studio_boleto_cancelado || false;

   105→    }
   106→
   107→    // Se nao tem nosso_numero, nao tem boleto emitido
   108→    if (!nossoNumero) {
   109→      console.log('[BOLETO-CHECK] Sem nosso_numero. Pode emitir.');
   110→      return { podeEmitir: true };
   111→    }
   112→
   113→    // Se situacao = emitido ou em_aberto, verificar no Itau se ainda ativo
   114→    if (situacao === 'emitido' || boletoStatus === 'em_aberto' || situacao === '' && nossoNumero) {
   115→      // Tentar consultar boleto no Itau
   116→      try {
   117→        var nnFirst = nossoNumero.split(' | ')[0].trim(); // Pega primeiro se multiplos
   118→        var consultaResult = await consultarBoletoPorNossoNumero(nnFirst);
   119→
   120→        if (consultaResult && consultaResult.dados) {
   121→          var dados = consultaResult.dados;
   122→          // Verificar situacao do boleto no Itau
   123→          var situacaoItau = '';
   124→          if (dados.data && dados.data.dado_boleto) {
   125→            situacaoItau = (dados.data.dado_boleto.situacao_boleto || '').toLowerCase();
   126→          }
   127→          console.log('[BOLETO-CHECK] Itau situacao_boleto: ' + situacaoItau);
   128→
   129→          // Se boleto esta ativo (em_aberto, registrado, etc), BLOQUEAR
   130→          if (situacaoItau === 'em_aberto' || situacaoItau === 'registrado' ||
   131→              situacaoItau === '' && dados.data) {
   132→            return {
   133→              podeEmitir: false,
   134→              motivo: 'Primeiro, cancele o boleto no Itau para nova emissao. Nosso Numero: ' + nnFirst,
   135→              nossoNumero: nnFirst,
   136→            };
   137→          }
   138→
   139→          // Se baixado/cancelado no Itau, pode emitir novo
   140→          if (situacaoItau.indexOf('cancelado') >= 0 || situacaoItau.indexOf('baixado') >= 0 ||
   141→              situacaoItau.indexOf('pago') >= 0) {
   142→            console.log('[BOLETO-CHECK] Boleto ' + situacaoItau + ' no Itau. Pode emitir novo.');
   143→            return { podeEmitir: true };
   144→          }
   145→        }
   146→      } catch (itauErr) {
   147→        console.warn('[BOLETO-CHECK] Erro ao consultar Itau: ' + itauErr.message);
   148→        // Se nao conseguiu consultar o Itau, e temos situacao=emitido no Odoo,
   149→        // BLOQUEIA por seguranca (nao sabemos se o boleto ainda esta ativo)
   150→        return {
   151→          podeEmitir: false,
   152→          motivo: 'Primeiro, cancele o boleto no Itau para nova emissao. Nosso Numero: ' + nossoNumero.split(' | ')[0] + ' (nao foi possivel verificar status no Itau)',
   153→          nossoNumero: nossoNumero.split(' | ')[0],
   154→        };
   155→      }
   156→    }
   157→
   158→    // Se situacao = pago, pode emitir novo (boleto ja foi pago)
   159→    if (situacao === 'pago' || boletoStatus === 'pago') {
   160→      console.log('[BOLETO-CHECK] Boleto anterior ja pago. Pode emitir novo.');
   161→      return { podeEmitir: true };
   162→    }
   163→
   164→    // Default: se tem nosso_numero e situacao nao e cancelado/expirado/pago, bloqueia
   165→    if (nossoNumero && situacao !== 'cancelado' && situacao !== 'expirado' && situacao !== 'pago' &&
   166→        boletoStatus !== 'cancelado' && boletoStatus !== 'expirado' && boletoStatus !== 'pago') {
   167→      return {
   168→        podeEmitir: false,
   169→        motivo: 'Primeiro, cancele o boleto no Itau para nova emissao. Nosso Numero: ' + nossoNumero.split(' | ')[0],
   170→        nossoNumero: nossoNumero.split(' | ')[0],
   171→      };
   172→    }
   173→
   174→    return { podeEmitir: true };
   175→  } catch (err) {
   176→    console.warn('[BOLETO-CHECK] Erro na verificacao: ' + err.message);
   177→    // Em caso de erro, permite emissao (falha aberta — melhor que bloquear tudo)
   178→    return { podeEmitir: true };
   179→  }
   180→}

   182→// --- Deteccao de cartao por bandeira ---
   183→var BANDEIRAS = ['VISA', 'MASTER', 'ELO', 'AMEX', 'HIPERCARD', 'HIPER'];
   184→
   185→function detectarCartao(formaPag) {
   186→  if (!formaPag) return null;
   187→  var n = formaPag.trim().toUpperCase();
   188→  for (var i = 0; i < BANDEIRAS.length; i++) {
   189→    var b = BANDEIRAS[i];
   190→    if (n.indexOf(b) === 0 || n.indexOf(b + ' ') === 0) {
   191→      var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
   192→      var parcelas = 1;
   193→      var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
   194→      if (match) parcelas = parseInt(match[1]);
   195→      return { bandeira: b, parcelas: parcelas, debito: debito };
   196→    }
   197→  }
   198→  // Generico: CARTAO, CARTAO CREDITO, CREDITO, etc.
   199→  if (n.indexOf('CART') === 0 || n.indexOf('CREDITO') === 0) {
   200→    var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
   201→    var parcelas = 1;
   202→    var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
   203→    if (match) parcelas = parseInt(match[1]);
   204→    return { bandeira: 'CARTAO', parcelas: parcelas, debito: debito };
   205→  }
   206→  return null;
   207→}

   210→async function handleCartao(req, res, d, cartaoInfo) {
   211→  var fat = d.fatura || {};
   212→  var faturaName = fat.name || fat.seu_numero || d.fatura_name || '';

   215→  var pag = d.pagador || {};
   216→
   217→  console.log('[API/CARTAO] Fatura:', faturaName, '| Valor:', valorTotal, '| Bandeira:', cartaoInfo.bandeira);
   218→
   219→  if (valorTotal <= 0) {
   220→    return res.json({ success: false, message: 'Valor invalido para pagamento em cartao' });
   221→  }
   222→
   223→  if (!config.rede.pv || !config.rede.chaveIntegracao) {
   224→    console.error('[API/CARTAO] Rede nao configurada. PV:', config.rede.pv ? 'OK' : 'faltando',
   225→      '| Chave:', config.rede.chaveIntegracao ? 'OK' : 'faltando');
   226→    return res.json({
   227→      success: false,
   228→      message: 'Rede nao configurada. Defina REDE_PV e REDE_CHAVE_INTEGRACAO nas variaveis de ambiente.',
   229→    });

   231→
   232→  try {
   233→    var linkResult = await criarLinkPagamento({
   234→      valor: valorTotal,
   235→      seu_numero: faturaName || String(Date.now()),
   236→      descricao: 'Fatura ' + faturaName,
   237→      parcelas: cartaoInfo.parcelas,
   238→      nome_pagador: pag.nome || d.pagador_nome || '',
   239→      cpf_cnpj_pagador: pag.cpf_cnpj || d.pagador_cpf || '',
   240→      email_pagador: pag.email || '',
   241→      expiracao: 7 * 86400,
   242→      host: req.protocol + '://' + req.get('host'),
   243→    });
   244→
   245→    var checkoutUrl = linkResult.link || '';
   246→
   247→    if (!checkoutUrl) {
   248→      console.error('[API/CARTAO] Checkout criado mas sem URL. Resposta bruta:', JSON.stringify(linkResult.raw));
   249→      return res.json({
   250→        success: false,
   251→        message: 'Checkout criado porem sem URL de pagamento. Verificar resposta da Rede.',
   252→        detail: linkResult.raw,
   253→      });
   254→    }
   255→
   256→    console.log('[API/CARTAO] Checkout URL gerada:', checkoutUrl.substring(0, 80) + '...');
   257→
   258→    res.json({
   259→      success: true,
   260→      data: {
   261→        forma_pagamento: d.forma_pagamento,
   262→        total_parcelas: 1,
   263→        valor_total: valorTotal.toFixed(2),
   264→        fatura_name: faturaName || '(nao informado)',
   265→        fatura_id: faturaId || 0,
   266→        pagamentos: [{
   267→          tipo: cartaoInfo.debito ? 'cartao_debito' : 'cartao_credito',
   268→          parcela: 1,
   269→          total_parcelas: 1,
   270→          valor_titulo: valorTotal.toFixed(2),
   271→          bandeira: cartaoInfo.bandeira,
   272→          parcelas: cartaoInfo.parcelas,
   273→          pix_copia_cola: checkoutUrl,
   274→          link_checkout: checkoutUrl,
   275→          checkout_id: linkResult.id || '',
   276→        }],
   277→        odoo_push: 'nao_aplicavel',
   278→      }
   279→    });
   280→
   281→  } catch (err) {
   282→    console.error('[API/CARTAO] ERRO:', err.message);
   283→    if (err.detail) console.error('[API/CARTAO] Detalhes:', JSON.stringify(err.detail));
   284→    res.json({
   285→      success: false,
   286→      message: 'Erro ao gerar link de pagamento: ' + (err.message || 'Erro desconhecido'),
   287→    });
   288→  }

   291→function calcDataVenc(base, dias) {
   292→  var db = base ? new Date(base + 'T12:00:00') : new Date();
   293→  var hj = new Date(); hj.setHours(0,0,0,0);
   294→  if (db <= hj) db = new Date();
   295→  db.setDate(db.getDate() + dias);
   296→  var a = db.getFullYear();
   297→  var m = String(db.getMonth()+1).padStart(2,'0');
   298→  var d = String(db.getDate()).padStart(2,'0');
   299→  return a+'-'+m+'-'+d;
   300→}

   679→// === PIX: handler para /pagar ===
   680→async function handlePix(req, res, d) {
   681→  var fat = d.fatura || {};
   682→  var faturaName = fat.name || fat.seu_numero || d.fatura_name || '';
   683→  var faturaId = (fat.id || d.fatura_id) ? parseInt(String(fat.id || d.fatura_id)) : 0;
   684→  var valorTotal = parseFloat(fat.valor_nominal || d.fatura_valor) || 0;
   685→  var pag = d.pagador || {};
   686→  var pagNome = pag.nome || d.pagador_nome || '';
   687→  var pagCpf = pag.cpf_cnpj || d.pagador_cpf || '';
   688→
   689→  console.log('[API/PIX] Fatura:', faturaName, '| Valor:', valorTotal);
   690→
   691→  if (valorTotal <= 0) {
   692→    return res.json({ success: false, message: 'Valor invalido para PIX' });
   693→  }
   694→
   695→  // === DEDUPLICACAO: verificar se ja tem requisicao em curso ou resultado recente ===
   696→  var lockKey = getPixLockKey(faturaId, faturaName);
   697→
   698→  // Se tem resultado cacheado (mesma fatura, menos de 60s atras), reusa
   699→  if (_pixCache[lockKey]) {
   700→    var cached = _pixCache[lockKey];
   701→    var age = Date.now() - cached.timestamp;
   702→    if (age < PIX_CACHE_TTL) {
   703→      console.log('[API/PIX] REUSANDO resultado cacheado (' + (age / 1000).toFixed(1) + 's atras) para', lockKey);
   704→      return res.json(cached.response);
   705→    }
   706→    delete _pixCache[lockKey];
   707→  }
   708→
   709→  // Se tem requisicao em curso, espera ela terminar
   710→  if (_pixLocks[lockKey]) {
   711→    console.log('[API/PIX] AGUARDANDO lock existente para', lockKey);
   712→    try {
   713→      var cachedResult = await _pixLocks[lockKey];
   714→      console.log('[API/PIX] Lock liberado, retornando resultado da requisicao original');
   715→      return res.json(cachedResult);
   716→    } catch (lockErr) {
   717→      console.log('[API/PIX] Requisicao original falhou, tentando novamente...');
   718→    }
   719→  }
   720→
   721→  // Criar lock
   722→  var lockResolve;
   723→  _pixLocks[lockKey] = new Promise(function(resolve) { lockResolve = resolve; });
   724→
   725→  // Auto-limpar lock apos TTL
   726→  setTimeout(function() {
   727→    if (_pixLocks[lockKey]) {
   728→      console.warn('[API/PIX] Lock expirou por TTL para', lockKey);
   729→      delete _pixLocks[lockKey];
   730→    }
   731→  }, PIX_LOCK_TTL);
   732→
   733→  try {
   734→    var devedor = buildDevedorFromDoc(pagNome, pagCpf);
   735→    if (!devedor && pagNome) {
   736→      console.warn('[API/PIX] Pagador com nome mas sem CPF/CNPJ valido, criando PIX sem devedor');
   737→    }
   738→    var pixResult = await criarCobrancaPix({
   739→      valor: valorTotal,
   740→      devedor: devedor,
   741→      solicitacaoPagador: faturaName || '',
   742→      expiracao: 86400,
   743→    });
   744→
   745→    console.log('[API/PIX] PIX criado: TXID=' + (pixResult.txid || 'N/A'));
   746→
   747→    var pixCopiaCola = pixResult.pixCopiaECola || '';
   748→    var qrcodeBase64 = '';
   749→    var htmlPix = '';
   750→

   752→    if (pixCopiaCola) {
   753→      try {
   754→        var qrPng = await bwipjs.toBuffer({
   755→          bcid: 'qrcode',
   756→          text: pixCopiaCola,
   757→          scale: 5,
   758→          width: 12,
   759→          height: 12,
   760→        });
   761→        qrcodeBase64 = 'data:image/png;base64,' + qrPng.toString('base64');
   762→        console.log('[API/PIX] QR code gerado:', (qrPng.length / 1024).toFixed(1) + 'KB');
   763→      } catch (qrErr) {
   764→        console.error('[API/PIX] Erro ao gerar QR code:', qrErr.message);
   765→      }
   766→    }
   767→
   768→    // Montar HTML para o campo Odoo
   769→    var valorFmt = 'R$ ' + valorTotal.toFixed(2).replace('.', ',');
   770→    htmlPix = '<div style="text-align:center; font-family:Arial,sans-serif; padding:10px;">' +
