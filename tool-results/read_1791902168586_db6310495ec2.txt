    33→var BANDEIRAS = ['VISA', 'MASTER', 'ELO', 'AMEX', 'HIPERCARD', 'HIPER'];

    35→function detectarCartao(formaPag) {
    36→  if (!formaPag) return null;
    37→  var n = formaPag.trim().toUpperCase();
    38→  for (var i = 0; i < BANDEIRAS.length; i++) {
    39→    var b = BANDEIRAS[i];
    40→    if (n.indexOf(b) === 0 || n.indexOf(b + ' ') === 0) {
    41→      var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
    42→      var parcelas = 1;
    43→      var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
    44→      if (match) parcelas = parseInt(match[1]);
    45→      return { bandeira: b, parcelas: parcelas, debito: debito };
    46→    }
    47→  }
    48→  // Generico: CARTAO, CARTAO CREDITO, CREDITO, etc.
    49→  if (n.indexOf('CART') === 0 || n.indexOf('CREDITO') === 0) {
    50→    var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
    51→    var parcelas = 1;
    52→    var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
    53→    if (match) parcelas = parseInt(match[1]);
    54→    return { bandeira: 'CARTAO', parcelas: parcelas, debito: debito };
    55→  }
    56→  return null;
    57→}

    77→      success: false,

    82→  try {
    83→    var linkResult = await criarLinkPagamento({

    85→      seu_numero: faturaName || String(Date.now()),
    86→      descricao: 'Fatura ' + faturaName,
    87→      parcelas: cartaoInfo.parcelas,

    89→      cpf_cnpj_pagador: pag.cpf_cnpj || d.pagador_cpf || '',
    90→      email_pagador: pag.email || '',
    91→      expiracao: 7 * 86400,

   100→        success: false,
   101→        message: 'Checkout criado porem sem URL de pagamento. Verificar resposta da Rede.',
   102→        detail: linkResult.raw,
   103→      });
   104→    }
   105→
   106→    console.log('[API/CARTAO] Checkout URL gerada:', checkoutUrl.substring(0, 80) + '...');
   107→
   108→    res.json({
   109→      success: true,
   110→      data: {
   111→        forma_pagamento: d.forma_pagamento,
   112→        total_parcelas: 1,
   113→        valor_total: valorTotal.toFixed(2),
   114→        fatura_name: faturaName || '(nao informado)',
   115→        fatura_id: faturaId || 0,
   116→        pagamentos: [{
   117→          tipo: cartaoInfo.debito ? 'cartao_debito' : 'cartao_credito',
   118→          parcela: 1,
   119→          total_parcelas: 1,
   120→          valor_titulo: valorTotal.toFixed(2),
   121→          bandeira: cartaoInfo.bandeira,
   122→          parcelas: cartaoInfo.parcelas,
   123→          pix_copia_cola: checkoutUrl,
   124→          link_checkout: checkoutUrl,

   128→      }

   131→  } catch (err) {

   141→function calcDataVenc(base, dias) {
   142→  var db = base ? new Date(base + 'T12:00:00') : new Date();
   143→  var hj = new Date(); hj.setHours(0,0,0,0);
   144→  if (db <= hj) db = new Date();
   145→  db.setDate(db.getDate() + dias);
   146→  var a = db.getFullYear();
   147→  var m = String(db.getMonth()+1).padStart(2,'0');
   148→  var d = String(db.getDate()).padStart(2,'0');
   149→  return a+'-'+m+'-'+d;
   150→}

   464→// === PIX: handler para /pagar ===
   465→async function handlePix(req, res, d) {
   466→  var fat = d.fatura || {};
   467→  var faturaName = fat.name || fat.seu_numero || d.fatura_name || '';
   468→  var faturaId = (fat.id || d.fatura_id) ? parseInt(String(fat.id || d.fatura_id)) : 0;
   469→  var valorTotal = parseFloat(fat.valor_nominal || d.fatura_valor) || 0;
   470→  var pag = d.pagador || {};
   471→  var pagNome = pag.nome || d.pagador_nome || '';
   472→  var pagCpf = pag.cpf_cnpj || d.pagador_cpf || '';
   473→
   474→  console.log('[API/PIX] Fatura:', faturaName, '| Valor:', valorTotal);
   475→
   476→  if (valorTotal <= 0) {
   477→    return res.json({ success: false, message: 'Valor invalido para PIX' });
   478→  }
   479→
   480→  // === DEDUPLICACAO: verificar se ja tem requisicao em curso ou resultado recente ===
   481→  var lockKey = getPixLockKey(faturaId, faturaName);
   482→
   483→  // Se tem resultado cacheado (mesma fatura, menos de 60s atras), reusa
   484→  if (_pixCache[lockKey]) {
   485→    var cached = _pixCache[lockKey];
   486→    var age = Date.now() - cached.timestamp;
   487→    if (age < PIX_CACHE_TTL) {
   488→      console.log('[API/PIX] REUSANDO resultado cacheado (' + (age / 1000).toFixed(1) + 's atras) para', lockKey);
   489→      return res.json(cached.response);
   490→    }
   491→    delete _pixCache[lockKey];
   492→  }
   493→
   494→  // Se tem requisicao em curso, espera ela terminar
   495→  if (_pixLocks[lockKey]) {
   496→    console.log('[API/PIX] AGUARDANDO lock existente para', lockKey);
   497→    try {
   498→      var cachedResult = await _pixLocks[lockKey];
   499→      console.log('[API/PIX] Lock liberado, retornando resultado da requisicao original');
   500→      return res.json(cachedResult);
   501→    } catch (lockErr) {
   502→      console.log('[API/PIX] Requisicao original falhou, tentando novamente...');
   503→    }
   504→  }
   505→
   506→  // Criar lock
   507→  var lockResolve;
   508→  _pixLocks[lockKey] = new Promise(function(resolve) { lockResolve = resolve; });
   509→
   510→  // Auto-limpar lock apos TTL
   511→  setTimeout(function() {
   512→    if (_pixLocks[lockKey]) {
   513→      console.warn('[API/PIX] Lock expirou por TTL para', lockKey);
   514→      delete _pixLocks[lockKey];
   515→    }
   516→  }, PIX_LOCK_TTL);
   517→
   518→  try {
   519→    var isCpf = pagCpf.replace(/\\D/g, '').length <= 11;
   520→    var pixResult = await criarCobrancaPix({
   521→      valor: valorTotal,
   522→      devedor: pagNome ? {
   523→        nome: pagNome,
   524→        cpf: isCpf ? pagCpf.replace(/\\D/g, '') : null,
   525→        cnpj: !isCpf ? pagCpf.replace(/\\D/g, '') : null,
   526→      } : undefined,
   527→      solicitacaoPagador: faturaName || '',
   528→      expiracao: 86400,
   529→    });
   530→
   531→    console.log('[API/PIX] PIX criado: TXID=' + (pixResult.txid || 'N/A'));
   532→
   533→    var pixCopiaCola = pixResult.pixCopiaECola || '';
   534→    var qrcodeBase64 = '';
   535→    var htmlPix = '';
   536→
   537→    // Gerar QR code a partir do pixCopiaECola (bwip-js v4 e async)
   538→    if (pixCopiaCola) {
   539→      try {
   540→        var qrPng = await bwipjs.toBuffer({
   541→          bcid: 'qrcode',
   542→          text: pixCopiaCola,
   543→          scale: 5,
   544→          width: 12,
   545→          height: 12,
   546→        });
   547→        qrcodeBase64 = 'data:image/png;base64,' + qrPng.toString('base64');
   548→        console.log('[API/PIX] QR code gerado:', (qrPng.length / 1024).toFixed(1) + 'KB');
   549→      } catch (qrErr) {
   550→        console.error('[API/PIX] Erro ao gerar QR code:', qrErr.message);
   551→      }
   552→    }
   553→
   554→    // Montar HTML para o campo Odoo
   555→    var valorFmt = 'R$ ' + valorTotal.toFixed(2).replace('.', ',');
   556→    htmlPix = '<div style="text-align:center; font-family:Arial,sans-serif; padding:10px;">' +
   557→      '<div style="font-size:18px; font-weight:bold; color:#333; margin-bottom:8px;">PIX</div>' +
   558→      '<div style="font-size:14px; color:#666; margin-bottom:12px;">' + faturaName + ' - ' + valorFmt + '</div>' +
   559→      (qrcodeBase64 ? '<img src="' + qrcodeBase64 + '" style="width:200px; height:200px; margin:0 auto 12px auto; display:block;" />' : '') +
   560→      (pixCopiaCola ? '<div style="font-size:10px; color:#999; word-break:break-all; max-width:400px; margin:0 auto; line-height:1.4;">' + pixCopiaCola + '</div>' : '') +
   561→      '</div>';
   562→
   563→    // Montar responseData ANTES do push (responde pro Odoo imediatamente)
   564→    var responseData = {
   565→      success: true,
   566→      data: {
   567→        forma_pagamento: 'PIX',
   568→        total_parcelas: 1,
   569→        valor_total: valorTotal.toFixed(2),
   570→        fatura_name: faturaName || '(nao informado)',
   571→        fatura_id: faturaId || 0,
   572→        pagamentos: [{
   573→          tipo: 'pix',
   574→          parcela: 1,
   575→          total_parcelas: 1,
   576→          valor_titulo: valorTotal.toFixed(2),
   577→          txid: pixResult.txid || '',
   578→          pix_copia_cola: pixCopiaCola,
   579→          qrcode_base64: qrcodeBase64,
   580→          html_pix: htmlPix,
   581→        }],
   582→        odoo_push: 'pending',
   583→      },
   584→    };
   585→
   586→    // Cache resultado e liberar lock
   587→    _pixCache[lockKey] = { timestamp: Date.now(), response: responseData };
   588→    delete _pixLocks[lockKey];
   589→    lockResolve(responseData);
   590→
   591→    // Responde pro Odoo IMEDIATAMENTE (o modulo Odoo grava os campos)
   592→    res.json(responseData);
   593→  } catch (err) {
   594→    console.error('[API/PIX] ERRO:', err.message);
   595→    var errResponse = { success: false, message: 'Erro ao criar PIX: ' + (err.message || 'Erro desconhecido') };
   596→    delete _pixLocks[lockKey];
   597→    lockResolve(errResponse);
   598→    res.json(errResponse);
   599→  }
   600→}

   644→// Le resposta_json do Odoo, cancela boletos no Itau, posta resultado no chatter e redireciona de volta
   645→router.get('/cancelar/web', async function(req, res) {
   646→  var odooId = req.query.odoo_id || '';
   647→  var oc = config.odoo;
   648→  var returnUrl = (oc && oc.url ? oc.url.replace(/\/+$/, '') : '') + '/web#id=' + odooId + '&model=account.move&view_type=form';
   649→
   650→  function odooRpc(model, method, args, kwargs) {
   651→    var xmlrpc = require('xmlrpc');
   652→    var base = oc.url.replace(/\/+$/, '');
   653→    var host = base.replace('https://', '');
   654→    var commonCli = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/common', port: 443 });
   655→    var modelsCli = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
   656→    return new Promise(function(ok, fail) {
   657→      commonCli.methodCall('authenticate', [oc.db, oc.user, oc.password, {}], function(e, uid) {
   658→        if (e || !uid) return fail(e || new Error('auth failed'));
   659→        var params = [oc.db, uid, oc.password, model, method, args || []];
   660→        if (kwargs) params.push(kwargs);
   661→        modelsCli.methodCall('execute_kw', params, function(e2, r) {
   662→          if (e2) fail(e2); else ok(r);
   663→        });
   664→      });
   665→    });
   666→  }
   667→
   668→  function postChatter(recordId, bodyHtml) {
   669→    return odooRpc('mail.message', 'create', [{
   670→      model: 'account.move',
   671→      res_id: parseInt(recordId),
   672→      body: bodyHtml,
   673→      message_type: 'comment',
   674→    }]);
   675→  }
   676→
   677→  if (!odooId || !oc || !oc.enabled || !oc.url || !oc.db || !oc.user || !oc.password) {
   678→    return res.redirect(returnUrl);
   679→  }
   680→
   681→  try {
   682→    var nossoNumeros = [];
   683→    console.log('[CANCELAR-WEB] Lendo resposta_json da fatura ID:', odooId);
   684→    var fields = await odooRpc('account.move', 'read', [[parseInt(odooId)], ['x_studio_itau_resposta_json']]);
   685→    var rawJson = '';
   686→    if (fields && fields[0] && fields[0].x_studio_itau_resposta_json) {
   687→      rawJson = fields[0].x_studio_itau_resposta_json;
   688→    }
   689→    if (!rawJson) {
   690→      await postChatter(odooId, '<b>Cancelar Boleto</b><br/><span style="color:red">Fatura nao possui dados de boleto.</span>');
   691→      return res.redirect(returnUrl);
   692→    }
   693→
   694→    var jsonStr = rawJson.replace(/'/g, '"').replace(/True/g, 'true').replace(/False/g, 'false').replace(/None/g, 'null');
   695→    var data = JSON.parse(jsonStr);
   696→    var pagamentos = data.pagamentos || [];
   697→    if (pagamentos.length === 0) {
   698→      await postChatter(odooId, '<b>Cancelar Boleto</b><br/><span style="color:red">Nenhum pagamento encontrado.</span>');
   699→      return res.redirect(returnUrl);
   700→    }
   701→    for (var i = 0; i < pagamentos.length; i++) {
   702→      if (pagamentos[i].nosso_numero) nossoNumeros.push(pagamentos[i].nosso_numero);
   703→    }
   704→    if (nossoNumeros.length === 0) {
   705→      await postChatter(odooId, '<b>Cancelar Boleto</b><br/><span style="color:red">Nenhum nosso numero encontrado.</span>');
   706→      return res.redirect(returnUrl);
   707→    }
   708→
   709→    console.log('[CANCELAR-WEB] Cancelando', nossoNumeros.length, 'boleto(s):', nossoNumeros.join(', '));
   710→    var resultados = [];
   711→    for (var i = 0; i < nossoNumeros.length; i++) {
   712→      try {
   713→        await cancelarBoleto(nossoNumeros[i]);
   714→        resultados.push({ nn: nossoNumeros[i], ok: true });
   715→      } catch (err) {
   716→        resultados.push({ nn: nossoNumeros[i], ok: false, erro: err.message });
   717→      }
   718→    }
   719→
   720→    var sucesso = resultados.filter(function(r) { return r.ok; }).length;
   721→    var falha = resultados.filter(function(r) { return !r.ok; }).length;
   722→    var chatBody = '<b>Cancelar Boleto</b><br/>';
   723→    if (falha === 0) {
   724→      chatBody += '<span style="color:green">' + sucesso + ' boleto(s) cancelado(s) com sucesso no Itau.</span>';
   725→    } else if (sucesso === 0) {
   726→      chatBody += '<span style="color:red">Falha ao cancelar todos os boletos:</span><br/>';
   727→      for (var i = 0; i < resultados.length; i++) chatBody += 'NN ' + resultados[i].nn + ': ' + resultados[i].erro + '<br/>';
   728→    } else {
   729→      chatBody += '<span style="color:orange">' + sucesso + ' cancelado(s), ' + falha + ' falha(s):</span><br/>';
   730→      for (var i = 0; i < resultados.length; i++) {
   731→        if (!resultados[i].ok) chatBody += 'NN ' + resultados[i].nn + ': ' + resultados[i].erro + '<br/>';
   732→      }
   733→    }
   734→    chatBody += '<br/><small>NN: ' + nossoNumeros.join(', ') + '</small>';
   735→
   736→    await postChatter(odooId, chatBody);
   737→
   738→    try {
   739→      await odooRpc('account.move', 'write', [[parseInt(odooId)], { 'x_studio_boleto_cancelado': true }]);
   740→    } catch (wErr) {
   741→      console.warn('[CANCELAR-WEB] Aviso:', wErr.message);
   742→    }
   743→
   744→    res.redirect(returnUrl);
   745→  } catch (err) {

   747→    try { await postChatter(odooId, '<b>Cancelar Boleto</b><br/><span style="color:red">Erro: ' + err.message + '</span>'); } catch(e) {}
