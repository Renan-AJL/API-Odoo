    16→ *     x_studio_sieg_serie_nfe   (Char, default 100)
    17→ *     x_studio_sieg_serie_nfse  (Char, default 1)

    19→ *     x_studio_c_trib_nac       (Char) — Codigo tributo nacional (NFS-e)

    34→function createClient(url) {
    35→  var base = url.replace(/\/+$/, '');
    36→  var host = base.replace('https://', '').replace('http://', '');
    37→  var port = base.startsWith('https') ? 443 : 80;
    38→  var isSecure = base.startsWith('https');
    39→  var createFn = isSecure ? xmlrpc.createSecureClient : xmlrpc.createClient;
    40→  return {
    41→    common: createFn({ host: host, path: '/xmlrpc/2/common', port: port }),
    42→    models: createFn({ host: host, path: '/xmlrpc/2/object', port: port }),
    43→  };
    44→}
    45→
    46→function authenticate(client, db, login, password) {
    47→  return new Promise(function(resolve, reject) {
    48→    client.common.methodCall('authenticate', [db, login, password, {}], function(err, uid) {
    49→      if (err) reject(new Error('Auth Odoo falhou: ' + (err.message || JSON.stringify(err))));
    50→      else if (uid === false || uid === null) reject(new Error('Credenciais Odoo invalidas'));
    51→      else resolve(uid);
    52→    });
    53→  });
    54→}
    55→
    56→function executeKw(client, db, uid, password, model, method, args, kwargs) {
    57→  return new Promise(function(resolve, reject) {
    58→    var params = [db, uid, password, model, method, args || []];
    59→    if (kwargs) params.push(kwargs);
    60→    client.models.methodCall('execute_kw', params, function(err, result) {
    61→      if (err) reject(new Error(model + '.' + method + ': ' + (err.message || JSON.stringify(err))));
    62→      else resolve(result);
    63→    });
    64→  });
    65→}

    67→// ============================================================
    68→// Main: Process Pending Emissions
    69→// ============================================================
    70→async function processPendingEmissions() {
    71→  var odoo = config.odoo;
    72→  if (!odoo || !odoo.enabled || !odoo.url) {
    73→    console.log('[SIEG-EMIT] Odoo nao configurado');
    74→    return { processed: 0, reason: 'odoo_not_configured' };
    75→  }
    76→
    77→  var client = createClient(odoo.url);
    78→  var uid = await authenticate(client, odoo.db, odoo.user, odoo.password);
    79→  var db = odoo.db;
    80→  var pwd = odoo.password;
    81→
    82→  console.log('[SIEG-EMIT] Buscando faturas pendentes...');
    83→
    84→  // Buscar NF-e pendentes
    85→  var nfeIds = await executeKw(client, db, uid, pwd, 'account.move', 'search', [[
    86→    ['move_type', '=', 'out_invoice'],
    87→    ['x_studio_nfe_status', '=', 'pendente'],
    88→  ]], { order: 'id asc', limit: 5 });
    89→
    90→  // Buscar NFS-e pendentes
    91→  var nfseIds = await executeKw(client, db, uid, pwd, 'account.move', 'search', [[
    92→    ['move_type', '=', 'out_invoice'],
    93→    ['x_studio_nfse_status', '=', 'pendente'],
    94→  ]], { order: 'id asc', limit: 5 });
    95→
    96→  var allPending = [];
    97→  for (var i = 0; i < nfeIds.length; i++) {
    98→    allPending.push({ id: nfeIds[i], tipo: 'nfe' });
    99→  }
   100→  for (var j = 0; j < nfseIds.length; j++) {
   101→    allPending.push({ id: nfseIds[j], tipo: 'nfse' });
   102→  }
   103→
   104→  if (!allPending.length) {
   105→    console.log('[SIEG-EMIT] Nenhuma fatura pendente');
   106→    return { processed: 0 };
   107→  }
   108→
   109→  console.log('[SIEG-EMIT] ' + allPending.length + ' fatura(s) pendente(s)');
   110→  var results = [];
   111→
   112→  for (var k = 0; k < allPending.length; k++) {
   113→    var item = allPending[k];
   114→    try {
   115→      var r = await processOne(client, db, uid, pwd, item.id, item.tipo);
   116→      results.push(r);
   117→    } catch (err) {
   118→      console.error('[SIEG-EMIT] ERRO fatura ' + item.id + ':', err.message);
   119→      await safeUpdateError(client, db, uid, pwd, item.id, item.tipo, err.message);
   120→      results.push({ move_id: item.id, sucesso: false, tipo: item.tipo, erro: err.message });
   121→    }
   122→  }
   123→
   124→  var ok = results.filter(function(r) { return r.sucesso; }).length;
   125→  console.log('[SIEG-EMIT] Concluido: ' + ok + '/' + results.length + ' sucesso');
   126→  return { processed: results.length, sucesso: ok, detalhes: results };
   127→}

   132→async function processOne(client, db, uid, pwd, moveId, tipo) {
   133→  console.log('[SIEG-EMIT] --- Fatura ID: ' + moveId + ' | Tipo: ' + tipo + ' ---');
   134→
   135→  // 1. Read account.move (sem invoice_line_ids — pode ser computed nao-stored no Odoo 19)
   136→  var moves = await executeKw(client, db, uid, pwd, 'account.move', 'read', [[moveId], [
   137→    'name', 'partner_id', 'company_id', 'invoice_date', 'date',
   138→    'amount_total', 'narration',
   139→    'x_studio_nfe_status', 'x_studio_nfse_status', 'payment_state',
   140→  ]]);
   141→  if (!moves || !moves.length) throw new Error('Fatura ' + moveId + ' nao encontrada');
   142→  var move = moves[0];
   143→  console.log('[SIEG-EMIT] Fatura: ' + move.name + ' | R$ ' + (move.amount_total || 0));
   144→
   145→  // 2. Read company
   146→  var companyId = tupId(move.company_id);
   147→  var company = await readCompany(client, db, uid, pwd, companyId);
   148→  console.log('[SIEG-EMIT] [EMITENTE] CNPJ=' + company.cnpj_cpf + ' IE=' + company.inscr_est + ' cMun=' + company.city_ibge_code + ' xMun=' + company.city + ' UF=' + company.state + ' xLgr=' + company.street + ' nro=' + company.number);
   149→
   150→  // 3. Read partner
   151→  var partnerId = tupId(move.partner_id);
   152→  if (!partnerId) throw new Error('Fatura sem parceiro');
   153→  var partner = await readPartner(client, db, uid, pwd, partnerId);
   154→  console.log('[SIEG-EMIT] [DESTINATARIO] CNPJ=' + partner.cnpj_cpf + ' xNome=' + partner.xNome + ' cMun=' + partner.city_ibge_code + ' xMun=' + partner.city + ' UF=' + partner.state + ' xLgr=' + partner.street + ' nro=' + partner.number);
   155→
   156→  // 4. Read invoice lines diretamente de account.move.line (evita computed field)
   157→  var allLineIds = await executeKw(client, db, uid, pwd, 'account.move.line', 'search', [[
   158→    ['move_id', '=', moveId],
   159→  ]], { order: 'id asc' });
   160→  console.log('[SIEG-EMIT] Linhas encontradas na fatura: ' + allLineIds.length);
   161→  var rawLines = await executeKw(client, db, uid, pwd, 'account.move.line', 'read', [allLineIds, [
   162→    'display_type', 'product_id', 'name', 'quantity', 'price_unit',
   163→    'price_subtotal', 'tax_ids', 'discount',
   164→  ]]);
   165→  // Debug: log todas as linhas
   166→  for (var li = 0; li < rawLines.length; li++) {
   167→    var l = rawLines[li];
   168→    var pName = 'N/A';
   169→    if (l.product_id) pName = Array.isArray(l.product_id) ? l.product_id[1] : String(l.product_id);
   170→    console.log('[SIEG-EMIT]   Line ' + li + ': display_type=' + JSON.stringify(l.display_type) + ' product=' + pName + ' qty=' + l.quantity + ' price=' + l.price_unit);
   171→  }
   172→  // Filtro principal: linhas com produto associado (mais confiavel que display_type)
   173→  var invoiceLines = rawLines.filter(function(l) { return l.product_id; });
   174→  // Fallback: se nenhuma tem product_id, tenta por display_type
   175→  if (!invoiceLines.length) {
   176→    invoiceLines = rawLines.filter(function(l) {
   177→      var dt = l.display_type;
   178→      return !dt || dt === 'product' || dt === 'service';
   179→    });
   180→  }
   181→  if (!invoiceLines.length) throw new Error('Fatura sem linhas de produto/servico (total: ' + rawLines.length + ', com product: 0)');
   182→  console.log('[SIEG-EMIT] Linhas de produto: ' + invoiceLines.length);
   183→
   184→  // 5. Get next NF number from company
   185→  var serie, numField, nextNum;
   186→  if (tipo === 'nfe') {
   187→    serie = String(company.x_studio_sieg_serie_nfe || '100');
   188→    numField = 'x_studio_sieg_ultimo_nfe';
   189→  } else {
   190→    serie = String(company.x_studio_sieg_serie_nfse || '1');
   191→    numField = 'x_studio_sieg_ultimo_nfse';
   192→  }
   193→  nextNum = (parseInt(company[numField]) || 0) + 1;
   194→  console.log('[SIEG-EMIT] Serie: ' + serie + ' | nNF: ' + nextNum);
   195→
   196→  // 6. Build line data with taxes + product x_studio_ fields
   197→  var linesData = [];
   198→  for (var i = 0; i < invoiceLines.length; i++) {
   199→    var ld = await buildLineData(client, db, uid, pwd, invoiceLines[i]);
   200→    linesData.push(ld);
   201→  }
   202→
   203→  // 7. Build emission payload
   204→  // Nota: usar amount_untaxed (sem impostos) para NF-e com CSOSN 103
   205→  // pois vNF deve ser igual a soma dos componentes (vProd + vICMS + ...)
   206→  // e para CSOSN 103 o ICMS e 0, entao vNF = vProd
   207→  var vProdSum = 0;
   208→  for (var li = 0; li < linesData.length; li++) {
   209→    vProdSum += parseFloat(linesData[li].price_subtotal) || 0;
   210→  }
   211→
   212→  // Pagamentos: NF-e 4.00 exige <pag> — usar valor total da fatura
   213→  var pagamentos = [{
   214→    tPag: '15', // PIX
   215→    vPag: String(vProdSum.toFixed(2)),
   216→  }];
   217→
   218→  var emitData = {
   219→    company: company,
   220→    partner: partner,
   221→    order: {
   222→      name: move.name,
   223→      number: String(nextNum),
   224→      date_order: move.invoice_date || move.date,
   225→      amount_total: vProdSum, // usar soma dos vProd (consistente com XML)
   226→      note: move.narration || '',
   227→    },
   228→    lines: linesData,
   229→    pagamentos: pagamentos,
   230→    config: {
   231→      serie: serie,
   232→      tpAmb: process.env.SIEG_TP_AMB || config.sieg.tpAmb || '2',
   233→      natOp: 'Venda de Mercadoria',
   234→      mod: tipo === 'nfe' ? '55' : '01',
   235→    },
   236→    tipo: tipo,
   237→  };
   238→
   239→  // NFS-e: build service block from product x_studio_ fields
   240→  if (tipo === 'nfse') {
   241→    emitData.service = buildServiceBlock(linesData, move);
   242→  }
   243→
   244→  // 8. Set status to 'processando'
   245→  var processVals = {};
   246→  if (tipo === 'nfe') processVals.x_studio_nfe_status = 'processando';
   247→  else processVals.x_studio_nfse_status = 'processando';
   248→  await executeKw(client, db, uid, pwd, 'account.move', 'write', [[moveId], processVals]);
   249→
   250→  // 9. Call SIEG API
   251→  console.log('[SIEG-EMIT] Enviando ao SIEG...');
   252→  var resultado;
   253→  try {
   254→    resultado = await emitirNota(emitData);
   255→  } catch (siegErr) {
   256→    // Captura erros HTTP (ex: 409 com detalhes no body)
   257→    var siegDetail = siegErr.response && siegErr.response.data;
   258→    var errMsg = 'SIEG HTTP ' + (siegErr.response ? siegErr.response.status : 'erro') + ': ';
   259→    if (siegDetail) {
   260→      // Tentar extrair ErrorMessage do formato SIEG (PascalCase)
   261→      var siegMsg = siegDetail.ErrorMessage || siegDetail.Message || siegDetail.message || '';
   262→      if (siegMsg) {
   263→        errMsg += siegMsg;
   264→      } else {
   265→        errMsg += typeof siegDetail === 'string' ? siegDetail : JSON.stringify(siegDetail).slice(0, 800);
   266→      }
   267→    } else {
   268→      errMsg += siegErr.message;
   269→    }
   270→    console.error('[SIEG-EMIT] Erro SIEG:', errMsg);
   271→    throw new Error(errMsg);
   272→  }
   273→  console.log('[SIEG-EMIT] SIEG retornou - sucesso: ' + resultado.sucesso + (resultado.httpStatus ? ' (HTTP ' + resultado.httpStatus + ')' : ''));
   274→
   275→  // Se SIEG retornou erro (IsSuccess=false), extrair mensagem detalhada
   276→  if (!resultado.sucesso && resultado.erro) {
   277→    console.error('[SIEG-EMIT] SIEG rejeitou o XML: ' + resultado.erro);
   278→  }
   279→
   280→  // 10. Parse result
   281→  var info = parseResult(resultado, tipo);
   282→  var xmlRetornoText = '';
   283→  var rawXmlAuth = resultado.resposta && (resultado.resposta.Xml || resultado.resposta.xml || resultado.resposta.xmlBase64);
   284→  if (rawXmlAuth) {
   285→    xmlRetornoText = (typeof rawXmlAuth === 'string' && rawXmlAuth.length > 200)
   286→      ? rawXmlAuth
   287→      : Buffer.from(rawXmlAuth, 'base64').toString('utf-8');
   288→  }
   289→
   290→  // 11. Update account.move with results
   291→  var updateVals = {};
   292→  if (info.sucesso) {
   293→    if (tipo === 'nfe') {
   294→      updateVals.x_studio_nfe_status = 'autorizada';
   295→      updateVals.x_studio_nfe_chave = info.chave;
   296→      updateVals.x_studio_nfe_protocolo = info.protocolo;
   297→    } else {
   298→      updateVals.x_studio_nfse_status = 'autorizada';
   299→      updateVals.x_studio_nfse_numero = String(nextNum);
   300→    }
   301→
   302→    // Increment NF number (only on authorization)
   303→    try {
   304→      var companyUpdate = {};
   305→      companyUpdate[numField] = nextNum;
   306→      await executeKw(client, db, uid, pwd, 'res.company', 'write', [[companyId], companyUpdate]);
   307→      console.log('[SIEG-EMIT] Sequencia ' + numField + ' = ' + nextNum);
   308→    } catch (seqErr) {
   309→      console.error('[SIEG-EMIT] Erro ao atualizar sequencia:', seqErr.message);
   310→    }
   311→  } else {
   312→    if (tipo === 'nfe') {
   313→      updateVals.x_studio_nfe_status = info.cStat + ' - ' + info.motivo;
   314→    } else {
   315→      updateVals.x_studio_nfse_status = info.motivo;
   316→    }
   317→  }
   318→
   319→  await executeKw(client, db, uid, pwd, 'account.move', 'write', [[moveId], updateVals]);
   320→  console.log('[SIEG-EMIT] Fatura atualizada');
   321→
   322→  // 12. Post XML + PDF to Chatter
   323→  await postChatterResult(client, db, uid, pwd, moveId, move.name, tipo, info, resultado, xmlRetornoText);
   324→
   325→  return {
   326→    move_id: moveId, sucesso: info.sucesso, tipo: tipo,
   327→    chave: info.chave, numero: nextNum, serie: serie,
   328→  };
   329→}

   331→// ============================================================
   332→// Post XML + PDF to Odoo Chatter
   333→// ============================================================
   334→async function postChatterResult(client, db, uid, pwd, moveId, moveName, tipo, info, resultado, xmlRetornoText) {
   335→  var attachIds = [];
   336→  var safeName = (moveName || 'Fatura').replace(/[^a-zA-Z0-9\-_]/g, '_');
   337→  var nfLabel = tipo === 'nfe' ? 'NF-e' : 'NFS-e';
   338→  var xmlContent = xmlRetornoText || resultado.xmlEnviado || '';
   339→  var pdfBase64 = resultado.pdfBase64 || '';
   340→
   341→  // Attach XML
   342→  if (xmlContent) {
   343→    try {
   344→      var xmlB64 = Buffer.from(xmlContent, 'utf-8').toString('base64');
   345→      var xmlAttach = await executeKw(client, db, uid, pwd, 'ir.attachment', 'create', [{
   346→        name: nfLabel + '_' + safeName + '.xml',
   347→        datas: xmlB64,
   348→        res_model: 'account.move',
   349→        res_id: moveId,
   350→        mimetype: 'application/xml',
   351→      }]);
   352→      attachIds.push(xmlAttach);
   353→      console.log('[SIEG-EMIT] XML anexado (ID: ' + xmlAttach + ')');
   354→    } catch (e) { console.error('[SIEG-EMIT] Erro anexar XML:', e.message); }
   355→  }
   356→
   357→  // Attach PDF (DANFE / DANFSE)
   358→  if (pdfBase64) {
   359→    try {
   360→      var pdfLabel = tipo === 'nfe' ? 'DANFE' : 'DANFSE';
   361→      var pdfAttach = await executeKw(client, db, uid, pwd, 'ir.attachment', 'create', [{
   362→        name: pdfLabel + '_' + safeName + '.pdf',
   363→        datas: pdfBase64,
   364→        res_model: 'account.move',
   365→        res_id: moveId,
   366→        mimetype: 'application/pdf',
   367→      }]);
   368→      attachIds.push(pdfAttach);
   369→      console.log('[SIEG-EMIT] PDF anexado (ID: ' + pdfAttach + ')');
   370→    } catch (e) { console.error('[SIEG-EMIT] Erro anexar PDF:', e.message); }
   371→  }
   372→
   373→  // Post chatter message
   374→  var body = '';
   375→  if (info.sucesso) {
   376→    body = '<b>' + nfLabel + ' Emitida com Sucesso!</b><br/>';
   377→    if (tipo === 'nfe') {
   378→      body += 'Chave: ' + info.chave + '<br/>';
   379→      body += 'Protocolo: ' + info.protocolo + '<br/>';
   380→    } else {
   381→      body += 'Numero: ' + info.numero + '<br/>';
   382→    }
   383→    body += 'Status: ' + info.cStat + ' - ' + info.motivo;
   384→    body += '<br/><br/><i>Arquivos anexados: XML e ' + (tipo === 'nfe' ? 'DANFE' : 'DANFSE') + ' (PDF)</i>';
   385→  } else {
   386→    body = '<b>Erro na Emissao de ' + nfLabel + '</b><br/>';
   387→    body += 'Status: ' + info.cStat + ' - ' + info.motivo;
   388→  }
   389→
   390→  try {
   391→    var msgVals = {
   392→      model: 'account.move',
   393→      res_id: moveId,
   394→      body: body,
   395→      message_type: 'comment',
   396→    };
   397→    if (attachIds.length > 0) {
   398→      msgVals.attachment_ids = [[6, 0, attachIds]];
   399→    }
   400→    await executeKw(client, db, uid, pwd, 'mail.message', 'create', [msgVals]);
   401→    console.log('[SIEG-EMIT] Mensagem postada no chatter');
   402→  } catch (e) { console.error('[SIEG-EMIT] Erro postar chatter:', e.message); }
   403→}

   408→async function readCompany(client, db, uid, pwd, companyId) {
   409→  var recs = await executeKw(client, db, uid, pwd, 'res.company', 'read', [[companyId], [
   410→    'name', 'partner_id', 'street', 'city', 'state_id', 'zip', 'phone', 'email', 'vat',
   411→    'x_studio_sieg_ultimo_nfe', 'x_studio_sieg_ultimo_nfse',
   412→    'x_studio_sieg_serie_nfe', 'x_studio_sieg_serie_nfse',
   413→  ]]);
   414→  if (!recs || !recs.length) throw new Error('Empresa ' + companyId + ' nao encontrada');
   415→  var c = recs[0];
   416→
   417→  var pId = tupId(c.partner_id);
   418→  var p = {};
   419→  if (pId) {
   420→    try { p = await readPartnerSafe(client, db, uid, pwd, pId); } catch (e) {
   421→      console.warn('[SIEG-EMIT] Erro ao ler partner da empresa:', e.message);
   422→    }
   423→  }
   424→
   425→  var stateCode = '', stateIbge = '';
   426→  var stateId = tupId(c.state_id || p.state_id);
   427→  if (stateId) {
   428→    try {
   429→      var sts = await executeKw(client, db, uid, pwd, 'res.country.state', 'read', [[stateId], ['code', 'ibge_code']]);
   430→      if (sts && sts[0]) { stateCode = sts[0].code || ''; stateIbge = sts[0].ibge_code || ''; }
   431→    } catch (e) {}
   432→  }
   433→
   434→  var cityIbge = '';
   435→  var cityRef = p.l10n_br_city_id || p.city_id;
   436→  if (cityRef && Array.isArray(cityRef)) cityIbge = await readCityIbge(client, db, uid, pwd, cityRef[0]);
   437→

   542→// ============================================================
   543→// NCM Extraction Helper
   544→// ============================================================
   545→/**
   546→ * Extrai codigo NCM (8 digitos) de um valor many2one do Odoo.
   547→ * Aceita: [id, "7308.90.90"], [id, "7308.90.90 - Descricao"], string pura, numero
   548→ */
   549→function extractNcmFromRef(ref) {
   550→  if (!ref) return '';
   551→  var str = '';
   552→  if (Array.isArray(ref)) {
   553→    str = String(ref[1] || ref[0] || '');
   554→  } else if (typeof ref === 'string') {
   555→    str = ref;
   556→  } else if (typeof ref === 'number') {
   557→    return String(ref).length === 8 ? String(ref) : '';
   558→  } else {
   559→    str = String(ref);
   560→  }
   561→  // Tentar regex NNNN.NN.NN
   562→  var m = str.match(/(\d{4})\.(\d{2})\.(\d{2})/);
   563→  if (m) return m[1] + m[2] + m[3];
   564→  // Fallback: so digitos
   565→  var digits = str.replace(/\D/g, '');
   566→  return digits.length === 8 ? digits : '';
   567→}

   800→async function extractTaxes(client, db, uid, pwd, line) {
   801→  var result = {
   802→    csosn: '103', cst_icms: '', vbc: 0, vicms: 0, picms: 0,
   803→    cst_pis: '01', vbc_pis: 0, ppis: 0, vpis: 0,

   805→  };
   806→
   807→  var taxIds = (line.tax_ids || []).map(function(t) { return Array.isArray(t) ? t[0] : t; }).filter(Boolean);
   808→  if (!taxIds.length) return result;
   809→
   810→  try {
   811→    var taxes = await executeKw(client, db, uid, pwd, 'account.tax', 'read', [taxIds, [
   812→      'name', 'amount', 'amount_type', 'description',
   813→    ]]);
   814→    if (!taxes || !taxes.length) return result;
   815→
   816→    var base = parseFloat(line.price_subtotal) || 0;
   817→    var totalIcms = 0, totalPis = 0, totalCofins = 0;
   818→
   819→    for (var i = 0; i < taxes.length; i++) {
   820→      var tax = taxes[i];
   821→      var n = ((tax.name || tax.description || '').toUpperCase());
   822→      var amt = parseFloat(tax.amount) || 0;
   823→      if (n.indexOf('ICMS') >= 0 || n.indexOf('CSOSN') >= 0) {
   824→        totalIcms += amt;
   825→        var m = n.match(/(10[0-9]|20[0-9]|300|400|500|900)/);
   826→        if (m) result.csosn = m[1];
   827→      } else if (n.indexOf('PIS') >= 0) { totalPis += amt; }
   828→      else if (n.indexOf('COFINS') >= 0) { totalCofins += amt; }
   829→    }
   830→
   831→    result.picms = totalIcms; result.vbc = base; result.vicms = base * (totalIcms / 100);
   832→    result.ppis = totalPis; result.vbc_pis = base; result.vpis = base * (totalPis / 100);
   833→    result.pcofins = totalCofins; result.vbc_cofins = base; result.vcofins = base * (totalCofins / 100);
   834→  } catch (e) { console.warn('[SIEG-EMIT] Erro ao extrair impostos:', e.message); }
   835→
   836→  return result;
   837→}

   842→function parseResult(resultado, tipo) {
   843→  var resp = resultado.resposta || {};
   844→  var chave = '', protocolo = '', cStat = '', motivo = '', numero = '', sucesso = false;
   845→
   846→  // Se SIEG retornou HTTP 4xx, usar ErrorMessage diretamente como motivo
   847→  var siegErro = resultado.erro || '';
   848→  if (resultado.httpStatus && resultado.httpStatus >= 400 && resultado.httpStatus < 500) {
   849→    // Erro de validacao do XML — extrair mensagem do SIEG
   850→    motivo = siegErro || 'Erro HTTP ' + resultado.httpStatus;
   851→    if (tipo === 'nfe') {
   852→      cStat = String(resultado.statusCode || resp.cStat || resp.status || '');
   853→    }
   854→    return { sucesso: false, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
   855→  }
   856→
   857→  if (tipo === 'nfe') {
   858→    chave = resp.chNFe || resp.chave || resp.ChaveXml || '';
   859→    protocolo = resp.nProt || resp.protocolo || '';
   860→    cStat = String(resp.cStat || resp.status || '');
   861→    motivo = resp.xMotivo || resp.motivo || (siegErro || 'Processado');
   862→    sucesso = (cStat === '100' || cStat === '104' || cStat === '150');
   863→  } else {
   864→    chave = resp.Chave || resp.chave || '';
   865→    protocolo = resp.Protocolo || resp.protocolo || '';
   866→    numero = resp.NumeroNfse || resp.nDFSe || resp.numero || '';
   867→    cStat = String(resp.CodigoVerificacao || resp.codigo || '');
   868→    motivo = resp.Motivo || resp.motivo || siegErro || (resultado.sucesso ? 'Autorizada' : 'Erro na emissao');
   869→    sucesso = resultado.sucesso;
   870→  }
   871→
   872→  return { sucesso: sucesso, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
   873→}
