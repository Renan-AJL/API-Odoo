    43→function formatCnpj(v) {
    44→  var s = String(v || '').replace(/\D/g, '');
    45→  if (s.length === 14) return s.replace(/^(\d{2})(\d{3})(\d{3})(\d{4})(\d{2})/, '$1.$2.$3/$4-$5');
    46→  if (s.length === 11) return s.replace(/^(\d{3})(\d{3})(\d{3})(\d{2})/, '$1.$2.$3-$4');
    47→  return v || '';
    48→}

    50→function fmtData(d) {
    51→  if (!d) return '';
    52→  var p = d.split('-');
    53→  return p.length === 3 ? p[2] + '/' + p[1] + '/' + p[0] : d;
    54→}

    56→function fmtValor(v) {
    57→  var s = String(v || '0').padStart(15, '0');
    58→  var c = parseInt(s.slice(-2), 10);
    59→  var r = parseInt(s.slice(0, -2), 10);
    60→  var val = (r + c / 100).toFixed(2);
    61→  var parts = val.split('.');
    62→  return parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, '.') + ',' + parts[1];
    63→}

    65→function fmtValorDecimal(v) {
    66→  // Formata valor decimal (ex: 4882.06 -> "4.882,06")
    67→  var n = parseFloat(v) || 0;
    68→  var val = n.toFixed(2);
    69→  var parts = val.split('.');
    70→  return parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, '.') + ',' + parts[1];
    71→}

    73→// Formata nosso numero com carteira e DV: 109/00051773-0
    74→function calcDvNN(nn) {
    75→  var s = String(nn).padStart(8, '0');
    76→  var weights = [2, 3, 4, 5, 6, 7, 8, 9];
    77→  var sum = 0;
    78→  for (var i = 0; i < 8 && i < s.length; i++) {
    79→    sum += parseInt(s[i]) * weights[i];
    80→  }
    81→  var dv = 11 - (sum % 11);
    82→  if (dv === 10 || dv === 11) dv = 0;
    83→  return dv;
    84→}

    86→function formatNN(nn, carteira) {
    87→  var c = carteira || '109';
    88→  var s = String(nn).padStart(8, '0');
    89→  var dv = calcDvNN(s);
    90→  return c + '/' + s + '-' + dv;
    91→}

    93→// Formata agencia/codigo: 7764/22338-9
    94→function formatAgCodigo(ag, idBen) {
    95→  var a = ag || '7764';
    96→  var id = idBen || '776400223389';
    97→  // Remove prefixo da agencia (primeiros 4 digitos)
    98→  var code = id.substring(4);
    99→  // Remove zeros a esquerda
   100→  code = code.replace(/^0+/, '') || '0';
   101→  // Insere hifen antes do ULTIMO digito: 223389 -> 22338-9
   102→  if (code.length >= 2) {
   103→    code = code.substring(0, code.length - 1) + '-' + code.substring(code.length - 1);
   104→  }
   105→  return a + '/' + code;
   106→}

   108→// Formata linha digitavel com pontos e espacos (estilo Itau)
   109→// Input: 47-digit string ou ja formatada
   110→function fmtLinhaDigitavel(ld) {
   111→  if (!ld) return '';
   112→  var s = String(ld).replace(/\s/g, '').replace(/\./g, '');
   113→  if (s.length >= 47) {
   114→    return s.substring(0, 5) + '.' + s.substring(5, 10) + ' ' +
   115→           s.substring(10, 15) + '.' + s.substring(15, 21) + ' ' +
   116→           s.substring(21, 26) + '.' + s.substring(26, 32) + ' ' +
   117→           s.substring(32, 33) + ' ' + s.substring(33);
   118→  }
   119→  return ld;
   120→}

   122→// Gera instrucoes padrao com juros e multa
   123→function getInstrucoes(valorTitulo) {
   124→  var vt = String(valorTitulo || '0').padStart(15, '0');
   125→  var cents = parseInt(vt.slice(-2), 10);
   126→  var reais = parseInt(vt.slice(0, -2), 10);
   127→  var val = reais + cents / 100;
   128→  // Juros: 0.03% ao dia sobre o valor
   129→  var juros = (val * 0.0003).toFixed(2).replace('.', ',');
   130→  // Multa: 2% do valor
   131→  var multa = (val * 0.02).toFixed(2).replace('.', ',');
   132→  return [
   133→    'APOS O VENCIMENTO COBRAR JUROS DE..........R$ ' + juros + ' AO DIA',
   134→    'APOS O VENCIMENTO COBRAR MULTA DE..........R$ ' + multa,
   135→    'Sujeito a inclusao no Serasa no 8 dias apos o vencimento.'
   136→  ];
   137→}

   154→
   155→function genBarcode(text) {
   156→  return new Promise(function (resolve) {
   157→    if (!bwipjs) { resolve(null); return; }
   158→    bwipjs.toBuffer({ bcid: 'code128', text: text, scale: 3, height: 20, includetext: false }, function (err, buf) {
   159→      resolve(err ? null : buf);
   160→    });
   161→  });
   162→}
   163→
   164→function genQRCode(text) {
   165→  return new Promise(function (resolve) {
   166→    if (!bwipjs) { resolve(null); return; }
   167→    bwipjs.toBuffer({ bcid: 'qrcode', text: text, scale: 5, includetext: false }, function (err, buf) {
   168→      resolve(err ? null : buf);
   169→    });
   170→  });
   171→}

   200→function drawVal(doc, x, y, w, text, opts) {

   209→// Desenha faixa de header (3 caixas: logo + banco + linha digitavel)
   210→function drawHeaderStrip(doc, y, title, linhaDigitavel, barcodeBuf) {
   211→  // Titulo da secao (canto superior direito)
   212→  if (title) {
   213→    doc.fillColor('#333').fontSize(8).font('Helvetica-Bold');
   214→    doc.text(title, LM, y - 16, { width: PW, align: 'right', lineBreak: false });
   215→  }
   216→
   217→  // Dimensoes das 3 caixas
   218→  var bw1 = 117;  // Logo
   219→  var bw2 = 52;   // Banco 341-7
   220→  var bw3 = PW - bw1 - bw2 - 2 * CELL_GAP;  // Linha digitavel ou barcode
   221→
   222→  // Caixa 1: Logo Itau
   223→  drawCell(doc, LM, y, bw1, BOX_H);
   224→  if (LOGO_LOADED && logoBuf) {
   225→    doc.image(logoBuf, LM + 3, y + (BOX_H - 20) / 2, { height: 20 });
   226→  } else {
   227→    // Fallback: desenhar logo vetorial
   228→    drawVectorLogo(doc, LM + 4, y + 2);
   229→  }
   230→
   231→  // Caixa 2: Codigo do banco 341-7
   232→  var x2 = LM + bw1 + CELL_GAP;
   233→  drawCell(doc, x2, y, bw2, BOX_H);
   234→  doc.fillColor('#000').fontSize(18).font('Helvetica-Bold');
   235→  doc.text('341-7', x2, y + 4, { width: bw2, align: 'center', lineBreak: false });
   236→
   237→  // Caixa 3: Linha digitavel OU codigo de barras
   238→  var x3 = x2 + bw2 + CELL_GAP;
   239→  drawCell(doc, x3, y, bw3, BOX_H);
   240→  if (barcodeBuf) {
   241→    // Barcode image
   242→    doc.image(barcodeBuf, x3 + 4, y + 4, { width: bw3 - 8, height: BOX_H - 8 });
   243→  } else {
   244→    // Linha digitavel formatada
   245→    var ld = fmtLinhaDigitavel(linhaDigitavel);
   246→    doc.fillColor('#000').fontSize(12).font('Helvetica-Bold');
   247→    doc.text(ld, x3 + 4, y + 8, { width: bw3 - 8, align: 'center', lineBreak: false });
   248→  }
   249→
   250→  return y + BOX_H;
   251→}

   631→async function drawBoleto(doc, dados) {
   632→  var barcodeText = dados.codigo_barras || '';
   633→  var barcodeBuf = await genBarcode(barcodeText);
   634→
   635→  var qrBuf = dados.qrcode_base64 ? Buffer.from(dados.qrcode_base64, 'base64') : null;
   636→  if (!qrBuf && (dados.pix_copia_cola || '')) {
   637→    qrBuf = await genQRCode(dados.pix_copia_cola);
   638→  }
   639→
   640→  // ===== RECIBO DO PAGADOR =====
   641→  var y = drawRecibo(doc, dados, barcodeBuf);
   642→
   643→  // ===== FICHA DE COMPENSAÇÃO =====
   644→  y = drawFicha(doc, y, dados, barcodeBuf, qrBuf);
   645→
   646→  return y;
   647→}

   651→async function buildPdfBuffer(dados) {
   652→  return new Promise(function (resolve, reject) {
   653→    try {
   654→      var doc = new PDFDocument({
   655→        size: 'A4',
   656→        margins: { top: 5, bottom: 5, left: LM, right: 27 },
   657→        info: { Title: 'Boleto ' + (dados.nosso_numero || ''), Author: 'AJL Ferro e Aco' }
   658→      });
   659→      var chunks = [];
   660→      doc.on('data', function (chunk) { chunks.push(chunk); });
   661→
   662→      drawBoleto(doc, dados).then(function (finalY) {
   663→        doc.fillColor('#999').fontSize(4);
   664→        doc.text('Emitido por AJL Ferro e Aco | Boleto gerado via API BoleCode Itau', LM, finalY + 4, { width: PW, align: 'center' });
   665→        doc.end();
   666→        doc.on('end', function () { resolve(Buffer.concat(chunks)); });
   667→      }).catch(reject);
   668→    } catch (err) { reject(err); }
   669→  });
   670→}
