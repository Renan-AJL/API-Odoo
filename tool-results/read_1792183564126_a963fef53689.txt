   104→// ============================================================
   105→// XML-RPC connection (singleton)
   106→// ============================================================
   107→var _cached = null;
   108→
   109→function getConn() {
   110→  if (_cached && _cached.uid) return _cached;
   111→  var odooConfig = config.odoo;
   112→  var base = odooConfig.url.replace(/\/+$/, '');
   113→  var host = base.replace('https://', '').replace('http://', '');
   114→  var common = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/common', port: 443 });
   115→  var models = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
   116→  _cached = { uid: null, common: common, models: models, db: odooConfig.db, password: odooConfig.password };
   117→  return _cached;
   118→}
   119→
   120→function authenticate() {
   121→  return new Promise(function(resolve, reject) {
   122→    var conn = getConn();
   123→    conn.common.methodCall('authenticate', [conn.db, config.odoo.user, conn.password, {}], function(err, uid) {
   124→      if (err) { reject(new Error('Auth Odoo falhou: ' + (err.message || JSON.stringify(err)))); }
   125→      else if (uid === false || uid === null) { reject(new Error('Auth Odoo: credenciais invalidas (UID=false)')); }
   126→      else { conn.uid = uid; logger.info('[ODOO-TE] Autenticado. UID: ' + uid); resolve(uid); }
   127→    });
   128→  });
   129→}
   130→
   131→function ensureAuth() {
   132→  var conn = getConn();
   133→  if (conn.uid) return Promise.resolve(conn.uid);
   134→  return authenticate();
   135→}
   136→
   137→function executeKw(model, method, args, kwargs) {
   138→  return new Promise(function(resolve, reject) {
   139→    ensureAuth().then(function(uid) {
   140→      var conn = getConn();
   141→      var params = [conn.db, uid, conn.password, model, method, args || []];
   142→      if (kwargs) params.push(kwargs);
   143→      // Debug log para metodos read/search (nao logar write/create com dados sensiveis)
   144→      if (method === 'read' || method === 'search' || method === 'search_read') {
   145→        logger.info('[ODOO-TE] executeKw: ' + model + '.' + method + ' args=' + JSON.stringify(args).substring(0, 200));
   146→      }
   147→      conn.models.methodCall('execute_kw', params, function(err8, result) {
   148→        if (err8) {
   149→          logger.error('[ODOO-TE] executeKw ERRO: ' + model + '.' + method + ': ' + (err8.message || JSON.stringify(err8)).substring(0, 300));
   150→          reject(new Error(model + '.' + method + ': ' + (err8.message || JSON.stringify(err8))));
   151→        }
   152→        else { resolve(result); }
   153→      });
   154→    }).catch(reject);
   155→  });
   156→}

   158→// ============================================================
   159→// flattenIds — normaliza IDs para Odoo XML-RPC read()
   160→// Odoo exige args = [[id1, id2, ...], [field1, field2, ...]]
   161→// IDs podem chegar como: 111, [111], [[111]], [[111,"name"]], [111,222,...]
   162→// Retorna sempre flat array de inteiros: [111, 222, ...]
   163→// ============================================================
   164→function flattenIds(ids) {
   165→  if (!ids) return [];
   166→  // Single integer
   167→  if (typeof ids === 'number') return [ids];
   168→  // Not an array — try parseInt
   169→  if (!Array.isArray(ids)) {
   170→    var n = parseInt(ids, 10);
   171→    return isNaN(n) ? [] : [n];
   172→  }
   173→  // Flat array — check first element
   174→  if (ids.length === 0) return [];
   175→  var first = ids[0];
   176→  // [111, 222, ...] — flat int array
   177→  if (typeof first === 'number') return ids;
   178→  // ["111", ...] — flat string array
   179→  if (typeof first === 'string') return ids.map(function(x) { return parseInt(x, 10); }).filter(function(n) { return !isNaN(n); });
   180→  // [[111, "name"], ...] or [[111], ...] — nested (Many2one pair or nested array)
   181→  if (Array.isArray(first)) {
   182→    return ids.map(function(item) {
   183→      if (typeof item === 'number') return item;
   184→      if (Array.isArray(item) && item.length > 0) {
   185→        var inner = typeof item[0] === 'number' ? item[0] : parseInt(item[0], 10);
   186→        return isNaN(inner) ? null : inner;
   187→      }
   188→      var n = parseInt(item, 10);
   189→      return isNaN(n) ? null : n;
   190→    }).filter(function(x) { return x !== null && x !== undefined; });
   191→  }
   192→  return ids;
   193→}

   195→// ============================================================
   196→// Safe read/write — tenta com x_studio, fallback sem
   197→// ============================================================
   198→var _customFieldsCache = {};
   199→
   200→async function safeReadCustom(model, ids, nativeFields, customFields) {
   201→  var flatIds = flattenIds(ids);
   202→  if (!flatIds.length) return [];
   203→  try {
   204→    return await executeKw(model, 'read', [flatIds, nativeFields.concat(customFields)]);
   205→  } catch (err) {
   206→    if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   207→      if (!_customFieldsCache[model]) {
   208→        _customFieldsCache[model] = { available: false, checked: true };
   209→        logger.warn('[ODOO-TE] Campos x_studio_* nao encontrados em ' + model + '. Usando campos nativos.');
   210→      }
   211→      return await executeKw(model, 'read', [flatIds, nativeFields]);
   212→    }
   213→    throw err;
   214→  }
   215→}
   216→
   217→async function safeWriteCustom(model, ids, vals) {
   218→  var customVals = {};
   219→  var nativeVals = {};
   220→  Object.keys(vals).forEach(function(k) {
   221→    if (k.indexOf('x_studio_') === 0) customVals[k] = vals[k];
   222→    else nativeVals[k] = vals[k];
   223→  });
   224→  if (Object.keys(nativeVals).length > 0) await executeKw(model, 'write', [ids, nativeVals]);
   225→  if (Object.keys(customVals).length > 0) {
   226→    try { await executeKw(model, 'write', [ids, customVals]); }
   227→    catch (err) {
   228→      if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   229→        if (!_customFieldsCache[model] || !_customFieldsCache[model].warned) {
   230→          logger.warn('[ODOO-TE] Campo x_studio_* nao existe em ' + model + '. Gravacao ignorada.');
   231→          if (!_customFieldsCache[model]) _customFieldsCache[model] = {};
   232→          _customFieldsCache[model].warned = true;
   233→        }
   234→      } else throw err;
   235→    }
   236→  }
   237→}

   242→async function read(model, ids, fields) {
   243→  var flatIds = flattenIds(ids);
   244→  if (!flatIds.length) return [];
   245→  return await executeKw(model, 'read', [flatIds, fields]);
   246→}

   248→async function searchRead(model, domain, fields, limit, offset, order) {
   249→  // Usa search_read nativo do Odoo — um unico XML-RPC call em vez de search+read
   250→  var kwargs = { fields: fields, limit: limit || 80, offset: offset || 0 };
   251→  if (order) kwargs.order = order;
   252→  return await executeKw(model, 'search_read', [domain], kwargs);
   253→}

   348→async function findSaleOrderByInvoice(invoiceId) {
   349→  console.log('[ODOO-TE] findSaleOrderByInvoice: invoiceId=' + invoiceId);
   350→
   351→  // Estrategia 1: sale.order.invoice_ids
   352→  try {
   353→    var saleIds = await executeKw('sale.order', 'search', [['invoice_ids', 'in', [invoiceId]]]);
   354→    if (saleIds && saleIds.length) {
   355→      var orders = await executeKw('sale.order', 'read', [saleIds, ['id', 'name']]);
   356→      if (orders && orders[0]) { console.log('[ODOO-TE] Estrategia 1 (invoice_ids) encontrou: ' + orders[0].name); return orders[0]; }
   357→    }
   358→    console.log('[ODOO-TE] Estrategia 1 (invoice_ids): sem resultados');
   359→  } catch (err) { console.warn('[ODOO-TE] Estrategia 1 falhou: ' + err.message); }
   360→
   361→  // Estrategia 2: sale_order_ids da fatura (campo computado)
   362→  try {
   363→    var invData = await executeKw('account.move', 'read', [[invoiceId], ['id', 'partner_id', 'sale_order_ids']]);
   364→    var inv = invData && invData[0];
   365→    console.log('[ODOO-TE] Fatura lida: id=' + (inv ? inv.id : 'null') + ', partner=' + JSON.stringify(inv ? inv.partner_id : null) + ', sale_order_ids=' + JSON.stringify(inv ? inv.sale_order_ids : null));
   366→    if (inv && inv.sale_order_ids && inv.sale_order_ids.length) {
   367→      var soId = Array.isArray(inv.sale_order_ids[0]) ? inv.sale_order_ids[0][0] : inv.sale_order_ids[0];
   368→      var orders = await executeKw('sale.order', 'read', [[soId], ['id', 'name']]);
   369→      if (orders && orders[0]) { console.log('[ODOO-TE] Estrategia 2 (sale_order_ids) encontrou: ' + orders[0].name); return orders[0]; }
   370→    }
   371→    console.log('[ODOO-TE] Estrategia 2 (sale_order_ids): sem resultados');
   372→  } catch (err) { console.warn('[ODOO-TE] Estrategia 2 falhou: ' + err.message); }
   373→
   374→  // Estrategia 3: account.move.line -> sale_line_ids -> order_id
   375→  try {
   376→    var invData = await executeKw('account.move', 'read', [[invoiceId], ['line_ids']]);
   377→    var inv = invData && invData[0];
   378→    var rawLineIds = (inv && inv.line_ids) || [];
   379→    var lineIds = flattenIds(rawLineIds);
   380→    console.log('[ODOO-TE] Estrategia 3: ' + lineIds.length + ' linhas da fatura (raw=' + rawLineIds.length + ')');
   381→    if (lineIds.length) {
   382→      var lines = await executeKw('account.move.line', 'read', [lineIds, ['sale_line_ids']]);
   383→      if (lines) {
   384→        for (var i = 0; i < lines.length; i++) {
   385→          var sli = lines[i].sale_line_ids;
   386→          if (sli && sli.length) {
   387→            var saleLineId = Array.isArray(sli[0]) ? sli[0][0] : sli[0];
   388→            var saleLines = await executeKw('sale.order.line', 'read', [[saleLineId], ['order_id']]);
   389→            if (saleLines && saleLines[0] && saleLines[0].order_id) {
   390→              var orderId = Array.isArray(saleLines[0].order_id) ? saleLines[0].order_id[0] : saleLines[0].order_id;
   391→              var orders = await executeKw('sale.order', 'read', [[orderId], ['id', 'name']]);
   392→              if (orders && orders[0]) { console.log('[ODOO-TE] Estrategia 3 (lines) encontrou: ' + orders[0].name); return orders[0]; }
   393→            }
   394→          }
   395→        }
   396→      }
   397→    }
   398→    console.log('[ODOO-TE] Estrategia 3 (lines): sem resultados');
   399→  } catch (err) { console.warn('[ODOO-TE] Estrategia 3 falhou: ' + err.message); }
   400→
   401→  // NOTA: Estrategia 4 (partner_id fallback) removida — achava vendas nao
   402→  // relacionadas ao mesmo cliente. Para send-invoice sem venda vinculada,
   403→  // o fluxo correto e usar as linhas da fatura diretamente.
   404→
   405→  console.error('[ODOO-TE] Nenhuma estrategia encontrou sale.order para invoice ' + invoiceId);
   406→  return null;

   450→async function getInvoiceLines(invoiceId) {
   451→  try {
   452→    // Evita account.move.line.search — Odoo 19.2 SaaS tem override de
   453→    // search_fetch com _get_recon_limit_from_domain que rejeita certos dominios.
   454→    // Em vez disso, le line_ids direto da fatura.
   455→    var invData = await executeKw('account.move', 'read', [[invoiceId], ['line_ids']]);
   456→    var inv = invData && invData[0];
   457→    if (!inv || !inv.line_ids || !inv.line_ids.length) return [];
   458→
   459→    // line_ids pode vir como [id, id, ...] ou [[id, name], ...] ou [[id], ...]
   460→    // flattenIds normaliza para flat array de inteiros
   461→    var lineIds = flattenIds(inv.line_ids);
   462→
   463→    var lines = await executeKw('account.move.line', 'read', [lineIds, INVOICE_LINE_FIELDS]);
   464→    if (!lines || !lines.length) return [];
   465→
   466→    // Filtra apenas linhas com produto (product_id != false)
   467→    return lines.filter(function(line) {
   468→      if (!line.product_id) return false;
   469→      // product_id pode ser [id, name] ou false
   470→      if (Array.isArray(line.product_id) && line.product_id[0]) return true;
   471→      if (typeof line.product_id === 'number' && line.product_id) return true;
   472→      return false;
   473→    });
   474→  } catch (err) {
   475→    logger.warn('[ODOO-TE] getInvoiceLines falhou: ' + err.message);
   476→    return [];
   477→  }
   478→}
