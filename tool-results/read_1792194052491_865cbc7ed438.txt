    40→  if (!formaPag) return null;
    41→  var n = formaPag.trim().toUpperCase();
    42→  for (var i = 0; i < BANDEIRAS.length; i++) {
    43→    var b = BANDEIRAS[i];
    44→    if (n.indexOf(b) === 0 || n.indexOf(b + ' ') === 0) {
    45→      var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
    46→      var parcelas = 1;
    47→      var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
    48→      if (match) parcelas = parseInt(match[1]);
    49→      return { bandeira: b, parcelas: parcelas, debito: debito };
    50→    }
    51→  }
    52→  // Generico: CARTAO, CARTAO CREDITO, CREDITO, etc.
    53→  if (n.indexOf('CART') === 0 || n.indexOf('CREDITO') === 0) {
    54→    var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;

   334→
   335→    var uid = await new Promise(function(ok, fail) {

   393→    var isPix = formaPag.trim().toUpperCase().indexOf('PIX') === 0;
   394→    if (!isPix && fatName) {
   395→      var boletoLockKey = fatName;

   402→        res.setHeader('Content-Type', 'text/plain; charset=utf-8');
   403→        return res.send(dedupResp);
   404→      }

   406→      // Se tem resultado cacheado (mesma fatura, menos de 120s atras), reusa
   407→      if (_boletoCache[boletoLockKey]) {

   410→        if (age < BOLETO_CACHE_TTL) {

   412→          res.setHeader('Content-Type', 'text/plain; charset=utf-8');
   413→          return res.send(cached.response);
   414→        }
   415→        delete _boletoCache[boletoLockKey];
   416→      }
   417→      // Se tem requisicao em curso, espera ela terminar
   418→      if (_boletoLocks[boletoLockKey]) {
   419→        console.log('[API/GERAR] AGUARDANDO lock boleto existente para', boletoLockKey);
   420→        try {
   421→          var cachedResult = await _boletoLocks[boletoLockKey];
   422→          console.log('[API/GERAR] Lock boleto liberado, retornando resultado da requisicao original');
   423→          res.setHeader('Content-Type', 'text/plain; charset=utf-8');

   814→    return new Promise(function(ok, fail) {
   815→      commonCli.methodCall('authenticate', [oc.db, oc.user, oc.password, {}], function(e, uid) {
   816→        if (e || !uid) return fail(e || new Error('auth failed'));
   817→        var params = [oc.db, uid, oc.password, model, method, args || []];
   818→        if (kwargs) params.push(kwargs);
   819→        modelsCli.methodCall('execute_kw', params, function(e2, r) {
