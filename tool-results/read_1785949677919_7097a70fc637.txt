   350→   549→   549→   549→// NCM Extraction Helper
   351→   550→   550→   550→// ============================================================
   352→   551→   551→   551→/**
   353→   552→   552→   552→ * Extrai codigo NCM (8 digitos) de um valor many2one do Odoo.
   354→   553→   553→   553→ * Aceita: [id, "7308.90.90"], [id, "7308.90.90 - Descricao"], string pura, numero
   355→   554→   554→   554→ */
   356→   555→   555→   555→function extractNcmFromRef(ref) {
   357→   556→   556→   556→  if (!ref) return '';
   358→   557→   557→   557→  var str = '';
   359→   558→   558→   558→  if (Array.isArray(ref)) {
   360→   559→   559→   559→    str = String(ref[1] || ref[0] || '');
   361→   560→   560→   560→  } else if (typeof ref === 'string') {
   362→   561→   561→   561→    str = ref;
   363→   562→   562→   562→  } else if (typeof ref === 'number') {
   364→   563→   563→   563→    return String(ref).length === 8 ? String(ref) : '';
   365→   564→   564→   564→  } else {
   366→   565→   565→   565→    str = String(ref);
   367→   566→   566→   566→  }
   368→   567→   567→   567→  // Tentar regex NNNN.NN.NN
   369→   568→   568→   568→  var m = str.match(/(\d{4})\.(\d{2})\.(\d{2})/);
   370→   569→   569→   569→  if (m) return m[1] + m[2] + m[3];
   371→   570→   570→   570→  // Fallback: so digitos
   372→   571→   571→   571→  var digits = str.replace(/\D/g, '');
   373→   572→   572→   572→  return digits.length === 8 ? digits : '';
   374→   573→   573→   573→}
