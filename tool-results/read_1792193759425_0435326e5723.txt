   105→// ============================================================
   106→// XML-RPC connection (singleton)
   107→// ============================================================
   108→var _cached = null;
   109→
   110→function getConn() {
   111→  if (_cached && _cached.uid) return _cached;
   112→  var odooConfig = config.odoo;
   113→  var base = odooConfig.url.replace(/\/+$/, '');
   114→  var host = base.replace('https://', '').replace('http://', '');
   115→  var common = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/common', port: 443 });
   116→  var models = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
   117→  _cached = { uid: null, common: common, models: models, db: odooConfig.db, password: odooConfig.password };
   118→  return _cached;
   119→}
   120→
   121→function authenticate() {
   122→  return new Promise(function(resolve, reject) {
   123→    var conn = getConn();
   124→    conn.common.methodCall('authenticate', [conn.db, config.odoo.user, conn.password, {}], function(err, uid) {
   125→      if (err) { reject(new Error('Auth Odoo falhou: ' + (err.message || JSON.stringify(err)))); }
   126→      else if (uid === false || uid === null) { reject(new Error('Auth Odoo: credenciais invalidas (UID=false)')); }
   127→      else { conn.uid = uid; logger.info('[ODOO-TE] Autenticado. UID: ' + uid); resolve(uid); }
   128→    });
   129→  });
   130→}
   131→
   132→function ensureAuth() {
   133→  var conn = getConn();
   134→  if (conn.uid) return Promise.resolve(conn.uid);
   135→  return authenticate();
   136→}
   137→
   138→function executeKw(model, method, args, kwargs) {
   139→  return new Promise(function(resolve, reject) {
   140→    ensureAuth().then(function(uid) {
   141→      var conn = getConn();
   142→      var params = [conn.db, uid, conn.password, model, method, args || []];
   143→      if (kwargs) params.push(kwargs);
   144→      conn.models.methodCall('execute_kw', params, function(err, result) {
   145→        if (err) { reject(new Error(model + '.' + method + ': ' + (err.message || JSON.stringify(err)))); }
   146→        else { resolve(result); }
   147→      });
   148→    }).catch(reject);
   149→  });
   150→}

   152→// ============================================================
   153→// Safe read/write — tenta com x_studio, fallback sem
   154→// ============================================================
   155→var _customFieldsCache = {};
   156→
   157→async function safeReadCustom(model, ids, nativeFields, customFields) {
   158→  try {
   159→    return await executeKw(model, 'read', [ids], { fields: nativeFields.concat(customFields) });
   160→  } catch (err) {
   161→    if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   162→      if (!_customFieldsCache[model]) {
   163→        _customFieldsCache[model] = { available: false, checked: true };
   164→        logger.warn('[ODOO-TE] Campos x_studio_* nao encontrados em ' + model + '. Usando campos nativos.');
   165→      }
   166→      return await executeKw(model, 'read', [ids], { fields: nativeFields });
   167→    }
   168→    throw err;
   169→  }
   170→}
   171→
   172→async function safeWriteCustom(model, ids, vals) {
   173→  var customVals = {};
   174→  var nativeVals = {};
   175→  Object.keys(vals).forEach(function(k) {
   176→    if (k.indexOf('x_studio_') === 0) customVals[k] = vals[k];
   177→    else nativeVals[k] = vals[k];
   178→  });
   179→  if (Object.keys(nativeVals).length > 0) await executeKw(model, 'write', [ids, nativeVals]);
   180→  if (Object.keys(customVals).length > 0) {
   181→    try { await executeKw(model, 'write', [ids, customVals]); }
   182→    catch (err) {
   183→      if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   184→        if (!_customFieldsCache[model] || !_customFieldsCache[model].warned) {
   185→          logger.warn('[ODOO-TE] Campo x_studio_* nao existe em ' + model + '. Gravacao ignorada.');
   186→          if (!_customFieldsCache[model]) _customFieldsCache[model] = {};
   187→          _customFieldsCache[model].warned = true;
   188→        }
   189→      } else throw err;
   190→    }
   191→  }
   192→}

   201→async function searchRead(model, domain, fields, limit, offset, order) {
   202→  var kwargs = { fields: fields, limit: limit || 0, offset: offset || 0 };
   203→  if (order) kwargs.order = order;
   204→  var ids = await executeKw(model, 'search', [domain], kwargs);
   205→  if (!ids || !ids.length) return [];
   206→  return await executeKw(model, 'read', [ids], { fields: fields });
   207→}

   302→async function findSaleOrderByInvoice(invoiceId) {
   303→  console.log('[ODOO-TE] findSaleOrderByInvoice: invoiceId=' + invoiceId);
   304→
   305→  // Estrategia 1: sale.order.invoice_ids
   306→  try {
   307→    var saleIds = await executeKw('sale.order', 'search', [['invoice_ids', 'in', [invoiceId]]]);
   308→    if (saleIds && saleIds.length) {
   309→      var orders = await executeKw('sale.order', 'read', [saleIds], { fields: ['id', 'name'] });
   310→      if (orders && orders[0]) { console.log('[ODOO-TE] Estrategia 1 (invoice_ids) encontrou: ' + orders[0].name); return orders[0]; }
   311→    }
   312→    console.log('[ODOO-TE] Estrategia 1 (invoice_ids): sem resultados');
   313→  } catch (err) { console.warn('[ODOO-TE] Estrategia 1 falhou: ' + err.message); }
   314→
   315→  // Estrategia 2: sale_order_ids da fatura (campo computado)
   316→  try {
   317→    var invData = await executeKw('account.move', 'read', [[invoiceId]], { fields: ['id', 'partner_id', 'sale_order_ids'] });
   318→    var inv = invData && invData[0];
   319→    console.log('[ODOO-TE] Fatura lida: id=' + (inv ? inv.id : 'null') + ', partner=' + JSON.stringify(inv ? inv.partner_id : null) + ', sale_order_ids=' + JSON.stringify(inv ? inv.sale_order_ids : null));
   320→    if (inv && inv.sale_order_ids && inv.sale_order_ids.length) {
   321→      var soId = Array.isArray(inv.sale_order_ids[0]) ? inv.sale_order_ids[0][0] : inv.sale_order_ids[0];
   322→      var orders = await executeKw('sale.order', 'read', [[soId]], { fields: ['id', 'name'] });
   323→      if (orders && orders[0]) { console.log('[ODOO-TE] Estrategia 2 (sale_order_ids) encontrou: ' + orders[0].name); return orders[0]; }
   324→    }
   325→    console.log('[ODOO-TE] Estrategia 2 (sale_order_ids): sem resultados');
   326→  } catch (err) { console.warn('[ODOO-TE] Estrategia 2 falhou: ' + err.message); }
   327→
   328→  // Estrategia 3: account.move.line -> sale_line_ids -> order_id
   329→  try {
   330→    var invData = await executeKw('account.move', 'read', [[invoiceId]], { fields: ['line_ids'] });
   331→    var inv = invData && invData[0];
   332→    var lineIds = (inv && inv.line_ids) || [];
   333→    console.log('[ODOO-TE] Estrategia 3: ' + lineIds.length + ' linhas da fatura');

   335→      var lines = await executeKw('account.move.line', 'read', [lineIds], { fields: ['sale_line_ids'] });
   336→      if (lines) {
   337→        for (var i = 0; i < lines.length; i++) {
   338→          var sli = lines[i].sale_line_ids;
   339→          if (sli && sli.length) {
   340→            var saleLineId = Array.isArray(sli[0]) ? sli[0][0] : sli[0];
   341→            var saleLines = await executeKw('sale.order.line', 'read', [[saleLineId]], { fields: ['order_id'] });
   342→            if (saleLines && saleLines[0] && saleLines[0].order_id) {
   343→              var orderId = Array.isArray(saleLines[0].order_id) ? saleLines[0].order_id[0] : saleLines[0].order_id;
   344→              var orders = await executeKw('sale.order', 'read', [[orderId]], { fields: ['id', 'name'] });
   345→              if (orders && orders[0]) { console.log('[ODOO-TE] Estrategia 3 (lines) encontrou: ' + orders[0].name); return orders[0]; }
   346→            }
   347→          }
   348→        }
   349→      }
   350→    }
   351→    console.log('[ODOO-TE] Estrategia 3 (lines): sem resultados');
   352→  } catch (err) { console.warn('[ODOO-TE] Estrategia 3 falhou: ' + err.message); }
   353→
   354→  // NOTA: Estrategia 4 (partner_id fallback) removida — achava vendas nao
   355→  // relacionadas ao mesmo cliente. Para send-invoice sem venda vinculada,
   356→  // o fluxo correto e usar as linhas da fatura diretamente.
   357→
   358→  console.error('[ODOO-TE] Nenhuma estrategia encontrou sale.order para invoice ' + invoiceId);
   359→  return null;
   360→}

   402→    if (!inv || !inv.line_ids || !inv.line_ids.length) return [];
   403→
   404→    // line_ids pode vir como [id, id, ...] ou [[id, name], ...]
   405→    var lineIds = inv.line_ids.map(function(item) {
   406→      return Array.isArray(item) ? item[0] : item;
   407→    });
   408→
   409→    var lines = await executeKw('account.move.line', 'read', [lineIds], { fields: INVOICE_LINE_FIELDS });
   410→    if (!lines || !lines.length) return [];
   411→
   412→    // Filtra apenas linhas com produto (product_id != false)
   413→    return lines.filter(function(line) {
   414→      if (!line.product_id) return false;
   415→      // product_id pode ser [id, name] ou false
   416→      if (Array.isArray(line.product_id) && line.product_id[0]) return true;
   417→      if (typeof line.product_id === 'number' && line.product_id) return true;
   418→      return false;
   419→    });
   420→  } catch (err) {
   421→    logger.warn('[ODOO-TE] getInvoiceLines falhou: ' + err.message);
   422→    return [];
   423→  }
   424→}
