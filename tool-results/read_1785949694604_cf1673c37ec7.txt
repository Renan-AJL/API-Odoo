   548→// ============================================================
   549→// NCM Extraction Helper
   550→// ============================================================
   551→/**
   552→ * Extrai codigo NCM (8 digitos) de um valor many2one do Odoo.
   553→ * Aceita: [id, "7308.90.90"], [id, "7308.90.90 - Descricao"], string pura, numero
   554→ */
   555→function extractNcmFromRef(ref) {
   556→  if (!ref) return '';
   557→  var str = '';
   558→  if (Array.isArray(ref)) {
   559→    str = String(ref[1] || ref[0] || '');
   560→  } else if (typeof ref === 'string') {
   561→    str = ref;
   562→  } else if (typeof ref === 'number') {
   563→    return String(ref).length === 8 ? String(ref) : '';
   564→  } else {
   565→    str = String(ref);
   566→  }
   567→  // Tentar regex NNNN.NN.NN
   568→  var m = str.match(/(\d{4})\.(\d{2})\.(\d{2})/);
   569→  if (m) return m[1] + m[2] + m[3];
   570→  // Fallback: so digitos
   571→  var digits = str.replace(/\D/g, '');
   572→  return digits.length === 8 ? digits : '';
   573→}

   811→async function extractTaxes(client, db, uid, pwd, line) {
   812→  // Padrão: Lucro Real (CRT 3) — CST 00, ICMS 18% PR interno
   813→  // PIS/COFINS não-cumulativo com base no valor do produto
   814→  var base = parseFloat(line.price_subtotal) || 0;
   815→  var result = {
   816→    csosn: '',
   817→    cst_icms: '00',
   818→    mod_bc: '3',
   819→    vbc: base,
   820→    vicms: base * 0.18,
   821→    picms: 18.00,
   822→    cst_pis: '01',
   823→    vbc_pis: base,
   824→    ppis: 1.65,
   825→    vpis: base * 0.0165,
   826→    cst_cofins: '01',
   827→    vbc_cofins: base,
   828→    pcofins: 7.60,
   829→    vcofins: base * 0.076,
   830→  };
   831→
   832→  var taxIds = (line.tax_ids || []).map(function(t) { return Array.isArray(t) ? t[0] : t; }).filter(Boolean);
   833→  if (!taxIds.length) return result;
   834→
   835→  try {
   836→    var taxes = await executeKw(client, db, uid, pwd, 'account.tax', 'read', [taxIds, [
   837→      'name', 'amount', 'amount_type', 'description',
   838→    ]]);
   839→    if (!taxes || !taxes.length) return result;
   840→
   841→    var totalIcms = 0, totalPis = 0, totalCofins = 0;
   842→    var foundCsosn = '';
   843→
   844→    for (var i = 0; i < taxes.length; i++) {
   845→      var tax = taxes[i];
   846→      var n = ((tax.name || tax.description || '').toUpperCase());
   847→      var amt = parseFloat(tax.amount) || 0;
   848→      if (n.indexOf('ICMS') >= 0 || n.indexOf('CSOSN') >= 0) {
   849→        totalIcms += amt;
   850→        var m = n.match(/(10[0-9]|20[0-9]|300|400|500|900)/);
   851→        if (m) foundCsosn = m[1];
   852→      } else if (n.indexOf('PIS') >= 0) { totalPis += amt; }
   853→      else if (n.indexOf('COFINS') >= 0) { totalCofins += amt; }
   854→    }
   855→
   856→    // Sobrescreve defaults apenas se o Odoo tiver impostos reais configurados (> 0)
   857→    if (foundCsosn) {
   858→      result.csosn = foundCsosn;
   859→      result.cst_icms = '';
   860→    }
   861→    if (totalIcms > 0) {
   862→      result.picms = totalIcms;
   863→      result.vicms = base * (totalIcms / 100);
   864→    }
   865→    if (totalPis > 0) {
   866→      result.ppis = totalPis;
   867→      result.vpis = base * (totalPis / 100);
   868→    }
   869→    if (totalCofins > 0) {
   870→      result.pcofins = totalCofins;
   871→      result.vcofins = base * (totalCofins / 100);
   872→    }
   873→  } catch (e) { console.warn('[SIEG-EMIT] Erro ao extrair impostos:', e.message); }
   874→
   875→  return result;
   876→}

   881→function parseResult(resultado, tipo) {
   882→  var resp = resultado.resposta || {};
   883→  var chave = '', protocolo = '', cStat = '', motivo = '', numero = '', sucesso = false;
   884→
   885→  // Se SIEG retornou HTTP 4xx, usar ErrorMessage diretamente como motivo
   886→  var siegErro = resultado.erro || '';
   887→  if (resultado.httpStatus && resultado.httpStatus >= 400 && resultado.httpStatus < 500) {
   888→    // Erro de validacao do XML — extrair mensagem do SIEG
   889→    motivo = siegErro || 'Erro HTTP ' + resultado.httpStatus;
   890→    if (tipo === 'nfe') {
   891→      cStat = String(resultado.statusCode || resp.cStat || resp.status || '');
   892→    }
   893→    return { sucesso: false, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
   894→  }
   895→
   896→  if (tipo === 'nfe') {
   897→    chave = resp.chNFe || resp.chave || resp.ChaveXml || '';
   898→    protocolo = resp.nProt || resp.protocolo || '';
   899→    cStat = String(resp.cStat || resp.status || '');
   900→    motivo = resp.xMotivo || resp.motivo || (siegErro || 'Processado');
   901→    sucesso = (cStat === '100' || cStat === '104' || cStat === '150');
   902→  } else {
   903→    chave = resp.Chave || resp.chave || '';
   904→    protocolo = resp.Protocolo || resp.protocolo || '';
   905→    numero = resp.NumeroNfse || resp.nDFSe || resp.numero || '';
   906→    cStat = String(resp.CodigoVerificacao || resp.codigo || '');
   907→    motivo = resp.Motivo || resp.motivo || siegErro || (resultado.sucesso ? 'Autorizada' : 'Erro na emissao');
   908→    sucesso = resultado.sucesso;
   909→  }
   910→
   911→  return { sucesso: sucesso, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
   912→}
