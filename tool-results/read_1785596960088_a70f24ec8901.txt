   132→async function processOne(client, db, uid, pwd, moveId, tipo) {

   139→    'x_studio_nfe_status', 'x_studio_nfse_status', 'payment_state',

   156→    ['move_id', '=', moveId],
   157→  ]], { order: 'id asc' });

   159→  var rawLines = await executeKw(client, db, uid, pwd, 'account.move.line', 'read', [allLineIds, [
   160→    'display_type', 'product_id', 'name', 'quantity', 'price_unit',
   161→    'price_subtotal', 'tax_ids', 'discount',

   167→    if (l.product_id) pName = Array.isArray(l.product_id) ? l.product_id[1] : String(l.product_id);

   169→  }

   171→  var invoiceLines = rawLines.filter(function(l) { return l.product_id; });
   172→  // Fallback: se nenhuma tem product_id, tenta por display_type

   175→      var dt = l.display_type;
   176→      return !dt || dt === 'product' || dt === 'service';
   177→    });
   178→  }
   179→  if (!invoiceLines.length) throw new Error('Fatura sem linhas de produto/servico (total: ' + rawLines.length + ', com product: 0)');
   180→  console.log('[SIEG-EMIT] Linhas de produto: ' + invoiceLines.length);
   181→

   185→    serie = String(company.x_studio_sieg_serie_nfe || '100');
   186→    numField = 'x_studio_sieg_ultimo_nfe';

   188→    serie = String(company.x_studio_sieg_serie_nfse || '1');
   189→    numField = 'x_studio_sieg_ultimo_nfse';

   193→

   195→  var linesData = [];
   196→  for (var i = 0; i < invoiceLines.length; i++) {
   197→    var ld = await buildLineData(client, db, uid, pwd, invoiceLines[i]);
   198→    linesData.push(ld);
   199→  }
   200→
   201→  // 7. Build emission payload
   202→  // Nota: usar amount_untaxed (sem impostos) para NF-e com CSOSN 103

   212→    tPag: '15', // PIX

   214→  }];
   215→
   216→  var emitData = {
   217→    company: company,
   218→    partner: partner,
   219→    order: {
   220→      name: move.name,
   221→      number: String(nextNum),
   222→      date_order: move.invoice_date || move.date,
   223→      amount_total: vProdSum, // usar soma dos vProd (consistente com XML)
   224→      note: move.narration || '',
   225→    },
   226→    lines: linesData,
   227→    pagamentos: pagamentos,
   228→    config: {

   234→    tipo: tipo,

   241→

   243→  var processVals = {};
   244→  if (tipo === 'nfe') processVals.x_studio_nfe_status = 'processando';
   245→  else processVals.x_studio_nfse_status = 'processando';
   246→  await executeKw(client, db, uid, pwd, 'account.move', 'write', [[moveId], processVals]);

   248→  // 9. Call SIEG API
   249→  console.log('[SIEG-EMIT] Enviando ao SIEG...');
   250→  var resultado;
   251→  try {

   256→    var errMsg = 'SIEG HTTP ' + (siegErr.response ? siegErr.response.status : 'erro') + ': ';
   257→    if (siegDetail) {
   258→      // Tentar extrair ErrorMessage do formato SIEG (PascalCase)
   259→      var siegMsg = siegDetail.ErrorMessage || siegDetail.Message || siegDetail.message || '';
   260→      if (siegMsg) {
   261→        errMsg += siegMsg;
   262→      } else {
   263→        errMsg += typeof siegDetail === 'string' ? siegDetail : JSON.stringify(siegDetail).slice(0, 800);
   264→      }
   265→    } else {
   266→      errMsg += siegErr.message;
   267→    }
   268→    console.error('[SIEG-EMIT] Erro SIEG:', errMsg);
   269→    throw new Error(errMsg);
   270→  }
   271→  console.log('[SIEG-EMIT] SIEG retornou - sucesso: ' + resultado.sucesso + (resultado.httpStatus ? ' (HTTP ' + resultado.httpStatus + ')' : ''));
   272→
   273→  // Se SIEG retornou erro (IsSuccess=false), extrair mensagem detalhada
   274→  if (!resultado.sucesso && resultado.erro) {
   275→    console.error('[SIEG-EMIT] SIEG rejeitou o XML: ' + resultado.erro);
   276→  }

   278→  // 10. Parse result
   279→  var info = parseResult(resultado, tipo);

   284→      ? rawXmlAuth
   285→      : Buffer.from(rawXmlAuth, 'base64').toString('utf-8');
   286→  }
   287→
   288→  // 11. Update account.move with results
   289→  var updateVals = {};
   290→  if (info.sucesso) {
   291→    if (tipo === 'nfe') {
   292→      updateVals.x_studio_nfe_status = 'autorizada';
   293→      updateVals.x_studio_nfe_chave = info.chave;
   294→      updateVals.x_studio_nfe_protocolo = info.protocolo;
   295→    } else {
   296→      updateVals.x_studio_nfse_status = 'autorizada';
   297→      updateVals.x_studio_nfse_numero = String(nextNum);
   298→    }
   299→
   300→    // Increment NF number (only on authorization)
   301→    try {
   302→      var companyUpdate = {};
   303→      companyUpdate[numField] = nextNum;
   304→      await executeKw(client, db, uid, pwd, 'res.company', 'write', [[companyId], companyUpdate]);
   305→      console.log('[SIEG-EMIT] Sequencia ' + numField + ' = ' + nextNum);
   306→    } catch (seqErr) {
   307→      console.error('[SIEG-EMIT] Erro ao atualizar sequencia:', seqErr.message);
   308→    }
   309→  } else {
   310→    if (tipo === 'nfe') {
   311→      updateVals.x_studio_nfe_status = info.cStat + ' - ' + info.motivo;
   312→    } else {
   313→      updateVals.x_studio_nfse_status = info.motivo;
   314→    }
   315→  }
   316→
   317→  await executeKw(client, db, uid, pwd, 'account.move', 'write', [[moveId], updateVals]);
   318→  console.log('[SIEG-EMIT] Fatura atualizada');
   319→
   320→  // 12. Post XML + PDF to Chatter
   321→  await postChatterResult(client, db, uid, pwd, moveId, move.name, tipo, info, resultado, xmlRetornoText);
   322→
   323→  return {
   324→    move_id: moveId, sucesso: info.sucesso, tipo: tipo,
   325→    chave: info.chave, numero: nextNum, serie: serie,
   326→  };
   327→}

   332→async function postChatterResult(client, db, uid, pwd, moveId, moveName, tipo, info, resultado, xmlRetornoText) {
   333→  var attachIds = [];
   334→  var safeName = (moveName || 'Fatura').replace(/[^a-zA-Z0-9\-_]/g, '_');
   335→  var nfLabel = tipo === 'nfe' ? 'NF-e' : 'NFS-e';
   336→  var xmlContent = xmlRetornoText || resultado.xmlEnviado || '';
   337→  var pdfBase64 = resultado.pdfBase64 || '';
   338→
   339→  // Attach XML
   340→  if (xmlContent) {
   341→    try {
   342→      var xmlB64 = Buffer.from(xmlContent, 'utf-8').toString('base64');
   343→      var xmlAttach = await executeKw(client, db, uid, pwd, 'ir.attachment', 'create', [{
   344→        name: nfLabel + '_' + safeName + '.xml',
   345→        datas: xmlB64,
   346→        res_model: 'account.move',
   347→        res_id: moveId,
   348→        mimetype: 'application/xml',
   349→      }]);
   350→      attachIds.push(xmlAttach);
   351→      console.log('[SIEG-EMIT] XML anexado (ID: ' + xmlAttach + ')');
   352→    } catch (e) { console.error('[SIEG-EMIT] Erro anexar XML:', e.message); }
   353→  }
   354→
   355→  // Attach PDF (DANFE / DANFSE)
   356→  if (pdfBase64) {
   357→    try {
   358→      var pdfLabel = tipo === 'nfe' ? 'DANFE' : 'DANFSE';
   359→      var pdfAttach = await executeKw(client, db, uid, pwd, 'ir.attachment', 'create', [{
   360→        name: pdfLabel + '_' + safeName + '.pdf',
   361→        datas: pdfBase64,
   362→        res_model: 'account.move',
   363→        res_id: moveId,
   364→        mimetype: 'application/pdf',
   365→      }]);
   366→      attachIds.push(pdfAttach);
   367→      console.log('[SIEG-EMIT] PDF anexado (ID: ' + pdfAttach + ')');
   368→    } catch (e) { console.error('[SIEG-EMIT] Erro anexar PDF:', e.message); }
   369→  }
   370→
   371→  // Post chatter message
   372→  var body = '';
   373→  if (info.sucesso) {
   374→    body = '<b>' + nfLabel + ' Emitida com Sucesso!</b><br/>';
   375→    if (tipo === 'nfe') {
   376→      body += 'Chave: ' + info.chave + '<br/>';
   377→      body += 'Protocolo: ' + info.protocolo + '<br/>';
   378→    } else {
   379→      body += 'Numero: ' + info.numero + '<br/>';
   380→    }
   381→    body += 'Status: ' + info.cStat + ' - ' + info.motivo;
   382→    body += '<br/><br/><i>Arquivos anexados: XML e ' + (tipo === 'nfe' ? 'DANFE' : 'DANFSE') + ' (PDF)</i>';
   383→  } else {
   384→    body = '<b>Erro na Emissao de ' + nfLabel + '</b><br/>';
   385→    body += 'Status: ' + info.cStat + ' - ' + info.motivo;
   386→  }
   387→
   388→  try {
   389→    var msgVals = {
   390→      model: 'account.move',
   391→      res_id: moveId,
   392→      body: body,
   393→      message_type: 'comment',
   394→    };
   395→    if (attachIds.length > 0) {
   396→      msgVals.attachment_ids = [[6, 0, attachIds]];
   397→    }
   398→    await executeKw(client, db, uid, pwd, 'mail.message', 'create', [msgVals]);
   399→    console.log('[SIEG-EMIT] Mensagem postada no chatter');
   400→  } catch (e) { console.error('[SIEG-EMIT] Erro postar chatter:', e.message); }
   401→}

   475→async function readPartnerSafe(client, db, uid, pwd, partnerId, extraFields) {

   482→  try {
   483→    var brRecs = await executeKw(client, db, uid, pwd, 'res.partner', 'read', [[partnerId], PARTNER_BR_FIELDS]);
   484→    if (brRecs && brRecs[0]) {
   485→      for (var k in brRecs[0]) { if (brRecs[0][k] !== undefined) p[k] = brRecs[0][k]; }
   486→    }
   487→  } catch (e) {
   488→    console.log('[SIEG-EMIT] Campos l10n_br nao disponiveis, usando vat como CNPJ/CPF');
   489→  }

   493→async function readPartner(client, db, uid, pwd, partnerId) {
   494→  var p = await readPartnerSafe(client, db, uid, pwd, partnerId);
   495→
   496→  var stateCode = '';
   497→  var stId = tupId(p.state_id);
   498→  if (stId) {
   499→    try {
   500→      var sts = await executeKw(client, db, uid, pwd, 'res.country.state', 'read', [[stId], ['code']]);
   501→      if (sts && sts[0]) stateCode = sts[0].code || '';
   502→    } catch (e) {}
   503→  }
   504→
   505→  var cityIbge = '';
   506→  var cityRef = p.l10n_br_city_id || p.city_id;
   507→  if (cityRef && Array.isArray(cityRef)) cityIbge = await readCityIbge(client, db, uid, pwd, cityRef[0]);
   508→
   509→  // Fallback 1: buscar res.city por nome + estado
   510→  if (!cityIbge && p.city && stateCode) {
   511→    cityIbge = await searchCityIbge(client, db, uid, pwd, p.city, stId);
   512→  }
   513→  // Fallback 2: tabela embutida de cidades principais
   514→  if (!cityIbge && p.city && stateCode) {
   515→    cityIbge = lookupIbgeFallback(p.city, stateCode);
   516→  }
   517→  if (!cityIbge) {
   518→    console.warn('[SIEG-EMIT] cMun vazio para parceiro ' + partnerId + ' (' + (p.name || '?') + '): cidade=' + (p.city || '?') + ' UF=' + (stateCode || '?'));
   519→  }
   520→
   521→  return {
   522→    cnpj_cpf: p.cnpj_cpf || p.vat || '',
   523→    legal_name: p.legal_name || p.name || '',
   524→    xNome: p.name || '',
   525→    inscr_est: p.inscr_est || '',
   526→    street: p.street || '',
   527→    number: p.number || 'S/N',
   528→    street2: p.street2 || p.district || '',
   529→    city: p.city || '',
   530→    state: stateCode,
   531→    zip: p.zip || '',
   532→    city_ibge_code: cityIbge,
   533→    phone: p.phone || '',
   534→    email: p.email || '',
   535→    is_consumer: !p.is_company,
   536→    district: p.district || p.street2 || '',
   537→  };
   538→}

   543→async function buildLineData(client, db, uid, pwd, line) {
   544→  var productId = tupId(line.product_id);
   545→  var productName = Array.isArray(line.product_id) ? line.product_id[1] : (line.name || 'Item');
   546→  var defaultCode = '', barcode = '', ncm = '', detailedType = 'product', uomName = 'UN';
   547→  var prodStudio = {}; // x_studio_ fields from product
   548→
   549→  if (productId) {
   550→    try {
   551→      // Campos core do produto (sempre existem) — SEM detailed_type (pode nao existir no SaaS)
   552→      var prods = await executeKw(client, db, uid, pwd, 'product.product', 'read', [[productId], [
   553→        'default_code', 'barcode', 'name', 'uom_id',
   554→      ]]);
   555→      if (prods && prods[0]) {
   556→        var pr = prods[0];
   557→        defaultCode = pr.default_code || '';
   558→        barcode = pr.barcode || '';
   559→        productName = pr.name || productName;
   560→        if (pr.uom_id && Array.isArray(pr.uom_id)) uomName = pr.uom_id[1] || 'UN';
   561→      }
   562→
   563→      // Campo detailed_type (pode nao existir no Odoo 19 SaaS)
   564→      try {

   693→async function extractTaxes(client, db, uid, pwd, line) {
   694→  var result = {
   695→    csosn: '103', cst_icms: '', vbc: 0, vicms: 0, picms: 0,
   696→    cst_pis: '99', vbc_pis: 0, ppis: 0, vpis: 0,
   697→    cst_cofins: '99', vbc_cofins: 0, pcofins: 0, vcofins: 0,
   698→  };
   699→
   700→  var taxIds = (line.tax_ids || []).map(function(t) { return Array.isArray(t) ? t[0] : t; }).filter(Boolean);
   701→  if (!taxIds.length) return result;
   702→
   703→  try {
   704→    var taxes = await executeKw(client, db, uid, pwd, 'account.tax', 'read', [taxIds, [
   705→      'name', 'amount', 'amount_type', 'description',
   706→    ]]);
   707→    if (!taxes || !taxes.length) return result;
   708→
   709→    var base = parseFloat(line.price_subtotal) || 0;
   710→    var totalIcms = 0, totalPis = 0, totalCofins = 0;
   711→
   712→    for (var i = 0; i < taxes.length; i++) {
   713→      var tax = taxes[i];
   714→      var n = ((tax.name || tax.description || '').toUpperCase());
   715→      var amt = parseFloat(tax.amount) || 0;
   716→      if (n.indexOf('ICMS') >= 0 || n.indexOf('CSOSN') >= 0) {
   717→        totalIcms += amt;
   718→        var m = n.match(/(10[0-9]|20[0-9]|300|400|500|900)/);
   719→        if (m) result.csosn = m[1];
   720→      } else if (n.indexOf('PIS') >= 0) { totalPis += amt; }
   721→      else if (n.indexOf('COFINS') >= 0) { totalCofins += amt; }
   722→    }
   723→
   724→    result.picms = totalIcms; result.vbc = base; result.vicms = base * (totalIcms / 100);
   725→    result.ppis = totalPis; result.vbc_pis = base; result.vpis = base * (totalPis / 100);
   726→    result.pcofins = totalCofins; result.vbc_cofins = base; result.vcofins = base * (totalCofins / 100);
   727→  } catch (e) { console.warn('[SIEG-EMIT] Erro ao extrair impostos:', e.message); }
   728→
   729→  return result;
   730→}

   735→function parseResult(resultado, tipo) {
   736→  var resp = resultado.resposta || {};
   737→  var chave = '', protocolo = '', cStat = '', motivo = '', numero = '', sucesso = false;
   738→
   739→  // Se SIEG retornou HTTP 4xx, usar ErrorMessage diretamente como motivo
   740→  var siegErro = resultado.erro || '';
   741→  if (resultado.httpStatus && resultado.httpStatus >= 400 && resultado.httpStatus < 500) {
   742→    // Erro de validacao do XML — extrair mensagem do SIEG
   743→    motivo = siegErro || 'Erro HTTP ' + resultado.httpStatus;
   744→    if (tipo === 'nfe') {
   745→      cStat = String(resultado.statusCode || resp.cStat || resp.status || '');
   746→    }
   747→    return { sucesso: false, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
   748→  }
   749→
   750→  if (tipo === 'nfe') {
   751→    chave = resp.chNFe || resp.chave || resp.ChaveXml || '';
   752→    protocolo = resp.nProt || resp.protocolo || '';
   753→    cStat = String(resp.cStat || resp.status || '');
   754→    motivo = resp.xMotivo || resp.motivo || (siegErro || 'Processado');
   755→    sucesso = (cStat === '100' || cStat === '104' || cStat === '150');
   756→  } else {
   757→    chave = resp.Chave || resp.chave || '';
   758→    protocolo = resp.Protocolo || resp.protocolo || '';
   759→    numero = resp.NumeroNfse || resp.nDFSe || resp.numero || '';
   760→    cStat = String(resp.CodigoVerificacao || resp.codigo || '');
   761→    motivo = resp.Motivo || resp.motivo || siegErro || (resultado.sucesso ? 'Autorizada' : 'Erro na emissao');
   762→    sucesso = resultado.sucesso;
   763→  }
   764→
   765→  return { sucesso: sucesso, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
   766→}
