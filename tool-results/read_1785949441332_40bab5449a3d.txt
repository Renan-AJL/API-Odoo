   150→   549→   549→   549→// NCM Extraction Helper
   151→   550→   550→   550→// ============================================================
   152→   551→   551→   551→/**
   153→   552→   552→   552→ * Extrai codigo NCM (8 digitos) de um valor many2one do Odoo.
   154→   553→   553→   553→ * Aceita: [id, "7308.90.90"], [id, "7308.90.90 - Descricao"], string pura, numero
   155→   554→   554→   554→ */
   156→   555→   555→   555→function extractNcmFromRef(ref) {
   157→   556→   556→   556→  if (!ref) return '';
   158→   557→   557→   557→  var str = '';
   159→   558→   558→   558→  if (Array.isArray(ref)) {
   160→   559→   559→   559→    str = String(ref[1] || ref[0] || '');
   161→   560→   560→   560→  } else if (typeof ref === 'string') {
   162→   561→   561→   561→    str = ref;
   163→   562→   562→   562→  } else if (typeof ref === 'number') {
   164→   563→   563→   563→    return String(ref).length === 8 ? String(ref) : '';
   165→   564→   564→   564→  } else {
   166→   565→   565→   565→    str = String(ref);
   167→   566→   566→   566→  }
   168→   567→   567→   567→  // Tentar regex NNNN.NN.NN
   169→   568→   568→   568→  var m = str.match(/(\d{4})\.(\d{2})\.(\d{2})/);
   170→   569→   569→   569→  if (m) return m[1] + m[2] + m[3];
   171→   570→   570→   570→  // Fallback: so digitos
   172→   571→   571→   571→  var digits = str.replace(/\D/g, '');
   173→   572→   572→   572→  return digits.length === 8 ? digits : '';
   174→   573→   573→   573→}

   457→   856→   856→   856→    // Sobrescreve defaults apenas se o Odoo tiver impostos reais configurados (> 0)

   459→   858→   858→   858→      result.csosn = foundCsosn;
   460→   859→   859→   859→      result.cst_icms = '';
