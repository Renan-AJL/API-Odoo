   549→   549→   549→// NCM Extraction Helper
   550→   550→   550→// ============================================================
   551→   551→   551→/**
   552→   552→   552→ * Extrai codigo NCM (8 digitos) de um valor many2one do Odoo.
   553→   553→   553→ * Aceita: [id, "7308.90.90"], [id, "7308.90.90 - Descricao"], string pura, numero
   554→   554→   554→ */
   555→   555→   555→function extractNcmFromRef(ref) {
   556→   556→   556→  if (!ref) return '';
   557→   557→   557→  var str = '';
   558→   558→   558→  if (Array.isArray(ref)) {
   559→   559→   559→    str = String(ref[1] || ref[0] || '');
   560→   560→   560→  } else if (typeof ref === 'string') {
   561→   561→   561→    str = ref;
   562→   562→   562→  } else if (typeof ref === 'number') {
   563→   563→   563→    return String(ref).length === 8 ? String(ref) : '';
   564→   564→   564→  } else {
   565→   565→   565→    str = String(ref);
   566→   566→   566→  }
   567→   567→   567→  // Tentar regex NNNN.NN.NN
   568→   568→   568→  var m = str.match(/(\d{4})\.(\d{2})\.(\d{2})/);
   569→   569→   569→  if (m) return m[1] + m[2] + m[3];
   570→   570→   570→  // Fallback: so digitos
   571→   571→   571→  var digits = str.replace(/\D/g, '');
   572→   572→   572→  return digits.length === 8 ? digits : '';
   573→   573→   573→}

   822→   822→   822→    cst_pis: '01',

   825→   825→   825→    vpis: base * 0.0165,
   826→   826→   826→    cst_cofins: '01',
