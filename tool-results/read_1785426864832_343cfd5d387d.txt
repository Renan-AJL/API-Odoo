   522→   561→   561→async function extractTaxes(client, db, uid, pwd, line) {
   523→   562→   562→  var result = {
   524→   563→   563→    csosn: '103', cst_icms: '', vbc: 0, vicms: 0, picms: 0,
   525→   564→   564→    cst_pis: '99', vbc_pis: 0, ppis: 0, vpis: 0,
   526→   565→   565→    cst_cofins: '99', vbc_cofins: 0, pcofins: 0, vcofins: 0,
   527→   566→   566→  };
   528→   567→   567→
   529→   568→   568→  var taxIds = (line.tax_ids || []).map(function(t) { return Array.isArray(t) ? t[0] : t; }).filter(Boolean);
   530→   569→   569→  if (!taxIds.length) return result;
   531→   570→   570→
   532→   571→   571→  try {
   533→   572→   572→    var taxes = await executeKw(client, db, uid, pwd, 'account.tax', 'read', [taxIds, [
   534→   573→   573→      'name', 'amount', 'amount_type', 'description',
   535→   574→   574→    ]]);
   536→   575→   575→    if (!taxes || !taxes.length) return result;
   537→   576→   576→
   538→   577→   577→    var base = parseFloat(line.price_subtotal) || 0;
   539→   578→   578→    var totalIcms = 0, totalPis = 0, totalCofins = 0;
   540→   579→   579→
   541→   580→   580→    for (var i = 0; i < taxes.length; i++) {
   542→   581→   581→      var tax = taxes[i];
   543→   582→   582→      var n = ((tax.name || tax.description || '').toUpperCase());
   544→   583→   583→      var amt = parseFloat(tax.amount) || 0;
   545→   584→   584→      if (n.indexOf('ICMS') >= 0 || n.indexOf('CSOSN') >= 0) {
   546→   585→   585→        totalIcms += amt;
   547→   586→   586→        var m = n.match(/(10[0-9]|20[0-9]|300|400|500|900)/);
   548→   587→   587→        if (m) result.csosn = m[1];
   549→   588→   588→      } else if (n.indexOf('PIS') >= 0) { totalPis += amt; }
   550→   589→   589→      else if (n.indexOf('COFINS') >= 0) { totalCofins += amt; }
   551→   590→   590→    }
   552→   591→   591→
   553→   592→   592→    result.picms = totalIcms; result.vbc = base; result.vicms = base * (totalIcms / 100);
   554→   593→   593→    result.ppis = totalPis; result.vbc_pis = base; result.vpis = base * (totalPis / 100);
   555→   594→   594→    result.pcofins = totalCofins; result.vbc_cofins = base; result.vcofins = base * (totalCofins / 100);
   556→   595→   595→  } catch (e) { console.warn('[SIEG-EMIT] Erro ao extrair impostos:', e.message); }
   557→   596→   596→
   558→   597→   597→  return result;
   559→   598→   598→}
