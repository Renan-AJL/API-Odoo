    17→// POST /api/v1/te/deliveries/send — Envia pedidos do ODOO ao TE
    18→// Body: { orderIds?: number[], pickingIds?: number[], orderType?: 1|2|3|4|5 }
    19→// -------------------------------------------------------
    20→router.post('/deliveries/send', async (req, res, next) => {
    21→  try {
    22→    const { orderIds, pickingIds, orderType = 1 } = req.body;
    23→
    24→    if (!orderIds?.length && !pickingIds?.length) {
    25→      return res.status(400).json({ success: false, error: 'Informe orderIds ou pickingIds' });
    26→    }
    27→
    28→    // 1. Busca pedidos ODOO
    29→    const pickings = [];
    30→    const saleOrders = [];
    31→
    32→    if (pickingIds?.length) {
    33→      const found = await odooTe.read('stock.picking', pickingIds, [
    34→        'id', 'name', 'partner_id', 'scheduled_date', 'move_line_count', 'origin',
    35→      ]);
    36→      pickings.push(...found);
    37→    }
    38→
    39→    if (orderIds?.length) {
    40→      const soFields = odooTe.getStudioFields('sale.order');
    41→      const found = await odooTe.read('sale.order', orderIds, [
    42→        'id', 'name', 'state', 'partner_id', 'amount_total', 'note', ...soFields,
    43→      ]);
    44→      saleOrders.push(...found);
    45→    }
    46→
    47→    // 2. Busca parceiros
    48→    const partnerIds = [
    49→      ...pickings.map(p => p.partner_id?.[0]),
    50→      ...saleOrders.map(o => o.partner_id?.[0]),
    51→    ].filter(Boolean);
    52→
    53→    const partners = partnerIds.length > 0
    54→      ? await odooTe.read('res.partner', [...new Set(partnerIds)], [
    55→          'id', 'name', 'cnpj_cpf', 'vat', 'phone', 'mobile', 'email',
    56→          'street', 'street_number', 'street2', 'zip', 'city',
    57→          'l10n_br_district', 'partner_latitude', 'partner_longitude',
    58→          'state_id', 'country_id',
    59→          ...odooTe.getStudioFields('res.partner'),
    60→        ])
    61→      : [];
    62→
    63→    const partnerMap = {};
    64→    partners.forEach(p => { partnerMap[p.id] = p; });
    65→
    66→    // 3. Mapeia para entregas TE
    67→    const teDeliveries = [];
    68→
    69→    for (const so of saleOrders) {
    70→      const partner = partnerMap[so.partner_id?.[0]] || {};
    71→      const picking = pickings.find(p => p.origin === so.name) || null;
    72→      const teDelivery = Mapper.odooToTeDelivery(so, partner, picking);
    73→      teDelivery.OrderType = orderType;
    74→      teDeliveries.push(teDelivery);
    75→    }
    76→
    77→    for (const picking of pickings) {
    78→      if (saleOrders.some(so => so.name === picking.origin)) continue; // ja mapeado
    79→      const partner = partnerMap[picking.partner_id?.[0]] || {};
    80→      const fakeSo = { name: picking.origin || `WH-${picking.id}`, note: '', id: 0 };
    81→      const teDelivery = Mapper.odooToTeDelivery(fakeSo, partner, picking);
    82→      teDelivery.OrderType = orderType;
    83→      teDeliveries.push(teDelivery);
    84→    }
    85→
    86→    if (teDeliveries.length === 0) {
    87→      return res.json({ success: true, sent: 0, message: 'Nenhuma entrega para enviar' });
    88→    }
    89→
    90→    // 4. Envia ao TE em lotes de 50
    91→    const results = [];
    92→    const BATCH = 50;
    93→
    94→    for (let i = 0; i < teDeliveries.length; i += BATCH) {
    95→      const batch = teDeliveries.slice(i, i + BATCH);
    96→      const result = await teClient.createDeliveries(batch);
    97→      results.push(result);
    98→
    99→      // 5. Marca como sincronizado no ODOO
   100→      const batchOrderNums = batch.map(d => d.OrderNumber);
   101→
   102→      const syncedSo = saleOrders.filter(so =>
   103→        batchOrderNums.includes(so.name) || batchOrderNums.includes(`WH-${so.id}`)
   104→      );
   105→      const syncedPickings = pickings.filter(p =>
   106→        batchOrderNums.includes(p.origin) || batchOrderNums.includes(`WH-${p.id}`)
   107→      );
   108→
   109→      if (syncedSo.length > 0) {
   110→        await odooTe.markSaleOrdersSynced(syncedSo.map(s => s.id), batchOrderNums[0], orderType);
   111→      }
   112→      if (syncedPickings.length > 0) {
   113→        await odooTe.markPickingsSynced(syncedPickings.map(p => p.id), batchOrderNums[0]);
   114→      }
   115→    }
   116→
   117→    console.log(`[TE-DELIVERY] Enviadas ${teDeliveries.length} entregas ao TE`);
   118→    res.json({
   119→      success: true,
   120→      sent: teDeliveries.length,
   121→      batches: results.length,
   122→      results,
   123→    });
   124→  } catch (err) {
   125→    console.error('[TE-DELIVERY] Erro:', err.message);
   126→    const status = err.isAxiosError ? (err.response?.status || 502) : 500;
   127→    res.status(status).json({ success: false, error: err.message });
   128→  }
   129→});

   173→

   175→    const picking = await odooTe.findPickingByTeId(orderNumber);

   178→    const saleOrder = await odooTe.findSaleOrderByTeId(orderNumber);
   179→    if (saleOrder) {
   180→      await odooTe.updateSaleOrderTeData(saleOrder.id, Mapper.teToOdooSaleOrder(delivery));
   181→    }
   182→
   183→    res.json({
   184→      success: true,
   185→      orderNumber,
   186→      situation: mapped.situation,
   187→      situationLabel: mapped.situationLabel,
   188→      trackingCode: mapped.trackingCode,
   189→      driverName: mapped.driverName,
   190→      odooUpdated: !!(picking || saleOrder),
   191→      raw: delivery,
   192→    });
   193→  } catch (err) {
   194→    console.error('[TE-DELIVERY] Erro status:', err.message);
   195→    res.status(500).json({ success: false, error: err.message });

   280→// GET /api/v1/te/deliveries/pull — Puxa status e atualiza ODOO
   281→// -------------------------------------------------------
   282→router.get('/deliveries/pull', async (req, res) => {
   283→  try {
   284→    const { dateFrom, dateTo, situation } = req.query;
   285→
   286→    const params = { page: 1 };
   287→    if (dateFrom) params.dateFrom = dateFrom;
   288→    if (dateTo) params.dateTo = dateTo;
   289→    if (situation) params.situation = parseInt(situation, 10);
   290→
   291→    const allDeliveries = await teClient.fetchAllPages('/api/Entregas', params);
   292→
   293→    let updated = 0;
   294→    let notFound = 0;
   295→
   296→    for (const d of allDeliveries) {
   297→      const orderNum = d.OrderNumber || d.orderNumber;
   298→      const picking = await odooTe.findPickingByTeId(orderNum);
   299→
   300→      if (picking) {
   301→        const mapped = Mapper.teToOdooPicking(d);
   302→        await odooTe.updatePickingTeData(picking.id, mapped);
   303→        updated++;
   304→      } else {
   305→        const so = await odooTe.findSaleOrderByTeId(orderNum);
   306→        if (so) {
   307→          await odooTe.updateSaleOrderTeData(so.id, Mapper.teToOdooSaleOrder(d));
   308→          updated++;
   309→        } else {
   310→          notFound++;
   311→        }
   312→      }
   313→    }
   314→
   315→    res.json({
   316→      success: true,
   317→      totalFetched: allDeliveries.length,
   318→      updatedInOdoo: updated,
   319→      notFoundInOdoo: notFound,
   320→    });
   321→  } catch (err) {
   322→    console.error('[TE-DELIVERY] Erro pull:', err.message);
   323→    res.status(500).json({ success: false, error: err.message });
   324→  }
   325→});

   359→function fmtDate(dateStr) {
   360→  if (!dateStr) return '';
   361→  try {
   362→    const d = new Date(dateStr);
   363→    if (isNaN(d.getTime())) return dateStr;
   364→    return d.toLocaleString('pt-BR', { timeZone: 'America/Sao_Paulo',
   365→      day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit' });
   366→  } catch { return dateStr; }
   367→}

   962→// ============================================================
   963→// MATCH MOTORISTA — Faz match do nome do TE com as opcoes do campo selection
   964→// O campo x_studio_motorista e um selection com 17 opcoes (key/label)
   965→// ============================================================
   966→const MOTORISTA_OPTIONS = [
   967→  'adelson', 'adriano', 'anderlucio', 'antonio', 'carlos', 'daniel',
   968→  'diego', 'eduardo', 'fellipe', 'gerson', 'glauber',
   969→  'henrique', 'italo', 'jeferson', 'joao_pedro', 'lucas',
   970→  'matheus', 'rafael',
   971→];
   972→
   973→function matchMotoristaSelection(driverNameFromTe) {
   974→  if (!driverNameFromTe) return null;
   975→  const name = driverNameFromTe.toUpperCase().trim();
   976→  // Remove sufixos comuns como "MOTORISTA", "FRETE", etc
   977→  const cleanName = name
   978→    .replace(/\b(MOTORISTA|FRETE|ENTREGAS|MOTOBOY)\b/g, '')
   979→    .replace(/[^A-ZÀ-Ú\s]/g, '')
   980→    .trim();
   981→  const parts = cleanName.split(/\s+/).filter(Boolean);
   982→
   983→  // Tenta match exato com a key ou com o primeiro/ultimo nome
   984→  for (const key of MOTORISTA_OPTIONS) {
   985→    const keyUpper = key.toUpperCase();
   986→    if (name.includes(keyUpper) || keyUpper.includes(name)) return key;
   987→  }
   988→  // Tenta match com qualquer parte do nome
   989→  for (const part of parts) {
   990→    if (part.length < 3) continue;
   991→    for (const key of MOTORISTA_OPTIONS) {
   992→      if (key.toUpperCase() === part || part.includes(key.toUpperCase())) return key;
   993→    }
   994→  }
   995→  return null;
   996→}

  1027→    var base = oc.url.replace(/\/+$/, '');
  1028→    var host = base.replace('https://', '');
  1029→
  1030→    function odooExec(model, method, args, kwargs) {
  1031→      return new Promise(function(ok, fail) {
  1032→        var common = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/common', port: 443 });
  1033→        common.methodCall('authenticate', [oc.db, oc.user, oc.password, {}], function(e, uid) {
  1034→          if (e || !uid) return fail(e || new Error('auth failed'));
  1035→          var params = [oc.db, uid, oc.password, model, method, args || []];
  1036→          if (kwargs) params.push(kwargs);
  1037→          var models = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
  1038→          models.methodCall('execute_kw', params, function(e2, r) {
  1039→            if (e2) return fail(e2);
  1040→            ok(r);
  1041→          });
  1042→        });
  1043→      });
  1044→    }
  1045→
  1046→    var modelIds = await odooExec('ir.model', 'search', [[['model', '=', 'account.move']]]);

  1048→      return res.status(500).json({ success: false, error: 'Model account.move nao encontrado' });
  1049→    }

  1051→

  1053→      [['name', '=', 'x_studio_motorista'], ['model_id', '=', modelId]]

  1055→

  1066→      } catch (writeErr) {
