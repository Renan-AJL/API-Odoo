     1→/**
     2→ * services/sieg-odoo-emit.js v2.0.0
     3→ * =============================================
     4→ * Processa emissoes pendentes de NF-e/NFS-e via polling Odoo XML-RPC.
     5→ * 
     6→ * CAMPOS USADOS (existentes no Odoo do cliente):
     7→ *   account.move:
     8→ *     x_studio_nfe_status     (Selection: vazio, pendente, processando, autorizada, erro)
     9→ *     x_studio_nfe_chave      (Char)
    10→ *     x_studio_nfe_protocolo  (Char)
    11→ *     x_studio_nfse_numero    (Char)
    12→ *     x_studio_nfse_status    (Selection: vazio, pendente, processando, autorizada, erro)
    13→ *   res.company:
    14→ *     x_studio_sieg_ultimo_nfe  (Integer)
    15→ *     x_studio_sieg_ultimo_nfse (Integer)
    16→ *     x_studio_sieg_serie_nfe   (Char, default 100)
    17→ *     x_studio_sieg_serie_nfse  (Char, default 1)
    18→ *   product.product:
    19→ *     x_studio_c_trib_nac       (Char) — Codigo tributo nacional (NFS-e)
    20→ *     x_studio_c_nbs            (Char) — Codigo NBS (NFS-e)
    21→ *     x_studio_aliquota_iss    (Char) — Aliquota ISS (NFS-e)
    22→ *     x_studio_ibge_code       (Char) — Codigo IBGE municipio
    23→ *
    24→ * Chatter: apos emissao, XML + PDF (DANFE/DANFSE) aparecem como anexos.
    25→ */

    36→function createClient(url) {
    37→  var base = url.replace(/\/+$/, '');
    38→  var host = base.replace('https://', '').replace('http://', '');
    39→  var port = base.startsWith('https') ? 443 : 80;
    40→  var isSecure = base.startsWith('https');
    41→  var createFn = isSecure ? xmlrpc.createSecureClient : xmlrpc.createClient;
    42→  return {
    43→    common: createFn({ host: host, path: '/xmlrpc/2/common', port: port }),
    44→    models: createFn({ host: host, path: '/xmlrpc/2/object', port: port }),
    45→  };
    46→}
    47→
    48→function authenticate(client, db, login, password) {
    49→  return new Promise(function(resolve, reject) {
    50→    client.common.methodCall('authenticate', [db, login, password, {}], function(err, uid) {
    51→      if (err) reject(new Error('Auth Odoo falhou: ' + (err.message || JSON.stringify(err))));
    52→      else if (uid === false || uid === null) reject(new Error('Credenciais Odoo invalidas'));
    53→      else resolve(uid);
    54→    });
    55→  });
    56→}
    57→
    58→function executeKw(client, db, uid, password, model, method, args, kwargs) {
    59→  return new Promise(function(resolve, reject) {
    60→    var params = [db, uid, password, model, method, args || []];
    61→    if (kwargs) params.push(kwargs);
    62→    client.models.methodCall('execute_kw', params, function(err, result) {
    63→      if (err) reject(new Error(model + '.' + method + ': ' + (err.message || JSON.stringify(err))));
    64→      else resolve(result);
    65→    });
    66→  });
    67→}

    70→// Main: Process Pending Emissions
    71→// ============================================================
    72→async function processPendingEmissions() {
    73→  var odoo = config.odoo;
    74→  if (!odoo || !odoo.enabled || !odoo.url) {
    75→    console.log('[SIEG-EMIT] Odoo nao configurado');
    76→    return { processed: 0, reason: 'odoo_not_configured' };
    77→  }
    78→
    79→  var client = createClient(odoo.url);
    80→  var uid = await authenticate(client, odoo.db, odoo.user, odoo.password);
    81→  var db = odoo.db;
    82→  var pwd = odoo.password;
    83→
    84→  console.log('[SIEG-EMIT] Buscando faturas pendentes...');
    85→
    86→  // Buscar NF-e pendentes
    87→  var nfeIds = await executeKw(client, db, uid, pwd, 'account.move', 'search', [[
    88→    ['move_type', '=', 'out_invoice'],
    89→    ['x_studio_nfe_status', '=', 'pendente'],
    90→  ]], { order: 'id asc', limit: 5 });
    91→
    92→  // Buscar NFS-e pendentes
    93→  var nfseIds = await executeKw(client, db, uid, pwd, 'account.move', 'search', [[
    94→    ['move_type', '=', 'out_invoice'],
    95→    ['x_studio_nfse_status', '=', 'pendente'],
    96→  ]], { order: 'id asc', limit: 5 });
    97→
    98→  var allPending = [];
    99→  for (var i = 0; i < nfeIds.length; i++) {
   100→    allPending.push({ id: nfeIds[i], tipo: 'nfe' });
   101→  }
   102→  for (var j = 0; j < nfseIds.length; j++) {
   103→    allPending.push({ id: nfseIds[j], tipo: 'nfse' });
   104→  }
   105→
   106→  if (!allPending.length) {
   107→    console.log('[SIEG-EMIT] Nenhuma fatura pendente');
   108→    return { processed: 0 };
   109→  }
   110→
   111→  console.log('[SIEG-EMIT] ' + allPending.length + ' fatura(s) pendente(s)');
   112→  var results = [];
   113→
   114→  for (var k = 0; k < allPending.length; k++) {
   115→    var item = allPending[k];
   116→    try {
   117→      var r = await processOne(client, db, uid, pwd, item.id, item.tipo);
   118→      results.push(r);
   119→    } catch (err) {
   120→      console.error('[SIEG-EMIT] ERRO fatura ' + item.id + ':', err.message);
   121→      await safeUpdateError(client, db, uid, pwd, item.id, item.tipo, err.message);
   122→      results.push({ move_id: item.id, sucesso: false, tipo: item.tipo, erro: err.message });
   123→    }
   124→  }
   125→
   126→  var ok = results.filter(function(r) { return r.sucesso; }).length;
   127→  console.log('[SIEG-EMIT] Concluido: ' + ok + '/' + results.length + ' sucesso');
   128→  return { processed: results.length, sucesso: ok, detalhes: results };
   129→}

   131→// ============================================================
   132→// Process Single Invoice
   133→// ============================================================
   134→async function processOne(client, db, uid, pwd, moveId, tipo) {
   135→  console.log('[SIEG-EMIT] --- Fatura ID: ' + moveId + ' | Tipo: ' + tipo + ' ---');
   136→
   137→  // 1. Read account.move (sem invoice_line_ids — pode ser computed nao-stored no Odoo 19)
   138→  var moves = await executeKw(client, db, uid, pwd, 'account.move', 'read', [[moveId], [
   139→    'name', 'partner_id', 'company_id', 'invoice_date', 'date',
   140→    'amount_total', 'narration',
   141→    'x_studio_nfe_status', 'x_studio_nfse_status', 'payment_state',
   142→    'x_studio_nosso_numero', 'x_studio_itau_resposta_json', 'x_studio_boleto_cancelado',
   143→    'invoice_line_ids', 'line_ids',
   144→  ]]);
   145→  if (!moves || !moves.length) throw new Error('Fatura ' + moveId + ' nao encontrada');
   146→  var move = moves[0];
   147→  console.log('[SIEG-EMIT] Fatura: ' + move.name + ' | R$ ' + (move.amount_total || 0));
   148→
   149→  // 1b. Read sale order origin (invoice_origin / sale_order_ids) for pedido number
   150→  var saleOrderNumber = '';
   151→  try {
   152→    // Try invoice_origin field first
   153→    var moveOrigin = await executeKw(client, db, uid, pwd, 'account.move', 'read', [[moveId], ['invoice_origin', 'ref']]);
   154→    if (moveOrigin && moveOrigin[0]) {
   155→      saleOrderNumber = moveOrigin[0].invoice_origin || moveOrigin[0].ref || '';
   156→    }
   157→    // If still empty, try sale_order_ids (M2M field)
   158→    if (!saleOrderNumber) {
   159→      var soIds = await executeKw(client, db, uid, pwd, 'account.move', 'read', [[moveId], ['sale_order_ids']]);
   160→      if (soIds && soIds[0] && soIds[0].sale_order_ids && soIds[0].sale_order_ids.length > 0) {
   161→        var soNames = await executeKw(client, db, uid, pwd, 'sale.order', 'read', [[soIds[0].sale_order_ids[0]], ['name']]);
   162→        if (soNames && soNames[0]) saleOrderNumber = soNames[0].name || '';
   163→      }
   164→    }
   165→  } catch (e) {
   166→    console.log('[SIEG-EMIT] Campo invoice_origin/sale_order_ids nao disponivel: ' + e.message);
   167→  }
   168→  if (saleOrderNumber) {
   169→    console.log('[SIEG-EMIT] Pedido de venda origem: ' + saleOrderNumber);
   170→  }
   171→
   172→  // 1c. Read boleto data from Itaú response JSON
   173→  var boletoInfo = '';
   174→  var boletoDataList = [];
   175→  try {
   176→    var rawJson = move.x_studio_itau_resposta_json || '';
   177→    if (rawJson) {
   178→      var jsonStr = rawJson.replace(/'/g, '"').replace(/True/g, 'true').replace(/False/g, 'false').replace(/None/g, 'null');
   179→      var parsed = JSON.parse(jsonStr);
   180→      var pagamentos = parsed.pagamentos || [];
   181→      var boletoParts = [];
   182→      for (var pi = 0; pi < pagamentos.length; pi++) {
   183→        var p = pagamentos[pi];
   184→        if (p.nosso_numero) {
   185→          var bolEntry = { nosso_numero: p.nosso_numero };
   186→          var part = 'Boleto: NN=' + p.nosso_numero;
   187→          if (p.linha_digitavel) {
   188→            part += ' LD=' + p.linha_digitavel;
   189→            bolEntry.linha_digitavel = p.linha_digitavel;
   190→          }
   191→          if (p.codigo_barras) {
   192→            part += ' CB=' + p.codigo_barras;
   193→            bolEntry.codigo_barras = p.codigo_barras;
   194→          }
   195→          boletoParts.push(part);
   196→          boletoDataList.push(bolEntry);
   197→        }
   198→      }
   199→      if (boletoParts.length > 0) {
   200→        boletoInfo = boletoParts.join(' | ');
   201→      }
   202→    }
   203→    // Also try nosso_numero directly if no JSON boletos found
   204→    if (boletoDataList.length === 0 && move.x_studio_nosso_numero) {
   205→      boletoInfo = 'Boleto: NN=' + move.x_studio_nosso_numero;
   206→      boletoDataList.push({ nosso_numero: move.x_studio_nosso_numero });
   207→    }
   208→  } catch (e) {
   209→    console.log('[SIEG-EMIT] Erro ao ler dados de boleto: ' + e.message);
   210→  }
   211→  if (boletoInfo) {
   212→    console.log('[SIEG-EMIT] Boleto info: ' + boletoInfo.substring(0, 200));
   213→  }
   214→
   215→  // 2. Read company
   216→  var companyId = tupId(move.company_id);
   217→  var company = await readCompany(client, db, uid, pwd, companyId);
   218→  console.log('[SIEG-EMIT] [EMITENTE] CNPJ=' + company.cnpj_cpf + ' IE=' + company.inscr_est + ' cMun=' + company.city_ibge_code + ' xMun=' + company.city + ' UF=' + company.state + ' xLgr=' + company.street + ' nro=' + company.number);
   219→
   220→  // 3. Read partner
   221→  var partnerId = tupId(move.partner_id);
   222→  if (!partnerId) throw new Error('Fatura sem parceiro');
   223→  var partner = await readPartner(client, db, uid, pwd, partnerId);
   224→  console.log('[SIEG-EMIT] [DESTINATARIO] CNPJ=' + partner.cnpj_cpf + ' IE=' + (partner.inscr_est || '(vazio)') + ' xNome=' + partner.xNome + ' cMun=' + partner.city_ibge_code + ' xMun=' + partner.city + ' UF=' + partner.state + ' xLgr=' + partner.street + ' nro=' + partner.number);
   225→
   226→  // 3b. Auto-lookup IE do destinatario se vazio (SEFAZ cStat 232 exige IE para CFOP 5xxx/6xxx)
   227→  if (!partner.inscr_est || String(partner.inscr_est).trim() === '') {
   228→    var partnerCnpj = (partner.cnpj_cpf || '').replace(/\D/g, '');
   229→    var partnerUf = partner.state || '';
   230→    if (partnerCnpj.length === 14) {
   231→      console.log('[SIEG-EMIT] [DEST-IE] IE vazia no Odoo. Tentando auto-lookup CNPJ=' + partnerCnpj + ' UF=' + partnerUf);
   232→      try {
   233→        var ieResult = await _ieService.obterIE(partnerCnpj, null, partnerUf);
   234→        if (ieResult.hasIE && ieResult.ie) {
   235→          partner.inscr_est = ieResult.ie;
   236→          console.log('[SIEG-EMIT] [DEST-IE] IE encontrada: ' + ieResult.ie + ' (UF=' + (ieResult.ieState || partnerUf) + ') via ' + ieResult.source);
   237→          // Tentar salvar IE no Odoo para futuras emissoes
   238→          try {
   239→            await executeKw(client, db, uid, pwd, 'res.partner', 'write', [[partnerId], { inscr_est: ieResult.ie }]);
   240→            console.log('[SIEG-EMIT] [DEST-IE] IE salva no parceiro Odoo (id=' + partnerId + ')');
   241→          } catch (writeErr) {
   242→            console.warn('[SIEG-EMIT] [DEST-IE] Nao conseguiu salvar IE no Odoo: ' + writeErr.message);
   243→          }
   244→        } else {
   245→          console.warn('[SIEG-EMIT] [DEST-IE] IE nao encontrada em nenhuma fonte. Motivos: ' + JSON.stringify(ieResult.reasons || []));
   246→          console.warn('[SIEG-EMIT] [DEST-IE] Emissao provavelmente sera rejeitada pela SEFAZ (cStat 232) se CFOP exigir IE.');
   247→        }
   248→      } catch (ieErr) {
   249→        console.error('[SIEG-EMIT] [DEST-IE] Erro no auto-lookup de IE: ' + ieErr.message);
   250→      }
   251→    } else if (partnerCnpj.length === 11) {
   252→      console.log('[SIEG-EMIT] [DEST-IE] Destinatario CPF (consumidor final) — IE nao necessaria');
   253→    } else {
   254→      console.warn('[SIEG-EMIT] [DEST-IE] CNPJ/CPF invalido (' + partnerCnpj.length + ' digitos) — sem auto-lookup de IE');
   255→    }
   256→  } else {
   257→    console.log('[SIEG-EMIT] [DEST-IE] IE ja preenchida no Odoo: ' + partner.inscr_est);
   258→  }
   259→
   260→  // 4. Read invoice lines diretamente de account.move.line (evita computed field)
   261→  var allLineIds = await executeKw(client, db, uid, pwd, 'account.move.line', 'search', [[
   262→    ['move_id', '=', moveId],
   263→  ]], { order: 'id asc' });
   264→  console.log('[SIEG-EMIT] Linhas encontradas na fatura: ' + allLineIds.length);
   265→  var rawLines = await executeKw(client, db, uid, pwd, 'account.move.line', 'read', [allLineIds, [
   266→    'display_type', 'product_id', 'name', 'quantity', 'price_unit',
   267→    'price_subtotal', 'tax_ids', 'discount',
   268→  ]]);
   269→  // Debug: log todas as linhas
   270→  for (var li = 0; li < rawLines.length; li++) {
   271→    var l = rawLines[li];
   272→    var pName = 'N/A';
   273→    if (l.product_id) pName = Array.isArray(l.product_id) ? l.product_id[1] : String(l.product_id);
   274→    console.log('[SIEG-EMIT]   Line ' + li + ': display_type=' + JSON.stringify(l.display_type) + ' product=' + pName + ' qty=' + l.quantity + ' price=' + l.price_unit);
   275→  }
   276→  // Filtro principal: linhas com produto associado (mais confiavel que display_type)
   277→  var invoiceLines = rawLines.filter(function(l) { return l.product_id; });
   278→  // Fallback: se nenhuma tem product_id, tenta por display_type
   279→  if (!invoiceLines.length) {
   280→    invoiceLines = rawLines.filter(function(l) {
   281→      var dt = l.display_type;
   282→      return !dt || dt === 'product' || dt === 'service';
   283→    });
   284→  }
   285→  if (!invoiceLines.length) throw new Error('Fatura sem linhas de produto/servico (total: ' + rawLines.length + ', com product: 0)');
   286→  console.log('[SIEG-EMIT] Linhas de produto: ' + invoiceLines.length);
   287→
   288→  // 5. Get next NF number from company
   289→  var serie, numField, nextNum;
   290→  if (tipo === 'nfe') {
   291→    serie = String(company.x_studio_sieg_serie_nfe || '100');
   292→    numField = 'x_studio_sieg_ultimo_nfe';
   293→  } else {
   294→    serie = String(company.x_studio_sieg_serie_nfse || '1');
   295→    numField = 'x_studio_sieg_ultimo_nfse';
   296→  }
   297→  nextNum = (parseInt(company[numField]) || 0) + 1;
   298→  console.log('[SIEG-EMIT] Serie: ' + serie + ' | nNF: ' + nextNum);
   299→
   300→  // 6. Build line data with taxes + product x_studio_ fields
   301→  var linesData = [];
   302→  for (var i = 0; i < invoiceLines.length; i++) {
   303→    var ld = await buildLineData(client, db, uid, pwd, invoiceLines[i]);
   304→    linesData.push(ld);
   305→  }
   306→
   307→  // 7. Build emission payload
   308→  // Nota: usar amount_untaxed (sem impostos) para NF-e com CSOSN 103
   309→  // pois vNF deve ser igual a soma dos componentes (vProd + vICMS + ...)
   310→  // e para CSOSN 103 o ICMS e 0, entao vNF = vProd
   311→  var vProdSum = 0;
   312→  for (var li = 0; li < linesData.length; li++) {
   313→    vProdSum += parseFloat(linesData[li].price_subtotal) || 0;
   314→  }
   315→
   316→  // Pagamentos: NF-e 4.00 — vPag deve ser igual a vNF (total SEM IBS/CBS).
   317→  // IBS/CBS sao tributos "por fora" e nao entram no bloco <pag>.
   318→  // Se vPag > vNF sem vTroco declarado -> cStat 866.
   319→  var pagamentos = [{
   320→    tPag: '15', // PIX
   321→    vPag: String(vProdSum.toFixed(2)),
   322→  }];
   323→
   324→  var emitData = {
   325→    company: company,
   326→    partner: partner,
   327→    order: {
   328→      name: move.name,
   329→      number: String(nextNum),
   330→      date_order: move.invoice_date || move.date,
   331→      amount_total: vProdSum, // vNF (sem IBS/CBS) — o XML calcula vNFTot internamente
   332→      note: move.narration || '',
   333→      sale_order_number: saleOrderNumber || '',
   334→      boleto_info: boletoInfo || '',
   335→    },
   336→    lines: linesData,
   337→    pagamentos: pagamentos,
   338→    boletoData: boletoDataList || [],
   339→    config: {
   340→      serie: serie,
   341→      tpAmb: process.env.SIEG_TP_AMB || config.sieg.tpAmb || '2',
   342→      natOp: 'Venda de Mercadoria',
   343→      mod: tipo === 'nfe' ? '55' : '01',
   344→    },
   345→    tipo: tipo,
   346→  };
   347→
   348→  // NFS-e: build service block from product x_studio_ fields
   349→  if (tipo === 'nfse') {
   350→    emitData.service = buildServiceBlock(linesData, move);
   351→  }
   352→
   353→  // 8. Set status to 'processando'
   354→  var processVals = {};
   355→  if (tipo === 'nfe') processVals.x_studio_nfe_status = 'processando';
   356→  else processVals.x_studio_nfse_status = 'processando';
   357→  await executeKw(client, db, uid, pwd, 'account.move', 'write', [[moveId], processVals]);
   358→
   359→  // 9. Call SIEG API
   360→  console.log('[SIEG-EMIT] Enviando ao SIEG...');
   361→  var resultado;
   362→  try {
   363→    resultado = await emitirNota(emitData);
   364→  } catch (siegErr) {
   365→    // Captura erros HTTP (ex: 409 com detalhes no body)
   366→    var siegDetail = siegErr.response && siegErr.response.data;
   367→    var errMsg = 'SIEG HTTP ' + (siegErr.response ? siegErr.response.status : 'erro') + ': ';
   368→    if (siegDetail) {
   369→      // Tentar extrair ErrorMessage do formato SIEG (PascalCase)
   370→      var siegMsg = siegDetail.ErrorMessage || siegDetail.Message || siegDetail.message || '';
   371→      if (siegMsg) {
   372→        errMsg += siegMsg;
   373→      } else {
   374→        errMsg += typeof siegDetail === 'string' ? siegDetail : JSON.stringify(siegDetail).slice(0, 800);
   375→      }
   376→    } else {
   377→      errMsg += siegErr.message;
   378→    }
   379→    console.error('[SIEG-EMIT] Erro SIEG:', errMsg);
   380→    throw new Error(errMsg);
   381→  }
   382→  console.log('[SIEG-EMIT] SIEG retornou - sucesso: ' + resultado.sucesso + (resultado.httpStatus ? ' (HTTP ' + resultado.httpStatus + ')' : ''));
   383→
   384→  // Se SIEG retornou erro (IsSuccess=false), extrair mensagem detalhada
   385→  if (!resultado.sucesso && resultado.erro) {
   386→    console.error('[SIEG-EMIT] SIEG rejeitou o XML: ' + resultado.erro);
   387→  }
   388→
   389→  // 10. Parse result
   390→  var info = parseResult(resultado, tipo);
   391→  var xmlRetornoText = '';
   392→  var rawXmlAuth = resultado.resposta && (resultado.resposta.Xml || resultado.resposta.xml || resultado.resposta.xmlBase64);
   393→  if (rawXmlAuth) {
   394→    xmlRetornoText = (typeof rawXmlAuth === 'string' && rawXmlAuth.length > 200)
   395→      ? rawXmlAuth
   396→      : Buffer.from(rawXmlAuth, 'base64').toString('utf-8');
   397→  }
   398→
   399→  // 11. Update account.move with results
   400→  var updateVals = {};
   401→  if (info.sucesso) {
   402→    if (tipo === 'nfe') {
   403→      updateVals.x_studio_nfe_status = 'autorizada';
   404→      updateVals.x_studio_nfe_chave = info.chave;
   405→      updateVals.x_studio_nfe_protocolo = info.protocolo;
   406→    } else {
   407→      updateVals.x_studio_nfse_status = 'autorizada';
   408→      updateVals.x_studio_nfse_numero = String(nextNum);
   409→    }
   410→
   411→    // Increment NF number (only on authorization)
   412→    try {
   413→      var companyUpdate = {};
   414→      companyUpdate[numField] = nextNum;
   415→      await executeKw(client, db, uid, pwd, 'res.company', 'write', [[companyId], companyUpdate]);
   416→      console.log('[SIEG-EMIT] Sequencia ' + numField + ' = ' + nextNum);

   780→// ============================================================
   781→// NCM Extraction Helper
   782→// ============================================================
   783→/**
   784→ * Extrai codigo NCM (8 digitos) de um valor many2one do Odoo.
   785→ * Aceita: [id, "7308.90.90"], [id, "7308.90.90 - Descricao"], string pura, numero
   786→ */
   787→function extractNcmFromRef(ref) {
   788→  if (!ref) return '';
   789→  var str = '';
   790→  if (Array.isArray(ref)) {
   791→    str = String(ref[1] || ref[0] || '');
   792→  } else if (typeof ref === 'string') {
   793→    str = ref;
   794→  } else if (typeof ref === 'number') {
   795→    return String(ref).length === 8 ? String(ref) : '';
   796→  } else {
   797→    str = String(ref);
   798→  }
   799→  // Tentar regex NNNN.NN.NN
   800→  var m = str.match(/(\d{4})\.(\d{2})\.(\d{2})/);
   801→  if (m) return m[1] + m[2] + m[3];
   802→  // Fallback: so digitos
   803→  var digits = str.replace(/\D/g, '');
   804→  return digits.length === 8 ? digits : '';
   805→}

  1087→async function extractTaxes(client, db, uid, pwd, line) {
  1088→  // Padrão: Lucro Real (CRT 3) — CST 00, ICMS 18% PR interno
  1089→  // PIS/COFINS não-cumulativo com base no valor do produto
  1090→  var base = parseFloat(line.price_subtotal) || 0;
  1091→  // Arredondamento: SEFAZ exige 2 casas decimais por item.
  1092→  // Se nao arredondar aqui, o total pode divergir (cStat 602/603).
  1093→  var r2 = function(n) { return Math.round((parseFloat(n) || 0) * 100) / 100; };
  1094→  var result = {
  1095→    csosn: '',
  1096→    cst_icms: '00',
  1097→    mod_bc: '3',
  1098→    vbc: r2(base),
  1099→    vicms: r2(base * 0.18),
  1100→    picms: 18.00,
  1101→    cst_pis: '01',
  1102→    vbc_pis: r2(base),
  1103→    ppis: 1.65,
  1104→    vpis: r2(base * 0.0165),
  1105→    cst_cofins: '01',
  1106→    vbc_cofins: r2(base),
  1107→    pcofins: 7.60,
  1108→    vcofins: r2(base * 0.076),
  1109→  };
  1110→
  1111→  var taxIds = (line.tax_ids || []).map(function(t) { return Array.isArray(t) ? t[0] : t; }).filter(Boolean);
  1112→  if (!taxIds.length) return result;
  1113→
  1114→  try {
  1115→    var taxes = await executeKw(client, db, uid, pwd, 'account.tax', 'read', [taxIds, [
  1116→      'name', 'amount', 'amount_type', 'description',
  1117→    ]]);
  1118→    if (!taxes || !taxes.length) return result;
  1119→
  1120→    var totalIcms = 0, totalPis = 0, totalCofins = 0;
  1121→    var foundCsosn = '';
  1122→
  1123→    for (var i = 0; i < taxes.length; i++) {
  1124→      var tax = taxes[i];
  1125→      var n = ((tax.name || tax.description || '').toUpperCase());
  1126→      var amt = parseFloat(tax.amount) || 0;
  1127→      if (n.indexOf('ICMS') >= 0 || n.indexOf('CSOSN') >= 0) {
  1128→        totalIcms += amt;
  1129→        var m = n.match(/(10[0-9]|20[0-9]|300|400|500|900)/);
  1130→        if (m) foundCsosn = m[1];
  1131→      } else if (n.indexOf('PIS') >= 0) { totalPis += amt; }
  1132→      else if (n.indexOf('COFINS') >= 0) { totalCofins += amt; }
  1133→    }
  1134→
  1135→    // Sobrescreve defaults apenas se o Odoo tiver impostos reais configurados (> 0)
  1136→    if (foundCsosn) {
  1137→      result.csosn = foundCsosn;
  1138→      result.cst_icms = '';
  1139→    }
  1140→    if (totalIcms > 0) {
  1141→      result.picms = totalIcms;
  1142→      result.vicms = r2(base * (totalIcms / 100));
  1143→    }
  1144→    if (totalPis > 0) {
  1145→      result.ppis = totalPis;
  1146→      result.vpis = r2(base * (totalPis / 100));
  1147→    }
  1148→    if (totalCofins > 0) {
  1149→      result.pcofins = totalCofins;
  1150→      result.vcofins = r2(base * (totalCofins / 100));
  1151→    }
  1152→  } catch (e) { console.warn('[SIEG-EMIT] Erro ao extrair impostos:', e.message); }
  1153→
  1154→  return result;
  1155→}

  1160→function parseResult(resultado, tipo) {
  1161→  var resp = resultado.resposta || {};
  1162→  var chave = '', protocolo = '', cStat = '', motivo = '', numero = '', sucesso = false;
  1163→
  1164→  // Se SIEG retornou HTTP 4xx, usar ErrorMessage diretamente como motivo
  1165→  var siegErro = resultado.erro || '';
  1166→  if (resultado.httpStatus && resultado.httpStatus >= 400 && resultado.httpStatus < 500) {
  1167→    // Erro de validacao do XML — extrair mensagem do SIEG
  1168→    motivo = siegErro || 'Erro HTTP ' + resultado.httpStatus;
  1169→    if (tipo === 'nfe') {
  1170→      cStat = String(resultado.statusCode || resp.cStat || resp.status || '');
  1171→    }
  1172→    return { sucesso: false, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
  1173→  }
  1174→
  1175→  if (tipo === 'nfe') {
  1176→    chave = resp.chNFe || resp.chave || resp.ChaveXml || '';
  1177→    protocolo = resp.nProt || resp.protocolo || '';
  1178→    cStat = String(resp.cStat || resp.status || '');
  1179→    motivo = resp.xMotivo || resp.motivo || (siegErro || 'Processado');
  1180→    sucesso = (cStat === '100' || cStat === '104' || cStat === '150');
  1181→  } else {
  1182→    chave = resp.Chave || resp.chave || '';
  1183→    protocolo = resp.Protocolo || resp.protocolo || '';
  1184→    numero = resp.NumeroNfse || resp.nDFSe || resp.numero || '';
  1185→    cStat = String(resp.CodigoVerificacao || resp.codigo || '');
  1186→    motivo = resp.Motivo || resp.motivo || siegErro || (resultado.sucesso ? 'Autorizada' : 'Erro na emissao');
  1187→    sucesso = resultado.sucesso;
  1188→  }
  1189→
  1190→  return { sucesso: sucesso, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
  1191→}

  1390→function normalizeCityName(name) {
  1391→  if (!name) return '';
  1392→  return name
  1393→    .normalize('NFD')
  1394→    .replace(/[\u0300-\u036f]/g, '') // remove diacriticos
  1395→    .replace(/\u00e7/g, 'c')          // cedilha -> c
  1396→    .replace(/\u00c7/g, 'C')
  1397→    .toUpperCase()
  1398→    .trim();

  1401→function lookupIbgeFallback(cityName, stateCode) {

  1409→  // Fallback: tenta sem preposicoes comuns ("do", "da", "de", "das", "dos")
  1410→  var simplified = key.replace(/\b(D|DA|DE|DO|DAS|DOS)\b/g, '').replace(/\s+/g, ' ').trim();
  1411→  code = IBGE_FALLBACK[simplified];
  1412→  if (code) {
  1413→    console.log('[SIEG-EMIT] IBGE via tabela fallback (simplificado): ' + simplified + ' = ' + code);
  1414→    return code;
  1415→  }
  1416→  return '';
  1417→}

  1471→// ============================================================
  1472→/**
  1473→ * Busca faturas com x_studio_nfe_status = 'cancelando' e envia o
  1474→ * evento de cancelamento à SEFAZ PR, depois atualiza o Odoo.
  1475→ * Chamado pelo runSiegPoll no server.js junto com processPendingEmissions.
  1476→ */
  1477→async function processPendingCancellations() {
  1478→  var odoo = config.odoo;
  1479→  if (!odoo || !odoo.enabled || !odoo.url) {
  1480→    return { processed: 0, reason: 'odoo_not_configured' };
  1481→  }
  1482→
  1483→  var client = createClient(odoo.url);
  1484→  var uid = await authenticate(client, odoo.db, odoo.user, odoo.password);
  1485→  var db = odoo.db;
  1486→  var pwd = odoo.password;
  1487→
  1488→  var ids = await executeKw(client, db, uid, pwd, 'account.move', 'search', [[
  1489→    ['move_type', '=', 'out_invoice'],
  1490→    ['x_studio_nfe_status', '=', 'cancelando'],
  1491→  ]], { order: 'id asc', limit: 5 });
  1492→
  1493→  if (!ids.length) return { processed: 0 };
  1494→
  1495→  console.log('[SIEG-CANCEL-POLL] ' + ids.length + ' fatura(s) aguardando cancelamento');
  1496→
  1497→  var { cancelarNFe } = require('./nfe-cancelamento');
  1498→
  1499→  var results = [];
  1500→  for (var i = 0; i < ids.length; i++) {
  1501→    var moveId = ids[i];
  1502→    try {
  1503→      var r = await cancelOneInvoice(client, db, uid, pwd, moveId, cancelarNFe);
  1504→      results.push(r);
  1505→    } catch (err) {
  1506→      console.error('[SIEG-CANCEL-POLL] ERRO fatura ' + moveId + ':', err.message);
  1507→      try {
  1508→        await executeKw(client, db, uid, pwd, 'account.move', 'write',
  1509→          [[moveId], { x_studio_nfe_status: 'erro' }]);
  1510→        await executeKw(client, db, uid, pwd, 'mail.message', 'create', [{
  1511→          model: 'account.move', res_id: moveId,
  1512→          body: '<b>Erro no Cancelamento</b><br/>' + err.message.substring(0, 500),
  1513→          message_type: 'comment',
  1514→        }]);
  1515→      } catch (e2) { console.error('[SIEG-CANCEL-POLL] Falha ao gravar erro:', e2.message); }
  1516→      results.push({ move_id: moveId, sucesso: false, erro: err.message });
  1517→    }
  1518→  }
  1519→
  1520→  var ok = results.filter(function(r) { return r.sucesso; }).length;
  1521→  console.log('[SIEG-CANCEL-POLL] Concluido: ' + ok + '/' + results.length + ' cancelada(s)');
  1522→  return { processed: results.length, sucesso: ok };
  1523→}

  1525→async function cancelOneInvoice(client, db, uid, pwd, moveId, cancelarNFe) {
  1526→  var moves = await executeKw(client, db, uid, pwd, 'account.move', 'read',
  1527→    [[moveId], ['name', 'x_studio_nfe_chave', 'x_studio_nfe_protocolo',
  1528→                'x_studio_nfe_status', 'company_id']]);
  1529→  if (!moves || !moves.length) throw new Error('Fatura ' + moveId + ' nao encontrada');
  1530→  var move = moves[0];
  1531→
  1532→  var chave     = (move.x_studio_nfe_chave || '').replace(/\D/g, '');
  1533→  var protocolo = String(move.x_studio_nfe_protocolo || '').trim();
  1534→  var justificativa = 'Cancelamento solicitado pelo emitente via Odoo';
  1535→  if (justificativa.length < 15) justificativa = 'Cancelamento solicitado pelo emitente via Odoo';
  1536→
  1537→  console.log('[SIEG-CANCEL-POLL] Fatura ' + move.name + ' | chave=' + chave.slice(0, 15) + '...');
  1538→
  1539→  if (!chave || chave.length !== 44) throw new Error('Chave de acesso invalida (' + chave.length + ' digitos). Nota precisa estar autorizada.');
  1540→  if (!protocolo) throw new Error('Protocolo de autorizacao ausente na fatura ' + move.name);
  1541→
  1542→  var companyId = Array.isArray(move.company_id) ? move.company_id[0] : move.company_id;
  1543→  var companies = await executeKw(client, db, uid, pwd, 'res.company', 'read', [[companyId], ['vat']]);
  1544→  var cnpj = ((companies && companies[0] && companies[0].vat) || '').replace(/\D/g, '');
  1545→  if (!cnpj) throw new Error('CNPJ do emitente nao encontrado na empresa');
  1546→
  1547→  // Marcar como processando para nao processar duas vezes em paralelo
  1548→  await executeKw(client, db, uid, pwd, 'account.move', 'write',
  1549→    [[moveId], { x_studio_nfe_status: 'processando' }]);
  1550→
  1551→  var resultado = await cancelarNFe({ chave, protocolo, cnpj, justificativa });
  1552→
  1553→  var statusNovo = resultado.sucesso ? 'cancelada' : 'erro';
  1554→  await executeKw(client, db, uid, pwd, 'account.move', 'write',
  1555→    [[moveId], {
  1556→      x_studio_nfe_status: statusNovo,
  1557→      x_studio_nfe_protocolo: resultado.nProt || protocolo,
  1558→    }]);
  1559→
  1560→  if (resultado.sucesso) {
  1561→    try {
  1562→      await executeKw(client, db, uid, pwd, 'account.move', 'button_draft', [[moveId]]);
  1563→      await executeKw(client, db, uid, pwd, 'account.move', 'button_cancel', [[moveId]]);
  1564→      console.log('[SIEG-CANCEL-POLL] Fatura ' + move.name + ' revertida no Odoo');
  1565→    } catch (e) {
  1566→      console.warn('[SIEG-CANCEL-POLL] Nao foi possivel reverter a fatura: ' + e.message);
  1567→    }
  1568→  }
  1569→
  1570→  var msgCorpo = resultado.sucesso
  1571→    ? '<b>NF-e Cancelada na SEFAZ</b><br/>'
  1572→      + 'Protocolo: ' + (resultado.nProt || protocolo) + '<br/>'
  1573→      + 'Data: ' + (resultado.dhRecbto || '') + '<br/>'
  1574→      + 'Justificativa: ' + justificativa
  1575→    : '<b>Cancelamento rejeitado pela SEFAZ</b><br/>'
  1576→      + 'Status: ' + resultado.cStat + ' - ' + resultado.xMotivo + '<br/>'
  1577→      + 'Justificativa enviada: ' + justificativa;
  1578→
  1579→  try {
  1580→    await executeKw(client, db, uid, pwd, 'mail.message', 'create', [{
  1581→      model: 'account.move', res_id: moveId,
  1582→      body: msgCorpo, message_type: 'comment',
  1583→    }]);
  1584→  } catch (e) { console.warn('[SIEG-CANCEL-POLL] Erro ao postar chatter:', e.message); }
  1585→
  1586→  return { move_id: moveId, sucesso: resultado.sucesso, fatura: move.name, cStat: resultado.cStat };
  1587→}
