   561→   561→async function extractTaxes(client, db, uid, pwd, line) {
   562→   562→  var result = {
   563→   563→    csosn: '103', cst_icms: '', vbc: 0, vicms: 0, picms: 0,
   564→   564→    cst_pis: '99', vbc_pis: 0, ppis: 0, vpis: 0,
   565→   565→    cst_cofins: '99', vbc_cofins: 0, pcofins: 0, vcofins: 0,
   566→   566→  };
   567→   567→
   568→   568→  var taxIds = (line.tax_ids || []).map(function(t) { return Array.isArray(t) ? t[0] : t; }).filter(Boolean);
   569→   569→  if (!taxIds.length) return result;
   570→   570→
   571→   571→  try {
   572→   572→    var taxes = await executeKw(client, db, uid, pwd, 'account.tax', 'read', [taxIds, [
   573→   573→      'name', 'amount', 'amount_type', 'description',
   574→   574→    ]]);
   575→   575→    if (!taxes || !taxes.length) return result;
   576→   576→
   577→   577→    var base = parseFloat(line.price_subtotal) || 0;
   578→   578→    var totalIcms = 0, totalPis = 0, totalCofins = 0;
   579→   579→
   580→   580→    for (var i = 0; i < taxes.length; i++) {
   581→   581→      var tax = taxes[i];
   582→   582→      var n = ((tax.name || tax.description || '').toUpperCase());
   583→   583→      var amt = parseFloat(tax.amount) || 0;
   584→   584→      if (n.indexOf('ICMS') >= 0 || n.indexOf('CSOSN') >= 0) {
   585→   585→        totalIcms += amt;
   586→   586→        var m = n.match(/(10[0-9]|20[0-9]|300|400|500|900)/);
   587→   587→        if (m) result.csosn = m[1];
   588→   588→      } else if (n.indexOf('PIS') >= 0) { totalPis += amt; }
   589→   589→      else if (n.indexOf('COFINS') >= 0) { totalCofins += amt; }
   590→   590→    }
   591→   591→
   592→   592→    result.picms = totalIcms; result.vbc = base; result.vicms = base * (totalIcms / 100);
   593→   593→    result.ppis = totalPis; result.vbc_pis = base; result.vpis = base * (totalPis / 100);
   594→   594→    result.pcofins = totalCofins; result.vbc_cofins = base; result.vcofins = base * (totalCofins / 100);
   595→   595→  } catch (e) { console.warn('[SIEG-EMIT] Erro ao extrair impostos:', e.message); }
   596→   596→
   597→   597→  return result;
   598→   598→}

   603→   603→function parseResult(resultado, tipo) {
   604→   604→  var resp = resultado.resposta || {};
   605→   605→  var chave = '', protocolo = '', cStat = '', motivo = '', numero = '', sucesso = false;
   606→   606→
   607→   607→  if (tipo === 'nfe') {
   608→   608→    chave = resp.chNFe || resp.chave || resp.ChaveXml || '';
   609→   609→    protocolo = resp.nProt || resp.protocolo || '';
   610→   610→    cStat = String(resp.cStat || resp.status || '');
   611→   611→    motivo = resp.xMotivo || resp.motivo || 'Processado';
   612→   612→    sucesso = (cStat === '100' || cStat === '104' || cStat === '150');
   613→   613→  } else {
   614→   614→    chave = resp.Chave || resp.chave || '';
   615→   615→    protocolo = resp.Protocolo || resp.protocolo || '';
   616→   616→    numero = resp.NumeroNfse || resp.nDFSe || resp.numero || '';
   617→   617→    cStat = String(resp.CodigoVerificacao || resp.codigo || '');
   618→   618→    motivo = resp.Motivo || resp.motivo || (resultado.sucesso ? 'Autorizada' : 'Erro na emissao');
   619→   619→    sucesso = resultado.sucesso;
   620→   620→  }
   621→   621→
   622→   622→  return { sucesso: sucesso, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
   623→   623→}
