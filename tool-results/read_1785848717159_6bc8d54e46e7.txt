   542→   542→// ============================================================
   543→   543→// NCM Extraction Helper
   544→   544→// ============================================================
   545→   545→/**
   546→   546→ * Extrai codigo NCM (8 digitos) de um valor many2one do Odoo.
   547→   547→ * Aceita: [id, "7308.90.90"], [id, "7308.90.90 - Descricao"], string pura, numero
   548→   548→ */
   549→   549→function extractNcmFromRef(ref) {
   550→   550→  if (!ref) return '';
   551→   551→  var str = '';
   552→   552→  if (Array.isArray(ref)) {
   553→   553→    str = String(ref[1] || ref[0] || '');
   554→   554→  } else if (typeof ref === 'string') {
   555→   555→    str = ref;
   556→   556→  } else if (typeof ref === 'number') {
   557→   557→    return String(ref).length === 8 ? String(ref) : '';
   558→   558→  } else {
   559→   559→    str = String(ref);
   560→   560→  }
   561→   561→  // Tentar regex NNNN.NN.NN
   562→   562→  var m = str.match(/(\d{4})\.(\d{2})\.(\d{2})/);
   563→   563→  if (m) return m[1] + m[2] + m[3];
   564→   564→  // Fallback: so digitos
   565→   565→  var digits = str.replace(/\D/g, '');
   566→   566→  return digits.length === 8 ? digits : '';
   567→   567→}

   813→   813→async function extractTaxes(client, db, uid, pwd, line) {
   814→   814→  // Padrão: Lucro Real (CRT 3) — CST 00, ICMS 18% PR interno
   815→   815→  // PIS/COFINS não-cumulativo com base no valor do produto
   816→   816→  var base = parseFloat(line.price_subtotal) || 0;
   817→   817→  var result = {
   818→   818→    csosn: '',
   819→   819→    cst_icms: '00',
   820→   820→    mod_bc: '3',
   821→   821→    vbc: base,
   822→   822→    vicms: base * 0.18,
   823→   823→    picms: 18.00,
   824→   824→    cst_pis: '01',
   825→   825→    vbc_pis: base,
   826→   826→    ppis: 1.65,
   827→   827→    vpis: base * 0.0165,
   828→   828→    cst_cofins: '01',
   829→   829→    vbc_cofins: base,
   830→   830→    pcofins: 7.60,
   831→   831→    vcofins: base * 0.076,
   832→   832→  };
   833→   833→
   834→   834→  var taxIds = (line.tax_ids || []).map(function(t) { return Array.isArray(t) ? t[0] : t; }).filter(Boolean);
   835→   835→  if (!taxIds.length) return result;
   836→   836→
   837→   837→  try {
   838→   838→    var taxes = await executeKw(client, db, uid, pwd, 'account.tax', 'read', [taxIds, [
   839→   839→      'name', 'amount', 'amount_type', 'description',
   840→   840→    ]]);
   841→   841→    if (!taxes || !taxes.length) return result;
   842→   842→
   843→   843→    var totalIcms = 0, totalPis = 0, totalCofins = 0;
   844→   844→    var foundCsosn = '';
   845→   845→
   846→   846→    for (var i = 0; i < taxes.length; i++) {
   847→   847→      var tax = taxes[i];
   848→   848→      var n = ((tax.name || tax.description || '').toUpperCase());
   849→   849→      var amt = parseFloat(tax.amount) || 0;
   850→   850→      if (n.indexOf('ICMS') >= 0 || n.indexOf('CSOSN') >= 0) {
   851→   851→        totalIcms += amt;
   852→   852→        var m = n.match(/(10[0-9]|20[0-9]|300|400|500|900)/);
   853→   853→        if (m) foundCsosn = m[1];
   854→   854→      } else if (n.indexOf('PIS') >= 0) { totalPis += amt; }
   855→   855→      else if (n.indexOf('COFINS') >= 0) { totalCofins += amt; }
   856→   856→    }
   857→   857→
   858→   858→    // Sobrescreve defaults apenas se o Odoo tiver impostos reais configurados (> 0)
   859→   859→    if (foundCsosn) {
   860→   860→      result.csosn = foundCsosn;
   861→   861→      result.cst_icms = '';
   862→   862→    }
   863→   863→    if (totalIcms > 0) {
   864→   864→      result.picms = totalIcms;
   865→   865→      result.vicms = base * (totalIcms / 100);
   866→   866→    }
   867→   867→    if (totalPis > 0) {
   868→   868→      result.ppis = totalPis;
   869→   869→      result.vpis = base * (totalPis / 100);
   870→   870→    }
   871→   871→    if (totalCofins > 0) {
   872→   872→      result.pcofins = totalCofins;
   873→   873→      result.vcofins = base * (totalCofins / 100);
   874→   874→    }
   875→   875→  } catch (e) { console.warn('[SIEG-EMIT] Erro ao extrair impostos:', e.message); }
   876→   876→
   877→   877→  return result;
   878→   878→}

   883→   883→function parseResult(resultado, tipo) {
   884→   884→  var resp = resultado.resposta || {};
   885→   885→  var chave = '', protocolo = '', cStat = '', motivo = '', numero = '', sucesso = false;
   886→   886→
   887→   887→  // Se SIEG retornou HTTP 4xx, usar ErrorMessage diretamente como motivo
   888→   888→  var siegErro = resultado.erro || '';
   889→   889→  if (resultado.httpStatus && resultado.httpStatus >= 400 && resultado.httpStatus < 500) {
   890→   890→    // Erro de validacao do XML — extrair mensagem do SIEG
   891→   891→    motivo = siegErro || 'Erro HTTP ' + resultado.httpStatus;
   892→   892→    if (tipo === 'nfe') {
   893→   893→      cStat = String(resultado.statusCode || resp.cStat || resp.status || '');
   894→   894→    }
   895→   895→    return { sucesso: false, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
   896→   896→  }
   897→   897→
   898→   898→  if (tipo === 'nfe') {
   899→   899→    chave = resp.chNFe || resp.chave || resp.ChaveXml || '';
   900→   900→    protocolo = resp.nProt || resp.protocolo || '';
   901→   901→    cStat = String(resp.cStat || resp.status || '');
   902→   902→    motivo = resp.xMotivo || resp.motivo || (siegErro || 'Processado');
   903→   903→    sucesso = (cStat === '100' || cStat === '104' || cStat === '150');
   904→   904→  } else {
   905→   905→    chave = resp.Chave || resp.chave || '';
   906→   906→    protocolo = resp.Protocolo || resp.protocolo || '';
   907→   907→    numero = resp.NumeroNfse || resp.nDFSe || resp.numero || '';
   908→   908→    cStat = String(resp.CodigoVerificacao || resp.codigo || '');
   909→   909→    motivo = resp.Motivo || resp.motivo || siegErro || (resultado.sucesso ? 'Autorizada' : 'Erro na emissao');
   910→   910→    sucesso = resultado.sucesso;
   911→   911→  }
   912→   912→
   913→   913→  return { sucesso: sucesso, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
   914→   914→}
