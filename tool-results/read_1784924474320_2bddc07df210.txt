    66→    // 3. Mapeia para entregas TE
    67→    const teDeliveries = [];
    68→
    69→    for (const so of saleOrders) {
    70→      const partner = partnerMap[so.partner_id?.[0]] || {};
    71→      const picking = pickings.find(p => p.origin === so.name) || null;
    72→      const teDelivery = Mapper.odooToTeDelivery(so, partner, picking);
    73→      teDelivery.OrderType = orderType;
    74→      teDeliveries.push(teDelivery);
    75→    }
    76→
    77→    for (const picking of pickings) {
    78→      if (saleOrders.some(so => so.name === picking.origin)) continue; // ja mapeado
    79→      const partner = partnerMap[picking.partner_id?.[0]] || {};
    80→      const fakeSo = { name: picking.origin || `WH-${picking.id}`, note: '', id: 0 };
    81→      const teDelivery = Mapper.odooToTeDelivery(fakeSo, partner, picking);
    82→      teDelivery.OrderType = orderType;
    83→      teDeliveries.push(teDelivery);
    84→    }
    85→
    86→    if (teDeliveries.length === 0) {
    87→      return res.json({ success: true, sent: 0, message: 'Nenhuma entrega para enviar' });
    88→    }
    89→
    90→    // 4. Envia ao TE em lotes de 50
    91→    const results = [];
    92→    const BATCH = 50;
    93→
    94→    for (let i = 0; i < teDeliveries.length; i += BATCH) {
    95→      const batch = teDeliveries.slice(i, i + BATCH);
    96→      const result = await teClient.createDeliveries(batch);
    97→      results.push(result);
    98→
    99→      // 5. Marca como sincronizado no ODOO
   100→      const batchOrderNums = batch.map(d => d.OrderNumber);
   101→
   102→      const syncedSo = saleOrders.filter(so =>
   103→        batchOrderNums.includes(so.name) || batchOrderNums.includes(`WH-${so.id}`)
   104→      );
   105→      const syncedPickings = pickings.filter(p =>
   106→        batchOrderNums.includes(p.origin) || batchOrderNums.includes(`WH-${p.id}`)
   107→      );
   108→
   109→      if (syncedSo.length > 0) {
   110→        await odooTe.markSaleOrdersSynced(syncedSo.map(s => s.id), batchOrderNums[0], orderType);
   111→      }
   112→      if (syncedPickings.length > 0) {
   113→        await odooTe.markPickingsSynced(syncedPickings.map(p => p.id), batchOrderNums[0]);
   114→      }
   115→    }
   116→
   117→    console.log(`[TE-DELIVERY] Enviadas ${teDeliveries.length} entregas ao TE`);
   118→    res.json({
   119→      success: true,
   120→      sent: teDeliveries.length,
   121→      batches: results.length,
   122→      results,
   123→    });
   124→  } catch (err) {
   125→    console.error('[TE-DELIVERY] Erro:', err.message);
   126→    const status = err.isAxiosError ? (err.response?.status || 502) : 500;
   127→    res.status(status).json({ success: false, error: err.message });
   128→  }

   359→function fmtDate(dateStr) {
   360→  if (!dateStr) return '';
   361→  try {
   362→    const d = new Date(dateStr);
   363→    if (isNaN(d.getTime())) return dateStr;
   364→    return d.toLocaleString('pt-BR', { timeZone: 'America/Sao_Paulo',
   365→      day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit' });
   366→  } catch { return dateStr; }

   535→// POST /api/v1/te/delivery-status — Busca status TE e grava card HTML
   536→// Body: { saleOrderId: number }
   537→// ============================================================
   538→router.post('/delivery-status', async (req, res) => {
   539→  try {
   540→    const { saleOrderId } = req.body;
   541→    if (!saleOrderId) {
   542→      return res.status(400).json({ success: false, error: 'Informe saleOrderId' });
   543→    }
   544→
   545→    // 1. Busca sale order no Odoo
   546→    const so = await odooTe.findSaleOrderById(saleOrderId);
   547→    if (!so) {
   548→      return res.status(404).json({ success: false, error: `Sale Order ${saleOrderId} nao encontrada` });
   549→    }
   550→
   551→    const f = odooTe.FIELDS['sale.order'];
   552→    const orderName = so.name;
   553→    const teOrderId = so[f.teOrderId];
   554→    const trackingCode = so[f.teTrackingCode];
   555→
   556→    if (!teOrderId && !trackingCode) {
   557→      // Nenhuma entrega enviada ao TE ainda
   558→      const noCard = buildErrorCard(
   559→        'Sem Entrega TE',
   560→        `Pedido ${orderName} ainda nao foi enviado ao TudoEntregue. Envie a entrega primeiro.`
   561→      );
   562→      await odooTe.updateSaleOrderStatusHtml(saleOrderId, noCard);
   563→      return res.json({ success: true, message: 'Pedido ainda nao enviado ao TE', card: noCard });
   564→    }
   565→
   566→    // 2. Busca dados no TE — /api/Entregas (dados completos da entrega)
   567→    let deliveryData = null;
   568→    if (teOrderId) {
   569→      const result = await teClient.getDeliveries({ orderNumber: teOrderId, page: 1 });
   570→      deliveryData = result?.Result?.[0] || null;
   571→    }
   572→
   573→    // 3. Busca situacao detalhada — /orders/situation
   574→    let situationData = null;
   575→    if (teOrderId) {
   576→      try {
   577→        situationData = await teClient.getOrderSituation({
   578→          orderType: 1,
   579→          orderID: teOrderId,
   580→        });
   581→      } catch (err) {
   582→        console.warn(`[TE-DELIVERY-STATUS] /orders/situation falhou: ${err.message}`);
   583→      }
   584→    }
   585→
   586→    // 4. Busca tracking detalhado (ocorrencias com fotos)
   587→    let trackingData = null;
   588→    if (trackingCode) {
   589→      try {
   590→        trackingData = await teClient.getTracking(trackingCode);
   591→      } catch (err) {
   592→        console.warn(`[TE-DELIVERY-STATUS] /tracking falhou: ${err.message}`);
   593→      }
   594→    }
   595→
   596→    // 5. Monta card HTML
   597→    const cardHtml = buildTeCard(deliveryData || {}, trackingData, situationData);
   598→
   599→    // 6. Tambem atualiza campos padrao do TE
   600→    if (deliveryData) {
   601→      const mapped = Mapper.teToOdooPicking(deliveryData);
   602→      const soMapped = Mapper.teToOdooSaleOrder(deliveryData);
   603→      await odooTe.updateSaleOrderTeData(saleOrderId, soMapped);
   604→    }
   605→
   606→    // 7. Atualiza campo de motorista (selection) a partir do TE
   607→    let motoristaAtualizado = null;
   608→    if (deliveryData) {
   609→      const driverName = deliveryData.DriverName || deliveryData.driverName || '';
   610→      if (driverName) {
   611→        motoristaAtualizado = matchMotoristaSelection(driverName);
   612→        if (motoristaAtualizado) {
   613→          await odooTe.updateSaleOrderMotorista(saleOrderId, motoristaAtualizado);
   614→          console.log(`[TE-DELIVERY-STATUS] Motorista: ${driverName} -> ${motoristaAtualizado}`);
   615→        }
   616→      }
   617→    }
   618→
   619→    // 8. Grava card no campo HTML
   620→    await odooTe.updateSaleOrderStatusHtml(saleOrderId, cardHtml);
   621→
   622→    console.log(`[TE-DELIVERY-STATUS] Card atualizado: SO ${orderName} (${saleOrderId})`);
   623→    res.json({
   624→      success: true,
   625→      orderName,
   626→      teOrderId,
   627→      trackingCode,
   628→      motorista: motoristaAtualizado || null,
   629→      hasDeliveryData: !!deliveryData,
   630→      hasSituationData: !!situationData,
   631→      hasTrackingData: !!trackingData,
   632→      card: cardHtml,
   633→    });
   634→  } catch (err) {
   635→    console.error('[TE-DELIVERY-STATUS] Erro:', err.message);
   636→    // Grava card de erro no campo
   637→    try {
   638→      await odooTe.updateSaleOrderStatusHtml(req.body.saleOrderId, buildErrorCard('Erro ao Buscar Status', err.message));
   639→    } catch {}
   640→    const status = err.isAxiosError ? (err.response?.status || 502) : 500;
   641→    res.status(status).json({ success: false, error: err.message });
   642→  }
   643→});

   646→// MATCH MOTORISTA — Faz match do nome do TE com as opcoes do campo selection
   647→// O campo x_studio_motorista e um selection com 17 opcoes (key/label)
   648→// ============================================================
   649→const MOTORISTA_OPTIONS = [
   650→  'adelson', 'anderlucio', 'antonio', 'carlos', 'daniel',
   651→  'diego', 'eduardo', 'fellipe', 'gerson', 'glauber',
   652→  'henrique', 'italo', 'jeferson', 'joao_pedro', 'lucas',
   653→  'matheus', 'rafael',
   654→];
   655→
   656→function matchMotoristaSelection(driverNameFromTe) {
   657→  if (!driverNameFromTe) return null;
   658→  const name = driverNameFromTe.toUpperCase().trim();
   659→  // Remove sufixos comuns como "MOTORISTA", "FRETE", etc
   660→  const cleanName = name
   661→    .replace(/\b(MOTORISTA|FRETE|ENTREGAS|MOTOBOY)\b/g, '')
   662→    .replace(/[^A-ZÀ-Ú\s]/g, '')
   663→    .trim();
   664→  const parts = cleanName.split(/\s+/).filter(Boolean);
   665→
   666→  // Tenta match exato com a key ou com o primeiro/ultimo nome
   667→  for (const key of MOTORISTA_OPTIONS) {
   668→    const keyUpper = key.toUpperCase();
   669→    if (name.includes(keyUpper) || keyUpper.includes(name)) return key;
   670→  }
   671→  // Tenta match com qualquer parte do nome
   672→  for (const part of parts) {
   673→    if (part.length < 3) continue;
   674→    for (const key of MOTORISTA_OPTIONS) {
   675→      if (key.toUpperCase() === part || part.includes(key.toUpperCase())) return key;
   676→    }
   677→  }
   678→  return null;
   679→}
