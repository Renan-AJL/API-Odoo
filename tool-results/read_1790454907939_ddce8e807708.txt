    74→def code_block(filepath):
    75→    """Read file and return list of Preformatted paragraphs."""
    76→    elements = []
    77→    try:
    78→        with open(filepath, 'r', encoding='utf-8') as f:
    79→            content = f.read()
    80→    except Exception as e:
    81→        content = f'// Arquivo nao encontrado: {e}'
    82→    
    83→    # Escape HTML entities for Paragraph
    84→    content_escaped = (content
    85→        .replace('&', '&amp;')
    86→        .replace('<', '&lt;')
    87→        .replace('>', '&gt;')
    88→        .replace('"', '&quot;'))
    89→    
    90→    # Split into lines and create paragraphs in chunks for better pagination
    91→    lines = content_escaped.split('\n')
    92→    chunk_size = 80
    93→    for i in range(0, len(lines), chunk_size):
    94→        chunk = '\n'.join(lines[i:i+chunk_size])
    95→        elements.append(Paragraph(chunk, s_code))
    96→        if i + chunk_size < len(lines):
    97→            elements.append(Spacer(1, 2))
    98→    return elements
