   878→// MATCH MOTORISTA — Faz match do nome do TE com as opcoes do campo selection
   879→// O campo x_studio_motorista e um selection com 17 opcoes (key/label)
   880→// ============================================================
   881→const MOTORISTA_OPTIONS = [
   882→  'adelson', 'adriano', 'anderlucio', 'antonio', 'carlos', 'daniel',
   883→  'diego', 'eduardo', 'fellipe', 'gerson', 'glauber',
   884→  'henrique', 'italo', 'jeferson', 'joao_pedro', 'lucas',
   885→  'matheus', 'rafael',
   886→];
   887→
   888→function matchMotoristaSelection(driverNameFromTe) {
   889→  if (!driverNameFromTe) return null;
   890→  const name = driverNameFromTe.toUpperCase().trim();
   891→  // Remove sufixos comuns como "MOTORISTA", "FRETE", etc
   892→  const cleanName = name
   893→    .replace(/\b(MOTORISTA|FRETE|ENTREGAS|MOTOBOY)\b/g, '')
   894→    .replace(/[^A-ZÀ-Ú\s]/g, '')
   895→    .trim();
   896→  const parts = cleanName.split(/\s+/).filter(Boolean);
   897→
   898→  // Tenta match exato com a key ou com o primeiro/ultimo nome
   899→  for (const key of MOTORISTA_OPTIONS) {
   900→    const keyUpper = key.toUpperCase();
   901→    if (name.includes(keyUpper) || keyUpper.includes(name)) return key;
   902→  }
   903→  // Tenta match com qualquer parte do nome
   904→  for (const part of parts) {
   905→    if (part.length < 3) continue;
   906→    for (const key of MOTORISTA_OPTIONS) {
   907→      if (key.toUpperCase() === part || part.includes(key.toUpperCase())) return key;
   908→    }
   909→  }
   910→  return null;
   911→}
