   105→   105→// XML-RPC connection (singleton)
   106→   106→// ============================================================
   107→   107→var _cached = null;
   108→   108→
   109→   109→function getConn() {
   110→   110→  if (_cached && _cached.uid) return _cached;
   111→   111→  var odooConfig = config.odoo;
   112→   112→  var base = odooConfig.url.replace(/\/+$/, '');
   113→   113→  var host = base.replace('https://', '').replace('http://', '');
   114→   114→  var common = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/common', port: 443 });
   115→   115→  var models = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
   116→   116→  _cached = { uid: null, common: common, models: models, db: odooConfig.db, password: odooConfig.password };
   117→   117→  return _cached;
   118→   118→}
   119→   119→
   120→   120→function authenticate() {
   121→   121→  return new Promise(function(resolve, reject) {
   122→   122→    var conn = getConn();
   123→   123→    conn.common.methodCall('authenticate', [conn.db, config.odoo.user, conn.password, {}], function(err, uid) {
   124→   124→      if (err) { reject(new Error('Auth Odoo falhou: ' + (err.message || JSON.stringify(err)))); }
   125→   125→      else if (uid === false || uid === null) { reject(new Error('Auth Odoo: credenciais invalidas (UID=false)')); }
   126→   126→      else { conn.uid = uid; logger.info('[ODOO-TE] Autenticado. UID: ' + uid); resolve(uid); }
   127→   127→    });
   128→   128→  });
   129→   129→}
   130→   130→
   131→   131→function ensureAuth() {
   132→   132→  var conn = getConn();
   133→   133→  if (conn.uid) return Promise.resolve(conn.uid);
   134→   134→  return authenticate();
   135→   135→}
   136→   136→
   137→   137→function executeKw(model, method, args, kwargs) {
   138→   138→  return new Promise(function(resolve, reject) {
   139→   139→    ensureAuth().then(function(uid) {
   140→   140→      var conn = getConn();
   141→   141→      var params = [conn.db, uid, conn.password, model, method, args || []];
   142→   142→      if (kwargs) params.push(kwargs);
   143→   143→      conn.models.methodCall('execute_kw', params, function(err, result) {
   144→   144→        if (err) { reject(new Error(model + '.' + method + ': ' + (err.message || JSON.stringify(err)))); }
   145→   145→        else { resolve(result); }
   146→   146→      });
   147→   147→    }).catch(reject);
   148→   148→  });
   149→   149→}

   152→   152→// Safe read/write — tenta com x_studio, fallback sem
   153→   153→// ============================================================
   154→   154→var _customFieldsCache = {};
   155→   155→
   156→   156→async function safeReadCustom(model, ids, nativeFields, customFields) {
   157→   157→  try {
   158→   158→    return await executeKw(model, 'read', [[ids, nativeFields.concat(customFields)]]);
   159→   159→  } catch (err) {
   160→   160→    if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   161→   161→      if (!_customFieldsCache[model]) {
   162→   162→        _customFieldsCache[model] = { available: false, checked: true };
   163→   163→        logger.warn('[ODOO-TE] Campos x_studio_* nao encontrados em ' + model + '. Usando campos nativos.');
   164→   164→      }
   165→   165→      return await executeKw(model, 'read', [[ids, nativeFields]]);
   166→   166→    }
   167→   167→    throw err;
   168→   168→  }
   169→   169→}
   170→   170→
   171→   171→async function safeWriteCustom(model, ids, vals) {
   172→   172→  var customVals = {};
   173→   173→  var nativeVals = {};
   174→   174→  Object.keys(vals).forEach(function(k) {
   175→   175→    if (k.indexOf('x_studio_') === 0) customVals[k] = vals[k];
   176→   176→    else nativeVals[k] = vals[k];
   177→   177→  });
   178→   178→  if (Object.keys(nativeVals).length > 0) await executeKw(model, 'write', [ids, nativeVals]);
   179→   179→  if (Object.keys(customVals).length > 0) {
   180→   180→    try { await executeKw(model, 'write', [ids, customVals]); }
   181→   181→    catch (err) {
   182→   182→      if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   183→   183→        if (!_customFieldsCache[model] || !_customFieldsCache[model].warned) {
   184→   184→          logger.warn('[ODOO-TE] Campo x_studio_* nao existe em ' + model + '. Gravacao ignorada.');
   185→   185→          if (!_customFieldsCache[model]) _customFieldsCache[model] = {};
   186→   186→          _customFieldsCache[model].warned = true;
   187→   187→        }
   188→   188→      } else throw err;
   189→   189→    }
   190→   190→  }
   191→   191→}

   394→   394→async function getInvoiceLines(invoiceId) {
   395→   395→  try {
   396→   396→    // Evita account.move.line.search — Odoo 19.2 SaaS tem override de
   397→   397→    // search_fetch com _get_recon_limit_from_domain que rejeita certos dominios.
   398→   398→    // Em vez disso, le line_ids direto da fatura.
   399→   399→    var invData = await executeKw('account.move', 'read', [[[invoiceId], ['line_ids']]]);
   400→   400→    var inv = invData && invData[0];
   401→   401→    if (!inv || !inv.line_ids || !inv.line_ids.length) return [];
   402→   402→
   403→   403→    // line_ids pode vir como [id, id, ...] ou [[id, name], ...]
   404→   404→    var lineIds = inv.line_ids.map(function(item) {
   405→   405→      return Array.isArray(item) ? item[0] : item;
   406→   406→    });
   407→   407→
   408→   408→    var lines = await executeKw('account.move.line', 'read', [[lineIds, INVOICE_LINE_FIELDS]]);
   409→   409→    if (!lines || !lines.length) return [];
   410→   410→
   411→   411→    // Filtra apenas linhas com produto (product_id != false)
   412→   412→    return lines.filter(function(line) {
   413→   413→      if (!line.product_id) return false;
   414→   414→      // product_id pode ser [id, name] ou false
   415→   415→      if (Array.isArray(line.product_id) && line.product_id[0]) return true;
   416→   416→      if (typeof line.product_id === 'number' && line.product_id) return true;
   417→   417→      return false;
   418→   418→    });
   419→   419→  } catch (err) {
   420→   420→    logger.warn('[ODOO-TE] getInvoiceLines falhou: ' + err.message);
   421→   421→    return [];
   422→   422→  }
   423→   423→}
