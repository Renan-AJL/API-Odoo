    10→ *     x_studio_nfe_protocolo  (Char)
    11→ *     x_studio_nfse_numero    (Char)
    12→ *     x_studio_nfse_status    (Selection: vazio, pendente, processando, autorizada, erro)

    15→ *     x_studio_sieg_ultimo_nfse (Integer)
    16→ *     x_studio_sieg_serie_nfe   (Char, default 100)
    17→ *     x_studio_sieg_serie_nfse  (Char, default 1)

    19→ *     x_studio_c_trib_nac       (Char) — Codigo tributo nacional (NFS-e)
    20→ *     x_studio_c_nbs            (Char) — Codigo NBS (NFS-e)

    34→function createClient(url) {
    35→  var base = url.replace(/\/+$/, '');
    36→  var host = base.replace('https://', '').replace('http://', '');
    37→  var port = base.startsWith('https') ? 443 : 80;
    38→  var isSecure = base.startsWith('https');
    39→  var createFn = isSecure ? xmlrpc.createSecureClient : xmlrpc.createClient;
    40→  return {
    41→    common: createFn({ host: host, path: '/xmlrpc/2/common', port: port }),
    42→    models: createFn({ host: host, path: '/xmlrpc/2/object', port: port }),
    43→  };
    44→}
    45→
    46→function authenticate(client, db, login, password) {
    47→  return new Promise(function(resolve, reject) {
    48→    client.common.methodCall('authenticate', [db, login, password, {}], function(err, uid) {
    49→      if (err) reject(new Error('Auth Odoo falhou: ' + (err.message || JSON.stringify(err))));
    50→      else if (uid === false || uid === null) reject(new Error('Credenciais Odoo invalidas'));
    51→      else resolve(uid);
    52→    });
    53→  });
    54→}
    55→
    56→function executeKw(client, db, uid, password, model, method, args, kwargs) {
    57→  return new Promise(function(resolve, reject) {
    58→    var params = [db, uid, password, model, method, args || []];
    59→    if (kwargs) params.push(kwargs);
    60→    client.models.methodCall('execute_kw', params, function(err, result) {
    61→      if (err) reject(new Error(model + '.' + method + ': ' + (err.message || JSON.stringify(err))));
    62→      else resolve(result);
    63→    });
    64→  });
    65→}

    67→// ============================================================
    68→// Main: Process Pending Emissions
    69→// ============================================================
    70→async function processPendingEmissions() {
    71→  var odoo = config.odoo;
    72→  if (!odoo || !odoo.enabled || !odoo.url) {
    73→    console.log('[SIEG-EMIT] Odoo nao configurado');
    74→    return { processed: 0, reason: 'odoo_not_configured' };
    75→  }
    76→
    77→  var client = createClient(odoo.url);
    78→  var uid = await authenticate(client, odoo.db, odoo.user, odoo.password);
    79→  var db = odoo.db;
    80→  var pwd = odoo.password;
    81→
    82→  console.log('[SIEG-EMIT] Buscando faturas pendentes...');
    83→
    84→  // Buscar NF-e pendentes
    85→  var nfeIds = await executeKw(client, db, uid, pwd, 'account.move', 'search', [[
    86→    ['move_type', '=', 'out_invoice'],
    87→    ['x_studio_nfe_status', '=', 'pendente'],
    88→  ]], { order: 'id asc', limit: 5 });
    89→
    90→  // Buscar NFS-e pendentes
    91→  var nfseIds = await executeKw(client, db, uid, pwd, 'account.move', 'search', [[
    92→    ['move_type', '=', 'out_invoice'],
    93→    ['x_studio_nfse_status', '=', 'pendente'],
    94→  ]], { order: 'id asc', limit: 5 });
    95→
    96→  var allPending = [];
    97→  for (var i = 0; i < nfeIds.length; i++) {
    98→    allPending.push({ id: nfeIds[i], tipo: 'nfe' });
    99→  }
   100→  for (var j = 0; j < nfseIds.length; j++) {
   101→    allPending.push({ id: nfseIds[j], tipo: 'nfse' });
   102→  }
   103→
   104→  if (!allPending.length) {
   105→    console.log('[SIEG-EMIT] Nenhuma fatura pendente');
   106→    return { processed: 0 };
   107→  }
   108→
   109→  console.log('[SIEG-EMIT] ' + allPending.length + ' fatura(s) pendente(s)');
   110→  var results = [];
   111→
   112→  for (var k = 0; k < allPending.length; k++) {
   113→    var item = allPending[k];
   114→    try {
   115→      var r = await processOne(client, db, uid, pwd, item.id, item.tipo);
   116→      results.push(r);
   117→    } catch (err) {
   118→      console.error('[SIEG-EMIT] ERRO fatura ' + item.id + ':', err.message);
   119→      await safeUpdateError(client, db, uid, pwd, item.id, item.tipo, err.message);
   120→      results.push({ move_id: item.id, sucesso: false, tipo: item.tipo, erro: err.message });
   121→    }
   122→  }
   123→
   124→  var ok = results.filter(function(r) { return r.sucesso; }).length;
   125→  console.log('[SIEG-EMIT] Concluido: ' + ok + '/' + results.length + ' sucesso');
   126→  return { processed: results.length, sucesso: ok, detalhes: results };
   127→}

   132→async function processOne(client, db, uid, pwd, moveId, tipo) {
   133→  console.log('[SIEG-EMIT] --- Fatura ID: ' + moveId + ' | Tipo: ' + tipo + ' ---');
   134→
   135→  // 1. Read account.move (sem invoice_line_ids — pode ser computed nao-stored no Odoo 19)
   136→  var moves = await executeKw(client, db, uid, pwd, 'account.move', 'read', [[moveId], [
   137→    'name', 'partner_id', 'company_id', 'invoice_date', 'date',
   138→    'amount_total', 'narration',
   139→    'x_studio_nfe_status', 'x_studio_nfse_status', 'payment_state',
   140→  ]]);
   141→  if (!moves || !moves.length) throw new Error('Fatura ' + moveId + ' nao encontrada');
   142→  var move = moves[0];
   143→  console.log('[SIEG-EMIT] Fatura: ' + move.name + ' | R$ ' + (move.amount_total || 0));
   144→
   145→  // 2. Read company
   146→  var companyId = tupId(move.company_id);
   147→  var company = await readCompany(client, db, uid, pwd, companyId);
   148→
   149→  // 3. Read partner
   150→  var partnerId = tupId(move.partner_id);
   151→  if (!partnerId) throw new Error('Fatura sem parceiro');
   152→  var partner = await readPartner(client, db, uid, pwd, partnerId);
   153→
   154→  // 4. Read invoice lines diretamente de account.move.line (evita computed field)
   155→  var allLineIds = await executeKw(client, db, uid, pwd, 'account.move.line', 'search', [[
   156→    ['move_id', '=', moveId],
   157→  ]], { order: 'id asc' });
   158→  console.log('[SIEG-EMIT] Linhas encontradas na fatura: ' + allLineIds.length);
   159→  var rawLines = await executeKw(client, db, uid, pwd, 'account.move.line', 'read', [allLineIds, [
   160→    'display_type', 'product_id', 'name', 'quantity', 'price_unit',
   161→    'price_subtotal', 'tax_ids', 'discount',
   162→  ]]);
   163→  // Debug: log todas as linhas
   164→  for (var li = 0; li < rawLines.length; li++) {
   165→    var l = rawLines[li];
   166→    var pName = 'N/A';
   167→    if (l.product_id) pName = Array.isArray(l.product_id) ? l.product_id[1] : String(l.product_id);
   168→    console.log('[SIEG-EMIT]   Line ' + li + ': display_type=' + JSON.stringify(l.display_type) + ' product=' + pName + ' qty=' + l.quantity + ' price=' + l.price_unit);
   169→  }
   170→  // Filtro principal: linhas com produto associado (mais confiavel que display_type)
   171→  var invoiceLines = rawLines.filter(function(l) { return l.product_id; });
   172→  // Fallback: se nenhuma tem product_id, tenta por display_type
   173→  if (!invoiceLines.length) {
   174→    invoiceLines = rawLines.filter(function(l) {
   175→      var dt = l.display_type;
   176→      return !dt || dt === 'product' || dt === 'service';
   177→    });
   178→  }
   179→  if (!invoiceLines.length) throw new Error('Fatura sem linhas de produto/servico (total: ' + rawLines.length + ', com product: 0)');
   180→  console.log('[SIEG-EMIT] Linhas de produto: ' + invoiceLines.length);
   181→
   182→  // 5. Get next NF number from company
   183→  var serie, numField, nextNum;
   184→  if (tipo === 'nfe') {
   185→    serie = String(company.x_studio_sieg_serie_nfe || '100');
   186→    numField = 'x_studio_sieg_ultimo_nfe';
   187→  } else {
   188→    serie = String(company.x_studio_sieg_serie_nfse || '1');
   189→    numField = 'x_studio_sieg_ultimo_nfse';
   190→  }
   191→  nextNum = (parseInt(company[numField]) || 0) + 1;
   192→  console.log('[SIEG-EMIT] Serie: ' + serie + ' | nNF: ' + nextNum);
   193→
   194→  // 6. Build line data with taxes + product x_studio_ fields
   195→  var linesData = [];
   196→  for (var i = 0; i < invoiceLines.length; i++) {
   197→    var ld = await buildLineData(client, db, uid, pwd, invoiceLines[i]);
   198→    linesData.push(ld);
   199→  }
   200→
   201→  // 7. Build emission payload
   202→  var emitData = {
   203→    company: company,
   204→    partner: partner,
   205→    order: {
   206→      name: move.name,
   207→      number: String(nextNum),
   208→      date_order: move.invoice_date || move.date,
   209→      amount_total: move.amount_total,
   210→      note: move.narration || '',
   211→    },
   212→    lines: linesData,
   213→    config: {
   214→      serie: serie,
   215→      tpAmb: process.env.SIEG_TP_AMB || config.sieg.tpAmb || '2',
   216→      natOp: 'Venda de Mercadoria',
   217→      mod: tipo === 'nfe' ? '55' : '01',
   218→    },
   219→    tipo: tipo,
   220→  };
   221→
   222→  // NFS-e: build service block from product x_studio_ fields
   223→  if (tipo === 'nfse') {
   224→    emitData.service = buildServiceBlock(linesData, move);
   225→  }
   226→
   227→  // 8. Set status to 'processando'
   228→  var processVals = {};
   229→  if (tipo === 'nfe') processVals.x_studio_nfe_status = 'processando';
   230→  else processVals.x_studio_nfse_status = 'processando';
   231→  await executeKw(client, db, uid, pwd, 'account.move', 'write', [[moveId], processVals]);
   232→
   233→  // 9. Call SIEG API
   234→  console.log('[SIEG-EMIT] Enviando ao SIEG...');
   235→  var resultado;
   236→  try {
   237→    resultado = await emitirNota(emitData);
   238→  } catch (siegErr) {
   239→    // Captura erros HTTP (ex: 409 com detalhes no body)
   240→    var siegDetail = siegErr.response && siegErr.response.data;
   241→    var errMsg = 'SIEG HTTP ' + (siegErr.response ? siegErr.response.status : 'erro') + ': ';
   242→    if (siegDetail) {
   243→      // Tentar extrair ErrorMessage do formato SIEG (PascalCase)
   244→      var siegMsg = siegDetail.ErrorMessage || siegDetail.Message || siegDetail.message || '';
   245→      if (siegMsg) {
   246→        errMsg += siegMsg;
   247→      } else {
   248→        errMsg += typeof siegDetail === 'string' ? siegDetail : JSON.stringify(siegDetail).slice(0, 800);
   249→      }
   250→    } else {
   251→      errMsg += siegErr.message;
   252→    }
   253→    console.error('[SIEG-EMIT] Erro SIEG:', errMsg);
   254→    throw new Error(errMsg);
   255→  }
   256→  console.log('[SIEG-EMIT] SIEG retornou - sucesso: ' + resultado.sucesso + (resultado.httpStatus ? ' (HTTP ' + resultado.httpStatus + ')' : ''));
   257→
   258→  // Se SIEG retornou erro (IsSuccess=false), extrair mensagem detalhada
   259→  if (!resultado.sucesso && resultado.erro) {
   260→    console.error('[SIEG-EMIT] SIEG rejeitou o XML: ' + resultado.erro);
   261→  }
   262→
   263→  // 10. Parse result
   264→  var info = parseResult(resultado, tipo);
   265→  var xmlRetornoText = '';
   266→  var rawXmlAuth = resultado.resposta && (resultado.resposta.Xml || resultado.resposta.xml || resultado.resposta.xmlBase64);
   267→  if (rawXmlAuth) {
   268→    xmlRetornoText = (typeof rawXmlAuth === 'string' && rawXmlAuth.length > 200)
   269→      ? rawXmlAuth
   270→      : Buffer.from(rawXmlAuth, 'base64').toString('utf-8');
   271→  }
   272→
   273→  // 11. Update account.move with results
   274→  var updateVals = {};
   275→  if (info.sucesso) {
   276→    if (tipo === 'nfe') {
   277→      updateVals.x_studio_nfe_status = 'autorizada';
   278→      updateVals.x_studio_nfe_chave = info.chave;
   279→      updateVals.x_studio_nfe_protocolo = info.protocolo;
   280→    } else {
   281→      updateVals.x_studio_nfse_status = 'autorizada';
   282→      updateVals.x_studio_nfse_numero = String(nextNum);
   283→    }
   284→
   285→    // Increment NF number (only on authorization)
   286→    try {
   287→      var companyUpdate = {};
   288→      companyUpdate[numField] = nextNum;
   289→      await executeKw(client, db, uid, pwd, 'res.company', 'write', [[companyId], companyUpdate]);
   290→      console.log('[SIEG-EMIT] Sequencia ' + numField + ' = ' + nextNum);
   291→    } catch (seqErr) {
   292→      console.error('[SIEG-EMIT] Erro ao atualizar sequencia:', seqErr.message);
   293→    }
   294→  } else {
   295→    if (tipo === 'nfe') {
   296→      updateVals.x_studio_nfe_status = info.cStat + ' - ' + info.motivo;
   297→    } else {
   298→      updateVals.x_studio_nfse_status = info.motivo;
   299→    }
   300→  }
   301→
   302→  await executeKw(client, db, uid, pwd, 'account.move', 'write', [[moveId], updateVals]);
   303→  console.log('[SIEG-EMIT] Fatura atualizada');
   304→
   305→  // 12. Post XML + PDF to Chatter
   306→  await postChatterResult(client, db, uid, pwd, moveId, move.name, tipo, info, resultado, xmlRetornoText);
   307→
   308→  return {
   309→    move_id: moveId, sucesso: info.sucesso, tipo: tipo,
   310→    chave: info.chave, numero: nextNum, serie: serie,
   311→  };
   312→}

   315→// Post XML + PDF to Odoo Chatter
   316→// ============================================================
   317→async function postChatterResult(client, db, uid, pwd, moveId, moveName, tipo, info, resultado, xmlRetornoText) {
   318→  var attachIds = [];
   319→  var safeName = (moveName || 'Fatura').replace(/[^a-zA-Z0-9\-_]/g, '_');
   320→  var nfLabel = tipo === 'nfe' ? 'NF-e' : 'NFS-e';
   321→  var xmlContent = xmlRetornoText || resultado.xmlEnviado || '';
   322→  var pdfBase64 = resultado.pdfBase64 || '';
   323→
   324→  // Attach XML
   325→  if (xmlContent) {
   326→    try {
   327→      var xmlB64 = Buffer.from(xmlContent, 'utf-8').toString('base64');
   328→      var xmlAttach = await executeKw(client, db, uid, pwd, 'ir.attachment', 'create', [{
   329→        name: nfLabel + '_' + safeName + '.xml',
   330→        datas: xmlB64,
   331→        res_model: 'account.move',
   332→        res_id: moveId,
   333→        mimetype: 'application/xml',
   334→      }]);
   335→      attachIds.push(xmlAttach);
   336→      console.log('[SIEG-EMIT] XML anexado (ID: ' + xmlAttach + ')');
   337→    } catch (e) { console.error('[SIEG-EMIT] Erro anexar XML:', e.message); }
   338→  }
   339→
   340→  // Attach PDF (DANFE / DANFSE)
   341→  if (pdfBase64) {
   342→    try {
   343→      var pdfLabel = tipo === 'nfe' ? 'DANFE' : 'DANFSE';
   344→      var pdfAttach = await executeKw(client, db, uid, pwd, 'ir.attachment', 'create', [{
   345→        name: pdfLabel + '_' + safeName + '.pdf',
   346→        datas: pdfBase64,
   347→        res_model: 'account.move',
   348→        res_id: moveId,
   349→        mimetype: 'application/pdf',
   350→      }]);
   351→      attachIds.push(pdfAttach);
   352→      console.log('[SIEG-EMIT] PDF anexado (ID: ' + pdfAttach + ')');
   353→    } catch (e) { console.error('[SIEG-EMIT] Erro anexar PDF:', e.message); }
   354→  }
   355→
   356→  // Post chatter message
   357→  var body = '';
   358→  if (info.sucesso) {
   359→    body = '<b>' + nfLabel + ' Emitida com Sucesso!</b><br/>';
   360→    if (tipo === 'nfe') {
   361→      body += 'Chave: ' + info.chave + '<br/>';
   362→      body += 'Protocolo: ' + info.protocolo + '<br/>';
   363→    } else {
   364→      body += 'Numero: ' + info.numero + '<br/>';
   365→    }
   366→    body += 'Status: ' + info.cStat + ' - ' + info.motivo;
   367→    body += '<br/><br/><i>Arquivos anexados: XML e ' + (tipo === 'nfe' ? 'DANFE' : 'DANFSE') + ' (PDF)</i>';
   368→  } else {
   369→    body = '<b>Erro na Emissao de ' + nfLabel + '</b><br/>';
   370→    body += 'Status: ' + info.cStat + ' - ' + info.motivo;
   371→  }
   372→
   373→  try {
   374→    var msgVals = {
   375→      model: 'account.move',
   376→      res_id: moveId,
   377→      body: body,
   378→      message_type: 'comment',
   379→    };
   380→    if (attachIds.length > 0) {
   381→      msgVals.attachment_ids = [[6, 0, attachIds]];
   382→    }
   383→    await executeKw(client, db, uid, pwd, 'mail.message', 'create', [msgVals]);
   384→    console.log('[SIEG-EMIT] Mensagem postada no chatter');
   385→  } catch (e) { console.error('[SIEG-EMIT] Erro postar chatter:', e.message); }
   386→}

   395→    'x_studio_sieg_serie_nfe', 'x_studio_sieg_serie_nfse',
   396→  ]]);
   397→  if (!recs || !recs.length) throw new Error('Empresa ' + companyId + ' nao encontrada');
   398→  var c = recs[0];
   399→
   400→  var pId = tupId(c.partner_id);
   401→  var p = {};
   402→  if (pId) {
   403→    try { p = await readPartnerSafe(client, db, uid, pwd, pId); } catch (e) {

   423→    legal_name: p.legal_name || c.name || 'AJL FERRO E ACO LTDA',

   425→    inscr_est: p.inscr_est || '9069585890',
   426→    street: c.street || p.street || '',
   427→    number: p.number || 'S/N',
   428→    street2: p.street2 || p.district || '',
   429→    city: c.city || p.city || 'Curitiba',
   430→    state: stateCode || 'PR',
   431→    zip: c.zip || p.zip || '',
   432→    city_ibge_code: cityIbge || '4106902',
   433→    state_ibge: stateIbge || '41',
   434→    crt: '1',
   435→    phone: c.phone || p.phone || '',
   436→    email: c.email || p.email || '',
   437→    x_studio_sieg_ultimo_nfe: c.x_studio_sieg_ultimo_nfe || 0,
   438→    x_studio_sieg_ultimo_nfse: c.x_studio_sieg_ultimo_nfse || 0,
   439→    x_studio_sieg_serie_nfe: c.x_studio_sieg_serie_nfe || '100',

   444→// ============================================================
   445→// Read Partner (safe for Odoo 19 SaaS — l10n_br fields may not exist)

   478→async function readPartner(client, db, uid, pwd, partnerId) {
   479→  var p = await readPartnerSafe(client, db, uid, pwd, partnerId);
   480→
   481→  var stateCode = '';
   482→  var stId = tupId(p.state_id);
   483→  if (stId) {
   484→    try {
   485→      var sts = await executeKw(client, db, uid, pwd, 'res.country.state', 'read', [[stId], ['code']]);
   486→      if (sts && sts[0]) stateCode = sts[0].code || '';
   487→    } catch (e) {}
   488→  }
   489→
   490→  var cityIbge = '';
   491→  var cityRef = p.l10n_br_city_id || p.city_id;
   492→  if (cityRef && Array.isArray(cityRef)) cityIbge = await readCityIbge(client, db, uid, pwd, cityRef[0]);
   493→
   494→  // Fallback 1: buscar res.city por nome + estado
   495→  if (!cityIbge && p.city && stateCode) {
   496→    cityIbge = await searchCityIbge(client, db, uid, pwd, p.city, stId);
   497→  }
   498→  // Fallback 2: tabela embutida de cidades principais
   499→  if (!cityIbge && p.city && stateCode) {
   500→    cityIbge = lookupIbgeFallback(p.city, stateCode);
   501→  }
   502→  if (!cityIbge) {
   503→    console.warn('[SIEG-EMIT] cMun vazio para parceiro ' + partnerId + ' (' + (p.name || '?') + '): cidade=' + (p.city || '?') + ' UF=' + (stateCode || '?'));
   504→  }
   505→
   506→  return {
   507→    cnpj_cpf: p.cnpj_cpf || p.vat || '',
   508→    legal_name: p.legal_name || p.name || '',
   509→    xNome: p.name || '',
   510→    inscr_est: p.inscr_est || '',
   511→    street: p.street || '',
   512→    number: p.number || 'S/N',
   513→    street2: p.street2 || p.district || '',
   514→    city: p.city || '',
   515→    state: stateCode,
   516→    zip: p.zip || '',
   517→    city_ibge_code: cityIbge,
   518→    phone: p.phone || '',
   519→    email: p.email || '',
   520→    is_consumer: !p.is_company,
   521→    district: p.district || p.street2 || '',
   522→  };

   663→async function extractTaxes(client, db, uid, pwd, line) {
   664→  var result = {
   665→    csosn: '103', cst_icms: '', vbc: 0, vicms: 0, picms: 0,
   666→    cst_pis: '99', vbc_pis: 0, ppis: 0, vpis: 0,
   667→    cst_cofins: '99', vbc_cofins: 0, pcofins: 0, vcofins: 0,
   668→  };
   669→
   670→  var taxIds = (line.tax_ids || []).map(function(t) { return Array.isArray(t) ? t[0] : t; }).filter(Boolean);
   671→  if (!taxIds.length) return result;
   672→
   673→  try {
   674→    var taxes = await executeKw(client, db, uid, pwd, 'account.tax', 'read', [taxIds, [
   675→      'name', 'amount', 'amount_type', 'description',
   676→    ]]);
   677→    if (!taxes || !taxes.length) return result;
   678→
   679→    var base = parseFloat(line.price_subtotal) || 0;
   680→    var totalIcms = 0, totalPis = 0, totalCofins = 0;
   681→
   682→    for (var i = 0; i < taxes.length; i++) {
   683→      var tax = taxes[i];
   684→      var n = ((tax.name || tax.description || '').toUpperCase());
   685→      var amt = parseFloat(tax.amount) || 0;
   686→      if (n.indexOf('ICMS') >= 0 || n.indexOf('CSOSN') >= 0) {
   687→        totalIcms += amt;
   688→        var m = n.match(/(10[0-9]|20[0-9]|300|400|500|900)/);
   689→        if (m) result.csosn = m[1];
   690→      } else if (n.indexOf('PIS') >= 0) { totalPis += amt; }
   691→      else if (n.indexOf('COFINS') >= 0) { totalCofins += amt; }
   692→    }
   693→
   694→    result.picms = totalIcms; result.vbc = base; result.vicms = base * (totalIcms / 100);
   695→    result.ppis = totalPis; result.vbc_pis = base; result.vpis = base * (totalPis / 100);
   696→    result.pcofins = totalCofins; result.vbc_cofins = base; result.vcofins = base * (totalCofins / 100);
   697→  } catch (e) { console.warn('[SIEG-EMIT] Erro ao extrair impostos:', e.message); }
   698→
   699→  return result;
   700→}

   705→function parseResult(resultado, tipo) {
   706→  var resp = resultado.resposta || {};
   707→  var chave = '', protocolo = '', cStat = '', motivo = '', numero = '', sucesso = false;
   708→
   709→  // Se SIEG retornou HTTP 4xx, usar ErrorMessage diretamente como motivo
   710→  var siegErro = resultado.erro || '';
   711→  if (resultado.httpStatus && resultado.httpStatus >= 400 && resultado.httpStatus < 500) {
   712→    // Erro de validacao do XML — extrair mensagem do SIEG
   713→    motivo = siegErro || 'Erro HTTP ' + resultado.httpStatus;
   714→    if (tipo === 'nfe') {
   715→      cStat = String(resultado.statusCode || resp.cStat || resp.status || '');
   716→    }
   717→    return { sucesso: false, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
   718→  }
   719→
   720→  if (tipo === 'nfe') {
   721→    chave = resp.chNFe || resp.chave || resp.ChaveXml || '';
   722→    protocolo = resp.nProt || resp.protocolo || '';
   723→    cStat = String(resp.cStat || resp.status || '');
   724→    motivo = resp.xMotivo || resp.motivo || (siegErro || 'Processado');
   725→    sucesso = (cStat === '100' || cStat === '104' || cStat === '150');
   726→  } else {
   727→    chave = resp.Chave || resp.chave || '';
   728→    protocolo = resp.Protocolo || resp.protocolo || '';
   729→    numero = resp.NumeroNfse || resp.nDFSe || resp.numero || '';
   730→    cStat = String(resp.CodigoVerificacao || resp.codigo || '');
   731→    motivo = resp.Motivo || resp.motivo || siegErro || (resultado.sucesso ? 'Autorizada' : 'Erro na emissao');
   732→    sucesso = resultado.sucesso;
   733→  }
   734→
   735→  return { sucesso: sucesso, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
   736→}
