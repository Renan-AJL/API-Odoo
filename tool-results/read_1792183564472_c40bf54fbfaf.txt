    17→// POST /api/v1/te/deliveries/send — Envia pedidos do ODOO ao TE
    18→// Body: { orderIds?: number[], pickingIds?: number[], orderType?: 1|2|3|4|5 }
    19→// -------------------------------------------------------
    20→router.post('/deliveries/send', async (req, res, next) => {
    21→  try {
    22→    const { orderIds, pickingIds, orderType = 1 } = req.body;
    23→
    24→    if (!orderIds?.length && !pickingIds?.length) {
    25→      return res.status(400).json({ success: false, error: 'Informe orderIds ou pickingIds' });
    26→    }
    27→
    28→    // 1. Busca pedidos ODOO
    29→    const pickings = [];
    30→    const saleOrders = [];
    31→
    32→    if (pickingIds?.length) {
    33→      const found = await odooTe.read('stock.picking', pickingIds, [
    34→        'id', 'name', 'partner_id', 'scheduled_date', 'move_line_count', 'origin',
    35→      ]);
    36→      pickings.push(...found);
    37→    }
    38→
    39→    if (orderIds?.length) {
    40→      const soFields = odooTe.getStudioFields('sale.order');
    41→      const found = await odooTe.read('sale.order', orderIds, [
    42→        'id', 'name', 'state', 'partner_id', 'amount_total', 'note', ...soFields,
    43→      ]);
    44→      saleOrders.push(...found);
    45→    }
    46→
    47→    // 2. Busca parceiros
    48→    const partnerIds = [
    49→      ...pickings.map(p => p.partner_id?.[0]),
    50→      ...saleOrders.map(o => o.partner_id?.[0]),
    51→    ].filter(Boolean);
    52→
    53→    const partners = partnerIds.length > 0
    54→      ? await odooTe.read('res.partner', [...new Set(partnerIds)], [
    55→          'id', 'name', 'cnpj_cpf', 'vat', 'phone', 'mobile', 'email',
    56→          'street', 'street_number', 'street2', 'zip', 'city',
    57→          'l10n_br_district', 'partner_latitude', 'partner_longitude',
    58→          'state_id', 'country_id',
    59→          ...odooTe.getStudioFields('res.partner'),
    60→        ])
    61→      : [];
    62→
    63→    const partnerMap = {};
    64→    partners.forEach(p => { partnerMap[p.id] = p; });
    65→
    66→    // 3. Mapeia para entregas TE
    67→    const teDeliveries = [];
    68→
    69→    for (const so of saleOrders) {
    70→      const partner = partnerMap[so.partner_id?.[0]] || {};
    71→      const picking = pickings.find(p => p.origin === so.name) || null;
    72→      const teDelivery = Mapper.odooToTeDelivery(so, partner, picking);
    73→      teDelivery.OrderType = orderType;
    74→      teDeliveries.push(teDelivery);
    75→    }
    76→
    77→    for (const picking of pickings) {
    78→      if (saleOrders.some(so => so.name === picking.origin)) continue; // ja mapeado
    79→      const partner = partnerMap[picking.partner_id?.[0]] || {};
    80→      const fakeSo = { name: picking.origin || `WH-${picking.id}`, note: '', id: 0 };
    81→      const teDelivery = Mapper.odooToTeDelivery(fakeSo, partner, picking);
    82→      teDelivery.OrderType = orderType;
    83→      teDeliveries.push(teDelivery);
    84→    }
    85→
    86→    if (teDeliveries.length === 0) {
    87→      return res.json({ success: true, sent: 0, message: 'Nenhuma entrega para enviar' });
    88→    }
    89→
    90→    // 4. Envia ao TE em lotes de 50
    91→    const results = [];
    92→    const BATCH = 50;
    93→
    94→    for (let i = 0; i < teDeliveries.length; i += BATCH) {
    95→      const batch = teDeliveries.slice(i, i + BATCH);
    96→      const result = await teClient.createDeliveries(batch);
    97→      results.push(result);
    98→
    99→      // 5. Marca como sincronizado no ODOO
   100→      const batchOrderNums = batch.map(d => d.OrderNumber);
   101→
   102→      const syncedSo = saleOrders.filter(so =>
   103→        batchOrderNums.includes(so.name) || batchOrderNums.includes(`WH-${so.id}`)
   104→      );
   105→      const syncedPickings = pickings.filter(p =>
   106→        batchOrderNums.includes(p.origin) || batchOrderNums.includes(`WH-${p.id}`)
   107→      );
   108→
   109→      if (syncedSo.length > 0) {
   110→        await odooTe.markSaleOrdersSynced(syncedSo.map(s => s.id), batchOrderNums[0], orderType);
   111→      }
   112→      if (syncedPickings.length > 0) {
   113→        await odooTe.markPickingsSynced(syncedPickings.map(p => p.id), batchOrderNums[0]);
   114→      }
   115→    }
   116→
   117→    console.log(`[TE-DELIVERY] Enviadas ${teDeliveries.length} entregas ao TE`);
   118→    res.json({
   119→      success: true,
   120→      sent: teDeliveries.length,
   121→      batches: results.length,
   122→      results,
   123→    });
   124→  } catch (err) {
   125→    console.error('[TE-DELIVERY] Erro:', err.message);
   126→    const status = err.isAxiosError ? (err.response?.status || 502) : 500;
   127→    res.status(status).json({ success: false, error: err.message });
   128→  }
   129→});

   183→    res.json({
   184→      success: true,

   186→      situation: mapped.situation,
   187→      situationLabel: mapped.situationLabel,
   188→      trackingCode: mapped.trackingCode,
   189→      driverName: mapped.driverName,
   190→      odooUpdated: !!(picking || saleOrder),
   191→      raw: delivery,
   192→    });
   193→  } catch (err) {
   194→    console.error('[TE-DELIVERY] Erro status:', err.message);

   280→// GET /api/v1/te/deliveries/pull — Puxa status e atualiza ODOO
   281→// -------------------------------------------------------
   282→router.get('/deliveries/pull', async (req, res) => {
   283→  try {
   284→    const { dateFrom, dateTo, situation } = req.query;
   285→
   286→    const params = { page: 1 };
   287→    if (dateFrom) params.dateFrom = dateFrom;
   288→    if (dateTo) params.dateTo = dateTo;
   289→    if (situation) params.situation = parseInt(situation, 10);
   290→
   291→    const allDeliveries = await teClient.fetchAllPages('/api/Entregas', params);
   292→
   293→    let updated = 0;
   294→    let notFound = 0;
   295→
   296→    for (const d of allDeliveries) {
   297→      const orderNum = d.OrderNumber || d.orderNumber;
   298→      const picking = await odooTe.findPickingByTeId(orderNum);
   299→
   300→      if (picking) {
   301→        const mapped = Mapper.teToOdooPicking(d);
   302→        await odooTe.updatePickingTeData(picking.id, mapped);
   303→        updated++;
   304→      } else {
   305→        const so = await odooTe.findSaleOrderByTeId(orderNum);
   306→        if (so) {
   307→          await odooTe.updateSaleOrderTeData(so.id, Mapper.teToOdooSaleOrder(d));
   308→          updated++;
   309→        } else {
   310→          notFound++;
   311→        }
   312→      }
   313→    }
   314→
   315→    res.json({
   316→      success: true,
   317→      totalFetched: allDeliveries.length,
   318→      updatedInOdoo: updated,
   319→      notFoundInOdoo: notFound,
   320→    });
   321→  } catch (err) {
   322→    console.error('[TE-DELIVERY] Erro pull:', err.message);
   323→    res.status(500).json({ success: false, error: err.message });
   324→  }
   325→});

   359→function fmtDate(dateStr) {
   360→  if (!dateStr) return '';
   361→  try {
   362→    const d = new Date(dateStr);
   363→    if (isNaN(d.getTime())) return dateStr;
   364→    return d.toLocaleString('pt-BR', { timeZone: 'America/Sao_Paulo',
   365→      day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit' });
   366→  } catch { return dateStr; }
   367→}

   818→router.post('/delivery-status', async (req, res) => {
   819→  try {
   820→    const { saleOrderId } = req.body;
   821→    if (!saleOrderId) {
   822→      return res.status(400).json({ success: false, error: 'Informe saleOrderId' });
   823→    }
   824→
   825→    // 1. Busca sale order no Odoo
   826→    const so = await odooTe.findSaleOrderById(saleOrderId);
   827→    if (!so) {
   828→      return res.status(404).json({ success: false, error: `Sale Order ${saleOrderId} nao encontrada` });
   829→    }
   830→
   831→    const f = odooTe.FIELDS['sale.order'];
   832→    const orderName = so.name;
   833→    const teOrderId = so[f.teOrderId];
   834→    const trackingCode = so[f.teTrackingCode];
   835→
   836→    if (!teOrderId && !trackingCode) {
   837→      // Nenhuma entrega enviada ao TE ainda
   838→      const noCard = buildErrorCard(
   839→        'Sem Entrega TE',
   840→        `Pedido ${orderName} ainda nao foi enviado ao TudoEntregue. Envie a entrega primeiro.`
   841→      );
   842→      await odooTe.updateSaleOrderStatusHtml(saleOrderId, noCard);
   843→      return res.json({ success: true, message: 'Pedido ainda nao enviado ao TE', card: noCard });
   844→    }
   845→
   846→    // 2. Busca dados no TE — /api/Entregas (dados completos da entrega)
   847→    let deliveryData = null;
   848→    if (teOrderId) {
   849→      const result = await teClient.getDeliveries({ orderNumber: teOrderId, page: 1 });
   850→      deliveryData = result?.Result?.[0] || null;
   851→    }
   852→
   853→    // 3. Busca situacao detalhada — /orders/situation
   854→    let situationData = null;
   855→    if (teOrderId) {
   856→      try {
   857→        situationData = await teClient.getOrderSituation({
   858→          orderType: 1,
   859→          orderID: teOrderId,
   860→        });
   861→      } catch (err) {
   862→        console.warn(`[TE-DELIVERY-STATUS] /orders/situation falhou: ${err.message}`);
   863→      }
   864→    }
   865→
   866→    // 4. Busca tracking detalhado (ocorrencias com fotos)
   867→    let trackingData = null;
   868→    if (trackingCode) {
   869→      try {
   870→        trackingData = await teClient.getTracking(trackingCode);
   871→      } catch (err) {
   872→        console.warn(`[TE-DELIVERY-STATUS] /tracking falhou: ${err.message}`);
   873→      }
   874→    }
   875→
   876→    // 5. Monta card HTML
   877→    const cardHtml = buildTeCard(deliveryData || {}, trackingData, situationData);
   878→
   879→    // 6. Tambem atualiza campos padrao do TE
   880→    if (deliveryData) {
   881→      const mapped = Mapper.teToOdooPicking(deliveryData);
   882→      const soMapped = Mapper.teToOdooSaleOrder(deliveryData);
   883→      await odooTe.updateSaleOrderTeData(saleOrderId, soMapped);
   884→    }
   885→
   886→    // 7. Atualiza campo de motorista (selection) a partir do TE
   887→    let motoristaAtualizado = null;
   888→    if (deliveryData) {
   889→      const driverName = deliveryData.DriverName || deliveryData.driverName || '';
   890→      if (driverName) {
   891→        motoristaAtualizado = matchMotoristaSelection(driverName);
   892→        if (motoristaAtualizado) {
   893→          await odooTe.updateSaleOrderMotorista(saleOrderId, motoristaAtualizado);
   894→          console.log(`[TE-DELIVERY-STATUS] Motorista: ${driverName} -> ${motoristaAtualizado}`);
   895→        }
   896→      }
   897→    }
   898→
   899→    // 8. Grava card no campo HTML
   900→    await odooTe.updateSaleOrderStatusHtml(saleOrderId, cardHtml);
   901→
   902→    console.log(`[TE-DELIVERY-STATUS] Card atualizado: SO ${orderName} (${saleOrderId})`);
   903→    res.json({
   904→      success: true,
   905→      orderName,
   906→      teOrderId,
   907→      trackingCode,
   908→      motorista: motoristaAtualizado || null,
   909→      hasDeliveryData: !!deliveryData,
   910→      hasSituationData: !!situationData,
   911→      hasTrackingData: !!trackingData,
   912→      card: cardHtml,
   913→    });
   914→  } catch (err) {
   915→    console.error('[TE-DELIVERY-STATUS] Erro:', err.message);
   916→    // Grava card de erro no campo
   917→    try {
   918→      await odooTe.updateSaleOrderStatusHtml(req.body.saleOrderId, buildErrorCard('Erro ao Buscar Status', err.message));
   919→    } catch {}
   920→    const status = err.isAxiosError ? (err.response?.status || 502) : 500;
   921→    res.status(status).json({ success: false, error: err.message });
   922→  }
   923→});

   925→// ============================================================
   926→// MATCH MOTORISTA — Faz match do nome do TE com as opcoes do campo selection
   927→// O campo x_studio_motorista e um selection com 17 opcoes (key/label)
   928→// ============================================================
   929→const MOTORISTA_OPTIONS = [
   930→  'adelson', 'adriano', 'anderlucio', 'antonio', 'carlos', 'daniel',
   931→  'diego', 'eduardo', 'fellipe', 'gerson', 'glauber',
   932→  'henrique', 'italo', 'jeferson', 'joao_pedro', 'lucas',
   933→  'matheus', 'rafael',
   934→];
   935→
   936→function matchMotoristaSelection(driverNameFromTe) {
   937→  if (!driverNameFromTe) return null;
   938→  const name = driverNameFromTe.toUpperCase().trim();
   939→  // Remove sufixos comuns como "MOTORISTA", "FRETE", etc
   940→  const cleanName = name
   941→    .replace(/\b(MOTORISTA|FRETE|ENTREGAS|MOTOBOY)\b/g, '')
   942→    .replace(/[^A-ZÀ-Ú\s]/g, '')
   943→    .trim();
   944→  const parts = cleanName.split(/\s+/).filter(Boolean);
   945→
   946→  // Tenta match exato com a key ou com o primeiro/ultimo nome
   947→  for (const key of MOTORISTA_OPTIONS) {
   948→    const keyUpper = key.toUpperCase();
   949→    if (name.includes(keyUpper) || keyUpper.includes(name)) return key;
   950→  }
   951→  // Tenta match com qualquer parte do nome
   952→  for (const part of parts) {
   953→    if (part.length < 3) continue;
   954→    for (const key of MOTORISTA_OPTIONS) {
   955→      if (key.toUpperCase() === part || part.includes(key.toUpperCase())) return key;
   956→    }
   957→  }
   958→  return null;
   959→}
