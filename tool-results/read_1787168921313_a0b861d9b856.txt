    35→    35→function detectarCartao(formaPag) {
    36→    36→  if (!formaPag) return null;
    37→    37→  var n = formaPag.trim().toUpperCase();
    38→    38→  for (var i = 0; i < BANDEIRAS.length; i++) {
    39→    39→    var b = BANDEIRAS[i];
    40→    40→    if (n.indexOf(b) === 0 || n.indexOf(b + ' ') === 0) {
    41→    41→      var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
    42→    42→      var parcelas = 1;
    43→    43→      var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
    44→    44→      if (match) parcelas = parseInt(match[1]);
    45→    45→      return { bandeira: b, parcelas: parcelas, debito: debito };
    46→    46→    }
    47→    47→  }
    48→    48→  // Generico: CARTAO, CARTAO CREDITO, CREDITO, etc.
    49→    49→  if (n.indexOf('CART') === 0 || n.indexOf('CREDITO') === 0) {
    50→    50→    var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
    51→    51→    var parcelas = 1;
    52→    52→    var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
    53→    53→    if (match) parcelas = parseInt(match[1]);
    54→    54→    return { bandeira: 'CARTAO', parcelas: parcelas, debito: debito };
    55→    55→  }
    56→    56→  return null;
    57→    57→}

   464→   464→// === PIX: handler para /pagar ===
   465→   465→async function handlePix(req, res, d) {
   466→   466→  var fat = d.fatura || {};
   467→   467→  var faturaName = fat.name || fat.seu_numero || d.fatura_name || '';
   468→   468→  var faturaId = (fat.id || d.fatura_id) ? parseInt(String(fat.id || d.fatura_id)) : 0;
   469→   469→  var valorTotal = parseFloat(fat.valor_nominal || d.fatura_valor) || 0;
   470→   470→  var pag = d.pagador || {};
   471→   471→  var pagNome = pag.nome || d.pagador_nome || '';
   472→   472→  var pagCpf = pag.cpf_cnpj || d.pagador_cpf || '';
   473→   473→
   474→   474→  console.log('[API/PIX] Fatura:', faturaName, '| Valor:', valorTotal);
   475→   475→
   476→   476→  if (valorTotal <= 0) {
   477→   477→    return res.json({ success: false, message: 'Valor invalido para PIX' });
   478→   478→  }
   479→   479→
   480→   480→  // === DEDUPLICACAO: verificar se ja tem requisicao em curso ou resultado recente ===
   481→   481→  var lockKey = getPixLockKey(faturaId, faturaName);
   482→   482→
   483→   483→  // Se tem resultado cacheado (mesma fatura, menos de 60s atras), reusa
   484→   484→  if (_pixCache[lockKey]) {
   485→   485→    var cached = _pixCache[lockKey];
   486→   486→    var age = Date.now() - cached.timestamp;
   487→   487→    if (age < PIX_CACHE_TTL) {
   488→   488→      console.log('[API/PIX] REUSANDO resultado cacheado (' + (age / 1000).toFixed(1) + 's atras) para', lockKey);
   489→   489→      return res.json(cached.response);
   490→   490→    }
   491→   491→    delete _pixCache[lockKey];
   492→   492→  }
   493→   493→
   494→   494→  // Se tem requisicao em curso, espera ela terminar
   495→   495→  if (_pixLocks[lockKey]) {
   496→   496→    console.log('[API/PIX] AGUARDANDO lock existente para', lockKey);
   497→   497→    try {
   498→   498→      var cachedResult = await _pixLocks[lockKey];
   499→   499→      console.log('[API/PIX] Lock liberado, retornando resultado da requisicao original');
   500→   500→      return res.json(cachedResult);
   501→   501→    } catch (lockErr) {
   502→   502→      console.log('[API/PIX] Requisicao original falhou, tentando novamente...');
   503→   503→    }
   504→   504→  }
   505→   505→
   506→   506→  // Criar lock
   507→   507→  var lockResolve;
   508→   508→  _pixLocks[lockKey] = new Promise(function(resolve) { lockResolve = resolve; });
   509→   509→
   510→   510→  // Auto-limpar lock apos TTL
   511→   511→  setTimeout(function() {
   512→   512→    if (_pixLocks[lockKey]) {
   513→   513→      console.warn('[API/PIX] Lock expirou por TTL para', lockKey);
   514→   514→      delete _pixLocks[lockKey];
   515→   515→    }
   516→   516→  }, PIX_LOCK_TTL);
   517→   517→
   518→   518→  try {
   519→   519→    var isCpf = pagCpf.replace(/\\D/g, '').length <= 11;
   520→   520→    var pixResult = await criarCobrancaPix({
   521→   521→      valor: valorTotal,
   522→   522→      devedor: pagNome ? {
   523→   523→        nome: pagNome,
   524→   524→        cpf: isCpf ? pagCpf.replace(/\\D/g, '') : null,
   525→   525→        cnpj: !isCpf ? pagCpf.replace(/\\D/g, '') : null,
   526→   526→      } : undefined,
   527→   527→      solicitacaoPagador: faturaName || '',
   528→   528→      expiracao: 86400,
   529→   529→    });
   530→   530→
   531→   531→    console.log('[API/PIX] PIX criado: TXID=' + (pixResult.txid || 'N/A'));
   532→   532→
   533→   533→    var pixCopiaCola = pixResult.pixCopiaECola || '';
   534→   534→    var qrcodeBase64 = '';
   535→   535→    var htmlPix = '';
   536→   536→
   537→   537→    // Gerar QR code a partir do pixCopiaECola (bwip-js v4 e async)
   538→   538→    if (pixCopiaCola) {
   539→   539→      try {
   540→   540→        var qrPng = await bwipjs.toBuffer({
   541→   541→          bcid: 'qrcode',
   542→   542→          text: pixCopiaCola,
   543→   543→          scale: 5,
   544→   544→          width: 12,
   545→   545→          height: 12,
   546→   546→        });
   547→   547→        qrcodeBase64 = 'data:image/png;base64,' + qrPng.toString('base64');
   548→   548→        console.log('[API/PIX] QR code gerado:', (qrPng.length / 1024).toFixed(1) + 'KB');
   549→   549→      } catch (qrErr) {
   550→   550→        console.error('[API/PIX] Erro ao gerar QR code:', qrErr.message);
   551→   551→      }
   552→   552→    }
   553→   553→
   554→   554→    // Montar HTML para o campo Odoo
   555→   555→    var valorFmt = 'R$ ' + valorTotal.toFixed(2).replace('.', ',');
   556→   556→    htmlPix = '<div style="text-align:center; font-family:Arial,sans-serif; padding:10px;">' +
   557→   557→      '<div style="font-size:18px; font-weight:bold; color:#333; margin-bottom:8px;">PIX</div>' +
   558→   558→      '<div style="font-size:14px; color:#666; margin-bottom:12px;">' + faturaName + ' - ' + valorFmt + '</div>' +
   559→   559→      (qrcodeBase64 ? '<img src="' + qrcodeBase64 + '" style="width:200px; height:200px; margin:0 auto 12px auto; display:block;" />' : '') +
   560→   560→      (pixCopiaCola ? '<div style="font-size:10px; color:#999; word-break:break-all; max-width:400px; margin:0 auto; line-height:1.4;">' + pixCopiaCola + '</div>' : '') +
   561→   561→      '</div>';
   562→   562→
   563→   563→    // Montar responseData ANTES do push (responde pro Odoo imediatamente)
   564→   564→    var responseData = {
   565→   565→      success: true,
   566→   566→      data: {
   567→   567→        forma_pagamento: 'PIX',
   568→   568→        total_parcelas: 1,
   569→   569→        valor_total: valorTotal.toFixed(2),
   570→   570→        fatura_name: faturaName || '(nao informado)',
   571→   571→        fatura_id: faturaId || 0,
   572→   572→        pagamentos: [{
   573→   573→          tipo: 'pix',
   574→   574→          parcela: 1,
   575→   575→          total_parcelas: 1,
   576→   576→          valor_titulo: valorTotal.toFixed(2),
   577→   577→          txid: pixResult.txid || '',
   578→   578→          pix_copia_cola: pixCopiaCola,
   579→   579→          qrcode_base64: qrcodeBase64,
   580→   580→          html_pix: htmlPix,
   581→   581→        }],
   582→   582→        odoo_push: 'pending',
   583→   583→      },
   584→   584→    };
   585→   585→
   586→   586→    // Cache resultado e liberar lock
   587→   587→    _pixCache[lockKey] = { timestamp: Date.now(), response: responseData };
   588→   588→    delete _pixLocks[lockKey];
   589→   589→    lockResolve(responseData);
   590→   590→
   591→   591→    // Responde pro Odoo IMEDIATAMENTE (o modulo Odoo grava os campos)
   592→   592→    res.json(responseData);
   593→   593→  } catch (err) {
   594→   594→    console.error('[API/PIX] ERRO:', err.message);
   595→   595→    var errResponse = { success: false, message: 'Erro ao criar PIX: ' + (err.message || 'Erro desconhecido') };
   596→   596→    delete _pixLocks[lockKey];
   597→   597→    lockResolve(errResponse);
   598→   598→    res.json(errResponse);
   599→   599→  }
   600→   600→}
