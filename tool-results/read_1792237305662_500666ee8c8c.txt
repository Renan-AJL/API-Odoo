    31→// --- Deteccao de cartao por bandeira ---
    32→var BANDEIRAS = ['VISA', 'MASTER', 'ELO', 'AMEX', 'HIPERCARD', 'HIPER'];

    34→function detectarCartao(formaPag) {
    35→  if (!formaPag) return null;
    36→  var n = formaPag.trim().toUpperCase();
    37→  for (var i = 0; i < BANDEIRAS.length; i++) {
    38→    var b = BANDEIRAS[i];
    39→    if (n.indexOf(b) === 0 || n.indexOf(b + ' ') === 0) {
    40→      var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
    41→      var parcelas = 1;
    42→      var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
    43→      if (match) parcelas = parseInt(match[1]);
    44→      return { bandeira: b, parcelas: parcelas, debito: debito };
    45→    }
    46→  }
    47→  // Generico: CARTAO, CARTAO CREDITO, CREDITO, etc.
    48→  if (n.indexOf('CART') === 0 || n.indexOf('CREDITO') === 0) {
    49→    var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
    50→    var parcelas = 1;
    51→    var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
    52→    if (match) parcelas = parseInt(match[1]);
    53→    return { bandeira: 'CARTAO', parcelas: parcelas, debito: debito };
    54→  }
    55→  return null;
    56→}

    59→async function handleCartao(req, res, d, cartaoInfo) {

    64→  var pag = d.pagador || {};
    65→
    66→  console.log('[API/CARTAO] Fatura:', faturaName, '| Valor:', valorTotal, '| Bandeira:', cartaoInfo.bandeira);
    67→
    68→  if (valorTotal <= 0) {
    69→    return res.json({ success: false, message: 'Valor invalido para pagamento em cartao' });
    70→  }
    71→
    72→  if (!config.rede.pv || !config.rede.chaveIntegracao) {
    73→    console.error('[API/CARTAO] Rede nao configurada. PV:', config.rede.pv ? 'OK' : 'faltando',
    74→      '| Chave:', config.rede.chaveIntegracao ? 'OK' : 'faltando');
    75→    return res.json({
    76→      success: false,
    77→      message: 'Rede nao configurada. Defina REDE_PV e REDE_CHAVE_INTEGRACAO nas variaveis de ambiente.',
    78→    });
    79→  }
    80→
    81→  try {
    82→    var linkResult = await criarLinkPagamento({
    83→      valor: valorTotal,
    84→      seu_numero: faturaName || String(Date.now()),
    85→      descricao: 'Fatura ' + faturaName,
    86→      parcelas: cartaoInfo.parcelas,
    87→      nome_pagador: pag.nome || d.pagador_nome || '',
    88→      cpf_cnpj_pagador: pag.cpf_cnpj || d.pagador_cpf || '',
    89→      email_pagador: pag.email || '',
    90→      expiracao: 7 * 86400,
    91→      host: req.protocol + '://' + req.get('host'),
    92→    });
    93→
    94→    var checkoutUrl = linkResult.link || '';
    95→
    96→    if (!checkoutUrl) {
    97→      console.error('[API/CARTAO] Checkout criado mas sem URL. Resposta bruta:', JSON.stringify(linkResult.raw));
    98→      return res.json({
    99→        success: false,
   100→        message: 'Checkout criado porem sem URL de pagamento. Verificar resposta da Rede.',
   101→        detail: linkResult.raw,
   102→      });
   103→    }
   104→
   105→    console.log('[API/CARTAO] Checkout URL gerada:', checkoutUrl.substring(0, 80) + '...');
   106→
   107→    res.json({
   108→      success: true,
   109→      data: {
   110→        forma_pagamento: d.forma_pagamento,
   111→        total_parcelas: 1,
   112→        valor_total: valorTotal.toFixed(2),
   113→        fatura_name: faturaName || '(nao informado)',
   114→        fatura_id: faturaId || 0,
   115→        pagamentos: [{
   116→          tipo: cartaoInfo.debito ? 'cartao_debito' : 'cartao_credito',
   117→          parcela: 1,
   118→          total_parcelas: 1,
   119→          valor_titulo: valorTotal.toFixed(2),
   120→          bandeira: cartaoInfo.bandeira,
   121→          parcelas: cartaoInfo.parcelas,
   122→          pix_copia_cola: checkoutUrl,
   123→          link_checkout: checkoutUrl,
   124→          checkout_id: linkResult.id || '',
   125→        }],
   126→        odoo_push: 'nao_aplicavel',
   127→      }
   128→    });
   129→
   130→  } catch (err) {
   131→    console.error('[API/CARTAO] ERRO:', err.message);
   132→    if (err.detail) console.error('[API/CARTAO] Detalhes:', JSON.stringify(err.detail));
   133→    res.json({
   134→      success: false,
   135→      message: 'Erro ao gerar link de pagamento: ' + (err.message || 'Erro desconhecido'),
   136→    });
   137→  }

   140→function calcDataVenc(base, dias) {
   141→  var db = base ? new Date(base + 'T12:00:00') : new Date();
   142→  var hj = new Date(); hj.setHours(0,0,0,0);
   143→  if (db <= hj) db = new Date();
   144→  db.setDate(db.getDate() + dias);
   145→  var a = db.getFullYear();
   146→  var m = String(db.getMonth()+1).padStart(2,'0');
   147→  var d = String(db.getDate()).padStart(2,'0');
   148→  return a+'-'+m+'-'+d;
   149→}

   434→    var sn = req.body.seu_numero || nn;
   435→    var parc = req.body.parcela || '';
   436→    var tp = req.body.total_parcelas || '';
   437→
   438→    if (!nn && !ld) {
   439→      res.setHeader('Content-Type', 'text/plain; charset=utf-8');
   440→      res.send('ERRO|Campos obrigatorios: nosso_numero ou linha_digitavel');
   441→      return;
   442→    }
   443→
   444→    var dados = {
   445→      nosso_numero: nn, linha_digitavel: ld, codigo_barras: cb,
   446→      pix_copia_cola: pix, valor_titulo: vd, data_vencimento: vc,
   447→      nome_pagador: pn, cpf_cnpj_pagador: pc, seu_numero: sn,
   448→      parcela: parc ? parseInt(parc) : 0, total_parcelas: tp ? parseInt(tp) : 0,
   449→    };
   450→
   451→    var pdfBuf = await generatePdfFromFields(dados);
   452→    var b64 = pdfBuf.toString('base64');
   453→
   454→    res.setHeader('Content-Type', 'text/plain; charset=utf-8');
   455→    res.send('OK|' + b64);

   457→    res.setHeader('Content-Type', 'text/plain; charset=utf-8');
   458→    res.send('ERRO|' + e.message);

   462→// === PIX: handler para /pagar ===
   463→async function handlePix(req, res, d) {
   464→  var fat = d.fatura || {};
   465→  var faturaName = fat.name || fat.seu_numero || d.fatura_name || '';
   466→  var faturaId = (fat.id || d.fatura_id) ? parseInt(String(fat.id || d.fatura_id)) : 0;
   467→  var valorTotal = parseFloat(fat.valor_nominal || d.fatura_valor) || 0;
   468→  var pag = d.pagador || {};
   469→  var pagNome = pag.nome || d.pagador_nome || '';
   470→  var pagCpf = pag.cpf_cnpj || d.pagador_cpf || '';
   471→
   472→  console.log('[API/PIX] Fatura:', faturaName, '| Valor:', valorTotal);
   473→
   474→  if (valorTotal <= 0) {
   475→    return res.json({ success: false, message: 'Valor invalido para PIX' });
   476→  }
   477→
   478→  // === DEDUPLICACAO: verificar se ja tem requisicao em curso ou resultado recente ===
   479→  var lockKey = getPixLockKey(faturaId, faturaName);
   480→
   481→  // Se tem resultado cacheado (mesma fatura, menos de 60s atras), reusa
   482→  if (_pixCache[lockKey]) {
   483→    var cached = _pixCache[lockKey];
   484→    var age = Date.now() - cached.timestamp;
   485→    if (age < PIX_CACHE_TTL) {
   486→      console.log('[API/PIX] REUSANDO resultado cacheado (' + (age / 1000).toFixed(1) + 's atras) para', lockKey);
   487→      return res.json(cached.response);
   488→    }
   489→    delete _pixCache[lockKey];
   490→  }
   491→
   492→  // Se tem requisicao em curso, espera ela terminar
   493→  if (_pixLocks[lockKey]) {
   494→    console.log('[API/PIX] AGUARDANDO lock existente para', lockKey);
   495→    try {
   496→      var cachedResult = await _pixLocks[lockKey];
   497→      console.log('[API/PIX] Lock liberado, retornando resultado da requisicao original');
   498→      return res.json(cachedResult);
   499→    } catch (lockErr) {
   500→      console.log('[API/PIX] Requisicao original falhou, tentando novamente...');
   501→    }
   502→  }
   503→
   504→  // Criar lock
   505→  var lockResolve;
   506→  _pixLocks[lockKey] = new Promise(function(resolve) { lockResolve = resolve; });
   507→
   508→  // Auto-limpar lock apos TTL
   509→  setTimeout(function() {
   510→    if (_pixLocks[lockKey]) {
   511→      console.warn('[API/PIX] Lock expirou por TTL para', lockKey);
   512→      delete _pixLocks[lockKey];
   513→    }
   514→  }, PIX_LOCK_TTL);
   515→
   516→  try {
   517→    var devedor = buildDevedorFromDoc(pagNome, pagCpf);
   518→    if (!devedor && pagNome) {
   519→      console.warn('[API/PIX] Pagador com nome mas sem CPF/CNPJ valido, criando PIX sem devedor');
   520→    }
   521→    var pixResult = await criarCobrancaPix({
   522→      valor: valorTotal,
   523→      devedor: devedor,
   524→      solicitacaoPagador: faturaName || '',
   525→      expiracao: 86400,
   526→    });
   527→
   528→    console.log('[API/PIX] PIX criado: TXID=' + (pixResult.txid || 'N/A'));
   529→
   530→    var pixCopiaCola = pixResult.pixCopiaECola || '';
   531→    var qrcodeBase64 = '';
   532→    var htmlPix = '';
   533→
   534→    // Gerar QR code a partir do pixCopiaECola (bwip-js v4 e async)
   535→    if (pixCopiaCola) {
   536→      try {
   537→        var qrPng = await bwipjs.toBuffer({
   538→          bcid: 'qrcode',
   539→          text: pixCopiaCola,
   540→          scale: 5,
   541→          width: 12,
   542→          height: 12,
   543→        });
   544→        qrcodeBase64 = 'data:image/png;base64,' + qrPng.toString('base64');
   545→        console.log('[API/PIX] QR code gerado:', (qrPng.length / 1024).toFixed(1) + 'KB');
   546→      } catch (qrErr) {
   547→        console.error('[API/PIX] Erro ao gerar QR code:', qrErr.message);
   548→      }
   549→    }
   550→
   551→    // Montar HTML para o campo Odoo
   552→    var valorFmt = 'R$ ' + valorTotal.toFixed(2).replace('.', ',');
   553→    htmlPix = '<div style="text-align:center; font-family:Arial,sans-serif; padding:10px;">' +
   554→      '<div style="font-size:18px; font-weight:bold; color:#333; margin-bottom:8px;">PIX</div>' +
   555→      '<div style="font-size:14px; color:#666; margin-bottom:12px;">' + faturaName + ' - ' + valorFmt + '</div>' +
   556→      (qrcodeBase64 ? '<img src="' + qrcodeBase64 + '" style="width:200px; height:200px; margin:0 auto 12px auto; display:block;" />' : '') +
   557→      (pixCopiaCola ? '<div style="font-size:10px; color:#999; word-break:break-all; max-width:400px; margin:0 auto; line-height:1.4;">' + pixCopiaCola + '</div>' : '') +
   558→      '</div>';
   559→
   560→    // Push PIX para Odoo (grava campos x_studio_*)
   561→    var pushResult = { pushed: false, reason: 'not_called' };
   562→    try {
   563→      pushResult = await pushPixToOdoo({
   564→        faturaId: faturaId,
   565→        faturaName: faturaName,
   566→        valor: valorTotal.toFixed(2).replace('.', ','),
   567→        pix: {
   568→          txid: pixResult.txid || '',
   569→          pix_copia_cola: pixCopiaCola,
   570→          qrcode_base64: qrcodeBase64,
   571→          html_pix: htmlPix,
   572→        },
   573→      });
   574→    } catch (pushErr) {
   575→      pushResult = { pushed: false, reason: pushErr.message };
   576→    }
   577→
   578→    var responseData = {
   579→      success: true,
   580→      data: {
   581→        forma_pagamento: 'PIX',
   582→        total_parcelas: 1,
   583→        valor_total: valorTotal.toFixed(2),
   584→        fatura_name: faturaName || '(nao informado)',
   585→        fatura_id: faturaId || 0,
   586→        pagamentos: [{
   587→          tipo: 'pix',
   588→          parcela: 1,
   589→          total_parcelas: 1,
   590→          valor_titulo: valorTotal.toFixed(2),
   591→          txid: pixResult.txid || '',
   592→          pix_copia_cola: pixCopiaCola,
   593→          qrcode_base64: qrcodeBase64,
   594→          html_pix: htmlPix,
   595→        }],
   596→        odoo_push: pushResult.pushed ? 'OK' : 'falhou_' + (pushResult.reason || 'unknown'),
   597→      },
   598→    };
   599→
   600→    // Cache resultado e liberar lock
   601→    _pixCache[lockKey] = { timestamp: Date.now(), response: responseData };
   602→    delete _pixLocks[lockKey];
   603→    lockResolve(responseData);
   604→
   605→    res.json(responseData);
   606→  } catch (err) {
   607→    console.error('[API/PIX] ERRO:', err.message);
   608→    var errResponse = { success: false, message: 'Erro ao criar PIX: ' + (err.message || 'Erro desconhecido') };
   609→    delete _pixLocks[lockKey];
   610→    lockResolve(errResponse);
   611→    res.json(errResponse);
   612→  }
   613→}
