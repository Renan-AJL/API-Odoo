    34→var BANDEIRAS = ['VISA', 'MASTER', 'ELO', 'AMEX', 'HIPERCARD', 'HIPER'];

    36→function detectarCartao(formaPag) {
    37→  if (!formaPag) return null;
    38→  var n = formaPag.trim().toUpperCase();
    39→  for (var i = 0; i < BANDEIRAS.length; i++) {
    40→    var b = BANDEIRAS[i];
    41→    if (n.indexOf(b) === 0 || n.indexOf(b + ' ') === 0) {
    42→      var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
    43→      var parcelas = 1;
    44→      var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
    45→      if (match) parcelas = parseInt(match[1]);
    46→      return { bandeira: b, parcelas: parcelas, debito: debito };
    47→    }
    48→  }
    49→  // Generico: CARTAO, CARTAO CREDITO, CREDITO, etc.
    50→  if (n.indexOf('CART') === 0 || n.indexOf('CREDITO') === 0) {
    51→    var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
    52→    var parcelas = 1;
    53→    var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
    54→    if (match) parcelas = parseInt(match[1]);
    55→    return { bandeira: 'CARTAO', parcelas: parcelas, debito: debito };
    56→  }
    57→  return null;
    58→}

    78→      success: false,
    79→      message: 'Rede nao configurada. Defina REDE_PV e REDE_CHAVE_INTEGRACAO nas variaveis de ambiente.',

    82→
    83→  try {
    84→    var linkResult = await criarLinkPagamento({
    85→      valor: valorTotal,
    86→      seu_numero: faturaName || String(Date.now()),
    87→      descricao: 'Fatura ' + faturaName,
    88→      parcelas: cartaoInfo.parcelas,
    89→      nome_pagador: pag.nome || d.pagador_nome || '',
    90→      cpf_cnpj_pagador: pag.cpf_cnpj || d.pagador_cpf || '',
    91→      email_pagador: pag.email || '',
    92→      expiracao: 7 * 86400,
    93→      host: req.protocol + '://' + req.get('host'),

    95→

    97→
    98→    if (!checkoutUrl) {
    99→      console.error('[API/CARTAO] Checkout criado mas sem URL. Resposta bruta:', JSON.stringify(linkResult.raw));
   100→      return res.json({
   101→        success: false,
   102→        message: 'Checkout criado porem sem URL de pagamento. Verificar resposta da Rede.',
   103→        detail: linkResult.raw,
   104→      });
   105→    }
   106→
   107→    console.log('[API/CARTAO] Checkout URL gerada:', checkoutUrl.substring(0, 80) + '...');
   108→
   109→    res.json({
   110→      success: true,
   111→      data: {
   112→        forma_pagamento: d.forma_pagamento,
   113→        total_parcelas: 1,
   114→        valor_total: valorTotal.toFixed(2),
   115→        fatura_name: faturaName || '(nao informado)',
   116→        fatura_id: faturaId || 0,
   117→        pagamentos: [{
   118→          tipo: cartaoInfo.debito ? 'cartao_debito' : 'cartao_credito',
   119→          parcela: 1,
   120→          total_parcelas: 1,
   121→          valor_titulo: valorTotal.toFixed(2),
   122→          bandeira: cartaoInfo.bandeira,
   123→          parcelas: cartaoInfo.parcelas,
   124→          pix_copia_cola: checkoutUrl,
   125→          link_checkout: checkoutUrl,
   126→          checkout_id: linkResult.id || '',

   128→        odoo_push: 'nao_aplicavel',
   129→      }

   131→
   132→  } catch (err) {

   142→function calcDataVenc(base, dias) {
   143→  var db = base ? new Date(base + 'T12:00:00') : new Date();
   144→  var hj = new Date(); hj.setHours(0,0,0,0);
   145→  if (db <= hj) db = new Date();
   146→  db.setDate(db.getDate() + dias);
   147→  var a = db.getFullYear();
   148→  var m = String(db.getMonth()+1).padStart(2,'0');
   149→  var d = String(db.getDate()).padStart(2,'0');
   150→  return a+'-'+m+'-'+d;
   151→}

   465→async function handlePix(req, res, d) {
   466→  var fat = d.fatura || {};
   467→  var faturaName = fat.name || fat.seu_numero || d.fatura_name || '';
   468→  var faturaId = (fat.id || d.fatura_id) ? parseInt(String(fat.id || d.fatura_id)) : 0;
   469→  var valorTotal = parseFloat(fat.valor_nominal || d.fatura_valor) || 0;
   470→  var pag = d.pagador || {};
   471→  var pagNome = pag.nome || d.pagador_nome || '';
   472→  var pagCpf = pag.cpf_cnpj || d.pagador_cpf || '';
   473→
   474→  console.log('[API/PIX] Fatura:', faturaName, '| Valor:', valorTotal);
   475→
   476→  if (valorTotal <= 0) {
   477→    return res.json({ success: false, message: 'Valor invalido para PIX' });
   478→  }
   479→
   480→  // === DEDUPLICACAO: verificar se ja tem requisicao em curso ou resultado recente ===
   481→  var lockKey = getPixLockKey(faturaId, faturaName);
   482→
   483→  // Se tem resultado cacheado (mesma fatura, menos de 60s atras), reusa
   484→  if (_pixCache[lockKey]) {
   485→    var cached = _pixCache[lockKey];
   486→    var age = Date.now() - cached.timestamp;
   487→    if (age < PIX_CACHE_TTL) {
   488→      console.log('[API/PIX] REUSANDO resultado cacheado (' + (age / 1000).toFixed(1) + 's atras) para', lockKey);
   489→      return res.json(cached.response);
   490→    }
   491→    delete _pixCache[lockKey];
   492→  }
   493→
   494→  // Se tem requisicao em curso, espera ela terminar
   495→  if (_pixLocks[lockKey]) {
   496→    console.log('[API/PIX] AGUARDANDO lock existente para', lockKey);
   497→    try {
   498→      var cachedResult = await _pixLocks[lockKey];
   499→      console.log('[API/PIX] Lock liberado, retornando resultado da requisicao original');
   500→      return res.json(cachedResult);
   501→    } catch (lockErr) {
   502→      console.log('[API/PIX] Requisicao original falhou, tentando novamente...');
   503→    }
   504→  }
   505→
   506→  // Criar lock
   507→  var lockResolve;
   508→  _pixLocks[lockKey] = new Promise(function(resolve) { lockResolve = resolve; });
   509→
   510→  // Auto-limpar lock apos TTL
   511→  setTimeout(function() {
   512→    if (_pixLocks[lockKey]) {
   513→      console.warn('[API/PIX] Lock expirou por TTL para', lockKey);
   514→      delete _pixLocks[lockKey];
   515→    }
   516→  }, PIX_LOCK_TTL);
   517→
   518→  try {
   519→    var devedor = buildDevedorFromDoc(pagNome, pagCpf);
   520→    if (!devedor && pagNome) {
   521→      console.warn('[API/PIX] Pagador com nome mas sem CPF/CNPJ valido, criando PIX sem devedor');
   522→    }
   523→    var pixResult = await criarCobrancaPix({
   524→      valor: valorTotal,
   525→      devedor: devedor,
   526→      solicitacaoPagador: faturaName || '',
   527→      expiracao: 86400,
   528→    });
   529→
   530→    console.log('[API/PIX] PIX criado: TXID=' + (pixResult.txid || 'N/A'));
   531→
   532→    var pixCopiaCola = pixResult.pixCopiaECola || '';
   533→    var qrcodeBase64 = '';
   534→    var htmlPix = '';
   535→
   536→    // Gerar QR code a partir do pixCopiaECola (bwip-js v4 e async)
   537→    if (pixCopiaCola) {
   538→      try {
   539→        var qrPng = await bwipjs.toBuffer({
   540→          bcid: 'qrcode',
   541→          text: pixCopiaCola,
   542→          scale: 5,
   543→          width: 12,
   544→          height: 12,
   545→        });
   546→        qrcodeBase64 = 'data:image/png;base64,' + qrPng.toString('base64');
   547→        console.log('[API/PIX] QR code gerado:', (qrPng.length / 1024).toFixed(1) + 'KB');
   548→      } catch (qrErr) {
   549→        console.error('[API/PIX] Erro ao gerar QR code:', qrErr.message);
   550→      }
   551→    }
   552→
   553→    // Montar HTML para o campo Odoo
   554→    var valorFmt = 'R$ ' + valorTotal.toFixed(2).replace('.', ',');
   555→    htmlPix = '<div style="text-align:center; font-family:Arial,sans-serif; padding:10px;">' +
   556→      '<div style="font-size:18px; font-weight:bold; color:#333; margin-bottom:8px;">PIX</div>' +
   557→      '<div style="font-size:14px; color:#666; margin-bottom:12px;">' + faturaName + ' - ' + valorFmt + '</div>' +
   558→      (qrcodeBase64 ? '<img src="' + qrcodeBase64 + '" style="width:200px; height:200px; margin:0 auto 12px auto; display:block;" />' : '') +
   559→      (pixCopiaCola ? '<div style="font-size:10px; color:#999; word-break:break-all; max-width:400px; margin:0 auto; line-height:1.4;">' + pixCopiaCola + '</div>' : '') +
   560→      '</div>';
   561→
   562→    // Montar responseData ANTES do push (responde pro Odoo imediatamente)
   563→    var responseData = {
   564→      success: true,
   565→      data: {
   566→        forma_pagamento: 'PIX',
   567→        total_parcelas: 1,
   568→        valor_total: valorTotal.toFixed(2),
   569→        fatura_name: faturaName || '(nao informado)',
   570→        fatura_id: faturaId || 0,
   571→        pagamentos: [{
   572→          tipo: 'pix',
   573→          parcela: 1,
   574→          total_parcelas: 1,
   575→          valor_titulo: valorTotal.toFixed(2),
   576→          txid: pixResult.txid || '',
   577→          pix_copia_cola: pixCopiaCola,
   578→          qrcode_base64: qrcodeBase64,
   579→          html_pix: htmlPix,
   580→        }],
   581→        odoo_push: 'pending',
   582→      },
   583→    };
   584→
   585→    // Cache resultado e liberar lock
   586→    _pixCache[lockKey] = { timestamp: Date.now(), response: responseData };
   587→    delete _pixLocks[lockKey];
   588→    lockResolve(responseData);
   589→
   590→    // Responde pro Odoo IMEDIATAMENTE (o modulo Odoo grava os campos)
   591→    res.json(responseData);
   592→  } catch (err) {
   593→    console.error('[API/PIX] ERRO:', err.message);
   594→    var errResponse = { success: false, message: 'Erro ao criar PIX: ' + (err.message || 'Erro desconhecido') };
   595→    delete _pixLocks[lockKey];
   596→    lockResolve(errResponse);
   597→    res.json(errResponse);
   598→  }
   599→}

   643→// Le resposta_json do Odoo, cancela boletos no Itau, posta resultado no chatter e redireciona de volta
   644→router.get('/cancelar/web', async function(req, res) {
   645→  var odooId = req.query.odoo_id || '';
   646→  var oc = config.odoo;
   647→  var returnUrl = (oc && oc.url ? oc.url.replace(/\/+$/, '') : '') + '/web#id=' + odooId + '&model=account.move&view_type=form';
   648→
   649→  function odooRpc(model, method, args, kwargs) {
   650→    var xmlrpc = require('xmlrpc');
   651→    var base = oc.url.replace(/\/+$/, '');
   652→    var host = base.replace('https://', '');
   653→    var commonCli = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/common', port: 443 });
   654→    var modelsCli = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
   655→    return new Promise(function(ok, fail) {
   656→      commonCli.methodCall('authenticate', [oc.db, oc.user, oc.password, {}], function(e, uid) {
   657→        if (e || !uid) return fail(e || new Error('auth failed'));
   658→        var params = [oc.db, uid, oc.password, model, method, args || []];
   659→        if (kwargs) params.push(kwargs);
   660→        modelsCli.methodCall('execute_kw', params, function(e2, r) {
   661→          if (e2) fail(e2); else ok(r);
   662→        });
   663→      });
   664→    });
   665→  }
   666→
   667→  function postChatter(recordId, bodyHtml) {
   668→    return odooRpc('mail.message', 'create', [{
   669→      model: 'account.move',
   670→      res_id: parseInt(recordId),
   671→      body: bodyHtml,
   672→      message_type: 'comment',
   673→    }]);
   674→  }
   675→
   676→  if (!odooId || !oc || !oc.enabled || !oc.url || !oc.db || !oc.user || !oc.password) {
   677→    return res.redirect(returnUrl);
   678→  }
   679→
   680→  try {
   681→    var nossoNumeros = [];
   682→    console.log('[CANCELAR-WEB] Lendo resposta_json da fatura ID:', odooId);
   683→    var fields = await odooRpc('account.move', 'read', [[parseInt(odooId)], ['x_studio_itau_resposta_json']]);
   684→    var rawJson = '';
   685→    if (fields && fields[0] && fields[0].x_studio_itau_resposta_json) {
   686→      rawJson = fields[0].x_studio_itau_resposta_json;
   687→    }
   688→    if (!rawJson) {
   689→      await postChatter(odooId, '<b>Cancelar Boleto</b><br/><span style="color:red">Fatura nao possui dados de boleto.</span>');
   690→      return res.redirect(returnUrl);
   691→    }
   692→
   693→    var jsonStr = rawJson.replace(/'/g, '"').replace(/True/g, 'true').replace(/False/g, 'false').replace(/None/g, 'null');
   694→    var data = JSON.parse(jsonStr);
   695→    var pagamentos = data.pagamentos || [];
   696→    if (pagamentos.length === 0) {
   697→      await postChatter(odooId, '<b>Cancelar Boleto</b><br/><span style="color:red">Nenhum pagamento encontrado.</span>');
   698→      return res.redirect(returnUrl);
   699→    }
   700→    for (var i = 0; i < pagamentos.length; i++) {
   701→      if (pagamentos[i].nosso_numero) nossoNumeros.push(pagamentos[i].nosso_numero);
   702→    }
   703→    if (nossoNumeros.length === 0) {
   704→      await postChatter(odooId, '<b>Cancelar Boleto</b><br/><span style="color:red">Nenhum nosso numero encontrado.</span>');
   705→      return res.redirect(returnUrl);
   706→    }
   707→
   708→    console.log('[CANCELAR-WEB] Cancelando', nossoNumeros.length, 'boleto(s):', nossoNumeros.join(', '));
   709→    var resultados = [];
   710→    for (var i = 0; i < nossoNumeros.length; i++) {
   711→      try {
   712→        await cancelarBoleto(nossoNumeros[i]);
   713→        resultados.push({ nn: nossoNumeros[i], ok: true });
   714→      } catch (err) {
   715→        resultados.push({ nn: nossoNumeros[i], ok: false, erro: err.message });
   716→      }
   717→    }
   718→
   719→    var sucesso = resultados.filter(function(r) { return r.ok; }).length;
   720→    var falha = resultados.filter(function(r) { return !r.ok; }).length;
   721→    var chatBody = '<b>Cancelar Boleto</b><br/>';
   722→    if (falha === 0) {
   723→      chatBody += '<span style="color:green">' + sucesso + ' boleto(s) cancelado(s) com sucesso no Itau.</span>';
   724→    } else if (sucesso === 0) {
   725→      chatBody += '<span style="color:red">Falha ao cancelar todos os boletos:</span><br/>';
   726→      for (var i = 0; i < resultados.length; i++) chatBody += 'NN ' + resultados[i].nn + ': ' + resultados[i].erro + '<br/>';
   727→    } else {
   728→      chatBody += '<span style="color:orange">' + sucesso + ' cancelado(s), ' + falha + ' falha(s):</span><br/>';
   729→      for (var i = 0; i < resultados.length; i++) {
   730→        if (!resultados[i].ok) chatBody += 'NN ' + resultados[i].nn + ': ' + resultados[i].erro + '<br/>';
   731→      }
   732→    }
   733→    chatBody += '<br/><small>NN: ' + nossoNumeros.join(', ') + '</small>';
   734→
   735→    await postChatter(odooId, chatBody);
   736→
   737→    try {
   738→      await odooRpc('account.move', 'write', [[parseInt(odooId)], { 'x_studio_boleto_cancelado': true }]);

   746→    try { await postChatter(odooId, '<b>Cancelar Boleto</b><br/><span style="color:red">Erro: ' + err.message + '</span>'); } catch(e) {}
