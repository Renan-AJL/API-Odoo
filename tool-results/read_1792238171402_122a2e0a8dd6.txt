   183→   183→var BANDEIRAS = ['VISA', 'MASTER', 'ELO', 'AMEX', 'HIPERCARD', 'HIPER'];
   184→   184→
   185→   185→function detectarCartao(formaPag) {
   186→   186→  if (!formaPag) return null;
   187→   187→  var n = formaPag.trim().toUpperCase();
   188→   188→  for (var i = 0; i < BANDEIRAS.length; i++) {
   189→   189→    var b = BANDEIRAS[i];
   190→   190→    if (n.indexOf(b) === 0 || n.indexOf(b + ' ') === 0) {
   191→   191→      var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
   192→   192→      var parcelas = 1;
   193→   193→      var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
   194→   194→      if (match) parcelas = parseInt(match[1]);
   195→   195→      return { bandeira: b, parcelas: parcelas, debito: debito };
   196→   196→    }
   197→   197→  }
   198→   198→  // Generico: CARTAO, CARTAO CREDITO, CREDITO, etc.
   199→   199→  if (n.indexOf('CART') === 0 || n.indexOf('CREDITO') === 0) {
   200→   200→    var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
   201→   201→    var parcelas = 1;
   202→   202→    var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
   203→   203→    if (match) parcelas = parseInt(match[1]);
   204→   204→    return { bandeira: 'CARTAO', parcelas: parcelas, debito: debito };
   205→   205→  }
   206→   206→  return null;
   207→   207→}
