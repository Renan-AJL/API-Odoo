   627→   646→// MATCH MOTORISTA — Faz match do nome do TE com as opcoes do campo selection
   628→   647→// O campo x_studio_motorista e um selection com 17 opcoes (key/label)
   629→   648→// ============================================================
   630→   649→const MOTORISTA_OPTIONS = [
   631→   650→  'adelson', 'anderlucio', 'antonio', 'carlos', 'daniel',
   632→   651→  'diego', 'eduardo', 'fellipe', 'gerson', 'glauber',
   633→   652→  'henrique', 'italo', 'jeferson', 'joao_pedro', 'lucas',
   634→   653→  'matheus', 'rafael',
   635→   654→];
   636→   655→
   637→   656→function matchMotoristaSelection(driverNameFromTe) {
   638→   657→  if (!driverNameFromTe) return null;
   639→   658→  const name = driverNameFromTe.toUpperCase().trim();
   640→   659→  // Remove sufixos comuns como "MOTORISTA", "FRETE", etc
   641→   660→  const cleanName = name
   642→   661→    .replace(/\b(MOTORISTA|FRETE|ENTREGAS|MOTOBOY)\b/g, '')
   643→   662→    .replace(/[^A-ZÀ-Ú\s]/g, '')
   644→   663→    .trim();
   645→   664→  const parts = cleanName.split(/\s+/).filter(Boolean);
   646→   665→
   647→   666→  // Tenta match exato com a key ou com o primeiro/ultimo nome
   648→   667→  for (const key of MOTORISTA_OPTIONS) {
   649→   668→    const keyUpper = key.toUpperCase();
   650→   669→    if (name.includes(keyUpper) || keyUpper.includes(name)) return key;
   651→   670→  }
   652→   671→  // Tenta match com qualquer parte do nome
   653→   672→  for (const part of parts) {
   654→   673→    if (part.length < 3) continue;
   655→   674→    for (const key of MOTORISTA_OPTIONS) {
   656→   675→      if (key.toUpperCase() === part || part.includes(key.toUpperCase())) return key;
   657→   676→    }
   658→   677→  }
   659→   678→  return null;
   660→   679→}
