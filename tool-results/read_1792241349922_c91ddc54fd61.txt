     1→/**
     2→ * services/odoo-push.js - v7.0.2
     3→ * =============================================
     4→ * Push de PDFs de boletos para Odoo via XML-RPC
     5→ * - Usa faturaId (record.id) diretamente - SEMPRE funciona, mesmo rascunho
     6→ * - Cria ir.attachment com PDF base64
     7→ * - Posta NOTA INTERNA no chatter com PDF anexado (mail.mt_note)
     8→ * - Cada boleto: 1 attachment + 1 nota com detalhes
     9→ * - Grava campos x_studio_* na fatura (situacao, tipo_pagamento, pix, nn, link)
    10→ *
    11→ * ENV VARS (Render):
    12→ *   ODOO_URL, ODOO_DB, ODOO_USER, ODOO_PASSWORD, ODOO_PUSH_ENABLED
    13→ * =============================================

    16→var xmlrpc = require('xmlrpc');
    17→
    18→function createClient(url) {
    19→  var base = url.replace(/\/+$/, '');
    20→  return {
    21→    common: xmlrpc.createSecureClient({ host: base.replace('https://', ''), path: '/xmlrpc/2/common', port: 443 }),
    22→    models: xmlrpc.createSecureClient({ host: base.replace('https://', ''), path: '/xmlrpc/2/object', port: 443 }),
    23→    baseUrl: base
    24→  };
    25→}
    26→
    27→function authenticate(client, db, login, password) {
    28→  return new Promise(function(resolve, reject) {
    29→    client.common.methodCall('authenticate', [db, login, password, {}], function(err, uid) {
    30→      if (err) {
    31→        reject(new Error('Erro ao autenticar no Odoo: ' + (err.message || JSON.stringify(err))));
    32→      } else if (uid === false || uid === null) {
    33→        reject(new Error('Falha na autenticacao Odoo: credenciais invalidas'));
    34→      } else {
    35→        console.log('[ODOO-PUSH] Autenticado. UID:', uid);
    36→        resolve(uid);
    37→      }
    38→    });
    39→  });
    40→}
    41→
    42→function executeKw(client, db, uid, password, model, method, args, kwargs) {
    43→  return new Promise(function(resolve, reject) {
    44→    var params = [db, uid, password, model, method, args || []];
    45→    if (kwargs) params.push(kwargs || {});
    46→    client.models.methodCall('execute_kw', params, function(err, result) {
    47→      if (err) {
    48→        reject(new Error('Erro Odoo ' + model + '.' + method + ': ' + (err.message || JSON.stringify(err))));
    49→      } else {
    50→        resolve(result);
    51→      }
    52→    });
    53→  });
    54→}

    56→/**
    57→ * Push completo
    58→ * @param {Object} pushData
    59→ *   - faturaId:    ID da fatura Odoo (record.id) - SEMPRE disponivel
    60→ *   - faturaName:  nome da fatura (pode ser vazio para rascunhos)
    61→ *   - boletos:     array de dados dos boletos
    62→ *   - pdfsBase64:  array de PDFs em base64 (mesma ordem)
    63→ */
    64→async function pushBoletosToOdoo(pushData) {
    65→  var config = require('../config');
    66→  var odooConfig = config.odoo;
    67→
    68→  if (!odooConfig || !odooConfig.enabled) {
    69→    console.log('[ODOO-PUSH] Desabilitado');
    70→    return { pushed: false, reason: 'disabled' };
    71→  }
    72→
    73→  if (!odooConfig.url || !odooConfig.db || !odooConfig.user || !odooConfig.password) {
    74→    console.warn('[ODOO-PUSH] Credenciais Odoo incompletas');
    75→    return { pushed: false, reason: 'missing_credentials' };
    76→  }
    77→
    78→  var faturaId = parseInt(pushData.faturaId) || 0;
    79→  var faturaName = pushData.faturaName || '';
    80→  var boletos = pushData.boletos || [];
    81→  var pdfsBase64 = pushData.pdfsBase64 || [];
    82→
    83→  console.log('[ODOO-PUSH] === INICIANDO ===');
    84→  console.log('[ODOO-PUSH] faturaId:', faturaId, '| faturaName:', faturaName || 'vazio');
    85→  console.log('[ODOO-PUSH] boletos:', boletos.length, '| pdfs:', pdfsBase64.length);
    86→
    87→  try {
    88→    var client = createClient(odooConfig.url);
    89→    var uid = await authenticate(client, odooConfig.db, odooConfig.user, odooConfig.password);
    90→
    91→    // Determinar recordId
    92→    var recordId;
    93→    if (faturaId > 0) {
    94→      // ID direto - sempre funciona, mesmo rascunho
    95→      recordId = faturaId;
    96→      console.log('[ODOO-PUSH] Usando faturaId direto:', recordId);
    97→    } else if (faturaName) {
    98→      // Fallback: buscar por nome
    99→      console.log('[ODOO-PUSH] Buscando fatura por nome:', faturaName);
   100→      var ids = await executeKw(client, odooConfig.db, uid, odooConfig.password, 'account.move', 'search', [[['name', '=', faturaName]]]);
   101→      if (!ids || ids.length === 0) {
   102→        console.warn('[ODOO-PUSH] Fatura nao encontrada por nome, buscando mais recente...');
   103→        var recentIds = await executeKw(client, odooConfig.db, uid, odooConfig.password, 'account.move', 'search', [[['move_type', '=', 'out_invoice']]], { order: 'id desc', limit: 1 });
   104→        if (!recentIds || recentIds.length === 0) {
   105→          return { pushed: false, reason: 'invoice_not_found' };
   106→        }
   107→        recordId = recentIds[0];
   108→      } else {
   109→        recordId = ids[0];
   110→      }
   111→    } else {
   112→      // Sem ID e sem nome - buscar mais recente
   113→      console.warn('[ODOO-PUSH] Sem ID e sem nome, buscando fatura mais recente...');
   114→      var recentIds = await executeKw(client, odooConfig.db, uid, odooConfig.password, 'account.move', 'search', [[['move_type', '=', 'out_invoice']]], { order: 'id desc', limit: 1 });
   115→      if (!recentIds || recentIds.length === 0) {
   116→        return { pushed: false, reason: 'no_invoices' };
   117→      }
   118→      recordId = recentIds[0];
   119→    }
   120→
   121→    console.log('[ODOO-PUSH] recordId final:', recordId);
   122→
   123→    // Buscar nome da fatura para o filename (se nao temos)
   124→    if (!faturaName) {
   125→      try {
   126→        var names = await executeKw(client, odooConfig.db, uid, odooConfig.password, 'account.move', 'read', [[recordId], ['name']]);
   127→        if (names && names[0] && names[0].name) {
   128→          faturaName = names[0].name;
   129→          console.log('[ODOO-PUSH] Nome encontrado:', faturaName);
   130→        }
   131→      } catch (e) {
   132→        console.warn('[ODOO-PUSH] Nao conseguiu ler nome:', e.message);
   133→      }
   134→    }
   135→
   136→    var faturaNameSafe = faturaName ? faturaName.replace(/[^a-zA-Z0-9\-_]/g, '_') : 'Fatura';
   137→    var totalAttachments = 0;
   138→
   139→    // Processar cada boleto
   140→    for (var i = 0; i < boletos.length; i++) {
   141→      var bol = boletos[i];
   142→      var pdfB64 = pdfsBase64[i];
   143→
   144→      console.log('[ODOO-PUSH] Processando boleto', (i + 1) + '/' + boletos.length, '- PDF:', pdfB64 ? (pdfB64.length / 1024).toFixed(0) + 'KB' : 'AUSENTE');
   145→
   146→      if (!pdfB64) {
   147→        console.warn('[ODOO-PUSH]   PDF ausente, pulando boleto', (i + 1));
   148→        continue;
   149→      }
   150→
   151→      var nn = bol.nosso_numero || '';
   152→      var vd = bol.valor_titulo || '0,00';
   153→      var vc = bol.data_vencimento || '';
   154→      var ld = bol.linha_digitavel || '';
   155→      var pix = bol.pix_copia_cola || '';
   156→      var p = bol.parcela || 1;
   157→      var t = bol.total_parcelas || 1;
   158→
   159→      // Nome do arquivo
   160→      var filename;
   161→      if (t > 1) {
   162→        filename = 'Boleto_' + faturaNameSafe + '_P' + p + 'de' + t + '_' + nn + '.pdf';
   163→      } else {
   164→        filename = 'Boleto_' + faturaNameSafe + '_' + nn + '.pdf';
   165→      }
   166→
   167→      try {
   168→        // 1. Criar attachment
   169→        var attachId = await executeKw(client, odooConfig.db, uid, odooConfig.password, 'ir.attachment', 'create', [{
   170→          name: filename,
   171→          raw: pdfB64,
   172→          res_model: 'account.move',
   173→          res_id: recordId,
   174→          mimetype: 'application/pdf',
   175→        }]);
   176→        console.log('[ODOO-PUSH]   Attachment OK:', filename, 'ID:', attachId);
   177→
   178→        // 2. Criar nota interna com PDF amarrado
   179→        var htmlBody = '<b>' + (t > 1 ? 'Boleto Parcela ' + p + '/' + t : 'Boleto') + '</b><br/>';
   180→        htmlBody += 'Nosso Numero: ' + nn + '<br/>';
   181→        htmlBody += 'Valor: R$ ' + vd + '<br/>';
   182→        htmlBody += 'Vencimento: ' + vc + '<br/>';
   183→        htmlBody += 'Linha Digitavel: ' + ld + '<br/>';
   184→        if (pix) {
   185→          htmlBody += 'PIX Copia e Cola: ' + pix;
   186→        }
   187→
   188→        await executeKw(client, odooConfig.db, uid, odooConfig.password, 'mail.message', 'create', [{
   189→          model: 'account.move',
   190→          res_id: recordId,
   191→          body: htmlBody,
   192→          message_type: 'comment',
   193→          subtype_xmlid: 'mail.mt_note',
   194→          attachment_ids: [[6, 0, [attachId]]],
   195→        }]);
   196→        console.log('[ODOO-PUSH]   Nota interna OK - boleto', (i + 1));
   197→        totalAttachments++;
   198→
   199→      } catch (err) {
   200→        console.error('[ODOO-PUSH]   ERRO boleto', (i + 1) + ':', err.message);
   201→      }
   202→    }
   203→
   204→    // === GRAVAR CAMPOS x_studio_* NA FATURA ===
   205→    // Preenche os campos customizados do Odoo com dados dos boletos emitidos
   206→    try {
   207→      var bol0 = boletos[0] || {};
   208→      var nnList = [];
   209→      var pixList = [];
   210→      var linkList = [];
   211→      for (var i = 0; i < boletos.length; i++) {
   212→        if (boletos[i].nosso_numero) nnList.push(boletos[i].nosso_numero);
   213→        if (boletos[i].pix_copia_cola) pixList.push(boletos[i].pix_copia_cola);
   214→        if (boletos[i].pdf_url_txid) linkList.push(boletos[i].pdf_url_txid);
   215→        if (boletos[i].pdf_url_nn) linkList.push(boletos[i].pdf_url_nn);
   216→      }
   217→
   218→      var camposWrite = {};
   219→
   220→      // x_studio_itau_situacao
   221→      if (totalAttachments > 0) {
   222→        camposWrite['x_studio_itau_situacao'] = 'emitido';
   223→      } else {
   224→        camposWrite['x_studio_itau_situacao'] = 'erro';
   225→      }
   226→
   227→      // x_studio_itau_tipo_pagamento
   228→      camposWrite['x_studio_itau_tipo_pagamento'] = 'BOLETO';
   229→
   230→      // x_studio_itau_pix_copia_cola
   231→      if (pixList.length > 0) {
   232→        camposWrite['x_studio_itau_pix_copia_cola'] = pixList.join(' | ');
   233→      }
   234→
   235→      // x_studio_itau_nosso_numero
   236→      if (nnList.length > 0) {
   237→        camposWrite['x_studio_itau_nosso_numero'] = nnList.join(' | ');
   238→      }
   239→
   240→      // x_studio_itau_link_status
   241→      if (linkList.length > 0) {
   242→        camposWrite['x_studio_itau_link_status'] = linkList[0]; // Link do primeiro boleto
   243→      }
   244→
   245→      var keysWrite = Object.keys(camposWrite);
   246→      if (keysWrite.length > 0) {
   247→        console.log('[ODOO-PUSH] Gravando campos x_studio_*:', keysWrite.join(', '));
   248→        try {
   249→          await executeKw(client, odooConfig.db, uid, odooConfig.password, 'account.move', 'write', [[recordId], camposWrite]);
   250→          console.log('[ODOO-PUSH] Campos x_studio_* gravados com sucesso na fatura', recordId);
   251→        } catch (fieldWriteErr) {
   252→          // Campo pode nao existir - tentar campo a campo
   253→          console.warn('[ODOO-PUSH] Erro ao gravar todos campos, tentando campo a campo:', fieldWriteErr.message);
   254→          var writtenFields = [];
   255→          for (var wk = 0; wk < keysWrite.length; wk++) {
   256→            var singleField = {};
   257→            singleField[keysWrite[wk]] = camposWrite[keysWrite[wk]];
   258→            try {
   259→              await executeKw(client, odooConfig.db, uid, odooConfig.password, 'account.move', 'write', [[recordId], singleField]);
   260→              writtenFields.push(keysWrite[wk]);
   261→            } catch (singleErr) {
   262→              console.warn('[ODOO-PUSH] Campo ausente/pulando:', keysWrite[wk]);
   263→            }
   264→          }
   265→          if (writtenFields.length > 0) {
   266→            console.log('[ODOO-PUSH] Campos gravados individualmente:', writtenFields.join(', '));
   267→          }
   268→        }
   269→      }
   270→    } catch (fieldsErr) {
   271→      console.error('[ODOO-PUSH] Erro ao gravar campos x_studio_*:', fieldsErr.message);
   272→      // Nao falhar o push inteiro por causa dos campos
   273→    }
   274→
   275→    console.log('[ODOO-PUSH] === PUSH COMPLETO: ' + totalAttachments + '/' + boletos.length + ' PDFs na fatura ' + faturaNameSafe + ' (ID: ' + recordId + ') ===');
   276→
   277→    // === PUSH BOLETOS PARA O PEDIDO DE VENDA ===
   278→    // Quando o Financeiro emite boleto, anexar tambem no chatter do pedido
   279→    var soResult = { pushed: false };
   280→    if (totalAttachments > 0) {
   281→      try {
   282→        soResult = await pushBoletosToSaleOrder({
   283→          client: client, db: odooConfig.db, uid: uid, password: odooConfig.password,
   284→          invoiceId: recordId, boletos: boletos, pdfsBase64: pdfsBase64, faturaName: faturaName,
   285→        });
   286→        if (soResult.pushed) {
   287→          console.log('[ODOO-PUSH] Boleto(s) tambem anexado(s) no pedido de venda:', soResult.sale_order_name);
   288→        }
   289→      } catch (soErr) {
   290→        console.warn('[ODOO-PUSH] Nao conseguiu anexar no pedido de venda:', soErr.message);
   291→      }
   292→    }
   293→
   294→    return { pushed: totalAttachments > 0, attachments: totalAttachments, record_id: recordId, sale_order_push: soResult };
   295→
   296→  } catch (err) {
   297→    console.error('[ODOO-PUSH] ERRO FATAL:', err.message);
   298→    return { pushed: false, reason: err.message };
   299→  }
   300→}

   302→/**

   351→        recordId = recentIds[0];
   352→      } else {
   353→        recordId = ids[0];
   354→      }
   355→    } else {
   356→      var recentIds = await executeKw(client, odooConfig.db, uid, odooConfig.password, 'account.move', 'search', [[['move_type', '=', 'out_invoice']]], { order: 'id desc', limit: 1 });
   357→      if (!recentIds || recentIds.length === 0) {

   427→    } catch (noteErr) {
   428→      console.warn('[ODOO-PUSH-PIX] Nota interna falhou (nao critico):', noteErr.message.substring(0, 100));
   429→    }
   430→
   431→    console.log('[ODOO-PUSH-PIX] === PUSH COMPLETO na fatura ' + faturaName + ' (ID: ' + recordId + ') ===');
   432→    return { pushed: true, record_id: recordId };
   433→
   434→  } catch (err) {
   435→    console.error('[ODOO-PUSH-PIX] ERRO:', err.message);
   436→    return { pushed: false, reason: err.message };
   437→  }
   438→}

   440→/**
   441→ * Busca o sale.order vinculado a uma invoice e anexa boleto/NF no chatter dele
   442→ * Chamado apos pushBoletosToOdoo() para espelhar os PDFs no pedido de venda
   443→ * @param {Object} params: { client, db, uid, password, invoiceId, boletos, pdfsBase64, faturaName }
   444→ */
   445→async function pushBoletosToSaleOrder(params) {
   446→  var client = params.client;
   447→  var db = params.db;
   448→  var uid = params.uid;
   449→  var password = params.password;
   450→  var invoiceId = params.invoiceId;
   451→  var boletos = params.boletos || [];
   452→  var pdfsBase64 = params.pdfsBase64 || [];
   453→  var faturaName = params.faturaName || '';
   454→
   455→  console.log('[ODOO-PUSH-SO] Buscando sale.order da fatura', invoiceId);
   456→
   457→  try {
   458→    // Buscar sale.order vinculado a esta invoice
   459→    // Metodo 1: via invoice_line_ids -> sale_line_ids -> order_id
   460→    // Metodo 2: search sale.order where invoice_ids contains invoiceId
   461→    var soIds = await executeKw(client, db, uid, password, 'sale.order', 'search', [[['invoice_ids', 'in', [invoiceId]]]], { limit: 5 });
   462→
   463→    if (!soIds || soIds.length === 0) {
   464→      // Fallback: tentar via invoice_origin (campo texto com nome do pedido)
   465→      console.log('[ODOO-PUSH-SO] Nenhum SO via invoice_ids, tentando invoice_origin...');
   466→      var invData = await executeKw(client, db, uid, password, 'account.move', 'read', [[invoiceId], ['invoice_origin']]);
   467→      var origin = (invData && invData[0] && invData[0].invoice_origin) || '';
   468→      if (origin) {
   469→        console.log('[ODOO-PUSH-SO] invoice_origin:', origin);
   470→        soIds = await executeKw(client, db, uid, password, 'sale.order', 'search', [[['name', 'ilike', origin]]], { limit: 5 });
   471→      }
   472→    }
   473→
   474→    if (!soIds || soIds.length === 0) {
   475→      console.warn('[ODOO-PUSH-SO] Nenhum pedido de venda vinculado a fatura', invoiceId);
   476→      return { pushed: false, reason: 'no_sale_order' };
   477→    }
   478→
   479→    // Usar o primeiro SO encontrado
   480→    var soId = soIds[0];
   481→    var soData = await executeKw(client, db, uid, password, 'sale.order', 'read', [[soId], ['name']]);
   482→    var soName = (soData && soData[0] && soData[0].name) || 'SO' + soId;
   483→    console.log('[ODOO-PUSH-SO] Pedido encontrado:', soName, '(ID:', soId, ')');
   484→
   485→    var faturaNameSafe = faturaName ? faturaName.replace(/[^a-zA-Z0-9\-_]/g, '_') : 'Fatura';
   486→    var soAttachments = 0;
   487→
   488→    for (var i = 0; i < boletos.length; i++) {
   489→      var bol = boletos[i];
   490→      var pdfB64 = pdfsBase64[i];
   491→      if (!pdfB64) continue;
   492→
   493→      var nn = bol.nosso_numero || '';
   494→      var vd = bol.valor_titulo || '0,00';
   495→      var vc = bol.data_vencimento || '';
   496→      var p = bol.parcela || 1;
   497→      var t = bol.total_parcelas || 1;
   498→
   499→      var filename;
   500→      if (t > 1) {
   501→        filename = 'Boleto_' + faturaNameSafe + '_P' + p + 'de' + t + '_' + nn + '.pdf';
   502→      } else {
   503→        filename = 'Boleto_' + faturaNameSafe + '_' + nn + '.pdf';
   504→      }
   505→
   506→      try {
   507→        // Criar attachment no sale.order
   508→        var attachId = await executeKw(client, db, uid, password, 'ir.attachment', 'create', [{
   509→          name: filename,
   510→          raw: pdfB64,
   511→          res_model: 'sale.order',
   512→          res_id: soId,
   513→          mimetype: 'application/pdf',
   514→        }]);
   515→        console.log('[ODOO-PUSH-SO]   Attachment OK:', filename, 'ID:', attachId);
   516→
   517→        // Nota interna no chatter do pedido
   518→        var htmlBody = '<b>' + (t > 1 ? 'Boleto Parcela ' + p + '/' + t : 'Boleto') + '</b> — Fatura: ' + (faturaName || 'N/A') + '<br/>';
   519→        htmlBody += 'Nosso Número: ' + nn + '<br/>';
   520→        htmlBody += 'Valor: R$ ' + vd + '<br/>';
   521→        htmlBody += 'Vencimento: ' + vc + '<br/>';
   522→
   523→        await executeKw(client, db, uid, password, 'mail.message', 'create', [{
   524→          model: 'sale.order',
   525→          res_id: soId,
   526→          body: htmlBody,
   527→          message_type: 'comment',
   528→          subtype_xmlid: 'mail.mt_note',
   529→          attachment_ids: [[6, 0, [attachId]]],
   530→        }]);
   531→        console.log('[ODOO-PUSH-SO]   Nota interna OK - boleto', (i + 1));
   532→        soAttachments++;
   533→
   534→      } catch (err) {
   535→        console.error('[ODOO-PUSH-SO]   ERRO boleto', (i + 1) + ':', err.message);
   536→      }
   537→    }
   538→
   539→    console.log('[ODOO-PUSH-SO] === PUSH COMPLETO: ' + soAttachments + '/' + boletos.length + ' PDFs no pedido ' + soName + ' ===');
   540→    return { pushed: soAttachments > 0, attachments: soAttachments, sale_order_id: soId, sale_order_name: soName };
   541→
   542→  } catch (err) {
   543→    console.error('[ODOO-PUSH-SO] ERRO:', err.message);
   544→    return { pushed: false, reason: err.message };
   545→  }
   546→}

   548→/**
   549→ * Copia attachments de boleto/NF da invoice para o sale.order vinculado
   550→ * Usado pelo script retroativo para pedidos que ja tinham boletos
   551→ * @param {Object} params: { client, db, uid, password, invoiceId }
   552→ */
   553→async function copyInvoiceAttachmentsToSaleOrder(params) {
   554→  var client = params.client;
   555→  var db = params.db;
   556→  var uid = params.uid;
   557→  var password = params.password;
   558→  var invoiceId = params.invoiceId;
   559→
   560→  try {
   561→    // Buscar SO vinculado
   562→    var soIds = await executeKw(client, db, uid, password, 'sale.order', 'search', [[['invoice_ids', 'in', [invoiceId]]]], { limit: 5 });
   563→    if (!soIds || soIds.length === 0) {
   564→      var invData = await executeKw(client, db, uid, password, 'account.move', 'read', [[invoiceId], ['invoice_origin', 'name']]);
   565→      var origin = (invData && invData[0] && invData[0].invoice_origin) || '';
   566→      if (origin) {
   567→        soIds = await executeKw(client, db, uid, password, 'sale.order', 'search', [[['name', 'ilike', origin]]], { limit: 5 });
   568→      }
   569→    }
   570→    if (!soIds || soIds.length === 0) return { pushed: false, reason: 'no_sale_order' };
   571→
   572→    var soId = soIds[0];
   573→
   574→    // Buscar attachments de boleto na invoice
   575→    var attachIds = await executeKw(client, db, uid, password, 'ir.attachment', 'search', [[
   576→      ['res_model', '=', 'account.move'],
   577→      ['res_id', '=', invoiceId],
   578→      ['name', 'ilike', 'Boleto'],
   579→      ['mimetype', '=', 'application/pdf'],
   580→    ]]);
   581→
   582→    if (!attachIds || attachIds.length === 0) {
   583→      console.log('[ODOO-COPY-SO] Fatura', invoiceId, ': sem attachments de boleto');
   584→      return { pushed: false, reason: 'no_boleto_attachments' };
   585→    }
   586→
   587→    // Ler dados dos attachments
   588→    var attachments = await executeKw(client, db, uid, password, 'ir.attachment', 'read', [attachIds, ['name', 'raw', 'mimetype']]);
   589→
   590→    // Verificar quais ja existem no SO (evitar duplicar)
   591→    var soExisting = await executeKw(client, db, uid, password, 'ir.attachment', 'search', [[
   592→      ['res_model', '=', 'sale.order'],
   593→      ['res_id', '=', soId],
   594→      ['name', 'ilike', 'Boleto'],
   595→    ]]);
   596→    var soExistingNames = [];
   597→    if (soExisting && soExisting.length > 0) {
   598→      var soExistingData = await executeKw(client, db, uid, password, 'ir.attachment', 'read', [soExisting, ['name']]);
   599→      soExistingNames = soExistingData.map(function(a) { return a.name; });
   600→    }
   601→
   602→    var copied = 0;
   603→    for (var i = 0; i < attachments.length; i++) {
   604→      var att = attachments[i];
   605→      // Pular se ja existe no SO
   606→      if (soExistingNames.indexOf(att.name) >= 0) {
   607→        console.log('[ODOO-COPY-SO]   Ja existe no SO:', att.name, '- pulando');
   608→        continue;
   609→      }
   610→
   611→      // Criar copia no sale.order
   612→      try {
   613→        var newAttId = await executeKw(client, db, uid, password, 'ir.attachment', 'create', [{
   614→          name: att.name,
   615→          raw: att.raw,
   616→          res_model: 'sale.order',
   617→          res_id: soId,
   618→          mimetype: att.mimetype || 'application/pdf',
   619→        }]);
   620→
   621→        // Postar chatter note
   622→        await executeKw(client, db, uid, password, 'mail.message', 'create', [{
   623→          model: 'sale.order',
   624→          res_id: soId,
   625→          body: '<b>Boleto</b> (cópia da fatura ID ' + invoiceId + ')<br/>Arquivo: ' + att.name,
   626→          message_type: 'comment',
   627→          subtype_xmlid: 'mail.mt_note',
   628→          attachment_ids: [[6, 0, [newAttId]]],
   629→        }]);
   630→        copied++;
   631→        console.log('[ODOO-COPY-SO]   Copiado:', att.name, '→ SO', soId);
   632→      } catch (copyErr) {
   633→        console.error('[ODOO-COPY-SO]   ERRO ao copiar', att.name + ':', copyErr.message);
   634→      }
   635→    }
   636→
   637→    return { pushed: copied > 0, copied: copied, sale_order_id: soId };
   638→
   639→  } catch (err) {
   640→    console.error('[ODOO-COPY-SO] ERRO:', err.message);
   641→    return { pushed: false, reason: err.message };
   642→  }
   643→}
