   106→// XML-RPC connection (singleton)
   107→// ============================================================
   108→var _cached = null;
   109→
   110→function getConn() {
   111→  if (_cached && _cached.uid) return _cached;
   112→  var odooConfig = config.odoo;
   113→  var base = odooConfig.url.replace(/\/+$/, '');
   114→  var host = base.replace('https://', '').replace('http://', '');
   115→  var common = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/common', port: 443 });
   116→  var models = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
   117→  _cached = { uid: null, common: common, models: models, db: odooConfig.db, password: odooConfig.password };
   118→  return _cached;
   119→}
   120→
   121→function authenticate() {
   122→  return new Promise(function(resolve, reject) {
   123→    var conn = getConn();
   124→    conn.common.methodCall('authenticate', [conn.db, config.odoo.user, conn.password, {}], function(err, uid) {
   125→      if (err) { reject(new Error('Auth Odoo falhou: ' + (err.message || JSON.stringify(err)))); }
   126→      else if (uid === false || uid === null) { reject(new Error('Auth Odoo: credenciais invalidas (UID=false)')); }
   127→      else { conn.uid = uid; logger.info('[ODOO-TE] Autenticado. UID: ' + uid); resolve(uid); }
   128→    });
   129→  });
   130→}
   131→
   132→function ensureAuth() {
   133→  var conn = getConn();
   134→  if (conn.uid) return Promise.resolve(conn.uid);
   135→  return authenticate();
   136→}
   137→
   138→function executeKw(model, method, args, kwargs) {
   139→  return new Promise(function(resolve, reject) {
   140→    ensureAuth().then(function(uid) {
   141→      var conn = getConn();
   142→      var params = [conn.db, uid, conn.password, model, method, args || []];
   143→      if (kwargs) params.push(kwargs);
   144→      // Debug log para metodos read/search (nao logar write/create com dados sensiveis)
   145→      if (method === 'read' || method === 'search' || method === 'search_read') {
   146→        logger.info('[ODOO-TE] executeKw: ' + model + '.' + method + ' args=' + JSON.stringify(args).substring(0, 200));
   147→      }
   148→      conn.models.methodCall('execute_kw', params, function(err8, result) {
   149→        if (err8) {
   150→          logger.error('[ODOO-TE] executeKw ERRO: ' + model + '.' + method + ': ' + (err8.message || JSON.stringify(err8)).substring(0, 300));
   151→          reject(new Error(model + '.' + method + ': ' + (err8.message || JSON.stringify(err8))));
   152→        }
   153→        else { resolve(result); }
   154→      });
   155→    }).catch(reject);
   156→  });
   157→}

   159→// ============================================================
   160→// flattenIds — normaliza IDs para Odoo XML-RPC read()
   161→// Odoo exige args = [[id1, id2, ...], [field1, field2, ...]]
   162→// IDs podem chegar como: 111, [111], [[111]], [[111,"name"]], [111,222,...]
   163→// Retorna sempre flat array de inteiros: [111, 222, ...]
   164→// ============================================================
   165→function flattenIds(ids) {
   166→  if (!ids) return [];
   167→  // Single integer
   168→  if (typeof ids === 'number') return [ids];
   169→  // Not an array — try parseInt
   170→  if (!Array.isArray(ids)) {
   171→    var n = parseInt(ids, 10);
   172→    return isNaN(n) ? [] : [n];
   173→  }
   174→  // Flat array — check first element
   175→  if (ids.length === 0) return [];
   176→  var first = ids[0];
   177→  // [111, 222, ...] — flat int array
   178→  if (typeof first === 'number') return ids;
   179→  // ["111", ...] — flat string array
   180→  if (typeof first === 'string') return ids.map(function(x) { return parseInt(x, 10); }).filter(function(n) { return !isNaN(n); });
   181→  // [[111, "name"], ...] or [[111], ...] — nested (Many2one pair or nested array)
   182→  if (Array.isArray(first)) {
   183→    return ids.map(function(item) {
   184→      if (typeof item === 'number') return item;
   185→      if (Array.isArray(item) && item.length > 0) {
   186→        var inner = typeof item[0] === 'number' ? item[0] : parseInt(item[0], 10);
   187→        return isNaN(inner) ? null : inner;
   188→      }
   189→      var n = parseInt(item, 10);
   190→      return isNaN(n) ? null : n;
   191→    }).filter(function(x) { return x !== null && x !== undefined; });
   192→  }
   193→  return ids;
   194→}

   196→// ============================================================
   197→// Safe read/write — tenta com x_studio, fallback sem
   198→// ============================================================
   199→var _customFieldsCache = {};
   200→
   201→async function safeReadCustom(model, ids, nativeFields, customFields) {
   202→  var flatIds = flattenIds(ids);
   203→  if (!flatIds.length) return [];
   204→  try {
   205→    return await executeKw(model, 'read', [flatIds, nativeFields.concat(customFields)]);
   206→  } catch (err) {
   207→    if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   208→      if (!_customFieldsCache[model]) {
   209→        _customFieldsCache[model] = { available: false, checked: true };
   210→        logger.warn('[ODOO-TE] Campos x_studio_* nao encontrados em ' + model + '. Usando campos nativos.');
   211→      }
   212→      return await executeKw(model, 'read', [flatIds, nativeFields]);
   213→    }
   214→    throw err;
   215→  }
   216→}
   217→
   218→async function safeWriteCustom(model, ids, vals) {
   219→  var customVals = {};
   220→  var nativeVals = {};
   221→  Object.keys(vals).forEach(function(k) {
   222→    if (k.indexOf('x_studio_') === 0) customVals[k] = vals[k];
   223→    else nativeVals[k] = vals[k];
   224→  });
   225→  if (Object.keys(nativeVals).length > 0) await executeKw(model, 'write', [ids, nativeVals]);
   226→  if (Object.keys(customVals).length > 0) {
   227→    try { await executeKw(model, 'write', [ids, customVals]); }
   228→    catch (err) {
   229→      if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   230→        if (!_customFieldsCache[model] || !_customFieldsCache[model].warned) {
   231→          logger.warn('[ODOO-TE] Campo x_studio_* nao existe em ' + model + '. Gravacao ignorada.');
   232→          if (!_customFieldsCache[model]) _customFieldsCache[model] = {};
   233→          _customFieldsCache[model].warned = true;
   234→        }
   235→      } else throw err;
   236→    }
   237→  }
   238→}

   250→  // Usa search_read nativo do Odoo — um unico XML-RPC call em vez de search+read
   251→  var kwargs = { fields: fields, limit: limit || 80, offset: offset || 0 };
   252→  if (order) kwargs.order = order;
   253→  return await executeKw(model, 'search_read', [domain], kwargs);

   451→async function getInvoiceLines(invoiceId) {
   452→  try {
   453→    // Evita account.move.line.search — Odoo 19.2 SaaS tem override de
   454→    // search_fetch com _get_recon_limit_from_domain que rejeita certos dominios.
   455→    // Em vez disso, le line_ids direto da fatura.
   456→    var invData = await executeKw('account.move', 'read', [[invoiceId], ['line_ids']]);
   457→    var inv = invData && invData[0];
   458→    if (!inv || !inv.line_ids || !inv.line_ids.length) return [];
   459→
   460→    // line_ids pode vir como [id, id, ...] ou [[id, name], ...] ou [[id], ...]
   461→    // flattenIds normaliza para flat array de inteiros
   462→    var lineIds = flattenIds(inv.line_ids);
   463→
   464→    var lines = await executeKw('account.move.line', 'read', [lineIds, INVOICE_LINE_FIELDS]);
   465→    if (!lines || !lines.length) return [];
   466→
   467→    // Filtra apenas linhas com produto (product_id != false)
   468→    return lines.filter(function(line) {
   469→      if (!line.product_id) return false;
   470→      // product_id pode ser [id, name] ou false
   471→      if (Array.isArray(line.product_id) && line.product_id[0]) return true;
   472→      if (typeof line.product_id === 'number' && line.product_id) return true;
   473→      return false;
   474→    });
   475→  } catch (err) {
   476→    logger.warn('[ODOO-TE] getInvoiceLines falhou: ' + err.message);
   477→    return [];
   478→  }
   479→}

   561→      return [];
   562→    }
   563→    throw err;
   564→  }
