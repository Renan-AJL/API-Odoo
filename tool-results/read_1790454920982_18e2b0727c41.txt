    11→// ── Cache ZIP em memória (chave -> xmlString, TTL 30min) ─────────
    12→var _zipCache = {}; // { chave: { xml: string, ts: number } }
    13→var ZIP_CACHE_TTL = 30 * 60 * 1000;
    14→function cacheSet(chave, xml) { _zipCache[chave] = { xml, ts: Date.now() }; }
    15→function cacheGet(chave) {
    16→  var e = _zipCache[chave];
    17→  if (!e) return null;
    18→  if (Date.now() - e.ts > ZIP_CACHE_TTL) { delete _zipCache[chave]; return null; }
    19→  return e.xml;
    20→}
    21→setInterval(function() {
    22→  var now = Date.now();
    23→  Object.keys(_zipCache).forEach(function(k) { if (now - _zipCache[k].ts > ZIP_CACHE_TTL) delete _zipCache[k]; });
    24→}, 10 * 60 * 1000);

    33→function parseCookies(req) {
    34→  var list = {};
    35→  var header = req.headers && req.headers.cookie;
    36→  if (!header) return list;
    37→  header.split(';').forEach(function(c) {
    38→    var idx = c.indexOf('=');
    39→    if (idx < 0) return;
    40→    var k = c.slice(0, idx).trim();
    41→    var v = c.slice(idx + 1).trim();
    42→    if (v[0] === '"') v = v.slice(1, -1);
    43→    list[k] = decodeURIComponent(v);
    44→  });
    45→  return list;
    46→}

    48→function makeToken(user) {
    49→  var exp     = Date.now() + COOKIE_TTL;
    50→  var payload = user + ':' + exp;
    51→  var sig     = crypto.createHmac('sha256', TOKEN_SECRET).update(payload).digest('hex');
    52→  return payload + ':' + sig;
    53→}
    54→
    55→function validateToken(token) {
    56→  if (!token) return false;
    57→  var idx = token.lastIndexOf(':');
    58→  if (idx < 0) return false;
    59→  var sig     = token.slice(idx + 1);
    60→  var payload = token.slice(0, idx);
    61→  var parts   = payload.split(':');
    62→  if (parts.length < 2) return false;
    63→  var exp     = parseInt(parts[parts.length - 1]);
    64→  if (isNaN(exp) || Date.now() > exp) return false;
    65→  var expected = crypto.createHmac('sha256', TOKEN_SECRET).update(payload).digest('hex');
    66→  return sig === expected;
    67→}

    69→function auth(req, res, next) {
    70→  var cookies = parseCookies(req);
    71→  if (validateToken(cookies[COOKIE_NAME])) return next();
    72→  var ah = req.headers['authorization'] || '';
    73→  if (ah.startsWith('Bearer ') && validateToken(ah.slice(7))) return next();
    74→  if (req.path.startsWith('/api/')) return res.status(401).json({ erro: 'Não autenticado' });
    75→  res.redirect('/admin/login');
    76→}

    78→function withTimeout(p, ms) {
    79→  return Promise.race([p, new Promise((_, rej) => setTimeout(() => rej(new Error('Timeout ' + ms + 'ms')), ms))]);
    80→}

    82→function odooClient(url) {
    83→  var base = (url || '').replace(/\/+$/, '');
    84→  var host = base.replace(/^https?:\/\//, '');
    85→  var secure = base.startsWith('https');
    86→  var fn = secure ? xmlrpc.createSecureClient : xmlrpc.createClient;
    87→  return {
    88→    common: fn({ host, path: '/xmlrpc/2/common', port: secure ? 443 : 80 }),
    89→    models: fn({ host, path: '/xmlrpc/2/object', port: secure ? 443 : 80 }),
    90→  };
    91→}

    93→function odooAuth(c, db, user, pass) {
    94→  return withTimeout(new Promise((res, rej) => {
    95→    c.common.methodCall('authenticate', [db, user, pass, {}], (err, uid) => {
    96→      if (err || !uid) rej(new Error('Auth falhou: ' + (err && (err.faultString || err.message) || 'uid nulo')));
    97→      else res(uid);
    98→    });
    99→  }), 15000);
   100→}

   102→function odooKw(c, db, uid, pass, model, method, args, kwargs) {
   103→  var params = [db, uid, pass, model, method, args || []];
   104→  if (kwargs) params.push(kwargs);
   105→  return withTimeout(new Promise((res, rej) => {
   106→    c.models.methodCall('execute_kw', params, (err, r) => {
   107→      if (err) rej(new Error(err.faultString || err.message)); else res(r);
   108→    });
   109→  }), 20000);
   110→}
