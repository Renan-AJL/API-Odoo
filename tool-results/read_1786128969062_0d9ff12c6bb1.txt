    17→    17→// POST /api/v1/te/deliveries/send — Envia pedidos do ODOO ao TE
    18→    18→// Body: { orderIds?: number[], pickingIds?: number[], orderType?: 1|2|3|4|5 }
    19→    19→// -------------------------------------------------------
    20→    20→router.post('/deliveries/send', async (req, res, next) => {
    21→    21→  try {
    22→    22→    const { orderIds, pickingIds, orderType = 1 } = req.body;
    23→    23→
    24→    24→    if (!orderIds?.length && !pickingIds?.length) {
    25→    25→      return res.status(400).json({ success: false, error: 'Informe orderIds ou pickingIds' });
    26→    26→    }
    27→    27→
    28→    28→    // 1. Busca pedidos ODOO
    29→    29→    const pickings = [];
    30→    30→    const saleOrders = [];
    31→    31→
    32→    32→    if (pickingIds?.length) {
    33→    33→      const found = await odooTe.read('stock.picking', pickingIds, [
    34→    34→        'id', 'name', 'partner_id', 'scheduled_date', 'move_line_count', 'origin',
    35→    35→      ]);
    36→    36→      pickings.push(...found);
    37→    37→    }
    38→    38→
    39→    39→    if (orderIds?.length) {
    40→    40→      const soFields = odooTe.getStudioFields('sale.order');
    41→    41→      const found = await odooTe.read('sale.order', orderIds, [
    42→    42→        'id', 'name', 'state', 'partner_id', 'amount_total', 'note', ...soFields,
    43→    43→      ]);
    44→    44→      saleOrders.push(...found);
    45→    45→    }
    46→    46→
    47→    47→    // 2. Busca parceiros
    48→    48→    const partnerIds = [
    49→    49→      ...pickings.map(p => p.partner_id?.[0]),
    50→    50→      ...saleOrders.map(o => o.partner_id?.[0]),
    51→    51→    ].filter(Boolean);
    52→    52→
    53→    53→    const partners = partnerIds.length > 0
    54→    54→      ? await odooTe.read('res.partner', [...new Set(partnerIds)], [
    55→    55→          'id', 'name', 'cnpj_cpf', 'vat', 'phone', 'mobile', 'email',
    56→    56→          'street', 'street_number', 'street2', 'zip', 'city',
    57→    57→          'l10n_br_district', 'partner_latitude', 'partner_longitude',
    58→    58→          'state_id', 'country_id',
    59→    59→          ...odooTe.getStudioFields('res.partner'),
    60→    60→        ])
    61→    61→      : [];
    62→    62→
    63→    63→    const partnerMap = {};
    64→    64→    partners.forEach(p => { partnerMap[p.id] = p; });
    65→    65→
    66→    66→    // 3. Mapeia para entregas TE
    67→    67→    const teDeliveries = [];
    68→    68→
    69→    69→    for (const so of saleOrders) {
    70→    70→      const partner = partnerMap[so.partner_id?.[0]] || {};
    71→    71→      const picking = pickings.find(p => p.origin === so.name) || null;
    72→    72→      const teDelivery = Mapper.odooToTeDelivery(so, partner, picking);
    73→    73→      teDelivery.OrderType = orderType;
    74→    74→      teDeliveries.push(teDelivery);
    75→    75→    }
    76→    76→
    77→    77→    for (const picking of pickings) {
    78→    78→      if (saleOrders.some(so => so.name === picking.origin)) continue; // ja mapeado
    79→    79→      const partner = partnerMap[picking.partner_id?.[0]] || {};
    80→    80→      const fakeSo = { name: picking.origin || `WH-${picking.id}`, note: '', id: 0 };
    81→    81→      const teDelivery = Mapper.odooToTeDelivery(fakeSo, partner, picking);
    82→    82→      teDelivery.OrderType = orderType;
    83→    83→      teDeliveries.push(teDelivery);
    84→    84→    }
    85→    85→
    86→    86→    if (teDeliveries.length === 0) {
    87→    87→      return res.json({ success: true, sent: 0, message: 'Nenhuma entrega para enviar' });
    88→    88→    }
    89→    89→
    90→    90→    // 4. Envia ao TE em lotes de 50
    91→    91→    const results = [];
    92→    92→    const BATCH = 50;
    93→    93→
    94→    94→    for (let i = 0; i < teDeliveries.length; i += BATCH) {
    95→    95→      const batch = teDeliveries.slice(i, i + BATCH);
    96→    96→      const result = await teClient.createDeliveries(batch);
    97→    97→      results.push(result);
    98→    98→
    99→    99→      // 5. Marca como sincronizado no ODOO
   100→   100→      const batchOrderNums = batch.map(d => d.OrderNumber);
   101→   101→
   102→   102→      const syncedSo = saleOrders.filter(so =>
   103→   103→        batchOrderNums.includes(so.name) || batchOrderNums.includes(`WH-${so.id}`)
   104→   104→      );
   105→   105→      const syncedPickings = pickings.filter(p =>
   106→   106→        batchOrderNums.includes(p.origin) || batchOrderNums.includes(`WH-${p.id}`)
   107→   107→      );
   108→   108→
   109→   109→      if (syncedSo.length > 0) {
   110→   110→        await odooTe.markSaleOrdersSynced(syncedSo.map(s => s.id), batchOrderNums[0], orderType);
   111→   111→      }
   112→   112→      if (syncedPickings.length > 0) {
   113→   113→        await odooTe.markPickingsSynced(syncedPickings.map(p => p.id), batchOrderNums[0]);
   114→   114→      }
   115→   115→    }
   116→   116→
   117→   117→    console.log(`[TE-DELIVERY] Enviadas ${teDeliveries.length} entregas ao TE`);
   118→   118→    res.json({
   119→   119→      success: true,
   120→   120→      sent: teDeliveries.length,
   121→   121→      batches: results.length,
   122→   122→      results,
   123→   123→    });
   124→   124→  } catch (err) {
   125→   125→    console.error('[TE-DELIVERY] Erro:', err.message);
   126→   126→    const status = err.isAxiosError ? (err.response?.status || 502) : 500;
   127→   127→    res.status(status).json({ success: false, error: err.message });
   128→   128→  }
   129→   129→});

   315→   315→    res.json({

   362→   362→    const d = new Date(dateStr);

   889→   889→// MATCH MOTORISTA — Faz match do nome do TE com as opcoes do campo selection
   890→   890→// O campo x_studio_motorista e um selection com 17 opcoes (key/label)
   891→   891→// ============================================================
   892→   892→const MOTORISTA_OPTIONS = [
   893→   893→  'adelson', 'adriano', 'anderlucio', 'antonio', 'carlos', 'daniel',
   894→   894→  'diego', 'eduardo', 'fellipe', 'gerson', 'glauber',
   895→   895→  'henrique', 'italo', 'jeferson', 'joao_pedro', 'lucas',
   896→   896→  'matheus', 'rafael',
   897→   897→];
   898→   898→
   899→   899→function matchMotoristaSelection(driverNameFromTe) {
   900→   900→  if (!driverNameFromTe) return null;
   901→   901→  const name = driverNameFromTe.toUpperCase().trim();
   902→   902→  // Remove sufixos comuns como "MOTORISTA", "FRETE", etc
   903→   903→  const cleanName = name
   904→   904→    .replace(/\b(MOTORISTA|FRETE|ENTREGAS|MOTOBOY)\b/g, '')
   905→   905→    .replace(/[^A-ZÀ-Ú\s]/g, '')
   906→   906→    .trim();
   907→   907→  const parts = cleanName.split(/\s+/).filter(Boolean);
   908→   908→
   909→   909→  // Tenta match exato com a key ou com o primeiro/ultimo nome
   910→   910→  for (const key of MOTORISTA_OPTIONS) {
   911→   911→    const keyUpper = key.toUpperCase();
   912→   912→    if (name.includes(keyUpper) || keyUpper.includes(name)) return key;
   913→   913→  }
   914→   914→  // Tenta match com qualquer parte do nome
   915→   915→  for (const part of parts) {
   916→   916→    if (part.length < 3) continue;
   917→   917→    for (const key of MOTORISTA_OPTIONS) {
   918→   918→      if (key.toUpperCase() === part || part.includes(key.toUpperCase())) return key;
   919→   919→    }
   920→   920→  }
   921→   921→  return null;
   922→   922→}
