    36→function detectarCartao(formaPag) {
    37→  if (!formaPag) return null;
    38→  var n = formaPag.trim().toUpperCase();
    39→  for (var i = 0; i < BANDEIRAS.length; i++) {
    40→    var b = BANDEIRAS[i];
    41→    if (n.indexOf(b) === 0 || n.indexOf(b + ' ') === 0) {
    42→      var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
    43→      var parcelas = 1;
    44→      var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
    45→      if (match) parcelas = parseInt(match[1]);
    46→      return { bandeira: b, parcelas: parcelas, debito: debito };
    47→    }
    48→  }
    49→  // Generico: CARTAO, CARTAO CREDITO, CREDITO, etc.
    50→  if (n.indexOf('CART') === 0 || n.indexOf('CREDITO') === 0) {
    51→    var debito = n.indexOf('DEBITO') >= 0 || n.indexOf('DEB') >= 0;
    52→    var parcelas = 1;
    53→    var match = n.match(/(\d+)\s*(VEZES|X|VEZ)/);
    54→    if (match) parcelas = parseInt(match[1]);
    55→    return { bandeira: 'CARTAO', parcelas: parcelas, debito: debito };
    56→  }
    57→  return null;
    58→}

   142→function calcDataVenc(base, dias) {
   143→  var db = base ? new Date(base + 'T12:00:00') : new Date();
   144→  var hj = new Date(); hj.setHours(0,0,0,0);
   145→  if (db <= hj) db = new Date();
   146→  db.setDate(db.getDate() + dias);
   147→  var a = db.getFullYear();
   148→  var m = String(db.getMonth()+1).padStart(2,'0');
   149→  var d = String(db.getDate()).padStart(2,'0');
   150→  return a+'-'+m+'-'+d;
   151→}

   521→async function handlePix(req, res, d) {
   522→  var fat = d.fatura || {};
   523→  var faturaName = fat.name || fat.seu_numero || d.fatura_name || '';
   524→  var faturaId = (fat.id || d.fatura_id) ? parseInt(String(fat.id || d.fatura_id)) : 0;
   525→  var valorTotal = parseFloat(fat.valor_nominal || d.fatura_valor) || 0;
   526→  var pag = d.pagador || {};
   527→  var pagNome = pag.nome || d.pagador_nome || '';
   528→  var pagCpf = pag.cpf_cnpj || d.pagador_cpf || '';
   529→
   530→  console.log('[API/PIX] Fatura:', faturaName, '| Valor:', valorTotal);
   531→
   532→  if (valorTotal <= 0) {
   533→    return res.json({ success: false, message: 'Valor invalido para PIX' });
   534→  }
   535→
   536→  // === DEDUPLICACAO: verificar se ja tem requisicao em curso ou resultado recente ===
   537→  var lockKey = getPixLockKey(faturaId, faturaName);
   538→
   539→  // Se tem resultado cacheado (mesma fatura, menos de 60s atras), reusa
   540→  if (_pixCache[lockKey]) {
   541→    var cached = _pixCache[lockKey];
   542→    var age = Date.now() - cached.timestamp;
   543→    if (age < PIX_CACHE_TTL) {
   544→      console.log('[API/PIX] REUSANDO resultado cacheado (' + (age / 1000).toFixed(1) + 's atras) para', lockKey);
   545→      return res.json(cached.response);
   546→    }
   547→    delete _pixCache[lockKey];
   548→  }
   549→
   550→  // Se tem requisicao em curso, espera ela terminar
   551→  if (_pixLocks[lockKey]) {
   552→    console.log('[API/PIX] AGUARDANDO lock existente para', lockKey);
   553→    try {
   554→      var cachedResult = await _pixLocks[lockKey];
   555→      console.log('[API/PIX] Lock liberado, retornando resultado da requisicao original');
   556→      return res.json(cachedResult);
   557→    } catch (lockErr) {
   558→      console.log('[API/PIX] Requisicao original falhou, tentando novamente...');
   559→    }
   560→  }
   561→
   562→  // Criar lock
   563→  var lockResolve;
   564→  _pixLocks[lockKey] = new Promise(function(resolve) { lockResolve = resolve; });
   565→
   566→  // Auto-limpar lock apos TTL
   567→  setTimeout(function() {
   568→    if (_pixLocks[lockKey]) {
   569→      console.warn('[API/PIX] Lock expirou por TTL para', lockKey);
   570→      delete _pixLocks[lockKey];
   571→    }
   572→  }, PIX_LOCK_TTL);
   573→
   574→  try {
   575→    var devedor = buildDevedorFromDoc(pagNome, pagCpf);
   576→    if (!devedor && pagNome) {
   577→      console.warn('[API/PIX] Pagador com nome mas sem CPF/CNPJ valido, criando PIX sem devedor');
   578→    }
   579→    var pixResult = await criarCobrancaPix({
   580→      valor: valorTotal,
   581→      devedor: devedor,
   582→      solicitacaoPagador: faturaName || '',
   583→      expiracao: 86400,
   584→    });
   585→
   586→    console.log('[API/PIX] PIX criado: TXID=' + (pixResult.txid || 'N/A'));
   587→
   588→    var pixCopiaCola = pixResult.pixCopiaECola || '';
   589→    var qrcodeBase64 = '';
   590→    var htmlPix = '';
   591→
   592→    // Gerar QR code a partir do pixCopiaECola (bwip-js v4 e async)
   593→    if (pixCopiaCola) {
   594→      try {
   595→        var qrPng = await bwipjs.toBuffer({
   596→          bcid: 'qrcode',
   597→          text: pixCopiaCola,
   598→          scale: 5,
   599→          width: 12,
   600→          height: 12,
   601→        });
   602→        qrcodeBase64 = 'data:image/png;base64,' + qrPng.toString('base64');
   603→        console.log('[API/PIX] QR code gerado:', (qrPng.length / 1024).toFixed(1) + 'KB');
   604→      } catch (qrErr) {
   605→        console.error('[API/PIX] Erro ao gerar QR code:', qrErr.message);
   606→      }
   607→    }
   608→
   609→    // Montar HTML para o campo Odoo
   610→    var valorFmt = 'R$ ' + valorTotal.toFixed(2).replace('.', ',');
   611→    htmlPix = '<div style="text-align:center; font-family:Arial,sans-serif; padding:10px;">' +
   612→      '<div style="font-size:18px; font-weight:bold; color:#333; margin-bottom:8px;">PIX</div>' +
   613→      '<div style="font-size:14px; color:#666; margin-bottom:12px;">' + faturaName + ' - ' + valorFmt + '</div>' +
   614→      (qrcodeBase64 ? '<img src="' + qrcodeBase64 + '" style="width:200px; height:200px; margin:0 auto 12px auto; display:block;" />' : '') +
   615→      (pixCopiaCola ? '<div style="font-size:10px; color:#999; word-break:break-all; max-width:400px; margin:0 auto; line-height:1.4;">' + pixCopiaCola + '</div>' : '') +
   616→      '</div>';
   617→
   618→    // Montar responseData ANTES do push (responde pro Odoo imediatamente)
   619→    var responseData = {
   620→      success: true,
   621→      data: {
   622→        forma_pagamento: 'PIX',
   623→        total_parcelas: 1,
   624→        valor_total: valorTotal.toFixed(2),
   625→        fatura_name: faturaName || '(nao informado)',
   626→        fatura_id: faturaId || 0,
   627→        pagamentos: [{
   628→          tipo: 'pix',
   629→          parcela: 1,
   630→          total_parcelas: 1,
   631→          valor_titulo: valorTotal.toFixed(2),
   632→          txid: pixResult.txid || '',
   633→          pix_copia_cola: pixCopiaCola,
   634→          qrcode_base64: qrcodeBase64,
   635→          html_pix: htmlPix,
   636→        }],
   637→        odoo_push: 'pending',
   638→      },
   639→    };
   640→
   641→    // Cache resultado e liberar lock
   642→    _pixCache[lockKey] = { timestamp: Date.now(), response: responseData };
   643→    delete _pixLocks[lockKey];
   644→    lockResolve(responseData);
   645→
   646→    // Responde pro Odoo IMEDIATAMENTE (o modulo Odoo grava os campos)
   647→    res.json(responseData);
   648→  } catch (err) {
   649→    console.error('[API/PIX] ERRO:', err.message);
   650→    var errResponse = { success: false, message: 'Erro ao criar PIX: ' + (err.message || 'Erro desconhecido') };
   651→    delete _pixLocks[lockKey];
   652→    lockResolve(errResponse);
   653→    res.json(errResponse);
   654→  }

   703→  var oc = config.odoo;

   705→
   706→  function odooRpc(model, method, args, kwargs) {
   707→    var xmlrpc = require('xmlrpc');
   708→    var base = oc.url.replace(/\/+$/, '');
   709→    var host = base.replace('https://', '');
   710→    var commonCli = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/common', port: 443 });
   711→    var modelsCli = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
   712→    return new Promise(function(ok, fail) {
   713→      commonCli.methodCall('authenticate', [oc.db, oc.user, oc.password, {}], function(e, uid) {
   714→        if (e || !uid) return fail(e || new Error('auth failed'));
   715→        var params = [oc.db, uid, oc.password, model, method, args || []];
   716→        if (kwargs) params.push(kwargs);
   717→        modelsCli.methodCall('execute_kw', params, function(e2, r) {
   718→          if (e2) fail(e2); else ok(r);
   719→        });
   720→      });
   721→    });
   722→  }
   723→
   724→  function postChatter(recordId, bodyHtml) {
   725→    return odooRpc('mail.message', 'create', [{
   726→      model: 'account.move',
   727→      res_id: parseInt(recordId),
   728→      body: bodyHtml,
   729→      message_type: 'comment',
   730→    }]);
   731→  }
   732→
   733→  if (!odooId || !oc || !oc.enabled || !oc.url || !oc.db || !oc.user || !oc.password) {

   737→  try {

   741→    var rawJson = '';

   896→
   897→  try {
   898→    var ontem = new Date(Date.now() - 24 * 3600000);
   899→    var di = ontem.toISOString().substring(0, 10);
   900→    var df = di;
   901→
   902→    console.log('[EXTRATO-CRON] Execucao automatica diaria: ' + di);
   903→    var transacoes = await buscarExtratoNormalizado(di, df);
   904→

   908→    }
   909→
   910→    var resultado = await criarExtrato(df, transacoes);
   911→    console.log('[EXTRATO-CRON] Extrato criado: ' + resultado.statementId + ' (' + transacoes.length + ' linhas)');
   912→
   913→    res.json({ success: true, transacoes: transacoes.length, statement_id: resultado.statementId, saldo: resultado.balance });
