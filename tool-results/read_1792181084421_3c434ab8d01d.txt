   104→// ============================================================
   105→// XML-RPC connection (singleton)
   106→// ============================================================
   107→var _cached = null;
   108→
   109→function getConn() {
   110→  if (_cached && _cached.uid) return _cached;
   111→  var odooConfig = config.odoo;
   112→  var base = odooConfig.url.replace(/\/+$/, '');
   113→  var host = base.replace('https://', '').replace('http://', '');
   114→  var common = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/common', port: 443 });
   115→  var models = xmlrpc.createSecureClient({ host: host, path: '/xmlrpc/2/object', port: 443 });
   116→  _cached = { uid: null, common: common, models: models, db: odooConfig.db, password: odooConfig.password };
   117→  return _cached;
   118→}
   119→
   120→function authenticate() {
   121→  return new Promise(function(resolve, reject) {
   122→    var conn = getConn();
   123→    conn.common.methodCall('authenticate', [conn.db, config.odoo.user, conn.password, {}], function(err, uid) {
   124→      if (err) { reject(new Error('Auth Odoo falhou: ' + (err.message || JSON.stringify(err)))); }
   125→      else if (uid === false || uid === null) { reject(new Error('Auth Odoo: credenciais invalidas (UID=false)')); }
   126→      else { conn.uid = uid; logger.info('[ODOO-TE] Autenticado. UID: ' + uid); resolve(uid); }
   127→    });
   128→  });
   129→}
   130→
   131→function ensureAuth() {
   132→  var conn = getConn();
   133→  if (conn.uid) return Promise.resolve(conn.uid);
   134→  return authenticate();
   135→}
   136→
   137→function executeKw(model, method, args, kwargs) {
   138→  return new Promise(function(resolve, reject) {
   139→    ensureAuth().then(function(uid) {
   140→      var conn = getConn();
   141→      var params = [conn.db, uid, conn.password, model, method, args || []];
   142→      if (kwargs) params.push(kwargs);
   143→      conn.models.methodCall('execute_kw', params, function(err, result) {
   144→        if (err) { reject(new Error(model + '.' + method + ': ' + (err.message || JSON.stringify(err)))); }
   145→        else { resolve(result); }
   146→      });
   147→    }).catch(reject);
   148→  });
   149→}

   151→// ============================================================
   152→// Safe read/write — tenta com x_studio, fallback sem
   153→// ============================================================
   154→var _customFieldsCache = {};
   155→
   156→async function safeReadCustom(model, ids, nativeFields, customFields) {
   157→  try {
   158→    return await executeKw(model, 'read', [[ids, nativeFields.concat(customFields)]]);
   159→  } catch (err) {
   160→    if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   161→      if (!_customFieldsCache[model]) {
   162→        _customFieldsCache[model] = { available: false, checked: true };
   163→        logger.warn('[ODOO-TE] Campos x_studio_* nao encontrados em ' + model + '. Usando campos nativos.');
   164→      }
   165→      return await executeKw(model, 'read', [[ids, nativeFields]]);
   166→    }
   167→    throw err;
   168→  }
   169→}
   170→
   171→async function safeWriteCustom(model, ids, vals) {
   172→  var customVals = {};
   173→  var nativeVals = {};
   174→  Object.keys(vals).forEach(function(k) {
   175→    if (k.indexOf('x_studio_') === 0) customVals[k] = vals[k];
   176→    else nativeVals[k] = vals[k];
   177→  });
   178→  if (Object.keys(nativeVals).length > 0) await executeKw(model, 'write', [ids, nativeVals]);
   179→  if (Object.keys(customVals).length > 0) {
   180→    try { await executeKw(model, 'write', [ids, customVals]); }
   181→    catch (err) {
   182→      if (err.message && (err.message.indexOf('Invalid field') !== -1 || err.message.indexOf('KeyError') !== -1)) {
   183→        if (!_customFieldsCache[model] || !_customFieldsCache[model].warned) {
   184→          logger.warn('[ODOO-TE] Campo x_studio_* nao existe em ' + model + '. Gravacao ignorada.');
   185→          if (!_customFieldsCache[model]) _customFieldsCache[model] = {};
   186→          _customFieldsCache[model].warned = true;
   187→        }
   188→      } else throw err;
   189→    }
   190→  }
   191→}

   200→async function searchRead(model, domain, fields, limit, offset, order) {
   201→  var kwargs = { fields: fields, limit: limit || 0, offset: offset || 0 };
   202→  if (order) kwargs.order = order;
   203→  var ids = await executeKw(model, 'search', [domain], kwargs);
   204→  if (!ids || !ids.length) return [];
   205→  return await executeKw(model, 'read', [[ids, fields]]);
   206→}

   301→async function findSaleOrderByInvoice(invoiceId) {
   302→  console.log('[ODOO-TE] findSaleOrderByInvoice: invoiceId=' + invoiceId);
   303→
   304→  // Estrategia 1: sale.order.invoice_ids
   305→  try {
   306→    var saleIds = await executeKw('sale.order', 'search', [['invoice_ids', 'in', [invoiceId]]]);
   307→    if (saleIds && saleIds.length) {
   308→      var orders = await executeKw('sale.order', 'read', [[saleIds, ['id', 'name']]]);
   309→      if (orders && orders[0]) { console.log('[ODOO-TE] Estrategia 1 (invoice_ids) encontrou: ' + orders[0].name); return orders[0]; }
   310→    }
   311→    console.log('[ODOO-TE] Estrategia 1 (invoice_ids): sem resultados');
   312→  } catch (err) { console.warn('[ODOO-TE] Estrategia 1 falhou: ' + err.message); }
   313→
   314→  // Estrategia 2: sale_order_ids da fatura (campo computado)
   315→  try {
   316→    var invData = await executeKw('account.move', 'read', [[[invoiceId], ['id', 'partner_id', 'sale_order_ids']]]);
   317→    var inv = invData && invData[0];
   318→    console.log('[ODOO-TE] Fatura lida: id=' + (inv ? inv.id : 'null') + ', partner=' + JSON.stringify(inv ? inv.partner_id : null) + ', sale_order_ids=' + JSON.stringify(inv ? inv.sale_order_ids : null));
   319→    if (inv && inv.sale_order_ids && inv.sale_order_ids.length) {
   320→      var soId = Array.isArray(inv.sale_order_ids[0]) ? inv.sale_order_ids[0][0] : inv.sale_order_ids[0];
   321→      var orders = await executeKw('sale.order', 'read', [[[soId], ['id', 'name']]]);
   322→      if (orders && orders[0]) { console.log('[ODOO-TE] Estrategia 2 (sale_order_ids) encontrou: ' + orders[0].name); return orders[0]; }
   323→    }
   324→    console.log('[ODOO-TE] Estrategia 2 (sale_order_ids): sem resultados');
   325→  } catch (err) { console.warn('[ODOO-TE] Estrategia 2 falhou: ' + err.message); }
   326→
   327→  // Estrategia 3: account.move.line -> sale_line_ids -> order_id
   328→  try {
   329→    var invData = await executeKw('account.move', 'read', [[[invoiceId], ['line_ids']]]);

   341→            if (saleLines && saleLines[0] && saleLines[0].order_id) {
   342→              var orderId = Array.isArray(saleLines[0].order_id) ? saleLines[0].order_id[0] : saleLines[0].order_id;

   344→              if (orders && orders[0]) { console.log('[ODOO-TE] Estrategia 3 (lines) encontrou: ' + orders[0].name); return orders[0]; }

   346→          }
   347→        }
   348→      }
   349→    }

   351→  } catch (err) { console.warn('[ODOO-TE] Estrategia 3 falhou: ' + err.message); }
   352→
   353→  // NOTA: Estrategia 4 (partner_id fallback) removida — achava vendas nao
   354→  // relacionadas ao mesmo cliente. Para send-invoice sem venda vinculada,
   355→  // o fluxo correto e usar as linhas da fatura diretamente.
   356→
   357→  console.error('[ODOO-TE] Nenhuma estrategia encontrou sale.order para invoice ' + invoiceId);
   358→  return null;
   359→}

   394→async function getInvoiceLines(invoiceId) {
   395→  try {
   396→    // Evita account.move.line.search — Odoo 19.2 SaaS tem override de
   397→    // search_fetch com _get_recon_limit_from_domain que rejeita certos dominios.
   398→    // Em vez disso, le line_ids direto da fatura.
   399→    var invData = await executeKw('account.move', 'read', [[[invoiceId], ['line_ids']]]);
   400→    var inv = invData && invData[0];
   401→    if (!inv || !inv.line_ids || !inv.line_ids.length) return [];
   402→
   403→    // line_ids pode vir como [id, id, ...] ou [[id, name], ...]
   404→    var lineIds = inv.line_ids.map(function(item) {
   405→      return Array.isArray(item) ? item[0] : item;
   406→    });
   407→
   408→    var lines = await executeKw('account.move.line', 'read', [[lineIds, INVOICE_LINE_FIELDS]]);
   409→    if (!lines || !lines.length) return [];
   410→
   411→    // Filtra apenas linhas com produto (product_id != false)
   412→    return lines.filter(function(line) {
   413→      if (!line.product_id) return false;
   414→      // product_id pode ser [id, name] ou false
   415→      if (Array.isArray(line.product_id) && line.product_id[0]) return true;
   416→      if (typeof line.product_id === 'number' && line.product_id) return true;
   417→      return false;
   418→    });
   419→  } catch (err) {
   420→    logger.warn('[ODOO-TE] getInvoiceLines falhou: ' + err.message);
   421→    return [];
   422→  }
   423→}
