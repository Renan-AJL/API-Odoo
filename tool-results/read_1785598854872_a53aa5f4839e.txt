    34→    34→function createClient(url) {
    35→    35→  var base = url.replace(/\/+$/, '');
    36→    36→  var host = base.replace('https://', '').replace('http://', '');
    37→    37→  var port = base.startsWith('https') ? 443 : 80;
    38→    38→  var isSecure = base.startsWith('https');
    39→    39→  var createFn = isSecure ? xmlrpc.createSecureClient : xmlrpc.createClient;
    40→    40→  return {
    41→    41→    common: createFn({ host: host, path: '/xmlrpc/2/common', port: port }),
    42→    42→    models: createFn({ host: host, path: '/xmlrpc/2/object', port: port }),
    43→    43→  };
    44→    44→}
    45→    45→
    46→    46→function authenticate(client, db, login, password) {
    47→    47→  return new Promise(function(resolve, reject) {
    48→    48→    client.common.methodCall('authenticate', [db, login, password, {}], function(err, uid) {
    49→    49→      if (err) reject(new Error('Auth Odoo falhou: ' + (err.message || JSON.stringify(err))));
    50→    50→      else if (uid === false || uid === null) reject(new Error('Credenciais Odoo invalidas'));
    51→    51→      else resolve(uid);
    52→    52→    });
    53→    53→  });
    54→    54→}
    55→    55→
    56→    56→function executeKw(client, db, uid, password, model, method, args, kwargs) {
    57→    57→  return new Promise(function(resolve, reject) {
    58→    58→    var params = [db, uid, password, model, method, args || []];
    59→    59→    if (kwargs) params.push(kwargs);
    60→    60→    client.models.methodCall('execute_kw', params, function(err, result) {
    61→    61→      if (err) reject(new Error(model + '.' + method + ': ' + (err.message || JSON.stringify(err))));
    62→    62→      else resolve(result);
    63→    63→    });
    64→    64→  });
    65→    65→}

    70→    70→async function processPendingEmissions() {
    71→    71→  var odoo = config.odoo;
    72→    72→  if (!odoo || !odoo.enabled || !odoo.url) {
    73→    73→    console.log('[SIEG-EMIT] Odoo nao configurado');
    74→    74→    return { processed: 0, reason: 'odoo_not_configured' };
    75→    75→  }
    76→    76→
    77→    77→  var client = createClient(odoo.url);
    78→    78→  var uid = await authenticate(client, odoo.db, odoo.user, odoo.password);
    79→    79→  var db = odoo.db;
    80→    80→  var pwd = odoo.password;
    81→    81→
    82→    82→  console.log('[SIEG-EMIT] Buscando faturas pendentes...');
    83→    83→
    84→    84→  // Buscar NF-e pendentes
    85→    85→  var nfeIds = await executeKw(client, db, uid, pwd, 'account.move', 'search', [[
    86→    86→    ['move_type', '=', 'out_invoice'],
    87→    87→    ['x_studio_nfe_status', '=', 'pendente'],
    88→    88→  ]], { order: 'id asc', limit: 5 });
    89→    89→
    90→    90→  // Buscar NFS-e pendentes
    91→    91→  var nfseIds = await executeKw(client, db, uid, pwd, 'account.move', 'search', [[
    92→    92→    ['move_type', '=', 'out_invoice'],
    93→    93→    ['x_studio_nfse_status', '=', 'pendente'],
    94→    94→  ]], { order: 'id asc', limit: 5 });
    95→    95→
    96→    96→  var allPending = [];
    97→    97→  for (var i = 0; i < nfeIds.length; i++) {
    98→    98→    allPending.push({ id: nfeIds[i], tipo: 'nfe' });
    99→    99→  }
   100→   100→  for (var j = 0; j < nfseIds.length; j++) {
   101→   101→    allPending.push({ id: nfseIds[j], tipo: 'nfse' });
   102→   102→  }
   103→   103→
   104→   104→  if (!allPending.length) {
   105→   105→    console.log('[SIEG-EMIT] Nenhuma fatura pendente');
   106→   106→    return { processed: 0 };
   107→   107→  }
   108→   108→
   109→   109→  console.log('[SIEG-EMIT] ' + allPending.length + ' fatura(s) pendente(s)');
   110→   110→  var results = [];
   111→   111→
   112→   112→  for (var k = 0; k < allPending.length; k++) {
   113→   113→    var item = allPending[k];
   114→   114→    try {
   115→   115→      var r = await processOne(client, db, uid, pwd, item.id, item.tipo);
   116→   116→      results.push(r);
   117→   117→    } catch (err) {
   118→   118→      console.error('[SIEG-EMIT] ERRO fatura ' + item.id + ':', err.message);
   119→   119→      await safeUpdateError(client, db, uid, pwd, item.id, item.tipo, err.message);
   120→   120→      results.push({ move_id: item.id, sucesso: false, tipo: item.tipo, erro: err.message });
   121→   121→    }
   122→   122→  }
   123→   123→
   124→   124→  var ok = results.filter(function(r) { return r.sucesso; }).length;
   125→   125→  console.log('[SIEG-EMIT] Concluido: ' + ok + '/' + results.length + ' sucesso');
   126→   126→  return { processed: results.length, sucesso: ok, detalhes: results };
   127→   127→}

   132→   132→async function processOne(client, db, uid, pwd, moveId, tipo) {
   133→   133→  console.log('[SIEG-EMIT] --- Fatura ID: ' + moveId + ' | Tipo: ' + tipo + ' ---');
   134→   134→
   135→   135→  // 1. Read account.move (sem invoice_line_ids — pode ser computed nao-stored no Odoo 19)
   136→   136→  var moves = await executeKw(client, db, uid, pwd, 'account.move', 'read', [[moveId], [
   137→   137→    'name', 'partner_id', 'company_id', 'invoice_date', 'date',
   138→   138→    'amount_total', 'narration',
   139→   139→    'x_studio_nfe_status', 'x_studio_nfse_status', 'payment_state',
   140→   140→  ]]);
   141→   141→  if (!moves || !moves.length) throw new Error('Fatura ' + moveId + ' nao encontrada');
   142→   142→  var move = moves[0];
   143→   143→  console.log('[SIEG-EMIT] Fatura: ' + move.name + ' | R$ ' + (move.amount_total || 0));
   144→   144→
   145→   145→  // 2. Read company
   146→   146→  var companyId = tupId(move.company_id);
   147→   147→  var company = await readCompany(client, db, uid, pwd, companyId);
   148→   148→
   149→   149→  // 3. Read partner
   150→   150→  var partnerId = tupId(move.partner_id);
   151→   151→  if (!partnerId) throw new Error('Fatura sem parceiro');
   152→   152→  var partner = await readPartner(client, db, uid, pwd, partnerId);
   153→   153→
   154→   154→  // 4. Read invoice lines diretamente de account.move.line (evita computed field)
   155→   155→  var allLineIds = await executeKw(client, db, uid, pwd, 'account.move.line', 'search', [[
   156→   156→    ['move_id', '=', moveId],
   157→   157→  ]], { order: 'id asc' });
   158→   158→  console.log('[SIEG-EMIT] Linhas encontradas na fatura: ' + allLineIds.length);
   159→   159→  var rawLines = await executeKw(client, db, uid, pwd, 'account.move.line', 'read', [allLineIds, [
   160→   160→    'display_type', 'product_id', 'name', 'quantity', 'price_unit',
   161→   161→    'price_subtotal', 'tax_ids', 'discount',
   162→   162→  ]]);
   163→   163→  // Debug: log todas as linhas
   164→   164→  for (var li = 0; li < rawLines.length; li++) {
   165→   165→    var l = rawLines[li];
   166→   166→    var pName = 'N/A';
   167→   167→    if (l.product_id) pName = Array.isArray(l.product_id) ? l.product_id[1] : String(l.product_id);
   168→   168→    console.log('[SIEG-EMIT]   Line ' + li + ': display_type=' + JSON.stringify(l.display_type) + ' product=' + pName + ' qty=' + l.quantity + ' price=' + l.price_unit);
   169→   169→  }
   170→   170→  // Filtro principal: linhas com produto associado (mais confiavel que display_type)
   171→   171→  var invoiceLines = rawLines.filter(function(l) { return l.product_id; });
   172→   172→  // Fallback: se nenhuma tem product_id, tenta por display_type
   173→   173→  if (!invoiceLines.length) {
   174→   174→    invoiceLines = rawLines.filter(function(l) {
   175→   175→      var dt = l.display_type;
   176→   176→      return !dt || dt === 'product' || dt === 'service';
   177→   177→    });
   178→   178→  }
   179→   179→  if (!invoiceLines.length) throw new Error('Fatura sem linhas de produto/servico (total: ' + rawLines.length + ', com product: 0)');
   180→   180→  console.log('[SIEG-EMIT] Linhas de produto: ' + invoiceLines.length);
   181→   181→
   182→   182→  // 5. Get next NF number from company
   183→   183→  var serie, numField, nextNum;
   184→   184→  if (tipo === 'nfe') {
   185→   185→    serie = String(company.x_studio_sieg_serie_nfe || '100');
   186→   186→    numField = 'x_studio_sieg_ultimo_nfe';
   187→   187→  } else {
   188→   188→    serie = String(company.x_studio_sieg_serie_nfse || '1');
   189→   189→    numField = 'x_studio_sieg_ultimo_nfse';
   190→   190→  }
   191→   191→  nextNum = (parseInt(company[numField]) || 0) + 1;
   192→   192→  console.log('[SIEG-EMIT] Serie: ' + serie + ' | nNF: ' + nextNum);
   193→   193→
   194→   194→  // 6. Build line data with taxes + product x_studio_ fields
   195→   195→  var linesData = [];
   196→   196→  for (var i = 0; i < invoiceLines.length; i++) {
   197→   197→    var ld = await buildLineData(client, db, uid, pwd, invoiceLines[i]);
   198→   198→    linesData.push(ld);
   199→   199→  }
   200→   200→
   201→   201→  // 7. Build emission payload
   202→   202→  // Nota: usar amount_untaxed (sem impostos) para NF-e com CSOSN 103
   203→   203→  // pois vNF deve ser igual a soma dos componentes (vProd + vICMS + ...)
   204→   204→  // e para CSOSN 103 o ICMS e 0, entao vNF = vProd
   205→   205→  var vProdSum = 0;
   206→   206→  for (var li = 0; li < linesData.length; li++) {
   207→   207→    vProdSum += parseFloat(linesData[li].price_subtotal) || 0;
   208→   208→  }
   209→   209→
   210→   210→  // Pagamentos: NF-e 4.00 exige <pag> — usar valor total da fatura
   211→   211→  var pagamentos = [{
   212→   212→    tPag: '15', // PIX
   213→   213→    vPag: String(vProdSum.toFixed(2)),
   214→   214→  }];
   215→   215→
   216→   216→  var emitData = {
   217→   217→    company: company,
   218→   218→    partner: partner,
   219→   219→    order: {
   220→   220→      name: move.name,
   221→   221→      number: String(nextNum),
   222→   222→      date_order: move.invoice_date || move.date,
   223→   223→      amount_total: vProdSum, // usar soma dos vProd (consistente com XML)
   224→   224→      note: move.narration || '',
   225→   225→    },
   226→   226→    lines: linesData,
   227→   227→    pagamentos: pagamentos,
   228→   228→    config: {
   229→   229→      serie: serie,
   230→   230→      tpAmb: process.env.SIEG_TP_AMB || config.sieg.tpAmb || '2',
   231→   231→      natOp: 'Venda de Mercadoria',
   232→   232→      mod: tipo === 'nfe' ? '55' : '01',
   233→   233→    },
   234→   234→    tipo: tipo,
   235→   235→  };
   236→   236→
   237→   237→  // NFS-e: build service block from product x_studio_ fields
   238→   238→  if (tipo === 'nfse') {
   239→   239→    emitData.service = buildServiceBlock(linesData, move);
   240→   240→  }
   241→   241→
   242→   242→  // 8. Set status to 'processando'
   243→   243→  var processVals = {};
   244→   244→  if (tipo === 'nfe') processVals.x_studio_nfe_status = 'processando';
   245→   245→  else processVals.x_studio_nfse_status = 'processando';
   246→   246→  await executeKw(client, db, uid, pwd, 'account.move', 'write', [[moveId], processVals]);
   247→   247→
   248→   248→  // 9. Call SIEG API
   249→   249→  console.log('[SIEG-EMIT] Enviando ao SIEG...');
   250→   250→  var resultado;
   251→   251→  try {
   252→   252→    resultado = await emitirNota(emitData);
   253→   253→  } catch (siegErr) {
   254→   254→    // Captura erros HTTP (ex: 409 com detalhes no body)
   255→   255→    var siegDetail = siegErr.response && siegErr.response.data;
   256→   256→    var errMsg = 'SIEG HTTP ' + (siegErr.response ? siegErr.response.status : 'erro') + ': ';
   257→   257→    if (siegDetail) {
   258→   258→      // Tentar extrair ErrorMessage do formato SIEG (PascalCase)
   259→   259→      var siegMsg = siegDetail.ErrorMessage || siegDetail.Message || siegDetail.message || '';
   260→   260→      if (siegMsg) {
   261→   261→        errMsg += siegMsg;
   262→   262→      } else {
   263→   263→        errMsg += typeof siegDetail === 'string' ? siegDetail : JSON.stringify(siegDetail).slice(0, 800);
   264→   264→      }
   265→   265→    } else {
   266→   266→      errMsg += siegErr.message;
   267→   267→    }
   268→   268→    console.error('[SIEG-EMIT] Erro SIEG:', errMsg);
   269→   269→    throw new Error(errMsg);
   270→   270→  }
   271→   271→  console.log('[SIEG-EMIT] SIEG retornou - sucesso: ' + resultado.sucesso + (resultado.httpStatus ? ' (HTTP ' + resultado.httpStatus + ')' : ''));
   272→   272→
   273→   273→  // Se SIEG retornou erro (IsSuccess=false), extrair mensagem detalhada
   274→   274→  if (!resultado.sucesso && resultado.erro) {
   275→   275→    console.error('[SIEG-EMIT] SIEG rejeitou o XML: ' + resultado.erro);
   276→   276→  }
   277→   277→
   278→   278→  // 10. Parse result
   279→   279→  var info = parseResult(resultado, tipo);
   280→   280→  var xmlRetornoText = '';
   281→   281→  var rawXmlAuth = resultado.resposta && (resultado.resposta.Xml || resultado.resposta.xml || resultado.resposta.xmlBase64);
   282→   282→  if (rawXmlAuth) {
   283→   283→    xmlRetornoText = (typeof rawXmlAuth === 'string' && rawXmlAuth.length > 200)
   284→   284→      ? rawXmlAuth
   285→   285→      : Buffer.from(rawXmlAuth, 'base64').toString('utf-8');
   286→   286→  }
   287→   287→
   288→   288→  // 11. Update account.move with results
   289→   289→  var updateVals = {};
   290→   290→  if (info.sucesso) {
   291→   291→    if (tipo === 'nfe') {
   292→   292→      updateVals.x_studio_nfe_status = 'autorizada';
   293→   293→      updateVals.x_studio_nfe_chave = info.chave;
   294→   294→      updateVals.x_studio_nfe_protocolo = info.protocolo;
   295→   295→    } else {
   296→   296→      updateVals.x_studio_nfse_status = 'autorizada';
   297→   297→      updateVals.x_studio_nfse_numero = String(nextNum);
   298→   298→    }
   299→   299→
   300→   300→    // Increment NF number (only on authorization)
   301→   301→    try {
   302→   302→      var companyUpdate = {};
   303→   303→      companyUpdate[numField] = nextNum;
   304→   304→      await executeKw(client, db, uid, pwd, 'res.company', 'write', [[companyId], companyUpdate]);
   305→   305→      console.log('[SIEG-EMIT] Sequencia ' + numField + ' = ' + nextNum);
   306→   306→    } catch (seqErr) {
   307→   307→      console.error('[SIEG-EMIT] Erro ao atualizar sequencia:', seqErr.message);
   308→   308→    }
   309→   309→  } else {
   310→   310→    if (tipo === 'nfe') {
   311→   311→      updateVals.x_studio_nfe_status = info.cStat + ' - ' + info.motivo;
   312→   312→    } else {
   313→   313→      updateVals.x_studio_nfse_status = info.motivo;
   314→   314→    }
   315→   315→  }
   316→   316→
   317→   317→  await executeKw(client, db, uid, pwd, 'account.move', 'write', [[moveId], updateVals]);
   318→   318→  console.log('[SIEG-EMIT] Fatura atualizada');
   319→   319→
   320→   320→  // 12. Post XML + PDF to Chatter
   321→   321→  await postChatterResult(client, db, uid, pwd, moveId, move.name, tipo, info, resultado, xmlRetornoText);
   322→   322→
   323→   323→  return {
   324→   324→    move_id: moveId, sucesso: info.sucesso, tipo: tipo,
   325→   325→    chave: info.chave, numero: nextNum, serie: serie,
   326→   326→  };
   327→   327→}

   329→   329→// ============================================================
   330→   330→// Post XML + PDF to Odoo Chatter
   331→   331→// ============================================================
   332→   332→async function postChatterResult(client, db, uid, pwd, moveId, moveName, tipo, info, resultado, xmlRetornoText) {
   333→   333→  var attachIds = [];
   334→   334→  var safeName = (moveName || 'Fatura').replace(/[^a-zA-Z0-9\-_]/g, '_');
   335→   335→  var nfLabel = tipo === 'nfe' ? 'NF-e' : 'NFS-e';
   336→   336→  var xmlContent = xmlRetornoText || resultado.xmlEnviado || '';
   337→   337→  var pdfBase64 = resultado.pdfBase64 || '';
   338→   338→
   339→   339→  // Attach XML
   340→   340→  if (xmlContent) {
   341→   341→    try {
   342→   342→      var xmlB64 = Buffer.from(xmlContent, 'utf-8').toString('base64');
   343→   343→      var xmlAttach = await executeKw(client, db, uid, pwd, 'ir.attachment', 'create', [{
   344→   344→        name: nfLabel + '_' + safeName + '.xml',
   345→   345→        datas: xmlB64,
   346→   346→        res_model: 'account.move',
   347→   347→        res_id: moveId,
   348→   348→        mimetype: 'application/xml',
   349→   349→      }]);
   350→   350→      attachIds.push(xmlAttach);
   351→   351→      console.log('[SIEG-EMIT] XML anexado (ID: ' + xmlAttach + ')');
   352→   352→    } catch (e) { console.error('[SIEG-EMIT] Erro anexar XML:', e.message); }
   353→   353→  }
   354→   354→
   355→   355→  // Attach PDF (DANFE / DANFSE)
   356→   356→  if (pdfBase64) {
   357→   357→    try {
   358→   358→      var pdfLabel = tipo === 'nfe' ? 'DANFE' : 'DANFSE';
   359→   359→      var pdfAttach = await executeKw(client, db, uid, pwd, 'ir.attachment', 'create', [{
   360→   360→        name: pdfLabel + '_' + safeName + '.pdf',
   361→   361→        datas: pdfBase64,
   362→   362→        res_model: 'account.move',
   363→   363→        res_id: moveId,
   364→   364→        mimetype: 'application/pdf',
   365→   365→      }]);
   366→   366→      attachIds.push(pdfAttach);
   367→   367→      console.log('[SIEG-EMIT] PDF anexado (ID: ' + pdfAttach + ')');
   368→   368→    } catch (e) { console.error('[SIEG-EMIT] Erro anexar PDF:', e.message); }
   369→   369→  }
   370→   370→
   371→   371→  // Post chatter message
   372→   372→  var body = '';
   373→   373→  if (info.sucesso) {
   374→   374→    body = '<b>' + nfLabel + ' Emitida com Sucesso!</b><br/>';
   375→   375→    if (tipo === 'nfe') {
   376→   376→      body += 'Chave: ' + info.chave + '<br/>';
   377→   377→      body += 'Protocolo: ' + info.protocolo + '<br/>';
   378→   378→    } else {
   379→   379→      body += 'Numero: ' + info.numero + '<br/>';
   380→   380→    }
   381→   381→    body += 'Status: ' + info.cStat + ' - ' + info.motivo;
   382→   382→    body += '<br/><br/><i>Arquivos anexados: XML e ' + (tipo === 'nfe' ? 'DANFE' : 'DANFSE') + ' (PDF)</i>';
   383→   383→  } else {

   693→   693→async function extractTaxes(client, db, uid, pwd, line) {
   694→   694→  var result = {
   695→   695→    csosn: '103', cst_icms: '', vbc: 0, vicms: 0, picms: 0,
   696→   696→    cst_pis: '99', vbc_pis: 0, ppis: 0, vpis: 0,
   697→   697→    cst_cofins: '99', vbc_cofins: 0, pcofins: 0, vcofins: 0,
   698→   698→  };
   699→   699→
   700→   700→  var taxIds = (line.tax_ids || []).map(function(t) { return Array.isArray(t) ? t[0] : t; }).filter(Boolean);
   701→   701→  if (!taxIds.length) return result;
   702→   702→
   703→   703→  try {
   704→   704→    var taxes = await executeKw(client, db, uid, pwd, 'account.tax', 'read', [taxIds, [
   705→   705→      'name', 'amount', 'amount_type', 'description',
   706→   706→    ]]);
   707→   707→    if (!taxes || !taxes.length) return result;
   708→   708→
   709→   709→    var base = parseFloat(line.price_subtotal) || 0;
   710→   710→    var totalIcms = 0, totalPis = 0, totalCofins = 0;
   711→   711→
   712→   712→    for (var i = 0; i < taxes.length; i++) {
   713→   713→      var tax = taxes[i];
   714→   714→      var n = ((tax.name || tax.description || '').toUpperCase());
   715→   715→      var amt = parseFloat(tax.amount) || 0;
   716→   716→      if (n.indexOf('ICMS') >= 0 || n.indexOf('CSOSN') >= 0) {
   717→   717→        totalIcms += amt;
   718→   718→        var m = n.match(/(10[0-9]|20[0-9]|300|400|500|900)/);
   719→   719→        if (m) result.csosn = m[1];
   720→   720→      } else if (n.indexOf('PIS') >= 0) { totalPis += amt; }
   721→   721→      else if (n.indexOf('COFINS') >= 0) { totalCofins += amt; }
   722→   722→    }
   723→   723→
   724→   724→    result.picms = totalIcms; result.vbc = base; result.vicms = base * (totalIcms / 100);
   725→   725→    result.ppis = totalPis; result.vbc_pis = base; result.vpis = base * (totalPis / 100);
   726→   726→    result.pcofins = totalCofins; result.vbc_cofins = base; result.vcofins = base * (totalCofins / 100);
   727→   727→  } catch (e) { console.warn('[SIEG-EMIT] Erro ao extrair impostos:', e.message); }
   728→   728→
   729→   729→  return result;
   730→   730→}
