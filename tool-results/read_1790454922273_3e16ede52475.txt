     7→ *   account.move:
     8→ *     x_studio_nfe_status     (Selection: vazio, pendente, processando, autorizada, erro)
     9→ *     x_studio_nfe_chave      (Char)
    10→ *     x_studio_nfe_protocolo  (Char)
    11→ *     x_studio_nfse_numero    (Char)
    12→ *     x_studio_nfse_status    (Selection: vazio, pendente, processando, autorizada, erro)
    13→ *   res.company:
    14→ *     x_studio_sieg_ultimo_nfe  (Integer)
    15→ *     x_studio_sieg_ultimo_nfse (Integer)
    16→ *     x_studio_sieg_serie_nfe   (Char, default 100)
    17→ *     x_studio_sieg_serie_nfse  (Char, default 1)
    18→ *   product.product:
    19→ *     x_studio_c_trib_nac       (Char) — Codigo tributo nacional (NFS-e)
    20→ *     x_studio_c_nbs            (Char) — Codigo NBS (NFS-e)
    21→ *     x_studio_aliquota_iss    (Char) — Aliquota ISS (NFS-e)
    22→ *     x_studio_ibge_code       (Char) — Codigo IBGE municipio

    24→ * Chatter: apos emissao, XML + PDF (DANFE/DANFSE) aparecem como anexos.

    36→function createClient(url) {
    37→  var base = url.replace(/\/+$/, '');
    38→  var host = base.replace('https://', '').replace('http://', '');
    39→  var port = base.startsWith('https') ? 443 : 80;
    40→  var isSecure = base.startsWith('https');
    41→  var createFn = isSecure ? xmlrpc.createSecureClient : xmlrpc.createClient;
    42→  return {
    43→    common: createFn({ host: host, path: '/xmlrpc/2/common', port: port }),
    44→    models: createFn({ host: host, path: '/xmlrpc/2/object', port: port }),
    45→  };
    46→}
    47→
    48→function authenticate(client, db, login, password) {
    49→  return new Promise(function(resolve, reject) {
    50→    client.common.methodCall('authenticate', [db, login, password, {}], function(err, uid) {
    51→      if (err) reject(new Error('Auth Odoo falhou: ' + (err.message || JSON.stringify(err))));
    52→      else if (uid === false || uid === null) reject(new Error('Credenciais Odoo invalidas'));
    53→      else resolve(uid);
    54→    });
    55→  });
    56→}
    57→
    58→function executeKw(client, db, uid, password, model, method, args, kwargs) {
    59→  return new Promise(function(resolve, reject) {
    60→    var params = [db, uid, password, model, method, args || []];
    61→    if (kwargs) params.push(kwargs);
    62→    client.models.methodCall('execute_kw', params, function(err, result) {
    63→      if (err) reject(new Error(model + '.' + method + ': ' + (err.message || JSON.stringify(err))));
    64→      else resolve(result);
    65→    });
    66→  });
    67→}

    69→// ============================================================
    70→// Main: Process Pending Emissions
    71→// ============================================================
    72→async function processPendingEmissions() {
    73→  var odoo = config.odoo;
    74→  if (!odoo || !odoo.enabled || !odoo.url) {
    75→    console.log('[SIEG-EMIT] Odoo nao configurado');
    76→    return { processed: 0, reason: 'odoo_not_configured' };
    77→  }
    78→
    79→  var client = createClient(odoo.url);
    80→  var uid = await authenticate(client, odoo.db, odoo.user, odoo.password);
    81→  var db = odoo.db;
    82→  var pwd = odoo.password;
    83→
    84→  console.log('[SIEG-EMIT] Buscando faturas pendentes...');
    85→
    86→  // Buscar NF-e pendentes
    87→  var nfeIds = await executeKw(client, db, uid, pwd, 'account.move', 'search', [[
    88→    ['move_type', '=', 'out_invoice'],
    89→    ['x_studio_nfe_status', '=', 'pendente'],
    90→  ]], { order: 'id asc', limit: 5 });
    91→
    92→  // Buscar NFS-e pendentes
    93→  var nfseIds = await executeKw(client, db, uid, pwd, 'account.move', 'search', [[
    94→    ['move_type', '=', 'out_invoice'],
    95→    ['x_studio_nfse_status', '=', 'pendente'],
    96→  ]], { order: 'id asc', limit: 5 });
    97→
    98→  var allPending = [];
    99→  for (var i = 0; i < nfeIds.length; i++) {
   100→    allPending.push({ id: nfeIds[i], tipo: 'nfe' });
   101→  }
   102→  for (var j = 0; j < nfseIds.length; j++) {
   103→    allPending.push({ id: nfseIds[j], tipo: 'nfse' });
   104→  }
   105→
   106→  if (!allPending.length) {
   107→    console.log('[SIEG-EMIT] Nenhuma fatura pendente');
   108→    return { processed: 0 };
   109→  }
   110→
   111→  console.log('[SIEG-EMIT] ' + allPending.length + ' fatura(s) pendente(s)');
   112→  var results = [];
   113→
   114→  for (var k = 0; k < allPending.length; k++) {
   115→    var item = allPending[k];
   116→    try {
   117→      var r = await processOne(client, db, uid, pwd, item.id, item.tipo);
   118→      results.push(r);
   119→    } catch (err) {
   120→      console.error('[SIEG-EMIT] ERRO fatura ' + item.id + ':', err.message);
   121→      await safeUpdateError(client, db, uid, pwd, item.id, item.tipo, err.message);
   122→      results.push({ move_id: item.id, sucesso: false, tipo: item.tipo, erro: err.message });
   123→    }
   124→  }
   125→
   126→  var ok = results.filter(function(r) { return r.sucesso; }).length;
   127→  console.log('[SIEG-EMIT] Concluido: ' + ok + '/' + results.length + ' sucesso');
   128→  return { processed: results.length, sucesso: ok, detalhes: results };
   129→}

   131→// ============================================================
   132→// Process Single Invoice

   134→async function processOne(client, db, uid, pwd, moveId, tipo) {
   135→  console.log('[SIEG-EMIT] --- Fatura ID: ' + moveId + ' | Tipo: ' + tipo + ' ---');
   136→
   137→  // 1. Read account.move (sem invoice_line_ids — pode ser computed nao-stored no Odoo 19)
   138→  var moves = await executeKw(client, db, uid, pwd, 'account.move', 'read', [[moveId], [
   139→    'name', 'partner_id', 'company_id', 'invoice_date', 'date',
   140→    'amount_total', 'narration',
   141→    'x_studio_nfe_status', 'x_studio_nfse_status', 'payment_state',
   142→  ]]);
   143→  if (!moves || !moves.length) throw new Error('Fatura ' + moveId + ' nao encontrada');
   144→  var move = moves[0];
   145→  console.log('[SIEG-EMIT] Fatura: ' + move.name + ' | R$ ' + (move.amount_total || 0));
   146→
   147→  // 2. Read company
   148→  var companyId = tupId(move.company_id);
   149→  var company = await readCompany(client, db, uid, pwd, companyId);
   150→  console.log('[SIEG-EMIT] [EMITENTE] CNPJ=' + company.cnpj_cpf + ' IE=' + company.inscr_est + ' cMun=' + company.city_ibge_code + ' xMun=' + company.city + ' UF=' + company.state + ' xLgr=' + company.street + ' nro=' + company.number);
   151→
   152→  // 3. Read partner
   153→  var partnerId = tupId(move.partner_id);
   154→  if (!partnerId) throw new Error('Fatura sem parceiro');
   155→  var partner = await readPartner(client, db, uid, pwd, partnerId);
   156→  console.log('[SIEG-EMIT] [DESTINATARIO] CNPJ=' + partner.cnpj_cpf + ' IE=' + (partner.inscr_est || '(vazio)') + ' xNome=' + partner.xNome + ' cMun=' + partner.city_ibge_code + ' xMun=' + partner.city + ' UF=' + partner.state + ' xLgr=' + partner.street + ' nro=' + partner.number);
   157→
   158→  // 3b. Auto-lookup IE do destinatario se vazio (SEFAZ cStat 232 exige IE para CFOP 5xxx/6xxx)
   159→  if (!partner.inscr_est || String(partner.inscr_est).trim() === '') {
   160→    var partnerCnpj = (partner.cnpj_cpf || '').replace(/\D/g, '');
   161→    var partnerUf = partner.state || '';
   162→    if (partnerCnpj.length === 14) {
   163→      console.log('[SIEG-EMIT] [DEST-IE] IE vazia no Odoo. Tentando auto-lookup CNPJ=' + partnerCnpj + ' UF=' + partnerUf);
   164→      try {
   165→        var ieResult = await _ieService.obterIE(partnerCnpj, null, partnerUf);
   166→        if (ieResult.hasIE && ieResult.ie) {
   167→          partner.inscr_est = ieResult.ie;
   168→          console.log('[SIEG-EMIT] [DEST-IE] IE encontrada: ' + ieResult.ie + ' (UF=' + (ieResult.ieState || partnerUf) + ') via ' + ieResult.source);
   169→          // Tentar salvar IE no Odoo para futuras emissoes
   170→          try {
   171→            await executeKw(client, db, uid, pwd, 'res.partner', 'write', [[partnerId], { inscr_est: ieResult.ie }]);
   172→            console.log('[SIEG-EMIT] [DEST-IE] IE salva no parceiro Odoo (id=' + partnerId + ')');
   173→          } catch (writeErr) {
   174→            console.warn('[SIEG-EMIT] [DEST-IE] Nao conseguiu salvar IE no Odoo: ' + writeErr.message);
   175→          }
   176→        } else {
   177→          console.warn('[SIEG-EMIT] [DEST-IE] IE nao encontrada em nenhuma fonte. Motivos: ' + JSON.stringify(ieResult.reasons || []));
   178→          console.warn('[SIEG-EMIT] [DEST-IE] Emissao provavelmente sera rejeitada pela SEFAZ (cStat 232) se CFOP exigir IE.');
   179→        }
   180→      } catch (ieErr) {
   181→        console.error('[SIEG-EMIT] [DEST-IE] Erro no auto-lookup de IE: ' + ieErr.message);
   182→      }
   183→    } else if (partnerCnpj.length === 11) {
   184→      console.log('[SIEG-EMIT] [DEST-IE] Destinatario CPF (consumidor final) — IE nao necessaria');
   185→    } else {
   186→      console.warn('[SIEG-EMIT] [DEST-IE] CNPJ/CPF invalido (' + partnerCnpj.length + ' digitos) — sem auto-lookup de IE');
   187→    }
   188→  } else {
   189→    console.log('[SIEG-EMIT] [DEST-IE] IE ja preenchida no Odoo: ' + partner.inscr_est);
   190→  }
   191→
   192→  // 4. Read invoice lines diretamente de account.move.line (evita computed field)
   193→  var allLineIds = await executeKw(client, db, uid, pwd, 'account.move.line', 'search', [[
   194→    ['move_id', '=', moveId],
   195→  ]], { order: 'id asc' });
   196→  console.log('[SIEG-EMIT] Linhas encontradas na fatura: ' + allLineIds.length);
   197→  var rawLines = await executeKw(client, db, uid, pwd, 'account.move.line', 'read', [allLineIds, [
   198→    'display_type', 'product_id', 'name', 'quantity', 'price_unit',
   199→    'price_subtotal', 'tax_ids', 'discount',
   200→  ]]);
   201→  // Debug: log todas as linhas
   202→  for (var li = 0; li < rawLines.length; li++) {
   203→    var l = rawLines[li];
   204→    var pName = 'N/A';
   205→    if (l.product_id) pName = Array.isArray(l.product_id) ? l.product_id[1] : String(l.product_id);
   206→    console.log('[SIEG-EMIT]   Line ' + li + ': display_type=' + JSON.stringify(l.display_type) + ' product=' + pName + ' qty=' + l.quantity + ' price=' + l.price_unit);
   207→  }
   208→  // Filtro principal: linhas com produto associado (mais confiavel que display_type)
   209→  var invoiceLines = rawLines.filter(function(l) { return l.product_id; });
   210→  // Fallback: se nenhuma tem product_id, tenta por display_type
   211→  if (!invoiceLines.length) {
   212→    invoiceLines = rawLines.filter(function(l) {
   213→      var dt = l.display_type;
   214→      return !dt || dt === 'product' || dt === 'service';
   215→    });
   216→  }
   217→  if (!invoiceLines.length) throw new Error('Fatura sem linhas de produto/servico (total: ' + rawLines.length + ', com product: 0)');
   218→  console.log('[SIEG-EMIT] Linhas de produto: ' + invoiceLines.length);
   219→
   220→  // 5. Get next NF number from company
   221→  var serie, numField, nextNum;
   222→  if (tipo === 'nfe') {
   223→    serie = String(company.x_studio_sieg_serie_nfe || '100');
   224→    numField = 'x_studio_sieg_ultimo_nfe';
   225→  } else {
   226→    serie = String(company.x_studio_sieg_serie_nfse || '1');
   227→    numField = 'x_studio_sieg_ultimo_nfse';
   228→  }
   229→  nextNum = (parseInt(company[numField]) || 0) + 1;
   230→  console.log('[SIEG-EMIT] Serie: ' + serie + ' | nNF: ' + nextNum);
   231→
   232→  // 6. Build line data with taxes + product x_studio_ fields
   233→  var linesData = [];
   234→  for (var i = 0; i < invoiceLines.length; i++) {
   235→    var ld = await buildLineData(client, db, uid, pwd, invoiceLines[i]);
   236→    linesData.push(ld);
   237→  }
   238→
   239→  // 7. Build emission payload
   240→  // Nota: usar amount_untaxed (sem impostos) para NF-e com CSOSN 103
   241→  // pois vNF deve ser igual a soma dos componentes (vProd + vICMS + ...)
   242→  // e para CSOSN 103 o ICMS e 0, entao vNF = vProd
   243→  var vProdSum = 0;
   244→  for (var li = 0; li < linesData.length; li++) {
   245→    vProdSum += parseFloat(linesData[li].price_subtotal) || 0;
   246→  }
   247→
   248→  // Pagamentos: NF-e 4.00 — vPag deve ser igual a vNF (total SEM IBS/CBS).
   249→  // IBS/CBS sao tributos "por fora" e nao entram no bloco <pag>.
   250→  // Se vPag > vNF sem vTroco declarado -> cStat 866.
   251→  var pagamentos = [{
   252→    tPag: '15', // PIX
   253→    vPag: String(vProdSum.toFixed(2)),
   254→  }];
   255→
   256→  var emitData = {
   257→    company: company,
   258→    partner: partner,
   259→    order: {
   260→      name: move.name,
   261→      number: String(nextNum),
   262→      date_order: move.invoice_date || move.date,
   263→      amount_total: vProdSum, // vNF (sem IBS/CBS) — o XML calcula vNFTot internamente
   264→      note: move.narration || '',
   265→    },
   266→    lines: linesData,
   267→    pagamentos: pagamentos,
   268→    config: {
   269→      serie: serie,
   270→      tpAmb: process.env.SIEG_TP_AMB || config.sieg.tpAmb || '2',
   271→      natOp: 'Venda de Mercadoria',
   272→      mod: tipo === 'nfe' ? '55' : '01',
   273→    },
   274→    tipo: tipo,
   275→  };
   276→
   277→  // NFS-e: build service block from product x_studio_ fields
   278→  if (tipo === 'nfse') {
   279→    emitData.service = buildServiceBlock(linesData, move);
   280→  }
   281→
   282→  // 8. Set status to 'processando'
   283→  var processVals = {};
   284→  if (tipo === 'nfe') processVals.x_studio_nfe_status = 'processando';
   285→  else processVals.x_studio_nfse_status = 'processando';
   286→  await executeKw(client, db, uid, pwd, 'account.move', 'write', [[moveId], processVals]);
   287→
   288→  // 9. Call SIEG API
   289→  console.log('[SIEG-EMIT] Enviando ao SIEG...');
   290→  var resultado;
   291→  try {
   292→    resultado = await emitirNota(emitData);
   293→  } catch (siegErr) {
   294→    // Captura erros HTTP (ex: 409 com detalhes no body)
   295→    var siegDetail = siegErr.response && siegErr.response.data;
   296→    var errMsg = 'SIEG HTTP ' + (siegErr.response ? siegErr.response.status : 'erro') + ': ';
   297→    if (siegDetail) {
   298→      // Tentar extrair ErrorMessage do formato SIEG (PascalCase)
   299→      var siegMsg = siegDetail.ErrorMessage || siegDetail.Message || siegDetail.message || '';
   300→      if (siegMsg) {
   301→        errMsg += siegMsg;
   302→      } else {
   303→        errMsg += typeof siegDetail === 'string' ? siegDetail : JSON.stringify(siegDetail).slice(0, 800);
   304→      }
   305→    } else {
   306→      errMsg += siegErr.message;
   307→    }
   308→    console.error('[SIEG-EMIT] Erro SIEG:', errMsg);
   309→    throw new Error(errMsg);
   310→  }
   311→  console.log('[SIEG-EMIT] SIEG retornou - sucesso: ' + resultado.sucesso + (resultado.httpStatus ? ' (HTTP ' + resultado.httpStatus + ')' : ''));
   312→
   313→  // Se SIEG retornou erro (IsSuccess=false), extrair mensagem detalhada
   314→  if (!resultado.sucesso && resultado.erro) {
   315→    console.error('[SIEG-EMIT] SIEG rejeitou o XML: ' + resultado.erro);
   316→  }
   317→
   318→  // 10. Parse result
   319→  var info = parseResult(resultado, tipo);
   320→  var xmlRetornoText = '';
   321→  var rawXmlAuth = resultado.resposta && (resultado.resposta.Xml || resultado.resposta.xml || resultado.resposta.xmlBase64);
   322→  if (rawXmlAuth) {
   323→    xmlRetornoText = (typeof rawXmlAuth === 'string' && rawXmlAuth.length > 200)
   324→      ? rawXmlAuth
   325→      : Buffer.from(rawXmlAuth, 'base64').toString('utf-8');
   326→  }
   327→
   328→  // 11. Update account.move with results
   329→  var updateVals = {};
   330→  if (info.sucesso) {
   331→    if (tipo === 'nfe') {
   332→      updateVals.x_studio_nfe_status = 'autorizada';
   333→      updateVals.x_studio_nfe_chave = info.chave;
   334→      updateVals.x_studio_nfe_protocolo = info.protocolo;
   335→    } else {
   336→      updateVals.x_studio_nfse_status = 'autorizada';
   337→      updateVals.x_studio_nfse_numero = String(nextNum);
   338→    }
   339→
   340→    // Increment NF number (only on authorization)
   341→    try {
   342→      var companyUpdate = {};
   343→      companyUpdate[numField] = nextNum;
   344→      await executeKw(client, db, uid, pwd, 'res.company', 'write', [[companyId], companyUpdate]);
   345→      console.log('[SIEG-EMIT] Sequencia ' + numField + ' = ' + nextNum);
   346→    } catch (seqErr) {
   347→      console.error('[SIEG-EMIT] Erro ao atualizar sequencia:', seqErr.message);
   348→    }
   349→  } else {
   350→    // Falha: NUNCA gravar texto livre num campo Selection (o Odoo descarta e a
   351→    // fatura some da fila). Grava um valor valido ('erro' ou 'pendente') e
   352→    // mantem o motivo detalhado no chatter.
   353→    var errStatus = statusOnError();
   354→    if (tipo === 'nfe') {
   355→      updateVals.x_studio_nfe_status = errStatus;
   356→    } else {
   357→      updateVals.x_studio_nfse_status = errStatus;
   358→    }
   359→    console.warn('[SIEG-EMIT] Emissao NAO autorizada -> status "' + errStatus
   360→      + '" (motivo: ' + (info.cStat ? info.cStat + ' - ' : '') + (info.motivo || 'n/d') + ')');
   361→  }
   362→
   363→  await writeStatusSafe(client, db, uid, pwd, moveId, tipo, updateVals);
   364→  console.log('[SIEG-EMIT] Fatura atualizada');
   365→
   366→  // 12. Post XML + PDF to Chatter
   367→  await postChatterResult(client, db, uid, pwd, moveId, move.name, tipo, info, resultado, xmlRetornoText);
   368→
   369→  return {
   370→    move_id: moveId, sucesso: info.sucesso, tipo: tipo,
   371→    chave: info.chave, numero: nextNum, serie: serie,
   372→  };
   373→}

   375→// ============================================================
   376→// Post XML + PDF to Odoo Chatter
   377→// ============================================================
   378→async function postChatterResult(client, db, uid, pwd, moveId, moveName, tipo, info, resultado, xmlRetornoText) {
   379→  var attachIds = [];
   380→  var safeName = (moveName || 'Fatura').replace(/[^a-zA-Z0-9\-_]/g, '_');
   381→  var nfLabel = tipo === 'nfe' ? 'NF-e' : 'NFS-e';
   382→  var xmlContent = xmlRetornoText || resultado.xmlEnviado || '';
   383→  var pdfBase64 = resultado.pdfBase64 || '';
   384→
   385→  // Attach XML
   386→  if (xmlContent) {
   387→    try {
   388→      var xmlB64 = Buffer.from(xmlContent, 'utf-8').toString('base64');
   389→      var xmlAttach = await executeKw(client, db, uid, pwd, 'ir.attachment', 'create', [{
   390→        name: nfLabel + '_' + safeName + '.xml',
   391→        raw: xmlB64,
   392→        res_model: 'account.move',
   393→        res_id: moveId,
   394→        mimetype: 'application/xml',
   395→      }]);
   396→      attachIds.push(xmlAttach);
   397→      console.log('[SIEG-EMIT] XML anexado (ID: ' + xmlAttach + ')');
   398→    } catch (e) { console.error('[SIEG-EMIT] Erro anexar XML:', e.message); }
   399→  }
   400→
   401→  // Attach PDF (DANFE / DANFSE)
   402→  if (pdfBase64) {
   403→    try {
   404→      var pdfLabel = tipo === 'nfe' ? 'DANFE' : 'DANFSE';
   405→      var pdfAttach = await executeKw(client, db, uid, pwd, 'ir.attachment', 'create', [{
   406→        name: pdfLabel + '_' + safeName + '.pdf',
   407→        raw: pdfBase64,
   408→        res_model: 'account.move',
   409→        res_id: moveId,
   410→        mimetype: 'application/pdf',
   411→      }]);
   412→      attachIds.push(pdfAttach);
   413→      console.log('[SIEG-EMIT] PDF anexado (ID: ' + pdfAttach + ')');
   414→    } catch (e) { console.error('[SIEG-EMIT] Erro anexar PDF:', e.message); }
   415→  }
   416→
   417→  // Post chatter message
   418→  var body = '';
   419→  if (info.sucesso) {
   420→    body = '<b>' + nfLabel + ' Emitida com Sucesso!</b><br/>';
   421→    if (tipo === 'nfe') {

   709→// ============================================================
   710→// NCM Extraction Helper
   711→// ============================================================
   712→/**
   713→ * Extrai codigo NCM (8 digitos) de um valor many2one do Odoo.
   714→ * Aceita: [id, "7308.90.90"], [id, "7308.90.90 - Descricao"], string pura, numero
   715→ */
   716→function extractNcmFromRef(ref) {
   717→  if (!ref) return '';
   718→  var str = '';
   719→  if (Array.isArray(ref)) {
   720→    str = String(ref[1] || ref[0] || '');
   721→  } else if (typeof ref === 'string') {
   722→    str = ref;
   723→  } else if (typeof ref === 'number') {
   724→    return String(ref).length === 8 ? String(ref) : '';
   725→  } else {
   726→    str = String(ref);
   727→  }
   728→  // Tentar regex NNNN.NN.NN
   729→  var m = str.match(/(\d{4})\.(\d{2})\.(\d{2})/);
   730→  if (m) return m[1] + m[2] + m[3];
   731→  // Fallback: so digitos
   732→  var digits = str.replace(/\D/g, '');
   733→  return digits.length === 8 ? digits : '';
   734→}

   972→async function extractTaxes(client, db, uid, pwd, line) {
   973→  // Padrão: Lucro Real (CRT 3) — CST 00, ICMS 18% PR interno
   974→  // PIS/COFINS não-cumulativo com base no valor do produto
   975→  var base = parseFloat(line.price_subtotal) || 0;
   976→  // Arredondamento: SEFAZ exige 2 casas decimais por item.
   977→  // Se nao arredondar aqui, o total pode divergir (cStat 602/603).
   978→  var r2 = function(n) { return Math.round((parseFloat(n) || 0) * 100) / 100; };
   979→  var result = {
   980→    csosn: '',
   981→    cst_icms: '00',
   982→    mod_bc: '3',
   983→    vbc: r2(base),
   984→    vicms: r2(base * 0.18),
   985→    picms: 18.00,
   986→    cst_pis: '01',
   987→    vbc_pis: r2(base),
   988→    ppis: 1.65,
   989→    vpis: r2(base * 0.0165),
   990→    cst_cofins: '01',
   991→    vbc_cofins: r2(base),
   992→    pcofins: 7.60,
   993→    vcofins: r2(base * 0.076),
   994→  };
   995→
   996→  var taxIds = (line.tax_ids || []).map(function(t) { return Array.isArray(t) ? t[0] : t; }).filter(Boolean);
   997→  if (!taxIds.length) return result;
   998→
   999→  try {
  1000→    var taxes = await executeKw(client, db, uid, pwd, 'account.tax', 'read', [taxIds, [
  1001→      'name', 'amount', 'amount_type', 'description',
  1002→    ]]);
  1003→    if (!taxes || !taxes.length) return result;
  1004→
  1005→    var totalIcms = 0, totalPis = 0, totalCofins = 0;
  1006→    var foundCsosn = '';
  1007→
  1008→    for (var i = 0; i < taxes.length; i++) {
  1009→      var tax = taxes[i];
  1010→      var n = ((tax.name || tax.description || '').toUpperCase());
  1011→      var amt = parseFloat(tax.amount) || 0;
  1012→      if (n.indexOf('ICMS') >= 0 || n.indexOf('CSOSN') >= 0) {
  1013→        totalIcms += amt;
  1014→        var m = n.match(/(10[0-9]|20[0-9]|300|400|500|900)/);
  1015→        if (m) foundCsosn = m[1];
  1016→      } else if (n.indexOf('PIS') >= 0) { totalPis += amt; }
  1017→      else if (n.indexOf('COFINS') >= 0) { totalCofins += amt; }
  1018→    }
  1019→
  1020→    // Sobrescreve defaults apenas se o Odoo tiver impostos reais configurados (> 0)
  1021→    if (foundCsosn) {
  1022→      result.csosn = foundCsosn;
  1023→      result.cst_icms = '';
  1024→    }
  1025→    if (totalIcms > 0) {
  1026→      result.picms = totalIcms;
  1027→      result.vicms = r2(base * (totalIcms / 100));
  1028→    }
  1029→    if (totalPis > 0) {
  1030→      result.ppis = totalPis;
  1031→      result.vpis = r2(base * (totalPis / 100));
  1032→    }
  1033→    if (totalCofins > 0) {
  1034→      result.pcofins = totalCofins;
  1035→      result.vcofins = r2(base * (totalCofins / 100));
  1036→    }
  1037→  } catch (e) { console.warn('[SIEG-EMIT] Erro ao extrair impostos:', e.message); }
  1038→
  1039→  return result;
  1040→}

  1045→function parseResult(resultado, tipo) {
  1046→  var resp = resultado.resposta || {};
  1047→  var chave = '', protocolo = '', cStat = '', motivo = '', numero = '', sucesso = false;
  1048→
  1049→  // Se SIEG retornou HTTP 4xx, usar ErrorMessage diretamente como motivo
  1050→  var siegErro = resultado.erro || '';
  1051→  if (resultado.httpStatus && resultado.httpStatus >= 400 && resultado.httpStatus < 500) {
  1052→    // Erro de validacao do XML — extrair mensagem do SIEG
  1053→    motivo = siegErro || 'Erro HTTP ' + resultado.httpStatus;
  1054→    if (tipo === 'nfe') {
  1055→      cStat = String(resultado.statusCode || resp.cStat || resp.status || '');
  1056→    }
  1057→    return { sucesso: false, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
  1058→  }
  1059→
  1060→  if (tipo === 'nfe') {
  1061→    chave = resp.chNFe || resp.chave || resp.ChaveXml || '';
  1062→    protocolo = resp.nProt || resp.protocolo || '';
  1063→    cStat = String(resp.cStat || resp.status || '');
  1064→    motivo = resp.xMotivo || resp.motivo || (siegErro || 'Processado');
  1065→    sucesso = (cStat === '100' || cStat === '104' || cStat === '150');
  1066→  } else {
  1067→    chave = resp.Chave || resp.chave || '';
  1068→    protocolo = resp.Protocolo || resp.protocolo || '';
  1069→    numero = resp.NumeroNfse || resp.nDFSe || resp.numero || '';
  1070→    cStat = String(resp.CodigoVerificacao || resp.codigo || '');
  1071→    motivo = resp.Motivo || resp.motivo || siegErro || (resultado.sucesso ? 'Autorizada' : 'Erro na emissao');
  1072→    sucesso = resultado.sucesso;
  1073→  }
  1074→
  1075→  return { sucesso: sucesso, chave: chave, protocolo: protocolo, cStat: cStat, motivo: motivo, numero: numero };
  1076→}

  1271→/**
  1272→ * Normaliza nome de cidade: remove acentos, cedilha, etc.
  1273→ * Para matching na tabela IBGE fallback.
  1274→ */
  1275→function normalizeCityName(name) {
  1276→  if (!name) return '';
  1277→  return name
  1278→    .normalize('NFD')
  1279→    .replace(/[\u0300-\u036f]/g, '') // remove diacriticos
  1280→    .replace(/\u00e7/g, 'c')          // cedilha -> c
  1281→    .replace(/\u00c7/g, 'C')
  1282→    .toUpperCase()
  1283→    .trim();
  1284→}
  1285→
  1286→function lookupIbgeFallback(cityName, stateCode) {
  1287→  if (!cityName || !stateCode) return '';
  1288→  var key = (normalizeCityName(cityName) + '/' + stateCode.toUpperCase().trim());
  1289→  var code = IBGE_FALLBACK[key];
  1290→  if (code) {
  1291→    console.log('[SIEG-EMIT] IBGE via tabela fallback: ' + key + ' = ' + code);
  1292→    return code;
  1293→  }
  1294→  // Fallback: tenta sem preposicoes comuns ("do", "da", "de", "das", "dos")
  1295→  var simplified = key.replace(/\b(D|DA|DE|DO|DAS|DOS)\b/g, '').replace(/\s+/g, ' ').trim();
  1296→  code = IBGE_FALLBACK[simplified];
  1297→  if (code) {
  1298→    console.log('[SIEG-EMIT] IBGE via tabela fallback (simplificado): ' + simplified + ' = ' + code);
  1299→    return code;
  1300→  }
  1301→  return '';
  1302→}

  1304→function tupId(val) {
  1305→  if (Array.isArray(val)) return val[0] || 0;
  1306→  if (typeof val === 'number') return val;
  1307→  return parseInt(val) || 0;

  1310→/**
  1311→ * Valor gravado no campo Selection quando a emissao falha.
  1312→ * NFE_STATUS_ON_ERROR=pendente -> a fatura volta para a fila e e retentada.
  1313→ * Padrao 'erro' -> exige reprocessamento manual.
  1314→ */
  1315→function statusOnError() {
  1316→  var v = String(process.env.NFE_STATUS_ON_ERROR || 'erro').trim().toLowerCase();
  1317→  return v === 'pendente' ? 'pendente' : 'erro';
  1318→}

  1320→/**
  1321→ * Grava o status com fallback: se o Odoo recusar o valor da Selection,
  1322→ * tenta 'pendente' para a fatura nunca ficar fora da fila silenciosamente.
  1323→ */
  1324→async function writeStatusSafe(client, db, uid, pwd, moveId, tipo, vals) {
  1325→  var field = tipo === 'nfe' ? 'x_studio_nfe_status' : 'x_studio_nfse_status';
  1326→  try {
  1327→    await executeKw(client, db, uid, pwd, 'account.move', 'write', [[moveId], vals]);
  1328→  } catch (e) {
  1329→    console.error('[SIEG-EMIT] Falha ao gravar status (' + vals[field] + '): ' + e.message);
  1330→    var fb = {};
  1331→    fb[field] = 'pendente';
  1332→    try {
  1333→      await executeKw(client, db, uid, pwd, 'account.move', 'write', [[moveId], fb]);
  1334→      console.warn('[SIEG-EMIT] Status revertido para "pendente" (retentativa possivel)');
  1335→    } catch (e2) { console.error('[SIEG-EMIT] Falha no fallback de status:', e2.message); }
  1336→  }
  1337→}

  1354→// ============================================================
  1355→// Process Pending Cancellations (polling para status 'cancelando')
  1356→// ============================================================
  1357→/**
  1358→ * Busca faturas com x_studio_nfe_status = 'cancelando' e envia o
  1359→ * evento de cancelamento à SEFAZ PR, depois atualiza o Odoo.
  1360→ * Chamado pelo runSiegPoll no server.js junto com processPendingEmissions.
  1361→ */
  1362→async function processPendingCancellations() {
  1363→  var odoo = config.odoo;
  1364→  if (!odoo || !odoo.enabled || !odoo.url) {
  1365→    return { processed: 0, reason: 'odoo_not_configured' };
  1366→  }
  1367→
  1368→  var client = createClient(odoo.url);
  1369→  var uid = await authenticate(client, odoo.db, odoo.user, odoo.password);
  1370→  var db = odoo.db;
  1371→  var pwd = odoo.password;
  1372→
  1373→  var ids = await executeKw(client, db, uid, pwd, 'account.move', 'search', [[
  1374→    ['move_type', '=', 'out_invoice'],
  1375→    ['x_studio_nfe_status', '=', 'cancelando'],
  1376→  ]], { order: 'id asc', limit: 5 });
  1377→
  1378→  if (!ids.length) return { processed: 0 };
  1379→
  1380→  console.log('[SIEG-CANCEL-POLL] ' + ids.length + ' fatura(s) aguardando cancelamento');
  1381→
  1382→  var { cancelarNFe } = require('./nfe-cancelamento');
  1383→
  1384→  var results = [];
  1385→  for (var i = 0; i < ids.length; i++) {
  1386→    var moveId = ids[i];
  1387→    try {
  1388→      var r = await cancelOneInvoice(client, db, uid, pwd, moveId, cancelarNFe);
  1389→      results.push(r);
  1390→    } catch (err) {
  1391→      console.error('[SIEG-CANCEL-POLL] ERRO fatura ' + moveId + ':', err.message);
  1392→      try {
  1393→        await executeKw(client, db, uid, pwd, 'account.move', 'write',
  1394→          [[moveId], { x_studio_nfe_status: 'erro' }]);
  1395→        await executeKw(client, db, uid, pwd, 'mail.message', 'create', [{
  1396→          model: 'account.move', res_id: moveId,
  1397→          body: '<b>Erro no Cancelamento</b><br/>' + err.message.substring(0, 500),
  1398→          message_type: 'comment',
  1399→        }]);
  1400→      } catch (e2) { console.error('[SIEG-CANCEL-POLL] Falha ao gravar erro:', e2.message); }
  1401→      results.push({ move_id: moveId, sucesso: false, erro: err.message });
  1402→    }
  1403→  }
  1404→
  1405→  var ok = results.filter(function(r) { return r.sucesso; }).length;
  1406→  console.log('[SIEG-CANCEL-POLL] Concluido: ' + ok + '/' + results.length + ' cancelada(s)');
  1407→  return { processed: results.length, sucesso: ok };
  1408→}

  1410→async function cancelOneInvoice(client, db, uid, pwd, moveId, cancelarNFe) {
  1411→  var moves = await executeKw(client, db, uid, pwd, 'account.move', 'read',
  1412→    [[moveId], ['name', 'x_studio_nfe_chave', 'x_studio_nfe_protocolo',
  1413→                'x_studio_nfe_status', 'company_id']]);
  1414→  if (!moves || !moves.length) throw new Error('Fatura ' + moveId + ' nao encontrada');
  1415→  var move = moves[0];
  1416→
  1417→  var chave     = (move.x_studio_nfe_chave || '').replace(/\D/g, '');
  1418→  var protocolo = String(move.x_studio_nfe_protocolo || '').trim();
  1419→  var justificativa = 'Cancelamento solicitado pelo emitente via Odoo';
  1420→  if (justificativa.length < 15) justificativa = 'Cancelamento solicitado pelo emitente via Odoo';
  1421→
  1422→  console.log('[SIEG-CANCEL-POLL] Fatura ' + move.name + ' | chave=' + chave.slice(0, 15) + '...');
  1423→
  1424→  if (!chave || chave.length !== 44) throw new Error('Chave de acesso invalida (' + chave.length + ' digitos). Nota precisa estar autorizada.');
  1425→  if (!protocolo) throw new Error('Protocolo de autorizacao ausente na fatura ' + move.name);
  1426→
  1427→  var companyId = Array.isArray(move.company_id) ? move.company_id[0] : move.company_id;
  1428→  var companies = await executeKw(client, db, uid, pwd, 'res.company', 'read', [[companyId], ['vat']]);
  1429→  var cnpj = ((companies && companies[0] && companies[0].vat) || '').replace(/\D/g, '');
  1430→  if (!cnpj) throw new Error('CNPJ do emitente nao encontrado na empresa');
  1431→
  1432→  // Marcar como processando para nao processar duas vezes em paralelo
  1433→  await executeKw(client, db, uid, pwd, 'account.move', 'write',
  1434→    [[moveId], { x_studio_nfe_status: 'processando' }]);
  1435→
  1436→  var resultado = await cancelarNFe({ chave, protocolo, cnpj, justificativa });
  1437→
  1438→  var statusNovo = resultado.sucesso ? 'cancelada' : 'erro';
  1439→  await executeKw(client, db, uid, pwd, 'account.move', 'write',
  1440→    [[moveId], {
  1441→      x_studio_nfe_status: statusNovo,
  1442→      x_studio_nfe_protocolo: resultado.nProt || protocolo,
  1443→    }]);
  1444→
  1445→  if (resultado.sucesso) {
  1446→    try {
  1447→      await executeKw(client, db, uid, pwd, 'account.move', 'button_draft', [[moveId]]);
  1448→      await executeKw(client, db, uid, pwd, 'account.move', 'button_cancel', [[moveId]]);
  1449→      console.log('[SIEG-CANCEL-POLL] Fatura ' + move.name + ' revertida no Odoo');
  1450→    } catch (e) {
  1451→      console.warn('[SIEG-CANCEL-POLL] Nao foi possivel reverter a fatura: ' + e.message);
  1452→    }
  1453→  }
  1454→
  1455→  var msgCorpo = resultado.sucesso
  1456→    ? '<b>NF-e Cancelada na SEFAZ</b><br/>'
  1457→      + 'Protocolo: ' + (resultado.nProt || protocolo) + '<br/>'
  1458→      + 'Data: ' + (resultado.dhRecbto || '') + '<br/>'
  1459→      + 'Justificativa: ' + justificativa
  1460→    : '<b>Cancelamento rejeitado pela SEFAZ</b><br/>'
  1461→      + 'Status: ' + resultado.cStat + ' - ' + resultado.xMotivo + '<br/>'
  1462→      + 'Justificativa enviada: ' + justificativa;
  1463→
  1464→  try {
  1465→    await executeKw(client, db, uid, pwd, 'mail.message', 'create', [{
  1466→      model: 'account.move', res_id: moveId,
  1467→      body: msgCorpo, message_type: 'comment',
  1468→    }]);
  1469→  } catch (e) { console.warn('[SIEG-CANCEL-POLL] Erro ao postar chatter:', e.message); }
  1470→
  1471→  return { move_id: moveId, sucesso: resultado.sucesso, fatura: move.name, cStat: resultado.cStat };
  1472→}
